\section{Problemas resueltos de estimación por intervalos y tamaño muestral}

\subsection{Enunciados de los problemas}

\begin{problema}
	\label{prob:dif_medias_indep_iguales}
	Un ingeniero de ciencia de datos evalúa el tiempo de latencia (en milisegundos) de dos arquitecturas de bases de datos en la nube (Arquitectura A y Arquitectura B) bajo cargas de trabajo similares. Extrae dos muestras aleatorias independientes con los siguientes resultados:
	\begin{itemize}
		\item \textbf{Arquitectura A:} $n_1 = 15$, media muestral $\bar{X}_1 = 48.5$ ms, desviación estándar $S_1 = 6.2$ ms.
		\item \textbf{Arquitectura B:} $n_2 = 12$, media muestral $\bar{X}_2 = 53.8$ ms, desviación estándar $S_2 = 5.8$ ms.
	\end{itemize}
	Asumiendo que las poblaciones son normales y poseen varianzas iguales ($\sigma_1^2 = \sigma_2^2$), construya un intervalo de confianza del $95\%$ para la verdadera diferencia de latencias medias $\mu_1 - \mu_2$.
\end{problema}

\begin{problema}
	\label{prob:dif_medias_welch}
	En un estudio sobre el rendimiento de dos algoritmos de optimización (Algoritmo 1 y Algoritmo 2) aplicados a redes neuronales convolucionales, se midió el tiempo de convergencia en segundos. Como la variabilidad de uno de ellos parece mucho mayor, se no se asumen varianzas iguales ($\sigma_1^2 \neq \sigma_2^2$). Los datos obtenidos son:
	\begin{itemize}
		\item \textbf{Algoritmo 1:} $n_1 = 10$, $\bar{X}_1 = 120.4$ s, $S_1 = 18.5$ s.
		\item \textbf{Algoritmo 2:} $n_2 = 14$, $\bar{X}_2 = 105.2$ s, $S_2 = 7.3$ s.
	\end{itemize}
	Construya un intervalo de confianza del $99\%$ para $\mu_1 - \mu_2$ utilizando la aproximación de Welch.
\end{problema}

\begin{problema}
	\label{prob:dif_proporciones_ab}
	En una prueba A/B para un sitio web de comercio electrónico, se muestran al azar dos versiones de un formulario de pago a los visitantes. De $n_1 = 800$ usuarios expuestos a la Versión A (diseño clásico), $X_1 = 144$ completaron la compra. De $n_2 = 850$ usuarios expuestos a la Versión B (diseño minimalista de un solo paso), $X_2 = 187$ completaron la compra.
	\begin{enumerate}
		\item Construya un intervalo de confianza del $95\%$ para la verdadera diferencia entre las tasas de conversión poblacionales $p_1 - p_2$.
		\item ¿Justifican los datos la decisión de adoptar definitivamente la Versión B?
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:ic_varianza}
	Un equipo de analítica financiera monitorea la volatilidad diaria de un índice bursátil de tecnología. Durante un mes bursátil de $n = 21$ días hábiles, los rendimientos logarítmicos muestran una desviación estándar muestral de $S = 1.45\%$. Asumiendo que los rendimientos diarios siguen una distribución normal, construya un intervalo de confianza del $90\%$ para la verdadera varianza poblacional $\sigma^2$ y para la desviación estándar $\sigma$.
\end{problema}

\begin{problema}
	\label{prob:ic_razon_varianzas}
	Para verificar si dos sensores IoT de temperatura industrial fabricados por distintos proveedores tienen exactamente la misma varianza y precisión en sus mediciones, se toman dos muestras independientes:
	\begin{itemize}
		\item \textbf{Proveedor 1:} $n_1 = 16$ mediciones, con varianza muestral $S_1^2 = 0.36$ $^\circ$C$^2$.
		\item \textbf{Proveedor 2:} $n_2 = 13$ mediciones, con varianza muestral $S_2^2 = 0.12$ $^\circ$C$^2$.
	\end{itemize}
	Construya un intervalo de confianza del $98\%$ para la razón de varianzas poblacionales $\frac{\sigma_1^2}{\sigma_2^2}$. ¿Existe evidencia estadística para afirmar que un proveedor tiene mayor varianza que el otro?
\end{problema}

\begin{problema}
	\label{prob:n_media}
	Un equipo de investigación desea estimar el tiempo promedio en que un usuario permanece interactuando en una plataforma educativa virtual antes de cerrar sesión. Por datos preliminares en plataformas análogas, se sabe que la desviación estándar de la duración de las sesiones ronda los $\sigma = 18$ minutos.
	\begin{enumerate}
		\item ¿Qué tamaño de muestra se requiere para estimar la media poblacional $\mu$ con un margen de error máximo tolerable de $E = 2$ minutos al nivel de confianza del $95\%$?
		\item Si el presupuesto solo permite monitorear un máximo de $n = 150$ sesiones, ¿cuál sería el margen de error real alcanzado con el $95\%$ de confianza?
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:n_proporcion}
	Una firma analítica proyecta realizar una encuesta nacional para determinar la proporción $p$ de hogares urbanos que utilizan servicios de inteligencia artificial generativa en su vida cotidiana.
	\begin{enumerate}
		\item Calcule el tamaño de muestra mínimo requerido para que la estimación tenga un margen de error no mayor a $3\%$ ($E = 0.03$) con un nivel de confianza del $99\%$, asumiendo que no se dispone de ninguna encuesta previa (escenario más conservador).
		\item Suponga que un estudio piloto anterior sugirió que dicha proporción se sitúa en torno al $20\%$ ($\hat{p}^* = 0.20$). ¿Cuántas encuestas se requerirían bajo este nuevo escenario con el mismo nivel de precisión y confianza? ¿Qué porcentaje de ahorro en el tamaño muestral representa contar con esta información previa?
	\end{enumerate}
\end{problema}


\subsection{Soluciones detalladas}

\begin{solucion}
	\textbf{Paso 1: Cálculo de la varianza muestral combinada ($S_p^2$).}
	Dado que asumimos homoscedasticidad, calculamos la desviación estándar agrupada o combinada $S_p$:
	\begin{align}
		S_p^2 &= \frac{(n_1 - 1)S_1^2 + (n_2 - 1)S_2^2}{n_1 + n_2 - 2} = \frac{(15 - 1)(6.2)^2 + (12 - 1)(5.8)^2}{15 + 12 - 2} \\
		&= \frac{14(38.44) + 11(33.64)}{25} = \frac{538.16 + 370.04}{25} = \frac{908.2}{25} = 36.328 \text{ ms}^2.
	\end{align}
	Por tanto, $S_p = \sqrt{36.328} \approx 6.027$ ms.
	
	\textbf{Paso 2: Obtención del valor crítico de $t$.}
	Los grados de libertad totales son $\nu = n_1 + n_2 - 2 = 25$. Para un nivel de confianza del $95\%$ ($\alpha = 0.05$), buscamos el valor crítico $t_{0.025, 25}$ en las tablas de la distribución $t$ de Student:
	\begin{align}
		t_{0.025, 25} = 2.060.
	\end{align}
	
	\textbf{Paso 3: Construcción del intervalo de confianza.}
	La diferencia de medias muestrales es $\bar{X}_1 - \bar{X}_2 = 48.5 - 53.8 = -5.3$ ms. El margen de error $E$ es:
	\begin{align}
		E = t_{0.025, 25} \cdot S_p \sqrt{\frac{1}{n_1} + \frac{1}{n_2}} = 2.060 \cdot (6.027) \sqrt{\frac{1}{15} + \frac{1}{12}}.
	\end{align}
	Calculamos el término radical: $\sqrt{0.0667 + 0.0833} = \sqrt{0.1500} \approx 0.3873$. Entonces:
	\begin{align}
		E = 2.060 \times 6.027 \times 0.3873 \approx 4.808 \text{ ms}.
	\end{align}
	
	El intervalo de confianza del $95\%$ resulta en:
	\begin{align}
		-5.3 - 4.808 \leq \mu_1 - \mu_2 \leq -5.3 + 4.808 \implies [-10.108, \, -0.492] \text{ ms}.
	\end{align}
	
	\textbf{Interpretación:} Con un $95\%$ de confianza, la verdadera diferencia de latencias medias entre la Arquitectura A y la B se encuentra entre $-10.11$ ms y $-0.49$ ms. Dado que todo el intervalo es estrictamente negativo y no incluye el cero, tenemos evidencia estadística significativa de que la Arquitectura A es más rápida (tiene menor latencia media) que la Arquitectura B.
\end{solucion}

\begin{solucion}
	\textbf{Paso 1: Cálculo de los grados de libertad de Welch-Satterthwaite ($\nu$).}
	Las varianzas muestrales divididas entre sus respectivos tamaños son:
	\begin{align}
		\frac{S_1^2}{n_1} &= \frac{18.5^2}{10} = \frac{342.25}{10} = 34.225, \\
		\frac{S_2^2}{n_2} &= \frac{7.3^2}{14} = \frac{53.29}{14} \approx 3.8064.
	\end{align}
	Sustituyendo en la fórmula de Welch:
	\begin{align}
		\nu &= \frac{(34.225 + 3.8064)^2}{\frac{34.225^2}{10 - 1} + \frac{3.8064^2}{14 - 1}} = \frac{(38.0314)^2}{\frac{1171.3506}{9} + \frac{14.4887}{13}} = \frac{1446.3874}{130.1501 + 1.1145} \\
		&= \frac{1446.3874}{131.2646} \approx 11.018.
	\end{align}
	Redondeando de forma conservadora al entero inmediato inferior, tomamos $\nu = 11$ grados de libertad.
	
	\textbf{Paso 2: Valor crítico de la distribución $t$.}
	Para un nivel del $99\%$ ($\alpha = 0.01 \implies \alpha/2 = 0.005$) y $\nu = 11$ grados de libertad, el valor crítico en la tabla $t$ es:
	\begin{align}
		t_{0.005, 11} = 3.106.
	\end{align}
	
	\textbf{Paso 3: Construcción del intervalo.}
	La diferencia puntual de las medias es $\bar{X}_1 - \bar{X}_2 = 120.4 - 105.2 = 15.2$ s. El error estándar es:
	\begin{align}
		SE(\bar{X}_1 - \bar{X}_2) = \sqrt{\frac{S_1^2}{n_1} + \frac{S_2^2}{n_2}} = \sqrt{34.225 + 3.8064} = \sqrt{38.0314} \approx 6.1670 \text{ s}.
	\end{align}
	El margen de error resulta: $E = 3.106 \times 6.1670 \approx 19.155$ s.
	
	El intervalo de confianza del $99\%$ es:
	\begin{align}
		15.2 - 19.155 \leq \mu_1 - \mu_2 \leq 15.2 + 19.155 \implies [-3.955, \, 34.355] \text{ s}.
	\end{align}
	
	\textbf{Interpretación:} Como el intervalo contiene al cero (va desde valores negativos hasta positivos), con un $99\%$ de confianza no podemos concluir que exista una diferencia estadísticamente significativa entre el tiempo de convergencia medio de ambos algoritmos.
\end{solucion}

\begin{solucion}
	\textbf{Parte 1: Cálculo de las proporciones muestrales y errores estándares.}
	Las proporciones empíricas de conversión son:
	\begin{align}
		\hat{p}_1 &= \frac{144}{800} = 0.18 \quad (18\%), \\
		\hat{p}_2 &= \frac{187}{850} = 0.22 \quad (22\%).
	\end{align}
	La estimación puntual de la diferencia es $\hat{p}_1 - \hat{p}_2 = 0.18 - 0.22 = -0.04$.
	
	Verificamos las condiciones de tamaño muestral: $800(0.18) = 144 \geq 10$, $800(0.82) = 656 \geq 10$, y de igual forma para la segunda muestra ($187$ y $663$). La aproximación normal es perfectamente válida.
	
	El error estándar de la diferencia de proporciones es:
	\begin{align}
		SE(\hat{p}_1 - \hat{p}_2) &= \sqrt{\frac{\hat{p}_1(1-\hat{p}_1)}{n_1} + \frac{\hat{p}_2(1-\hat{p}_2)}{n_2}} = \sqrt{\frac{0.18(0.82)}{800} + \frac{0.22(0.78)}{850}} \\
		&= \sqrt{\frac{0.1476}{800} + \frac{0.1716}{850}} = \sqrt{0.0001845 + 0.0002019} = \sqrt{0.0003864} \approx 0.01966.
	\end{align}
	
	Para una confianza del $95\%$ ($Z_{0.025} = 1.96$), el margen de error es:
	\begin{align}
		E = 1.96 \times 0.01966 \approx 0.0385 \quad (3.85\%).
	\end{align}
	
	El intervalo de confianza del $95\%$ resulta en:
	\begin{align}
		-0.04 - 0.0385 \leq p_1 - p_2 \leq -0.04 + 0.0385 \implies [-0.0785, \, -0.0015].
	\end{align}
	
	\textbf{Parte 2: Decisión sobre la adopción del diseño.}
	El intervalo de confianza para $p_1 - p_2$ es $[-7.85\%, -0.15\%]$. Dado que el límite superior del intervalo es estrictamente negativo (es decir, el intervalo excluye al cero), se demuestra que la tasa de conversión poblacional $p_2$ es significativamente superior a $p_1$ al nivel del $95\%$ de confianza. Por lo tanto, los datos justifican fehacientemente la adopción de la Versión B.
\end{solucion}

	\textbf{Paso 1: Grados de libertad y cuantiles chi-cuadrada.}
	Los grados de libertad asociados al estadístico muestral son $\nu = n - 1 = 21 - 1 = 20$.
	Para un nivel de confianza del $90\%$ ($\alpha = 0.10$), las colas superior e inferior dejan un área de $\alpha/2 = 0.05$ y $1-\alpha/2 = 0.95$, respectivamente. Consultando las tablas de la distribución $\chi^2$ con $\nu = 20$:
	\begin{align}
		\chi^2_{0.05, 20} &= 31.410, \\
		\chi^2_{0.95, 20} &= 10.851.
	\end{align}
	
	\textbf{Paso 2: Intervalo para la varianza $\sigma^2$.}
	El término del numerador $(n-1)S^2$ es:
	\begin{align}
		(20)(1.45)^2 = 20 \times 2.1025 = 42.05.
	\end{align}
	
	Sustituyendo en los límites teóricos para $\sigma^2$:
	\begin{align}
		\frac{42.05}{31.410} \leq \sigma^2 \leq \frac{42.05}{10.851} \implies [1.3387, \, 3.8752] \text{ (\%}^2\text{)}.
	\end{align}
	
	\textbf{Paso 3: Intervalo para la desviación estándar $\sigma$.}
	Tomando la raíz cuadrada positiva en todos los miembros de la desigualdad:
	\begin{align}
		\sqrt{1.3387} \leq \sigma \leq \sqrt{3.8752} \implies [1.157, \, 1.969] \%.
	\end{align}
	
	\textbf{Interpretación:} Con un $90\%$ de confianza, la verdadera volatilidad diaria del índice bursátil de tecnología se sitúa entre $1.16\%$ y $1.97\%$.
\end{solucion}

	\textbf{Paso 1: Grados de libertad y puntos críticos $F$.}
	Los grados de libertad son $\nu_1 = n_1 - 1 = 15$ (numerador) y $\nu_2 = n_2 - 1 = 12$ (denominador).
	Para el nivel de confianza del $98\%$ ($\alpha = 0.02 \implies \alpha/2 = 0.01$), el cuantil superior de la distribución de Fisher-Snedecor es:
	\begin{align}
		F_{0.01, 15, 12} = 3.96.
	\end{align}
	Para el cuantil inferior, aplicamos la propiedad de reciprocidad de la distribución $F$:
	\begin{align}
		F_{0.99, 15, 12} = \frac{1}{F_{0.01, 12, 15}} = \frac{1}{3.67} \approx 0.2725.
	\end{align}
	
	\textbf{Paso 2: Cálculo del intervalo para el cociente de varianzas.}
	La razón de las varianzas muestrales observadas es:
	\begin{align}
		\frac{S_1^2}{S_2^2} = \frac{0.36}{0.12} = 3.00.
	\end{align}
	
	Sustituyendo en la fórmula del intervalo de confianza:
	\begin{align}
		3.00 \cdot \frac{1}{3.96} \leq \frac{\sigma_1^2}{\sigma_2^2} \leq 3.00 \cdot \frac{1}{0.2725} \implies [0.7576, \, 11.009].
	\end{align}
	
	\textbf{Interpretación y conclusión:} El intervalo de confianza del $98\%$ para la razón de varianzas va desde $0.758$ hasta $11.009$. Dado que el intervalo contiene explícitamente el valor $1.0$ (que correspondería al escenario en el que $\sigma_1^2 = \sigma_2^2$), no existe evidencia estadística al nivel del $98\%$ para afirmar que el Proveedor 1 tenga una variabilidad poblacional diferente o superior a la del Proveedor 2.
\end{solucion}

\begin{solucion}
	\textbf{Parte 1: Tamaño de muestra requerido ($n$).}
	Para un nivel del $95\%$ de confianza, el valor normal estandarizado es $Z_{0.025} = 1.96$. La dispersión estimada es $\sigma = 18$ min y el margen de error máximo es $E = 2$ min.
	
	Sustituyendo en la fórmula fundamental para medias:
	\begin{align}
		n = \left( \frac{Z_{\alpha/2} \cdot \sigma}{E} \right)^2 = \left( \frac{1.96 \times 18}{2} \right)^2 = (1.96 \times 9)^2 = (17.64)^2 = 311.1696.
	\end{align}
	Dado que los tamaños muestrales representan conteos de personas o sesiones enteras y debemos garantizar que el error nunca exceda el límite pactado, siempre se redondea hacia el entero superior inmediato:
	\begin{align}
		n \geq 312 \text{ sesiones}.
	\end{align}
	
	\textbf{Parte 2: Margen de error para un tamaño fijo de $n = 150$.}
	Despejando $E$ de la fórmula del intervalo de confianza para la media:
	\begin{align}
		E = Z_{\alpha/2} \frac{\sigma}{\sqrt{n}} = 1.96 \times \frac{18}{\sqrt{150}} = \frac{35.28}{12.2474} \approx 2.8806 \text{ minutos}.
	\end{align}
	Por tanto, con un presupuesto limitado a $150$ observaciones, el margen de error del estudio aumentaría a $2.88$ minutos al mismo $95\%$ de confianza.
\end{solucion}

\begin{solucion}
	\textbf{Parte 1: Escenario conservador ($\hat{p}^* = 0.5$).}
	Para un nivel de confianza del $99\%$, el cuantil normal es $Z_{0.005} = 2.576$. El margen de error fijado es $E = 0.03$. Al no tener información a priori, utilizamos la varianza máxima $p^*(1-p^*) = 0.5 \times 0.5 = 0.25$:
	\begin{align}
		n_{\text{max}} = \frac{Z_{\alpha/2}^2}{4 E^2} = \frac{(2.576)^2}{4 (0.03)^2} = \frac{6.635776}{4 \times 0.0009} = \frac{6.635776}{0.0036} = 1843.27.
	\end{align}
	Redondeando al entero superior, se necesitan exactamente $n = 1844$ hogares en la encuesta.
	
	\textbf{Parte 2: Escenario con información previa ($\hat{p}^* = 0.20$).}
	Ahora utilizamos la estimación de varianza $p^*(1-p^*) = 0.20(0.80) = 0.16$:
	\begin{align}
		n = \left( \frac{Z_{\alpha/2}}{E} \right)^2 p^*(1-p^*) = \left( \frac{2.576}{0.03} \right)^2 (0.16) = (85.8667)^2 \times 0.16 = 7373.084 \times 0.16 = 1179.69.
	\end{align}
	Redondeando al entero superior, se precisan $n = 1180$ hogares.
	
	\textbf{Porcentaje de ahorro:}
	El ahorro absoluto en encuestas por recolectar es $1844 - 1180 = 664$ hogares. El porcentaje de reducción resulta en:
	\begin{align}
		\text{Ahorro} = \frac{664}{1844} \times 100\% \approx 36.01\%.
	\end{align}
	Contar con un estudio piloto o conocimiento experto que acote la proporción lejos de $0.5$ reduce los costos de recolección de datos en más de un $36\%$.
\end{solucion}
