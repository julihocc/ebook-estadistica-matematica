\begin{problema}
	\label{prob:2.5.6}
	En un grupo de estudiantes de estadística, el 60\% son mujeres ($M$) y el 40\% son hombres ($H$). Se sabe que el 10\% de las mujeres son zurdas ($Z$) y el 15\% de los hombres son zurdos. Si se selecciona una persona al azar y resulta ser zurda, calcula aplicando el Teorema de Bayes la probabilidad de que sea mujer.
\end{problema}

\begin{problema}
	\label{prob:2.5.4}
	Un estudiante se prepara para un examen general. Si estudia debidamente ($E$), la probabilidad de aprobar ($A$) es de 0.90. Si no estudia ($E'$), la probabilidad de aprobar es de sólo 0.30. La probabilidad de que el estudiante estudie es de 0.70.
	\begin{enumerate}
		\item ¿Cuál es la probabilidad total de que el estudiante apruebe el examen?
		\item Si el estudiante aprobó, ¿cuál es la probabilidad condicional de que haya estudiado?
		\item Si el estudiante no aprobó, ¿cuál es la probabilidad de que no haya estudiado?
	\end{enumerate}
\end{problema}

\subsubsection*{2. Nivel Operativo (Intermedios / De Desarrollo)}

\begin{problema}
	\label{prob:2.5.7}
	La Caja 1 contiene 3 canicas rojas y 2 azules, mientras que la Caja 2 contiene 2 canicas rojas y 8 azules.
	Se lanzan dos dados balanceados y se calcula la suma de sus caras: si se obtiene una suma de a lo más seis puntos, se elige una canica de la Caja 1; en cualquier otro caso, se elige una canica de la Caja 2.
	\begin{enumerate}
		\item Calcula la probabilidad marginal de que la canica elegida sea roja.
		\item Si quien lanzó los dados no revela la suma obtenida, pero sí muestra que la canica elegida fue roja, ¿cuál es la probabilidad a posteriori de que se haya extraído de la Caja 1?
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:2.5.2}
	Un test médico de diagnóstico clínico para detectar una patología tiene las siguientes métricas condicionales:
	\begin{itemize}
		\item Sensibilidad ($P(+|E)$): Si la persona tiene la enfermedad, el test da positivo con probabilidad 0.95.
		\item Especificidad ($P(-|E')$): Si la persona no tiene la enfermedad, el test da negativo con probabilidad 0.90.
	\end{itemize}
	Si la prevalencia de la enfermedad en la población es del 2\% ($P(E)=0.02$), calcula:
	\begin{enumerate}
		\item La probabilidad de que una persona elegida al azar obtenga un resultado positivo en el test.
		\item El Valor Predictivo Positivo (VPP): la probabilidad de que quien dio positivo realmente tenga la enfermedad.
		\item El Valor Predictivo Negativo (VPN): la probabilidad de que quien dio negativo realmente esté sano.
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:2.5.5}
	Un canal digital de comunicación transmite señales binarias (0 o 1). Por ruido en el canal, la probabilidad de que un bit 0 transmitido se reciba erróneamente como 1 es de 0.02, y la probabilidad de que un bit 1 transmitido se reciba como 0 es de 0.03. Si el 60\% de los bits transmitidos desde la fuente son 0:
	\begin{enumerate}
		\item ¿Cuál es la probabilidad total de que un bit recibido sea 1?
		\item Si en el receptor se registra un 1, ¿cuál es la probabilidad exacta de que realmente se haya transmitido un 1 desde la fuente?
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:2.5.8}
	Demuestra rigurosamente el Teorema de Bayes para una partición finita y exhaustiva $\{B_1, B_2, \dots, B_k\}$ del espacio muestral $S$ dado un evento observable $A$ con $P(A)>0$:
	\begin{align}
		P(B_j | A) = \frac{P(A|B_j)P(B_j)}{\sum_{i=1}^k P(A|B_i)P(B_i)}, \quad \text{para todo } j \in \{1, \dots, k\}.
	\end{align}
\end{problema}

\subsubsection*{4. Nivel Desafiante (Expertos / Deducción Profunda)}

\begin{problema}
	\label{prob:2.5.9}
	\textbf{La Falacia del Fiscal y la razón de momios:} En una investigación forense se recupera una muestra biológica en la escena del crimen. El perfil de ADN extraído tiene una probabilidad de coincidencia aleatoria en la población de $\alpha = 10^{-6}$ (es decir, $P(\text{coincidencia} | \text{inocente}) = 10^{-6}$). Un sospechoso es identificado mediante un rastreo masivo y su ADN coincide perfectamente con la muestra ($P(\text{coincidencia} | \text{culpable}) = 1$). El fiscal afirma ante el jurado que, puesto que la probabilidad de coincidencia aleatoria es de apenas una en un millón, la probabilidad de que el sospechoso sea inocente es del $0.0001\%$.
	
	Demuestra mediante la formulación en momios (odds ratio) del Teorema de Bayes que el razonamiento del fiscal es erróneo. Si en la región habitan $N = 5,000,000$ de personas con posible acceso a la escena y no existe ninguna otra evidencia contra el sospechoso, calcula la verdadera probabilidad a posteriori de culpabilidad $P(\text{culpable} | \text{coincidencia})$.
\end{problema}

\begin{problema}
	\label{prob:2.5.10}
	\textbf{Filtro Bayesiano de Spam (Clasificador Naive Bayes):} Un servidor de correo procesa un flujo donde el 60\% de los mensajes entrantes es correo no deseado o spam ($S$) y el 40\% es correo legítimo ($S'$). Un algoritmo analiza la presencia de tres palabras clave ($W_1, W_2, W_3$). Bajo el supuesto de independencia condicional (Naive Bayes), la aparición de las palabras dado la clase del correo es independiente:
	\begin{align}
		P(W_1 \cap W_2 \cap W_3 | S) &= P(W_1|S) \cdot P(W_2|S) \cdot P(W_3|S) \\
		P(W_1 \cap W_2 \cap W_3 | S') &= P(W_1|S') \cdot P(W_2|S') \cdot P(W_3|S')
	\end{align}
	Las probabilidades condicionales empíricas en spam son: $P(W_1|S)=0.80$, $P(W_2|S)=0.70$, $P(W_3|S)=0.60$; y en correo legítimo son: $P(W_1|S')=0.10$, $P(W_2|S')=0.20$, $P(W_3|S')=0.10$.
	
	Demuestra la fórmula general de decisión del clasificador Naive Bayes y calcula la probabilidad exacta a posteriori de que un correo que contenga simultáneamente las tres palabras ($W = W_1 \cap W_2 \cap W_3$) sea en realidad spam.
\end{problema}

\subsubsection*{1. Nivel Fundamental (Sencillos / Directos)}

\begin{solproblema}[prob:2.5.1]
	\begin{enumerate}
		\item \textbf{Sin información:} El espacio muestral completo tiene 6 eventos equiprobables $S=\{1,2,3,4,5,6\}$.
		Los resultados estrictamente menores que 4 son el subconjunto $A=\{1,2,3\}$.
		Por tanto, $P(A) = \frac{\card{A}}{\card{S}} = \frac{3}{6} = \frac{1}{2} = 0.50$.
		
		\item \textbf{Con información de número impar:} El nuevo espacio muestral condicionado es el evento $I=\{1,3,5\}$ con $\card{I}=3$.
		Los elementos de $A$ que también están en $I$ son la intersección $A \cap I = \{1,3\}$, con $\card{A \cap I}=2$.
		Aplicando la definición de probabilidad condicional (o de Bayes simplificado por equiprobabilidad):
		\begin{align}
			P(A|I) = \frac{P(A \cap I)}{P(I)} = \frac{2/6}{3/6} = \frac{2}{3} \approx 0.6667.
		\end{align}
	\end{enumerate}
\end{solproblema}

\begin{solproblema}[prob:2.5.6]
	Sean los eventos $M$ (mujer) con $P(M)=0.60$ y $H$ (hombre) con $P(H)=0.40$.
	Las tasas de zurdería son $P(Z|M)=0.10$ y $P(Z|H)=0.15$.
	
	Por la regla de la probabilidad total, la probabilidad de que una persona elegida al azar sea zurda es:
	\begin{align}
		P(Z) &= P(Z|M)P(M) + P(Z|H)P(H) = (0.10)(0.60) + (0.15)(0.40) \\
		&= 0.060 + 0.060 = 0.120 \text{ (o 12\%).}
	\end{align}
	Por el Teorema de Bayes, la probabilidad a posteriori de que la persona zurda sea mujer es:
	\begin{align}
		P(M|Z) = \frac{P(Z|M)P(M)}{P(Z)} = \frac{0.060}{0.120} = \frac{1}{2} = 0.50 \text{ (50\%).}
	\end{align}
\end{solproblema}

\begin{solproblema}[prob:2.5.4]
	Sea $A$ el evento de aprobar y $E$ el de estudiar ($P(E)=0.70$, $P(E')=0.30$).
	Las probabilidades condicionales del examen son $P(A|E)=0.90$ y $P(A|E')=0.30$.
	\begin{enumerate}
		\item Probabilidad marginal total de aprobar:
		\begin{align}
			P(A) &= P(A|E)P(E) + P(A|E')P(E') = (0.90)(0.70) + (0.30)(0.30) \\
			&= 0.63 + 0.09 = 0.72 \text{ (72\%).}
		\end{align}
		
		\item Por el Teorema de Bayes, si aprobó, la probabilidad de haber estudiado es:
		\begin{align}
			P(E|A) = \frac{P(A|E)P(E)}{P(A)} = \frac{0.63}{0.72} = \frac{7}{8} = 0.875 \text{ (87.5\%).}
		\end{align}
		
		\item Para el evento de no aprobar ($A'$), tenemos $P(A') = 1 - 0.72 = 0.28$.
		Las probabilidades de suspender dado si estudió o no son $P(A'|E)=1-0.90=0.10$ y $P(A'|E')=1-0.30=0.70$.
		Por el Teorema de Bayes sobre los complementos:
		\begin{align}
			P(E'|A') = \frac{P(A'|E')P(E')}{P(A')} = \frac{(0.70)(0.30)}{0.28} = \frac{0.21}{0.28} = \frac{3}{4} = 0.75 \text{ (75\%).}
		\end{align}
	\end{enumerate}
\end{solproblema}

\subsubsection*{2. Nivel Operativo (Intermedios / De Desarrollo)}

\begin{solproblema}[prob:2.5.7]
	El espacio muestral del lanzamiento de dos dados tiene 36 puntos. Una suma de a lo más 6 incluye los resultados con suma 2, 3, 4, 5 y 6, cuyo conteo es $1+2+3+4+5 = 15$ eventos simples.
	Por tanto, la probabilidad de elegir la Caja 1 es $P(C_1) = \frac{15}{36} = \frac{5}{12}$, y la de elegir la Caja 2 es $P(C_2) = 1 - \frac{5}{12} = \frac{7}{12}$.
	Las proporciones de canicas rojas ($R$) en las cajas son $P(R|C_1) = \frac{3}{5}$ y $P(R|C_2) = \frac{2}{10} = \frac{1}{5}$.
	
	\begin{enumerate}
		\item Probabilidad total de extraer una canica roja:
		\begin{align}
			P(R) &= P(R|C_1)P(C_1) + P(R|C_2)P(C_2) = \left(\frac{3}{5} \cdot \frac{5}{12}\right) + \left(\frac{1}{5} \cdot \frac{7}{12}\right) \\
			&= \frac{15}{60} + \frac{7}{60} = \frac{22}{60} = \frac{11}{30} \approx 0.3667.
		\end{align}
		
		\item Probabilidad condicional inversa (Bayes) de que provenga de la Caja 1:
		\begin{align}
			P(C_1|R) = \frac{P(R|C_1)P(C_1)}{P(R)} = \frac{15/60}{22/60} = \frac{15}{22} \approx 0.6818 \text{ (68.18\%).}
		\end{align}
	\end{enumerate}
\end{solproblema}

\begin{solproblema}[prob:2.5.2]
	Con prevalencia $P(E)=0.02$ (y $P(E')=0.98$), sensibilidad $P(+|E)=0.95$, y especificidad $P(-|E')=0.90$ (de donde la tasa de falsos positivos es $P(+|E')=1-0.90=0.10$):
	\begin{enumerate}
		\item Probabilidad total de resultado positivo:
		\begin{align}
			P(+) &= P(+|E)P(E) + P(+|E')P(E') = (0.95)(0.02) + (0.10)(0.98) \\
			&= 0.019 + 0.098 = 0.117 \text{ (11.7\%).}
		\end{align}
		
		\item Valor Predictivo Positivo (VPP) por el Teorema de Bayes:
		\begin{align}
			P(E|+) = \frac{P(+|E)P(E)}{P(+)} = \frac{0.019}{0.117} = \frac{19}{117} \approx 0.1624 \text{ (16.24\%).}
		\end{align}
		\emph{Interpretación:} Pese a la alta sensibilidad del test (95\%), debido a la baja prevalencia, el 83.76\% de los positivos son falsos positivos.
		
		\item Valor Predictivo Negativo (VPN):
		\begin{align}
			P(-) &= 1 - P(+) = 1 - 0.117 = 0.883 \\
			P(E'|-) &= \frac{P(-|E')P(E')}{P(-)} = \frac{(0.90)(0.98)}{0.883} = \frac{0.882}{0.883} \approx 0.9989 \text{ (99.89\%).}
		\end{align}
	\end{enumerate}
\end{solproblema}

\begin{solproblema}[prob:2.5.5]
	Sea $T_0, T_1$ la transmisión y $R_0, R_1$ la recepción, con $P(T_0)=0.60$ y $P(T_1)=0.40$.
	Las probabilidades del canal son $P(R_1|T_0)=0.02$ (error) y $P(R_1|T_1)=1-0.03=0.97$ (acierto).
	\begin{enumerate}
		\item Probabilidad marginal de recibir el bit 1:
		\begin{align}
			P(R_1) &= P(R_1|T_0)P(T_0) + P(R_1|T_1)P(T_1) = (0.02)(0.60) + (0.97)(0.40) \\
			&= 0.012 + 0.388 = 0.400 \text{ (40\%).}
		\end{align}
		
		\item Probabilidad a posteriori de que el 1 recibido sea auténtico:
		\begin{align}
			P(T_1|R_1) = \frac{P(R_1|T_1)P(T_1)}{P(R_1)} = \frac{0.388}{0.400} = 0.970 \text{ (97\%).}
		\end{align}
	\end{enumerate}
\end{solproblema}

\begin{solproblema}[prob:2.5.3]
	Sea $D$ la pieza defectuosa y las tres líneas de producción $M_1, M_2, M_3$.
	\begin{enumerate}
		\item Primero obtenemos la probabilidad marginal de defecto en toda la planta:
		\begin{align}
			P(D) &= P(D|M_1)P(M_1) + P(D|M_2)P(M_2) + P(D|M_3)P(M_3) \\
			&= (0.03)(0.50) + (0.05)(0.30) + (0.07)(0.20) = 0.015 + 0.015 + 0.014 = 0.044.
		\end{align}
		Calculamos las tres probabilidades a posteriori (teorema de Bayes generalizado):
		\begin{align}
			P(M_1|D) &= \frac{0.015}{0.044} = \frac{15}{44} \approx 0.3409 \text{ (34.09\%)} \\
			P(M_2|D) &= \frac{0.015}{0.044} = \frac{15}{44} \approx 0.3409 \text{ (34.09\%)} \\
			P(M_3|D) &= \frac{0.014}{0.044} = \frac{14}{44} = \frac{7}{22} \approx 0.3182 \text{ (31.82\%)}
		\end{align}
		
		\item Las líneas $M_1$ y $M_2$ comparten exactamente la máxima probabilidad condicional del 34.09\% cada una. Aunque $M_1$ tiene una tasa de defectos muy baja (3\%) frente a $M_2$ (5\%) o $M_3$ (7\%), su masivo volumen de producción (50\% del total de la planta) hace que iguale en número absoluto de piezas defectuosas generadas a la línea $M_2$.
	\end{enumerate}
\end{solproblema}

\begin{solproblema}[prob:2.5.8]
	Sea $\{B_1, B_2, \dots, B_k\}$ una partición exhaustiva ($\bigcup_{i=1}^k B_i = S$) y mutuamente excluyente ($B_i \cap B_m = \emptyset$ para todo $i \neq m$).
	Por la regla del producto de la probabilidad condicional, para cualquier índice particular $j$:
	\begin{align}
		P(B_j \cap A) = P(B_j | A) P(A) = P(A | B_j) P(B_j).
	\end{align}
	Al despejar la probabilidad condicional inversa $P(B_j|A)$ dado que $P(A)>0$:
	\begin{align}
		P(B_j | A) = \frac{P(A | B_j) P(B_j)}{P(A)}. \label{eq:bayes_num}
	\end{align}
	Por otra parte, como los eventos $B_i$ forman una partición de $S$, podemos expresar el evento $A$ como la unión disjunta de sus intersecciones sobre cada celda de la partición:
	\begin{align}
		A = A \cap S = A \cap \left( \bigcup_{i=1}^k B_i \right) = \bigcup_{i=1}^k (A \cap B_i).
	\end{align}
	Al aplicar el tercer axioma de Kolmogorov (aditividad para eventos disjuntos), la probabilidad total de $A$ es:
	\begin{align}
		P(A) = \sum_{i=1}^k P(A \cap B_i) = \sum_{i=1}^k P(A | B_i) P(B_i). \label{eq:prob_total_k}
	\end{align}
	Sustituyendo el denominador (\ref{eq:prob_total_k}) en la ecuación (\ref{eq:bayes_num}), se obtiene de manera rigurosa la fórmula general del Teorema de Bayes para $k$ estados.
\end{solproblema}

\subsubsection*{4. Nivel Desafiante (Expertos / Deducción Profunda)}

\begin{solproblema}[prob:2.5.9]
	Sea $I$ el evento de ser el culpable real del crimen, y $E$ la coincidencia en el perfil de ADN.
	El fiscal incurre en una falacia condicional al igualar erróneamente $P(E|I') = 10^{-6}$ con la probabilidad inversa $P(I'|E)$.
	
	Evaluemos la veracidad formal mediante la versión del Teorema de Bayes en razón de momios (Odds Form):
	\begin{align}
		\frac{P(I|E)}{P(I'|E)} = \underbrace{\frac{P(E|I)}{P(E|I')}}_{\text{Likelihood Ratio (LR)}} \times \underbrace{\frac{P(I)}{P(I')}}_{\text{Momios a priori}}.
	\end{align}
	Sin más evidencias que la ubicación geográfica en la población de $N = 5,000,000$ habitantes, la probabilidad a priori de que un individuo arbitrario sea culpable es $P(I) = \frac{1}{N} = \frac{1}{5,000,000} = 2 \times 10^{-7}$, mientras que $P(I') \approx 1$.
	
	El cociente de verosimilitud de la prueba es:
	\begin{align}
		\text{LR} = \frac{P(E|I)}{P(E|I')} = \frac{1}{10^{-6}} = 1,000,000.
	\end{align}
	Calculamos ahora los momios a posteriori multiplicando la verosimilitud por el momio a priori:
	\begin{align}
		\text{Momios}(I|E) = 1,000,000 \times \frac{1 / 5,000,000}{1} = \frac{1,000,000}{5,000,000} = \frac{1}{5} = 0.20.
	\end{align}
	Convirtiendo el momio a posteriori $\frac{P(I|E)}{1-P(I|E)} = \frac{1}{5}$ a probabilidad exacta:
	\begin{align}
		P(I|E) = \frac{0.20}{1 + 0.20} = \frac{1/5}{6/5} = \frac{1}{6} \approx 0.1667 \text{ (16.67\%).}
	\end{align}
	\textbf{Demostración y refutación:} La probabilidad de que el sospechoso sea culpable es de sólo 16.67\% (y no del 99.9999\% como alegaba el fiscal). El error radica en no considerar que, al evaluar un perfil en una población de 5 millones donde la frecuencia de coincidencia aleatoria es 1 en 1 millón, se esperan estadísticamente unos 5 falsos positivos puramente aleatorios en la población.
\end{solproblema}

\begin{solproblema}[prob:2.5.10]
	Sea el vector de palabras observadas $W = W_1 \cap W_2 \cap W_3$. Por la regla del Teorema de Bayes para clases dicotómicas ($S$ y $S'$):
	\begin{align}
		P(S|W) = \frac{P(W|S)P(S)}{P(W|S)P(S) + P(W|S')P(S')}.
	\end{align}
	Bajo la hipótesis de independencia condicional (Naive Bayes), sustituimos la probabilidad conjunta por el producto de marginales condicionales:
	\begin{align}
		P(W|S) &= P(W_1|S) \cdot P(W_2|S) \cdot P(W_3|S) = (0.80)(0.70)(0.60) = 0.336. \\
		P(W|S') &= P(W_1|S') \cdot P(W_2|S') \cdot P(W_3|S') = (0.10)(0.20)(0.10) = 0.002.
	\end{align}
	Con la proporción a priori del servidor $P(S)=0.60$ y $P(S')=0.40$, ponderamos ambos numeradores:
	\begin{align}
		P(W|S)P(S) &= (0.336)(0.60) = 0.2016 \\
		P(W|S')P(S') &= (0.002)(0.40) = 0.0008
	\end{align}
	La probabilidad marginal de recibir este conjunto exacto de palabras en un correo cualquiera es:
	\begin{align}
		P(W) = 0.2016 + 0.0008 = 0.2024.
	\end{align}
	Por lo tanto, la probabilidad a posteriori de que el mensaje entrante sea spam es:
	\begin{align}
		P(S|W) = \frac{0.2016}{0.2024} = \frac{252}{253} \approx 0.996047 \text{ (99.60\%).}
	\end{align}
	El clasificador Naive Bayes logra filtrar y categorizar el mensaje como spam con una certeza superior al 99.6\%.
\end{solproblema}
