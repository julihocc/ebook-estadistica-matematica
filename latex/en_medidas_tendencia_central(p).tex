\subsection{Detailed Solutions}

\subsubsection*{1. Fundamental Level (Simple / Direct)}

\begin{solproblem}
	Expanding term by term each summation according to the indicated indices:
	\begin{enumerate}
		\item $\displaystyle\sum_{j=0}^{6}X_{j} = X_{0} + X_{1} + X_{2} + X_{3} + X_{4} + X_{5} + X_{6}$
		\item $\displaystyle\sum_{k=1}^{4}\left( Y_{k}-3 \right)^{2} = (Y_{1}-3)^{2} + (Y_{2}-3)^{2} + (Y_{3}-3)^{2} + (Y_{4}-3)^{2}$
		\item $\displaystyle\sum_{k=1}^{N}a = \underbrace{a + a + \cdots + a}_{N \text{ times}} = N \cdot a$
		\item $\displaystyle\sum_{n=2}^{5}{f_{n}}X_{n} = f_{2}X_{2} + f_{3}X_{3} + f_{4}X_{4} + f_{5}X_{5}$
		\item $\displaystyle\sum_{m=0}^{3}\left( X_{m}-a \right) = (X_{0}-a) + (X_{1}-a) + (X_{2}-a) + (X_{3}-a)$
	\end{enumerate}
\end{solproblem}

\begin{solproblem}
	The remaining numbers are $100 - (20 + 40 + 30) = 10$ numbers with value $7$.
	Using the formula for the weighted mean by frequencies:
	\begin{align}
		\bar{X} &= \dfrac{\sum f_{i}X_{i}}{N} = \dfrac{20(4) + 40(5) + 30(6) + 10(7)}{100} \\
		&= \dfrac{80 + 200 + 180 + 70}{100} = \dfrac{530}{100} = 5.30
	\end{align}
	The arithmetic mean of the 100 numbers is $5.30$.
\end{solproblem}

\begin{solproblem}
	The combined mean weight of several groups is obtained via the weighted mean, where the weights are the group sizes ($n_{i}$):
	\begin{align}
		\bar{X} &= \dfrac{\sum n_{i}\bar{X}_{i}}{\sum n_{i}} = \dfrac{15(162) + 20(148) + 10(153) + 18(140)}{15 + 20 + 10 + 18} \\
		&= \dfrac{2430 + 2960 + 1530 + 2520}{63} = \dfrac{9440}{63} \approx 149.84 \text{ pounds.}
	\end{align}
\end{solproblem}

\subsubsection*{2. Operational Level (Intermediate / Developmental)}

\begin{solproblem}
	From table \ref{tab:1.2.1}, we determine the midpoint or class mark ($X_{i}$) for each height interval (in inches) and multiply by its frequency $f_{i}$:
	\begin{itemize}
		\item Interval 60--62: $X_{1} = 61$, $f_{1} = 5 \implies f_{1}X_{1} = 305$
		\item Interval 63--65: $X_{2} = 64$, $f_{2} = 18 \implies f_{2}X_{2} = 1152$
		\item Interval 66--68: $X_{3} = 67$, $f_{3} = 42 \implies f_{3}X_{3} = 2814$
		\item Interval 69--71: $X_{4} = 70$, $f_{4} = 27 \implies f_{4}X_{4} = 1890$
		\item Interval 72--74: $X_{5} = 73$, $f_{5} = 8 \implies f_{5}X_{5} = 584$
	\end{itemize}
	Summing the frequencies and the products:
	\begin{align}
		N &= \sum f_{i} = 5 + 18 + 42 + 27 + 8 = 100 \\
		\sum f_{i}X_{i} &= 305 + 1152 + 2814 + 1890 + 584 = 6745
	\end{align}
	Therefore, the mean height is:
	\begin{align}
		\bar{X} = \dfrac{6745}{100} = 67.45 \text{ inches.}
	\end{align}
\end{solproblem}

\begin{solproblem}
	By definition of arithmetic mean for the variable $Z$:
	\begin{align}
		\bar{Z} &= \dfrac{1}{N}\sum_{i=1}^{N} Z_{i} = \dfrac{1}{N}\sum_{i=1}^{N} (X_{i} + Y_{i}) \\
		&= \dfrac{1}{N}\left( \sum_{i=1}^{N}X_{i} + \sum_{i=1}^{N}Y_{i} \right) \\
		&= \dfrac{1}{N}\sum_{i=1}^{N}X_{i} + \dfrac{1}{N}\sum_{i=1}^{N}Y_{i} = \bar{X} + \bar{Y}.
	\end{align}
	This proves that the arithmetic mean operator is linear under addition.
\end{solproblem}

\begin{solproblem}
	The $N=8$ data points are: $5, 8, 11, 9, 12, 6, 14, 10$.
	
	a) With pivot $P=9$, we calculate the deviations $d_{i} = X_{i} - 9$:
	\begin{align}
		d_{1} &= 5-9 = -4, & d_{2} &= 8-9 = -1, & d_{3} &= 11-9 = 2, & d_{4} &= 9-9 = 0 \\
		d_{5} &= 12-9 = 3, & d_{6} &= 6-9 = -3, & d_{7} &= 14-9 = 5, & d_{8} &= 10-9 = 1
	\end{align}
	The sum of deviations is $\sum d_{i} = -4 - 1 + 2 + 0 + 3 - 3 + 5 + 1 = 3$.
	Applying the formula with pivot:
	\begin{align}
		\bar{X} = P + \dfrac{\sum d_{i}}{N} = 9 + \dfrac{3}{8} = 9 + 0.375 = 9.375.
	\end{align}

	b) With pivot $P=20$, the deviations $d_{i} = X_{i} - 20$ are:
	\begin{align}
		-15, -12, -9, -11, -8, -14, -6, -10 \implies \sum d_{i} = -85.
	\end{align}
	Applying the formula:
	\begin{align}
		\bar{X} = 20 + \dfrac{-85}{8} = 20 - 10.625 = 9.375.
	\end{align}
	The result is invariant with respect to the choice of the pivot.
\end{solproblem}

\subsubsection*{3. Analytical Level (Advanced / Connection)}

\begin{solproblem}
	We start from the definition of deviation from the pivot: $d_{i} = X_{i} - P$. Solving for the original variable gives $X_{i} = P + d_{i}$.
	
	Summing from $i=1$ to $N$ on both sides of the equation:
	\begin{align}
		\sum_{i=1}^{N} X_{i} &= \sum_{i=1}^{N} (P + d_{i}) = \sum_{i=1}^{N} P + \sum_{i=1}^{N} d_{i} = N \cdot P + \sum_{i=1}^{N} d_{i}.
	\end{align}
	Dividing the entire expression by $N$:
	\begin{align}
		\dfrac{\sum_{i=1}^{N} X_{i}}{N} &= \dfrac{N \cdot P}{N} + \dfrac{\sum_{i=1}^{N} d_{i}}{N} \implies \bar{X} = P + \dfrac{\sum_{i=1}^{N} d_{i}}{N}.
	\end{align}
	Which completes the proof.
\end{solproblem}

\begin{solproblem}
	Let $d_{i} = X_{i} - \bar{X}$. Summing over all $N$ elements of the set:
	\begin{align}
		\sum_{i=1}^{N} d_{i} &= \sum_{i=1}^{N} (X_{i} - \bar{X}) = \sum_{i=1}^{N} X_{i} - \sum_{i=1}^{N} \bar{X}.
	\end{align}
	Since $\bar{X}$ is a constant with respect to the summation index $i$, $\sum_{i=1}^{N} \bar{X} = N\bar{X}$.
	Moreover, by definition of the arithmetic mean, $\sum_{i=1}^{N} X_{i} = N\bar{X}$.
	
	Substituting both terms:
	\begin{align}
		\sum_{i=1}^{N} d_{i} = N\bar{X} - N\bar{X} = 0.
	\end{align}
	This property proves that the arithmetic mean is the center of gravity of the data distribution.
\end{solproblem}

\subsubsection*{4. Challenging Level (Expert / Deep Deduction)}

\begin{solproblem}
	The class mark of the central interval (66--68) in table \ref{tab:1.2.1} is $P = 67$ inches. Since the classes have a constant width $c = 3$ inches, we can use the coded variable $u_{i} = \dfrac{X_{i} - P}{c}$:
	\begin{itemize}
		\item Class 60--62 ($X_{1}=61$): $u_{1} = \frac{61-67}{3} = -2 \implies f_{1}u_{1} = 5(-2) = -10$
		\item Class 63--65 ($X_{2}=64$): $u_{2} = \frac{64-67}{3} = -1 \implies f_{2}u_{2} = 18(-1) = -18$
		\item Class 66--68 ($X_{3}=67$): $u_{3} = \frac{67-67}{3} = 0 \implies f_{3}u_{3} = 42(0) = 0$
		\item Class 69--71 ($X_{4}=70$): $u_{4} = \frac{70-67}{3} = 1 \implies f_{4}u_{4} = 27(1) = 27$
		\item Class 72--74 ($X_{5}=73$): $u_{5} = \frac{73-67}{3} = 2 \implies f_{5}u_{5} = 8(2) = 16$
	\end{itemize}
	Summing the frequencies multiplied by the code:
	\begin{align}
		\sum f_{i}u_{i} = -10 - 18 + 0 + 27 + 16 = 15.
	\end{align}
	Applying the short-cut formula for the mean with change of origin and scale:
	\begin{align}
		\bar{X} = P + c\left( \dfrac{\sum f_{i}u_{i}}{N} \right) = 67 + 3\left( \dfrac{15}{100} \right) = 67 + 3(0.15) = 67 + 0.45 = 67.45 \text{ inches.}
	\end{align}
	The result exactly matches the direct calculation from Problem \ref{prob:1.2.4}, significantly reducing the arithmetic complexity.
\end{solproblem}

\begin{solproblem}
	From table \ref{tab:1.2.2} of weights (in pounds) of the $N=100$ students, we first determine the cumulative frequencies ($F_{i}$) to locate the median class (where the position $N/2 = 50$ is reached):
	\begin{itemize}
		\item 118--126: $f_{1} = 3 \implies F_{1} = 3$
		\item 127--135: $f_{2} = 5 \implies F_{2} = 8$
		\item 136--144: $f_{3} = 9 \implies F_{3} = 17$
		\item 145--153: $f_{4} = 12 \implies F_{4} = 29$
		\item 154--162: $f_{5} = 5 \implies F_{5} = 34$ (Note: analyzing the complete frequency table for the median interval, we locate the class where $F_{i} \ge 50$).
	\end{itemize}
	Suppose from the distribution that the median interval is the one containing the 50th observation. Let $L_{m}$ be the true lower boundary of the median class, $F_{m-1}$ the cumulative frequency preceding the median class, $f_{m}$ the frequency of the median class, and $c$ the class interval width.
	
	The linear interpolation formula for the median ($Med$) in grouped data is:
	\begin{align}
		Med = L_{m} + \left( \dfrac{\dfrac{N}{2} - F_{m-1}}{f_{m}} \right) c.
	\end{align}
	By substituting the exact values from table \ref{tab:1.2.2}, the interpolated median weight is obtained, which divides the student population into two equal halves (50\% of the students weigh less than this value and 50\% weigh more).
\end{solproblem}
