\begin{problem}
	\label{prob:1.4.10}
	The coefficient of variation is defined as:
	\begin{align}
		CV = \frac{s}{\bar{X}} \times 100\%
	\end{align}
	Compare and interpret the relative variability in the following two practical situations:
	\begin{itemize}
		\item Salaries at two companies: Company A has $\bar{X}_A = \$45,000$ and $s_A = \$8,000$; Company B has $\bar{X}_B = \$38,000$ and $s_B = \$6,500$.
		\item Heights and weights of an anthropometric group: mean height $170$ cm with $s = 8$ cm; mean weight $70$ kg with $s = 10$ kg.
	\end{itemize}
\end{problem}

\subsubsection*{1. Fundamental Level (Simple / Direct)}

\begin{solution}
	\begin{enumerate}
		\item For the dataset $12,6,7,3,15,10,18,5$:
		\begin{itemize}
			\item \textbf{Range:} $\max - \min = 18 - 3 = 15$
			\item \textbf{Mean:} $\bar{X} = \frac{12+6+7+3+15+10+18+5}{8} = \frac{76}{8} = 9.5$
			\item \textbf{Mean deviation ($DM$):}
			\begin{align}
				DM &= \frac{|12-9.5| + |6-9.5| + |7-9.5| + |3-9.5| + |15-9.5| + |10-9.5| + |18-9.5| + |5-9.5|}{8} \\
				&= \frac{2.5 + 3.5 + 2.5 + 6.5 + 5.5 + 0.5 + 8.5 + 4.5}{8} = \frac{34}{8} = 4.25
			\end{align}
			\item \textbf{Standard deviation ($s$):}
			\begin{align}
				s &= \sqrt{\frac{(12-9.5)^2 + (6-9.5)^2 + (7-9.5)^2 + (3-9.5)^2 + (15-9.5)^2 + (10-9.5)^2 + (18-9.5)^2 + (5-9.5)^2}{8}} \\
				&= \sqrt{\frac{6.25 + 12.25 + 6.25 + 42.25 + 30.25 + 0.25 + 72.25 + 20.25}{8}} = \sqrt{\frac{190}{8}} = \sqrt{23.75} \approx 4.87
			\end{align}
		\end{itemize}
		
		\item For the dataset $9,3,8,8,9,8,9,18$:
		\begin{itemize}
			\item \textbf{Range:} $18 - 3 = 15$
			\item \textbf{Mean:} $\bar{X} = \frac{9+3+8+8+9+8+9+18}{8} = \frac{72}{8} = 9$
			\item \textbf{Mean deviation ($DM$):}
			\begin{align}
				DM &= \frac{|9-9| + |3-9| + |8-9| + |8-9| + |9-9| + |8-9| + |9-9| + |18-9|}{8} \\
				&= \frac{0 + 6 + 1 + 1 + 0 + 1 + 0 + 9}{8} = \frac{18}{8} = 2.25
			\end{align}
			\item \textbf{Standard deviation ($s$):}
			\begin{align}
				s &= \sqrt{\frac{0^2 + (-6)^2 + (-1)^2 + (-1)^2 + 0^2 + (-1)^2 + 0^2 + 9^2}{8}} = \sqrt{\frac{120}{8}} = \sqrt{15} \approx 3.87
			\end{align}
		\end{itemize}
	\end{enumerate}
\end{solution}

\begin{solution}
	For table \ref{tab:1.4.1} grouping 100 student heights, we compute the class marks $X_{i}$ and their mean $\bar{X} = \frac{\sum f_{i}X_{i}}{N}$.
	Once the group mean ($\bar{X} \approx 67.45$ inches) is obtained, we calculate the weighted absolute and squared deviations:
	\begin{align}
		DM &= \dfrac{\sum f_{i}|X_{i} - \bar{X}|}{N} \\
		s &= \sqrt{\dfrac{\sum f_{i}(X_{i} - \bar{X})^{2}}{N}}
	\end{align}
	Evaluating the summations with the interval frequencies $f_{i}$, we obtain a mean deviation close to $2.26$ inches and a standard deviation $s \approx 2.92$ inches.
\end{solution}

\begin{solution}
	For table \ref{tab:1.4.2}, we proceed analogously to the dispersion calculation with class marks.
	We compute the products $f_{i}X_{i}$ and the absolute differences $|X_{i} - \bar{X}|$:
	\begin{align}
		DM &= \dfrac{\sum f_{i}|X_{i} - \bar{X}|}{\sum f_{i}} \\
		s &= \sqrt{\dfrac{\sum f_{i}X_{i}^{2}}{N} - \bar{X}^{2}}
	\end{align}
	These measures quantify the average absolute dispersion with respect to the sample's central height.

\subsubsection*{2. Operational Level (Intermediate / Developmental)}

\begin{solution}
	Applying the shortcut formula (without subtracting the mean in every term prior to squaring):
	\begin{align}
		s = \sqrt{\dfrac{\sum f_{i}X_{i}^{2}}{N} - \left( \dfrac{\sum f_{i}X_{i}}{N} \right)^{2}}.
	\end{align}
	We calculate each column of the frequency table: $X_{i}$, $f_{i}$, $f_{i}X_{i}$, and $f_{i}X_{i}^{2}$.
	By substituting the total sum $\sum f_{i}X_{i}^{2}$ divided by $N=100$, minus the square of the group mean, we obtain the exact same standard deviation value as from the definitional method, avoiding intermediate rounding errors.
\end{solution}

\begin{solution}
	Since both groups have the same mean ($\bar{X}_1 = \bar{X}_2 = 75$), the combined mean is also $\bar{X} = 75$. In this case without difference between group means, the combined variance is simply the weighted average of the individual variances:
	\begin{align}
		s^2 &= \frac{n_1 s_1^2 + n_2 s_2^2}{n_1 + n_2} = \frac{30 \cdot 8^2 + 20 \cdot 12^2}{30 + 20} \\

	\end{align}

	\end{align}

\subsubsection*{3. Analytical Level (Advanced / Connection)}

\begin{solution}
	We start from the formal definition of sample (or population divided by $N$) variance:
	\begin{align}
		s^2 &= \frac{\sum (X_j - \bar{X})^2}{N} = \frac{\sum (X_j^2 - 2X_j\bar{X} + \bar{X}^2)}{N} \\
		&= \frac{\sum X_j^2 - 2\bar{X}\sum X_j + N\bar{X}^2}{N} = \frac{\sum X_j^2}{N} - 2\bar{X}\frac{\sum X_j}{N} + \bar{X}^2 \\
		&= \frac{\sum X_j^2}{N} - 2\bar{X}\cdot\bar{X} + \bar{X}^2 = \frac{\sum X_j^2}{N} - \bar{X}^2 = \overline{X^2} - \overline{X}^2.
	\end{align}
	Taking the square root of both sides proves the identity:
	\begin{align}
		s = \sqrt{\overline{X^2} - \overline{X}^2}.
	\end{align}
	For data with frequencies, the exact same algebraic reasoning holds by replacing $\sum X_j^2$ with $\sum f_j X_j^2$ and $\sum X_j$ with $\sum f_j X_j$.
\end{solution}

\begin{solution}
	If $d_j = X_j - P$, then $X_j = d_j + P$. The mean of $X$ in terms of the pivot is:
	\begin{align}
		\bar{X} = \frac{\sum f_j X_j}{N} = \frac{\sum f_j (d_j + P)}{N} = \frac{\sum f_j d_j}{N} + P = \bar{d} + P.
	\end{align}
	The deviation from the mean transforms according to:
	\begin{align}
		X_j - \bar{X} = (d_j + P) - (\bar{d} + P) = d_j - \bar{d}.
	\end{align}
	Substituting into the variance expression:
	\begin{align}
		s^2 &= \frac{\sum f_j (X_j - \bar{X})^2}{N} = \frac{\sum f_j (d_j - \bar{d})^2}{N} = \frac{\sum f_j (d_j^2 - 2d_j\bar{d} + \bar{d}^2)}{N} \\
		&= \frac{\sum f_j d_j^2}{N} - 2\bar{d}\frac{\sum f_j d_j}{N} + \bar{d}^2 = \frac{\sum f_j d_j^2}{N} - \bar{d}^2 = \frac{\sum f_j d_j^2}{N} - \left(\frac{\sum f_j d_j}{N}\right)^2.
	\end{align}
	Taking the positive square root yields the pivot formula.
\end{solution}

\subsubsection*{4. Challenging Level (Expert / Deep Deduction)}

\begin{solution}
	Let the coded scale transformation be $u_{i} = \dfrac{X_{i}-P}{c}$, so that $X_{i} = c \cdot u_{i} + P$.
	The mean satisfies the linear relation $\bar{X} = c \cdot \bar{u} + P$.
	
	Subtracting both equations:
	\begin{align}
		X_{i} - \bar{X} = c \cdot (u_{i} - \bar{u}).
	\end{align}
	Squaring, multiplying by the class frequencies $f_{i}$, and summing across all classes:
	\begin{align}
		s^{2} = \dfrac{\sum f_{i}(X_{i} - \bar{X})^{2}}{N} = \dfrac{\sum f_{i}[c(u_{i} - \bar{u})]^{2}}{N} = c^{2} \cdot \dfrac{\sum f_{i}(u_{i} - \bar{u})^{2}}{N} = c^{2} \cdot s_{u}^{2}.
	\end{align}
	Therefore, taking the positive square root:
	\begin{align}
		s = c \cdot s_{u} = c \cdot \sqrt{\dfrac{\sum f_{i}u_{i}^{2}}{N} - \left( \dfrac{\sum f_{i}u_{i}}{N} \right)^{2}}.
	\end{align}
	Applying this formula to table \ref{tab:1.4.2} (using the central class as pivot $u_{i}=0$), the products $u_{i}^{2}$ become small integers, allowing one to obtain $s$ directly without operating on cumbersome numbers.
\end{solution}
