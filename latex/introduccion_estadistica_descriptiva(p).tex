\section*{Problemas}

\subsection{Enunciados de los problemas}

\subsubsection*{1. Nivel Fundamental (Sencillos / Directos)}

\begin{problema}
	\label{prob:1.1.1}
	Clasifique cada una de las siguientes variables según su escala de medición (\emph{Nominal, Ordinal, de Intervalo o de Razón}):
	\begin{enumerate}
		\item El código postal de la residencia de un cliente en una plataforma de comercio electrónico.
		\item El nivel de satisfacción de un usuario respecto a una aplicación móvil (Muy Insatisfecho, Insatisfecho, Neutral, Satisfecho, Muy Satisfecho).
		\item La temperatura en grados Celsius (°C) registrada por los servidores de un centro de datos.
		\item El salario mensual en pesos mexicanos de un ingeniero en ciencia de datos.
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:1.1.2}
	En un estudio de mercadotecnia, se desea investigar el gasto promedio anual en suscripciones de \emph{streaming} por hogar en México. Para ello, se envía una encuesta digital aleatoria a $2,500$ usuarios registrados en una empresa proveedora de fibra óptica en la Ciudad de México.
	\begin{enumerate}
		\item Identifique la \emph{Población objetivo} y la \emph{Muestra} de este estudio.
		\item ¿Incurre el estudio en algún tipo de sesgo de cobertura? Explique brevemente por qué.
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:1.1.3}
	Determine si la métrica descrita en los siguientes enunciados representa un \textbf{Parámetro} ($\mu, \sigma^2, p$) o un \textbf{Estadístico} ($\bar{X}, S^2, \hat{p}$):
	\begin{enumerate}
		\item Tras censar las 32 entidades federativas, el INEGI determinó que el promedio de personas por vivienda en el país es de $3.6$.
		\item Un ingeniero de calidad selecciona un lote aleatorio de 80 procesadores y calcula una varianza en la frecuencia de reloj de $0.015 \text{ GHz}^2$.
		\item En un subconjunto de prueba de 5,000 transacciones bancarias, la tasa de fraude detectada fue del $0.84\%$.
	\end{enumerate}
\end{problema}

\subsubsection*{2. Nivel Operativo (Intermedios / De Desarrollo)}

\begin{problema}
	\label{prob:1.1.4}
	Un administrador de bases de datos sabe por auditorías históricas que el tiempo promedio verdadero de consulta en un servidor principal es de $\mu = 45 \text{ ms}$. Durante una hora de alta concurrencia, toma una muestra aleatoria de $n = 60$ consultas y obtiene los siguientes tiempos (en ms): la media muestral resulta ser $\bar{X} = 51.2 \text{ ms}$.
	\begin{enumerate}
		\item Calcule el \emph{error de muestreo} (o error de estimación simple) para esta muestra en particular.
		\item Si el administrador toma una segunda muestra aleatoria de tamaño $n = 60$ en la misma hora, ¿necesariamente obtendrá el mismo valor de $\bar{X} = 51.2 \text{ ms}$? Explique el concepto de variabilidad muestral.
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:1.1.5}
	Una corporación de consultoría tecnológica cuenta con 4,000 empleados divididos en cuatro divisiones: Desarrollo ($2,000$), Infraestructura ($1,000$), Ventas ($600$) y Recursos Humanos ($400$). Se requiere seleccionar una muestra aleatoria de $n = 200$ empleados para evaluar el clima laboral.
	\begin{enumerate}
		\item Si se utiliza un muestreo aleatorio simple sobre toda la nómina, ¿cuál es el riesgo potencial respecto a la división de Recursos Humanos?
		\item Diseñe un esquema de \emph{muestreo aleatorio estratificado proporcional} y determine exactamente cuántos empleados deben entrevistarse en cada una de las cuatro divisiones para garantizar una representatividad equitativa.
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:1.1.6}
	Para evaluar el éxito de una nueva plataforma de banca web, un departamento de experiencia de usuario manda un correo masivo a 100,000 clientes invitándolos a contestar voluntariamente un cuestionario en línea sobre usabilidad. Responden 1,200 clientes, y el $88\%$ de ellos manifiesta que la plataforma es extremadamente confusa.
	\begin{enumerate}
		\item ¿Qué sesgo metodológico predomina en esta recolección de datos y por qué distorsiona la proporción real del total de clientes ($p$)?
		\item Proponga un rediseño del protocolo de muestreo para obtener una estimación insesgada y confiable de la usabilidad web sin incrementar el presupuesto de recolección.
	\end{enumerate}
\end{problema}

\subsubsection*{3. Nivel Analítico (Avanzados / De Conexión)}

\begin{problema}
	\label{prob:1.1.7}
	En 1936, la revista \emph{Literary Digest} realizó un muestreo masivo enviando 10 millones de cuestionarios por correo para predecir la elección presidencial de Estados Unidos, utilizando como marco muestral directorios telefónicos y registros de propietarios de automóviles. A pesar de obtener una muestra gigantesca de $n \approx 2.4 \text{ millones}$ de respuestas, predijeron un triunfo contundente de Alf Landon, cuando en realidad Roosevelt ganó por avalancha. Al mismo tiempo, George Gallup pronosticó correctamente la victoria de Roosevelt encuestando a apenas $n = 50,000$ personas.
	
	Demuestre conceptual y algebraicamente cómo, si el mecanismo de selección muestral posee un sesgo sistemático $B = \mathbb{E}[\bar{X}] - \mu \neq 0$, aumentar el tamaño de muestra $n \to \infty$ no reduce el error total, sino que en cambio hace que la estimación converja con certeza hacia un valor incorrecto.
\end{problema}

\begin{problema}
	\label{prob:1.1.8}
	En estadística inferencial, el error estándar de la media muestral al estimar una media poblacional $\mu$ está dado por la relación:
	\begin{align}
		\sigma_{\bar{X}} = \dfrac{\sigma}{\sqrt{n}},
	\end{align}
	donde $\sigma$ es la desviación estándar poblacional y $n$ es el tamaño de la muestra.
	\begin{enumerate}
		\item Si un equipo de analítica desea reducir el error estándar muestral exactamente a la mitad, ¿por qué factor debe multiplicar su tamaño de muestra actual $n$? Demuéstrelo matemáticamente.
		\item Discuta la implicación de la de ley de rendimientos decrecientes en el presupuesto de recolección de datos de Big Data a partir de este resultado.
	\end{enumerate}
\end{problema}

\subsubsection*{4. Nivel Desafiante (Expertos / De Exploración)}

\begin{problema}
	\label{prob:1.1.9}
	En un proyecto de Machine Learning financiero, un equipo extrae registros automáticos de servidores para calcular la duración de sesiones de usuarios en un portal de transacciones. Sin embargo, el sistema interrumpe y descarta automáticamente los registros (\emph{timeout logs}) de cualquier sesión que exceda los 15 minutos sin actividad.
	\begin{enumerate}
		\item Demuestre cómo este sesgo de truncamiento o supervivencia (\emph{survival bias}) afecta la distribución empírica observada y el cálculo de la media muestral $\bar{X}$ respecto a la verdadera media poblacional $\mu$ de duración de sesiones.
		\item Si además el conjunto de datos contiene \emph{outliers} extremos causados por robots de telemetría que generan sesiones de $0.001 \text{ segundos}$, ¿por qué la estadística descriptiva tradicional basada exclusivamente en la media aritmética resulta frágil antes de modelar? Justifique qué medidas robustas deberían preferirse.
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:1.1.10}
	Suponga que se ha recolectado una muestra aleatoria estratificada no proporcional de tamaño total $n$ proveniente de $K$ estratos poblacionales. Sea $N_k$ el tamaño real del estrato $k$ en el censo total ($N = \sum_{k=1}^K N_k$), y $n_k$ el tamaño de muestra tomado en ese estrato ($n = \sum_{k=1}^K n_k$). Si $\bar{X}_k$ es la media muestral del estrato $k$, demuestre algebraicamente que el estimador insesgado post-estratificado de la media poblacional $\mu$ es el promedio ponderado:
	\begin{align}
		\bar{X}_{est} = \sum_{k=1}^K w_k \bar{X}_k, \quad \text{donde } w_k = \dfrac{N_k}{N}.
	\end{align}
	Verifique qué ocurre con los pesos $w_k$ cuando el diseño original es estrictamente proporcional al tamaño del estrato ($n_k = n \cdot \frac{N_k}{N}$).
\end{problema}

\subsection{Respuestas y sugerencias de los problemas}

\subsubsection*{1. Nivel Fundamental (Sencillos / Directos)}

\begin{solproblema}
	La clasificación por nivel o escala de medición se justifica de la siguiente forma:
	\begin{enumerate}
		\item \textbf{Código postal $\to$ Nominal:} Los números actúan exclusivamente como etiquetas o categorías de ubicación geográfica. No existe un orden natural ni sentido matemático en sumar o restar códigos postales.
		\item \textbf{Satisfacción de usuario $\to$ Ordinal:} Existe una jerarquía o clasificación intrínseca (Muy Insatisfecho $<$ Insatisfecho $<$ Neutral $<$ Satisfecho $<$ Muy Satisfecho), pero la distancia conceptual entre una categoría y otra no es necesariamente uniforme ni cuantificable.
		\item \textbf{Temperatura en °C $\to$ De Intervalo:} La diferencia numérica de grados tiene un significado físico constante ($30^\circ\text{C} - 20^\circ\text{C} = 20^\circ\text{C} - 10^\circ\text{C}$), pero el cero es arbitrario (el $0^\circ\text{C}$ representa el punto de congelación del agua, no la ausencia absoluta de energía térmica). Por tanto, no podemos afirmar que $40^\circ\text{C}$ sea el "doble de calor" que $20^\circ\text{C}$.
		\item \textbf{Salario mensual $\to$ De Razón:} Posee intervalos uniformes y un cero absoluto real ($0$ pesos implica carencia total de ingresos salariales). Las razones son válidas: un salario de $\$60,000$ es exactamente el doble de uno de $\$30,000$.
	\end{enumerate}
\end{solproblema}

\begin{solproblema}
	Del análisis conceptual del diseño de muestreo:
	\begin{enumerate}
		\item \textbf{Población objetivo ($N$):} Todos los hogares en México que consumen o son susceptibles de consumir servicios de suscripción de \emph{streaming}. \textbf{Muestra ($n$):} Los $2,500$ usuarios registrados de la empresa proveedora de fibra óptica en la Ciudad de México que recibieron (o respondieron) la encuesta.
		\item \textbf{Sesgo de cobertura:} Sí incurre en un grave sesgo de cobertura y de selección. Al encuestar únicamente a clientes de internet residencial de alta velocidad (fibra óptica) en la capital del país, se excluye sistemáticamente a zonas rurales, ciudades medianas, hogares sin fibra óptica o consumidores que acceden a plataformas exclusivamente mediante datos móviles. El gasto promedio estimado será superior al verdadero promedio nacional.
	\end{enumerate}
\end{solproblema}

\begin{solproblema}
	La distinción entre parámetro y estadístico se basa en si la métrica describe al universo total censado o a una porción del mismo:
	\begin{enumerate}
		\item \textbf{Promedio de $3.6$ personas por vivienda $\to$ Parámetro ($\mu$):} Proviene de un censo nacional exhaustivo que abarcó la totalidad de las 32 entidades federativas.
		\item \textbf{Varianza de $0.015 \text{ GHz}^2$ $\to$ Estadístico ($S^2$):} Es una medida calculada exclusivamente sobre la muestra aleatoria de 80 procesadores inspeccionados.
		\item \textbf{Tasa de fraude del $0.84\%$ $\to$ Estadístico ($\hat{p}$):} Es una proporción estimada sobre el subconjunto muestral de 5,000 transacciones extraídas para prueba.
	\end{enumerate}
\end{solproblema}

\subsubsection*{2. Nivel Operativo (Intermedios / De Desarrollo)}

\begin{solproblema}
	Para el cálculo y variabilidad del muestreo:
	\begin{enumerate}
		\item El error de estimación simple (discrepancia muestral) se calcula formalmente como:
		\begin{align}
			e = \bar{X} - \mu = 51.2 \text{ ms} - 45.0 \text{ ms} = +6.2 \text{ ms}.
		\end{align}
		La muestra en particular sobrestima el tiempo promedio poblacional en $6.2 \text{ ms}$.
		\item No, una segunda muestra de tamaño $n = 60$ casi con certeza arrojará un valor de $\bar{X}$ diferente (por ejemplo, $43.8 \text{ ms}$ o $47.5 \text{ ms}$). Esto es la esencia de la \textbf{variabilidad muestral}: dado que los elementos elegidos al azar cambian de una muestra a otra, el estadístico $\bar{X}$ es en realidad una variable aleatoria cuya distribución oscila en torno al parámetro poblacional $\mu$.
	\end{enumerate}
\end{solproblema}

\begin{solproblema}
	Del diseño del muestreo estratificado:
	\begin{enumerate}
		\item Si se utiliza un muestreo aleatorio simple ($n = 200$ sobre $N = 4,000$), al ser Recursos Humanos el departamento más pequeño ($400/4000 = 10\%$ de la empresa), existe un riesgo probabilístico de que por fluctuación aleatoria queden muy pocos o ningún empleado de RH en la muestra elegida, perdiendo su representatividad en el clima laboral.
		\item En un muestreo aleatorio estratificado proporcional, el tamaño de muestra de cada estrato $k$ es proporcional a su peso en la nómina: $n_k = n \cdot \left(\frac{N_k}{N}\right) = 200 \cdot \left(\frac{N_k}{4,000}\right)$. Sustituimos:
		\begin{itemize}
			\item \textbf{Desarrollo:} $n_1 = 200 \cdot \left(\frac{2,000}{4,000}\right) = 200(0.50) = 100 \text{ empleados.}$
			\item \textbf{Infraestructura:} $n_2 = 200 \cdot \left(\frac{1,000}{4,000}\right) = 200(0.25) = 50 \text{ empleados.}$
			\item \textbf{Ventas:} $n_3 = 200 \cdot \left(\frac{600}{4,000}\right) = 200(0.15) = 30 \text{ empleados.}$
			\item \textbf{Recursos Humanos:} $n_4 = 200 \cdot \left(\frac{400}{4,000}\right) = 200(0.10) = 20 \text{ empleados.}$
		\end{itemize}
		La suma total verifica que $100 + 50 + 30 + 20 = 200$ entrevistas en total con paridad estructural garantizada.
	\end{enumerate}
\end{solproblema}

\begin{solproblema}
	Del análisis del correo de usabilidad:
	\begin{enumerate}
		\item Predomina el \textbf{sesgo de autoselección o respuesta voluntaria}. Aquellos usuarios que experimentaron frustración técnica grave con la plataforma tienen una motivación psicológica mucho mayor para abrir la encuesta y quejarse que aquellos usuarios satisfechos o indiferentes que completaron sus operaciones sin problema. Por tanto, el $88\%$ es una sobrestimación masiva del nivel real de confusión poblacional ($p$).
		\item \textbf{Rediseño sin costo extra:} En lugar de mandar un correo pasivo a 100,000 personas, se debe seleccionar una muestra aleatoria pura mucho más pequeña (por ejemplo, $n = 500$ clientes) y aplicar un seguimiento activo intermitente en la misma sesión web (\emph{pop-up} aleatorio controlado al finalizar una transacción) o contacto directo programado, asegurando una tasa de respuesta alta y no sesgada por frustración personal.
	\end{enumerate}
\end{solproblema}

\subsubsection*{3. Nivel Analítico (Avanzados / De Conexión)}

\begin{solproblema}
	Sea $\bar{X}_n$ el estimador muestral basado en una muestra de tamaño $n$. Por definición, el error cuadrático medio (MSE) de un estimador descompone la imprecisión total en dos componentes algebraicos: varianza muestral y sesgo al cuadrado:
	\begin{align}
		\text{MSE}(\bar{X}_n) = \mathbb{E}\left[(\bar{X}_n - \mu)^2\right] = \text{Var}(\bar{X}_n) + (\text{Sesgo})^2 = \dfrac{\sigma^2}{n} + B^2,
	\end{align}
	donde $B = \mathbb{E}[\bar{X}_n] - \mu$ representa el sesgo metodológico derivado del marco de selección (en \emph{Literary Digest}, encuestar solo a personas pudientes con teléfono y automóvil en plena Gran Depresión).
	
	Al incrementar el tamaño de muestra de manera indefinida ($n \to \infty$), el componente de variabilidad muestral tiende a cero ($\lim_{n \to \infty} \frac{\sigma^2}{n} = 0$), por la Ley de los Grandes Números. Sin embargo, el componente del sesgo sistemático permanece exactamente igual:
	\begin{align}
		\lim_{n \to \infty} \text{MSE}(\bar{X}_n) = 0 + B^2 = B^2 > 0.
	\end{align}
	Esto demuestra que una muestra gigante de millones de registros sesgados ($n = 2.4 \text{ millones}$) únicamente reduce la varianza en torno a un valor incorrecto, generando una falsa ilusión de precisión absoluta sobre una estimación errónea. En contraste, una muestra pequeña como la de Gallup ($n = 50,000$), donde $B = 0$, alcanza una convergencia insesgada y exacta hacia el verdadero parámetro $\mu$.
\end{solproblema}

\begin{solproblema}
	Del análisis del error estándar y rendimientos decrecientes:
	\begin{enumerate}
		\item Queremos un nuevo error estándar $\sigma_{\bar{X}}'$ que cumpla $\sigma_{\bar{X}}' = \frac{1}{2}\sigma_{\bar{X}}$. Sea $k$ el factor multiplicativo del tamaño de muestra original tal que el nuevo tamaño sea $n' = k \cdot n$. Entonces:
		\begin{align}
			\sigma_{\bar{X}}' = \dfrac{\sigma}{\sqrt{n'}} = \dfrac{\sigma}{\sqrt{k \cdot n}} = \dfrac{1}{\sqrt{k}} \cdot \left(\dfrac{\sigma}{\sqrt{n}}\right) = \dfrac{1}{\sqrt{k}} \sigma_{\bar{X}}.
		\end{align}
		Igualando los factores:
		\begin{align}
			\dfrac{1}{\sqrt{k}} = \dfrac{1}{2} \implies \sqrt{k} = 2 \implies k = 4.
		\end{align}
		Por tanto, para reducir el error de muestreo a la mitad, es indispensable \textbf{cuadruplicar} el tamaño de muestra ($n' = 4n$).
		\item \textbf{Implicación en Big Data:} La relación inversa cuadrada indica una ley de rendimientos decrecientes drástica. Si pasar de $n=1,000$ a $n=4,000$ (añadiendo $3,000$ registros) reduce el error a la mitad, lograr otra reducción equivalente a la mitad requeriría pasar a $n=16,000$ (añadiendo $12,000$ registros más). En arquitecturas masivas, recolectar y procesar petabytes adicionales tiene un costo computacional y financiero exponencial que aporta reducciones marginales en el error estándar, siendo mucho más valioso invertir en mejorar la calidad y representatividad (eliminar sesgo $B$) que en inflar ciegamente el volumen bruto de datos.
	\end{enumerate}
\end{solproblema}

\subsubsection*{4. Nivel Desafiante (Expertos / De Exploración)}

\begin{solproblema}
	Del análisis algorítmico y de truncamiento en \emph{logs}:
	\begin{enumerate}
		\item Al truncar las sesiones con duración $X > 15 \text{ min}$, la distribución empírica observada por el sistema ya no es la distribución poblacional original $f(x)$, sino la distribución condicional truncada a la derecha:
		\begin{align}
			f_{\text{obs}}(x) = f(x \mid X \le 15) = \dfrac{f(x)}{P(X \le 15)} \quad \text{para } 0 \le x \le 15.
		\end{align}
		Dado que la verdadera esperanza poblacional integra sobre todo el dominio ($0 \le x < \infty$):
		\begin{align}
			\mu = \int_{0}^{15} x f(x) \, dx + \int_{15}^{\infty} x f(x) \, dx,
		\end{align}
		la media muestral calculada exclusivamente sobre los registros observados ($\bar{X}_{\text{obs}}$) converge a la esperanza truncada $\mu_{\text{obs}} = \mathbb{E}[X \mid X \le 15]$, cuya integral omite la cola derecha pesada de usuarios prolongados. Consecuentemente, el estadístico subestima sistemáticamente la verdadera duración promedio poblacional ($\mathbb{E}[\bar{X}_{\text{obs}}] < \mu$).
		\item La media aritmética es un operador estadístico de \textbf{punto de quiebre cero} (\emph{zero breakdown point}), lo que significa que un solo valor atípico artificial o aberrante ($x_i \to 0$ por ráfagas de telemetría de $0.001 \text{ s}$ o un registro erróneo extremadamente grande) arrastra el promedio sin cota. Por ello, en flujos masivos de datos con ruido se deben emplear \textbf{medidas descriptivas robustas}, tales como la \textbf{mediana muestral} (cuyo punto de quiebre es del $50\%$) o la \textbf{media truncada/recortada} (\emph{trimmed mean} al $5\%$ o $10\%$), las cuales aíslan los extremos aberrantes sin perder la tendencia central general.
	\end{enumerate}
\end{solproblema}

\begin{solproblema}
	Del análisis de la ponderación post-estratificada:
	
	Por definición matemática, la media poblacional total $\mu$ de una población de tamaño $N$ dividida en $K$ estratos mutuamente excluyentes se expresa como el promedio ponderado de las medias poblacionales de cada estrato ($\mu_k$):
	\begin{align}
		\mu = \dfrac{1}{N} \sum_{i=1}^N X_i = \sum_{k=1}^K \dfrac{N_k}{N} \mu_k = \sum_{k=1}^K w_k \mu_k,
	\end{align}
	donde $w_k = \frac{N_k}{N}$ representa la verdadera proporción del estrato $k$ en el universo censal ($\sum w_k = 1$).
	
	Dado que en cada estrato la media muestral $\bar{X}_k = \frac{1}{n_k} \sum_{j=1}^{n_k} X_{kj}$ es un estimador insesgado de la media intra-estrato ($\mathbb{E}[\bar{X}_k] = \mu_k$), aplicamos el operador valor esperado por linealidad al estimador propuesto $\bar{X}_{est}$:
	\begin{align}
		\mathbb{E}[\bar{X}_{est}] = \mathbb{E}\left[ \sum_{k=1}^K w_k \bar{X}_k \right] = \sum_{k=1}^K w_k \mathbb{E}[\bar{X}_k] = \sum_{k=1}^K w_k \mu_k = \mu.
	\end{align}
	Esto demuestra formalmente que $\bar{X}_{est}$ es un estimador insesgado de la media poblacional total, incluso si la muestra se obtuvo de forma completamente desproporcionada en algunos estratos.
	
	\textbf{Verificación en muestreo proporcional:}
	Si por diseño el muestreo es estrictamente proporcional, se cumple que $n_k = n \left(\frac{N_k}{N}\right) = n \cdot w_k$, por lo cual el peso ponderador equivale exactamente a la fracción de muestra tomada: $w_k = \frac{n_k}{n}$. Sustituyendo esto en la fórmula post-estratificada:
	\begin{align}
		\bar{X}_{est} = \sum_{k=1}^K \left( \dfrac{n_k}{n} \right) \bar{X}_k = \dfrac{1}{n} \sum_{k=1}^K n_k \left( \dfrac{1}{n_k} \sum_{j=1}^{n_k} X_{kj} \right) = \dfrac{1}{n} \sum_{k=1}^K \sum_{j=1}^{n_k} X_{kj} = \bar{X}.
	\end{align}
	Es decir, bajo asignación proporcional, el promedio ponderado post-estratificado colapsa elegantemente a la media aritmética simple de todos los $n$ datos muestrales.
\end{solproblema}
