\subsubsection*{1. Fundamental Level (Simple / Direct)}

\begin{problema}
	\label{prob:3.9.1}
	A regular die is rolled 120 times in a random experiment, and the following observed frequencies are recorded for each of its six faces:
	\begin{center}
		\begin{tabular}{|l|c|c|c|c|c|c|}\hline
			Face & 1 & 2 & 3 & 4 & 5 & 6 \\\hline
			Observed frequency ($O_i$) & 18 & 22 & 20 & 19 & 23 & 18 \\\hline
		\end{tabular}
	\end{center}
	\begin{enumerate}
		\item Formulate the null ($H_0$) and alternative ($H_a$) hypotheses to formally evaluate whether the die is fair (equiprobable).
		\item Calculate the expected frequency under the null hypothesis ($E_i$) for each face of the die.
		\item Calculate the value of Pearson's $\chi^2$ test statistic.
		\item Determine the degrees of freedom associated with the experiment.
		\item Should $H_0$ be rejected at a significance level of $\alpha = 0.05$? (The critical quantile in the tables is $\chi^2_{0.05, \, 5} = 11.07$).
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:chi-desaf-contingencia-2x2}
	\textbf{Analytical derivation of the exact contingency statistic for $2 \times 2$ matrices:}
	Consider a general contingency table of order $2 \times 2$ where the inner cells contain the observed absolute frequencies $a, b, c, d$, with marginal row totals $R_1, R_2$, column totals $C_1, C_2$, and overall sample size $N = a+b+c+d$:
	\begin{center}
		\begin{tabular}{|c|c|c|}\hline
			$a$ & $b$ & $R_1 = a+b$ \\\hline
			$c$ & $d$ & $R_2 = c+d$ \\\hline
			$C_1 = a+c$ & $C_2 = b+d$ & $N$ \\\hline
		\end{tabular}
	\end{center}
	\begin{enumerate}
		\item Starting from Pearson's general definition under independence $\chi^2 = \sum_{i=1}^2 \sum_{j=1}^2 \dfrac{(O_{ij} - E_{ij})^2}{E_{ij}}$ where expected frequencies are $E_{ij} = \dfrac{R_i C_j}{N}$, prove using rigorous algebra that the general formula collapses to the direct closed expression:
		\begin{align}
			\chi^2 = \frac{N(ad - bc)^2}{(a+b)(c+d)(a+c)(b+d)}.
		\end{align}
		\item Analyze the elementary geometry and statistical significance of the factor $(ad - bc)$ in the numerator, relating it to the determinant of the cross-frequency matrix and the odds ratio ($\text{OR} = \dfrac{ad}{bc}$).
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:chi-desaf-teorema-pearson}
	\textbf{Asymptotic multinomial foundation of Pearson's Theorem and analysis of Yates' continuity correction:}
	Let there be an experiment with $N$ independent trials falling into mutually exclusive categories $k$ with random count vector $\mathbf{O} = (O_1, \dots, O_k)$ and theoretical parameter vector $\mathbf{p} = (p_1, \dots, p_k)$ such that $\sum_{i=1}^k p_i = 1$ and $\sum_{i=1}^k O_i = N$.
	\begin{enumerate}
		\item Each marginal cell follows a Binomial distribution $O_i \sim \text{Bin}(N, p_i)$. By the Central Limit Theorem, the standardized variables $Z_i = \dfrac{O_i - Np_i}{\sqrt{Np_i(1-p_i)}}$ converge to standard normals. However, Pearson's test uses the non-standardized sum relative to the full marginal variance: $Q = \sum_{i=1}^k \dfrac{(O_i - Np_i)^2}{Np_i}$. Analytically justify why $Q$ is not distributed as $\chi^2_k$, but exactly as $\chi^2_{k-1}$, explaining the fundamental role of the linear dependence constraint $\sum_{i=1}^k (O_i - Np_i) = 0$ in the vector space of the statistic.
		\item In $2 \times 2$ tables (with $\nu = 1$ degree of freedom) applied to small or medium-sized samples, Frank Yates introduced **Yates' Continuity Correction**:
		\begin{align}
			\chi^2_{\text{Yates}} = \frac{N \left( |ad - bc| - \frac{N}{2} \right)^2}{(a+b)(c+d)(a+c)(b+d)}.
		\end{align}
		Explain epistemologically and from the perspective of discrete perturbation theory why the original Pearson $\chi^2$ statistic tends to be systematically optimistic (overestimating statistical significance and increasing Type I error probability) when continuous distributions approximate discrete hypergeometrics, and demonstrate how the penalty term $-\dfrac{N}{2}$ stabilizes $p$-values in the tails of Fisher's exact distribution.
	\end{enumerate}
\end{problema}

	\begin{enumerate}
		\item \textbf{Formulation of hypotheses:}
		\begin{align}
			H_0&: p_1 = p_2 = p_3 = p_4 = p_5 = p_6 = \frac{1}{6} \quad (\text{The die is perfectly symmetric and equiprobable}). \\
			H_a&: \text{At least one of the six probabilities } p_i \text{ is strictly different from } \frac{1}{6}.
		\end{align}
		
		\item \textbf{Expected frequencies under the null hypothesis ($E_i$):}
		If the die is fair, for each of the 6 faces across $N = 120$ rolls, we expect equiprobably:
		\begin{align}
			E_i = N \times p_i = 120 \times \frac{1}{6} = 20 \text{ occurrences.}
		\end{align}
		
		\item \textbf{Calculation of the $\chi^2$ test statistic:}
		We sum over the 6 faces the ratio of the squared discrepancy to the expected theoretical value:
		\begin{align}
			\chi^2 &= \sum_{i=1}^6 \frac{(O_i - E_i)^2}{E_i} \\
			&= \frac{(18-20)^2}{20} + \frac{(22-20)^2}{20} + \frac{(20-20)^2}{20} + \frac{(19-20)^2}{20} + \frac{(23-20)^2}{20} + \frac{(18-20)^2}{20} \\
			&= \frac{(-2)^2}{20} + \frac{(2)^2}{20} + \frac{0^2}{20} + \frac{(-1)^2}{20} + \frac{(3)^2}{20} + \frac{(-2)^2}{20} \\
			&= \frac{4}{20} + \frac{4}{20} + 0 + \frac{1}{20} + \frac{9}{20} + \frac{4}{20} = \frac{22}{20} = 1.10.
		\end{align}
		
		\item \textbf{Degrees of freedom ($\nu$):}
		In a goodness-of-fit test across $k = 6$ mutually exclusive categories where theoretical probabilities are known a priori (no parameters were estimated), the degrees of freedom are:
		\begin{align}
			\nu = k - 1 = 6 - 1 = 5.
		\end{align}
		
		\item \textbf{Decision rule and statistical conclusion:}
		For a significance level of $\alpha = 0.05$ with $\nu = 5$ degrees of freedom, Pearson's quantile table sets the upper critical threshold at $\chi^2_{0.05, \, 5} = 11.07$.
		Since our sample statistic is far below the threshold ($1.10 < 11.07$), **there is no statistical evidence to reject the null hypothesis ($H_0$)**.
		
		**Conclusion:** Small variations in observed frequencies (such as rolling 23 fives versus 18 ones) are completely natural and attributable to random sampling variation. The die behaves formally as a fair instrument.
	\end{enumerate}
\end{solucion}

\begin{solucion}
	\begin{enumerate}
		\item \textbf{Violation of Cochran's Rule and validity of the statistic:}
		The exact distribution of the random variable $Q = \sum \dfrac{(O_i - E_i)^2}{E_i}$ in finite samples is discrete and highly skewed. Only when observed frequencies are sufficiently large via the De Moivre-Laplace Central Limit Theorem does the true distribution of $Q$ asymptotically converge to Pearson's continuous distribution $\chi^2_\nu$.
		
		William G. Cochran formally demonstrated that for this integral approximation to be mathematically reliable (not distorting $p$-values or causing excessive false positive rates in the tails), **no individual expected frequency ($E_i$) in the study should be less than $1.0$, and no more than $20\%$ of all cells should have expected frequencies less than $5.0$** ($E_i < 5$).
		
		In our design with $k=4$ cells, we observe two categories in severe violation: $E_3 = 3.2 < 5$ and $E_4 = 1.1 < 5$. This implies that $50\%$ of the cells ($2 / 4$) violate the threshold of 5, completely invalidating the use of standard $\chi^2$ tables.
		
		\item \textbf{Methodological corrective procedure:}
		The canonical statistical solution consists of **pooling or collapsing adjacent low-probability categories** into a single combined category until all cells in the new design vector satisfy $E_i \ge 5$.
		
		In this case, we merge classes 3 and 4 into a new aggregated category ($3+4$). Its new combined expected frequency will be:
		\begin{align}
			E_{3+4} = E_3 + E_4 = 3.2 + 1.1 = 4.3.
		\end{align}
		If strict rigor requires (since $4.3$ is still slightly below 5), we could merge it with category 2 ($E_{2+3+4} = 30.5 + 4.3 = 34.8$). However, if we accept the double merge into $k' = 3$ categories ($E_1 = 45.2, E_2 = 30.5, E_{3+4} = 4.3$), since only one cell is slightly below 5 representing a minor fraction of a design satisfying $E_i \ge 1$, the analysis is acceptable.
		Upon collapsing to $k' = 3$ categories, the new theoretical degrees of freedom of the test necessarily decrease from $3$ to:
		\begin{align}
			\nu_{\text{corrected}} = k' - 1 = 3 - 1 = 2 \text{ degrees of freedom.}
		\end{align}
	\end{enumerate}
\end{solucion}

\subsubsection*{4. Challenging Level (Experts / Deep Deduction)}

\begin{solucion}
	\begin{enumerate}
		\item \textbf{Complete algebraic derivation of the simplified formula for $2 \times 2$ tables:}
		In a general cell of a $2 \times 2$ table, the observed frequency is denoted by $O_{ij}$ and the expected count under independence by $E_{ij} = \dfrac{R_i C_j}{N}$. The difference between observation and expectation in the first cell ($O_{11} = a$, where $R_1 = a+b$ and $C_1 = a+c$) is:
		\begin{align}
			O_{11} - E_{11} = a - \frac{(a+b)(a+c)}{N} = \frac{a(a+b+c+d) - (a^2 + ac + ab + bc)}{N}.
		\end{align}
		Expanding the algebraic product in the numerator:
		\begin{align}
			O_{11} - E_{11} = \frac{a^2 + ab + ac + ad - a^2 - ac - ab - bc}{N} = \frac{ad - bc}{N}.
		\end{align}
		Since marginal observed and expected totals for rows and columns match exactly by design, the simple deviations across all four cells share identical absolute magnitude in the numerator, varying only by alternating signs in a checkerboard pattern:
		\begin{align}
			O_{11} - E_{11} &= \frac{ad - bc}{N}, & O_{12} - E_{12} &= -\frac{ad - bc}{N}. \\
			O_{21} - E_{21} &= -\frac{ad - bc}{N}, & O_{22} - E_{22} &= \frac{ad - bc}{N}.
		\end{align}
		Squaring each of these four deviations inside Pearson's statistic yields a universal quadratic numerator across all four cells: $(O_{ij} - E_{ij})^2 = \dfrac{(ad - bc)^2}{N^2}$.
		
		Substituting this common factor into the general definition $\chi^2 = \sum \sum \dfrac{(O_{ij} - E_{ij})^2}{E_{ij}}$:
		\begin{align}
			\chi^2 &= \frac{(ad - bc)^2}{N^2} \left[ \frac{1}{E_{11}} + \frac{1}{E_{12}} + \frac{1}{E_{21}} + \frac{1}{E_{22}} \right] \\
			&= \frac{(ad - bc)^2}{N^2} \left[ \frac{N}{(a+b)(a+c)} + \frac{N}{(a+b)(b+d)} + \frac{N}{(c+d)(a+c)} + \frac{N}{(c+d)(b+d)} \right].
		\end{align}
		Factoring out $N$ from all four terms simplifies with $N^2$ in the outer denominator:
		\begin{align}
			\chi^2 = \frac{(ad - bc)^2}{N} \left[ \frac{1}{R_1 C_1} + \frac{1}{R_1 C_2} + \frac{1}{R_2 C_1} + \frac{1}{R_2 C_2} \right].
		\end{align}
		We group terms inside the bracket by row common factors ($R_1$ and $R_2$):
		\begin{align}
			\frac{1}{R_1 C_1} + \frac{1}{R_1 C_2} + \frac{1}{R_2 C_1} + \frac{1}{R_2 C_2} &= \frac{1}{R_1}\left( \frac{1}{C_1} + \frac{1}{C_2} \right) + \frac{1}{R_2}\left( \frac{1}{C_1} + \frac{1}{C_2} \right) \\
			&= \left( \frac{1}{R_1} + \frac{1}{R_2} \right) \left( \frac{1}{C_1} + \frac{1}{C_2} \right).
		\end{align}
		Performing elementary algebraic addition inside both parentheses:
		\begin{align}
			\left( \frac{R_2 + R_1}{R_1 R_2} \right) \left( \frac{C_2 + C_1}{C_1 C_2} \right).
		\end{align}
		Recalling that the total sum of the two rows ($R_1 + R_2$) or two columns ($C_1 + C_2$) equals exactly the overall sample $N$, the entire bracket equals $\dfrac{N \cdot N}{R_1 R_2 C_1 C_2} = \dfrac{N^2}{R_1 R_2 C_1 C_2}$.
		
		Substituting the simplified bracket back into the statistic equation:
		\begin{align}
			\chi^2 = \frac{(ad - bc)^2}{N} \left[ \frac{N^2}{(a+b)(c+d)(a+c)(b+d)} \right] = \frac{N(ad - bc)^2}{(a+b)(c+d)(a+c)(b+d)}.
		\end{align}
		This elegantly derives Pearson's exact computational formula for $2 \times 2$ tables.
		
		\item \textbf{Significance of the factor $(ad - bc)$ in linear algebra and odds ratio theory:}
		The factor $(ad - bc)$ governs the behavior of the entire statistic. If we structure the original contingency table as a square matrix of order 2:
		\begin{align}
			\mathbf{M} = \begin{pmatrix} a & b \\ c & d \end{pmatrix},
		\end{align}
		we immediately recognize that the term $(ad - bc)$ is the **exact determinant of the contingency matrix ($\det(\mathbf{M}) = ad - bc$)**.
		\begin{itemize}
			\item If the determinant is zero ($\det(\mathbf{M}) = ad - bc = 0 \iff ad = bc$), the rows are linearly dependent (proportional to each other). In probability, this equates to **absolute statistical independence between the qualitative variables** ($\chi^2 = 0$).
			\item Dividing both sides of the null point $ad = bc$ by $bc$ yields the definition of the **Odds Ratio ($\text{OR}$)**:
			\begin{align}
				\text{OR} = \frac{a / c}{b / d} = \frac{ad}{bc}.
			\end{align}
			When $\text{OR} = 1$, the hypothesis of independence holds ($\det = 0$). As $\text{OR}$ diverges from $1$ (toward infinity for strong positive association or toward zero for inverse association), the modulus of the determinant $|ad - bc|$ grows quadratically, vigorously driving the $\chi^2$ statistic toward the right-hand rejection region of the distribution.
		\end{itemize}
	\end{enumerate}
\end{solucion}

		\item \textbf{Foundation of degrees of freedom in Pearson's Theorem ($k-1$ instead of $k$):}
		Let the random vector of observed counts be $\mathbf{O} = (O_1, \dots, O_k)^T$ following a multinomial distribution with parameters $(N, \, p_1, \dots, p_k)$. Pearson's non-standardized statistic is the sum of $k$ quadratic components:
		\begin{align}
			Q = \sum_{i=1}^k \frac{(O_i - Np_i)^2}{Np_i} = \sum_{i=1}^k Z_i^2, \quad \text{where } Z_i = \frac{O_i - Np_i}{\sqrt{Np_i}}.
		\end{align}
		If the variables $Z_i$ were standard normals and **stochastically independent** of each other, their sum of squares $Q$ would be distributed as a direct sum of squared normals, namely $\chi^2_k$ with exactly $k$ degrees of freedom.
		
		However, in any closed experiment or survey, the total sample size is a fixed finite constant ($N$). This imposes a **strict, unbreakable linear constraint** upon individual deviations:
		\begin{align}
			\sum_{i=1}^k O_i = N \implies \sum_{i=1}^k (O_i - Np_i) = \sum_{i=1}^k O_i - N \sum_{i=1}^k p_i = N - N(1) = 0.
		\end{align}
		Since the $k$ sampling deviations must necessarily sum to zero, the count of the $k$-th cell is not a free variable; it is deterministically pre-assigned as soon as the first $k-1$ counts are known:
		\begin{align}
			O_k - Np_k = -\sum_{i=1}^{k-1} (O_i - Np_i).
		\end{align}
		In terms of linear algebra geometry, the random deviation vector $\mathbf{Z} = (Z_1, \dots, Z_k)^T$ cannot freely explore the entire Euclidean space $\mathbb{R}^k$. Instead, it is confined to live inside a degenerate affine hyperplane of strictly reduced topological dimension by one unit: $\mathbb{R}^{k-1}$.
		By orthogonally projecting the multivariate normal vector onto this $(k-1)$-dimensional hyperplane via the idempotent spectral covariance matrix, the eigenvalues of the normalized quadratic form yield exactly $k-1$ ones and a single zero. Consequently, the asymptotic statistic $Q$ is distributed as a $\chi^2$ variable with the true spectral rank of the matrix:
		\begin{align}
			\nu = k - 1 \text{ degrees of freedom.}
		\end{align}
		
		\item \textbf{Epistemology of positive bias in Pearson and Yates' correction mechanism:}
		\begin{itemize}
			\item \textbf{Origin of optimistic bias (overestimation of Type I error):}
			The exact theoretical distribution of the test statistic for a finite $2 \times 2$ discrete table is dictated by **Fisher's Exact Hypergeometric Distribution**. As a counting function of integer subjects or successes, its sampling probability function is purely **discrete (step-like)**.
			In contrast, the reference distribution $\chi^2_1$ against which we compare Pearson's statistic is a **smooth, absolutely continuous** mathematical curve.
			When attempting to approximate discrete step jumps using a continuous integral curve, the area under the continuous right tail of the $\chi^2_1$ distribution (which represents the evaluated $p$-value) is systematically smaller than the true exact probability of observing that extreme discrepancy in the discrete hypergeometric world. Therefore, the classic Pearson formula yields artificially low $p$-values ($p_{\text{Pearson}} < p_{\text{exact}}$), inducing the researcher to commit an excess of **false positives (Type I error)** and declare spurious significance in finite samples that true exact probability does not support.
			
			\item \textbf{Mathematical mechanism of Yates' stabilization ($-\frac{N}{2}$):}
			To bridge this quantization gap and align the continuous asymptotic curve with discrete hypergeometric steps (analogous to De Moivre's $\pm 0.5$ continuity correction for binomial-normal approximations), Frank Yates introduced a constant contraction that reduces the discrepancy between observation and expectation by exactly half a discrete unit ($0.5$).
			Since for an elementary cell the correction is $|O_{ij} - E_{ij}| - 0.5$, translating this adjustment into the expanded general determinant formula for the full table multiplies the reduction factor by exactly the total cell sample size $N$:
			\begin{align}
				|ad - bc| \longrightarrow |ad - bc| - \frac{N}{2}.
			\end{align}
			By subtracting the constant quantity $\dfrac{N}{2}$ from the absolute magnitude of the determinant before squaring it, the final statistic $\chi^2_{\text{Yates}}$ **decreases in numerical value**:
			\begin{align}
				\chi^2_{\text{Yates}} = \frac{N \left( |ad - bc| - \frac{N}{2} \right)^2}{(a+b)(c+d)(a+c)(b+d)} < \chi^2_{\text{Pearson}}.
			\end{align}
			Lowering the statistic increases the associated $p$-value in the right tail of the continuous distribution, bringing it into perfect epistemological alignment with the strict $p$-value of Fisher's Exact Test. Yates thus transforms the test into a conservative tool free of false positives for discrete contingency analysis in data science and clinical statistics.
		\end{itemize}
	\end{enumerate}
\end{solucion}
