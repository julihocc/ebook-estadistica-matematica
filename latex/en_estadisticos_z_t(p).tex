\begin{problema}\label{prob:c7ea911}
A battery factory claims that the mean lifetime is $500$ hours. The population standard deviation is known to be $\sigma=40$. A sample of $n=64$ batteries has $\bar x=490$. State the left-tailed hypotheses, compute the $Z$ statistic and $p$-value, and decide whether to reject at $\alpha=0.05$.
\end{problema}

\begin{problema}\label{prob:f6ad523}
A tire manufacturer claims a mean lifetime of $50{,}000$ km. A sample of $n=16$ tires gives $\bar x=48{,}500$ and $s=3{,}000$. Decide whether to use $Z$ or $t$, state the hypotheses for a left-tailed test, compute the test statistic, approximate the $p$-value with $15$ degrees of freedom, and state the conclusion at $\alpha=0.05$.
\end{problema}

\begin{problema}\label{prob:6c535e4}
Derive the scale-invariance of Student's $t$ statistic. Show how $\sigma$ cancels from
\[
	T=\frac{Z}{\sqrt{V/\nu}},
\]
where
\[
	Z=\frac{\bar X-\mu}{\sigma/\sqrt n},\qquad
	V=\frac{(n-1)S^2}{\sigma^2},\qquad
	\nu=n-1.
\]
\end{problema}

\begin{problema}\label{prob:a9a27c5}
A telecom company claims that mean waiting time is $8$ minutes. The standard deviation is known to be $\sigma=2$. A sample of $n=49$ calls gives $\bar x=8.5$. Compute the $Z$ statistic for $H_a:\mu>8$, the one-sided $p$-value, and the decisions at $\alpha=0.05$ and $\alpha=0.01$.
\end{problema}

\begin{problema}\label{prob:8843fed}
Create a bilateral known-$\sigma$ $Z$-test example about sodium content in food. Use $\mu_0=800$, $n=36$, $\bar x=815$, and $\sigma=45$. Specify whether the test uses $Z$ or $t$, state the hypotheses, compute the statistic and $p$-value, and conclude at $\alpha=0.05$.
\end{problema}

\begin{solproblema}[prob:c7ea911]
The hypotheses are
\[
	H_0:\mu=500,\qquad H_a:\mu<500.
\]
The statistic is
\[
	Z=\frac{490-500}{40/\sqrt{64}}=\frac{-10}{5}=-2.
\]
The left-tailed $p$-value is $P(Z<-2)\approx 0.0228$. Since $0.0228<0.05$, reject $H_0$.
\end{solproblema}

\begin{solproblema}[prob:f6ad523]
Use Student's $t$ distribution because $\sigma$ is unknown and the sample is small. The hypotheses are
\[
	H_0:\mu=50000,\qquad H_a:\mu<50000.
\]
The statistic is
\[
	t=\frac{48500-50000}{3000/\sqrt{16}}=-2.
\]
With $df=15$, the one-sided $p$-value is approximately $0.032$. Since it is below $0.05$, reject $H_0$.
\end{solproblema}

\begin{solproblema}[prob:4072ece]
The sample mean is $\bar x=100$. The squared deviations sum to $36$, so
\[
	s^2=\frac{36}{8}=4.5,\qquad s\approx 2.1213.
\]
The hypotheses are
\[
	H_0:\mu=100,\qquad H_a:\mu\ne 100.
\]
The statistic is
\[
	t=\frac{100-100}{2.1213/\sqrt9}=0,
\]
with $df=8$. The $p$-value is $1$, so do not reject $H_0$.
\end{solproblema}

\begin{solproblema}[prob:6c535e4]
Substitute the definitions:
\[
	T=\frac{(\bar X-\mu)/(\sigma/\sqrt n)}
	{\sqrt{\left((n-1)S^2/\sigma^2\right)/(n-1)}}.
\]
The denominator simplifies to $S/\sigma$, so
\[
	T=\frac{(\bar X-\mu)/(\sigma/\sqrt n)}{S/\sigma}
	=\frac{\bar X-\mu}{S/\sqrt n}.
\]
Thus the unknown population scale $\sigma$ cancels, which is why the statistic follows a $t_{n-1}$ reference distribution under normality.
\end{solproblema}

\begin{solproblema}[prob:a9a27c5]
The statistic is
\[
	Z=\frac{8.5-8}{2/\sqrt{49}}=1.75.
\]
The one-sided $p$-value is
\[
	P(Z>1.75)\approx 0.0401.
\]
At $\alpha=0.05$, reject $H_0$; at $\alpha=0.01$, do not reject. The stricter level requires stronger evidence and reduces the tolerated probability of a Type I error.
\end{solproblema}

\begin{solproblema}[prob:8843fed]
Use a $Z$ test because $\sigma$ is known. The hypotheses are
\[
	H_0:\mu=800,\qquad H_a:\mu\ne 800.
\]
The statistic is
\[
	Z=\frac{815-800}{45/\sqrt{36}}=\frac{15}{7.5}=2.0.
\]
The two-sided $p$-value is about $0.0456$. Since $0.0456<0.05$, reject $H_0$.
\end{solproblema}
