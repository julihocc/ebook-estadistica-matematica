\section*{Problems}

\subsection{Problem Statements}

\subsubsection*{1. Fundamental Level (Simple / Direct)}

\begin{problema}
	\label{prob-z-1}
	A battery manufacturer claims that the average lifespan of its batteries is 500 hours with a known population standard deviation of $\sigma = 40$ hours. A quality control inspector takes a random sample of $n = 64$ batteries and finds a sample average lifespan of $\bar{X} = 490$ hours.
	\begin{enumerate}
		\item Formulate the null ($H_0$) and alternative ($H_a$) hypotheses to test whether the average lifespan is statistically less than 500 hours.
		\item Calculate the $Z$ test statistic.
		\item Determine the $p$-value associated with the test.
		\item Should the null hypothesis be rejected at the significance level $\alpha = 0.05$?
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob-t-1}
	A tire manufacturer claims that its tires last an average of $\mu_0 = 50,000$ km. A consumer takes a random sample of $n = 16$ tires and finds a sample mean $\bar{X} = 48,500$ km with a sample standard deviation $s = 3,000$ km.
	\begin{enumerate}
		\item Which distribution should be used for the hypothesis test ($Z$ or Student's $t$)? Explain why.
		\item Formulate the hypotheses to test whether the population average lifespan is less than 50,000 km.
		\item Calculate the $t$-statistic.
		\item Determine the approximate $p$-value using $\nu = n-1 = 15$ degrees of freedom.
		\item Should $H_0$ be rejected at the significance level $\alpha = 0.05$?
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob-z-t-new1}
	A random sample of size $n = 25$ taken from a normal population yields a sample standard deviation $s = 12$.
	\begin{enumerate}
		\item Calculate the standard error of the sample mean $\text{SE}(\bar{X}) = \dfrac{s}{\sqrt{n}}$.
		\item Determine the critical values of Student's $t$-statistic for a two-tailed test at the significance levels $\alpha = 0.05$ and $\alpha = 0.01$ with $\nu = 24$ degrees of freedom.
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob-z-t-new2}
	\textbf{The Central Limit Theorem and Standardization:} Let $X_1, X_2, \dots, X_n$ be a simple random sample drawn from a population whose probability distribution is not normal, but which has finite population mean $\mu$ and variance $\sigma^2$.
	\begin{enumerate}
		\item Prove using the linear operators of expectation $E[\cdot]$ and variance $\Var[\cdot]$ that the expected value and variance of the sample mean $\bar{X}$ are $E[\bar{X}] = \mu$ and $\Var[\bar{X}] = \dfrac{\sigma^2}{n}$.
		\item Prove algebraically that when applying linear standardization $Z_n = \dfrac{\bar{X} - E[\bar{X}]}{\sqrt{\Var[\bar{X}]}}$, the resulting statistic has exact mean 0 and exact variance 1 for any sample size $n$.
		\item Formally explain why for a large sample size ($n \ge 30$) the hypothesis test on $\mu$ using $Z_n = \dfrac{\bar{X} - \mu_0}{\sigma/\sqrt{n}}$ is asymptotically valid even in the presence of population skewness in the original data.
	\end{enumerate}
\end{problema}

\subsubsection*{4. Challenging Level (Experts / Deep Deduction)}

\begin{problema}
	\label{prob-z-t-new3}
	\textbf{Theoretical derivation of scale invariance of Student's $t$-statistic:}
	In exact sampling theory from a normal distribution, if $Z \sim N(0,1)$ and $V \sim \chi^2(\nu)$ are two independent random variables, Student's statistic is defined as the transformed ratio:
	\begin{align}
		T = \frac{Z}{\sqrt{V / \nu}} \sim t_\nu.
	\end{align}
	Let $X_1, \dots, X_n$ be a normal random sample $N(\mu, \sigma^2)$. We know that the standardized statistic is $Z = \dfrac{\bar{X} - \mu}{\sigma/\sqrt{n}} \sim N(0,1)$ and that the normalized sum of sample squares satisfies $V = \dfrac{(n-1)S^2}{\sigma^2} \sim \chi^2(n-1)$, with $\bar{X}$ and $S^2$ being independent by Cochran's Theorem.
	
	Prove algebraically step-by-step that when substituting these expressions into the definition of the $T$ statistic, the unknown population scale parameter ($\sigma$) cancels exactly, demonstrating formally why the pivotal quantity $\displaystyle T = \frac{\bar{X} - \mu}{S/\sqrt{n}}$ allows performing exact tests on the location parameter $\mu$ without requiring prior knowledge of the dispersion $\sigma$.
\end{problema}

\begin{problema}
	\label{prob-z-t-new4}
	\textbf{Heavy-tail properties of Student's $t$-distribution vs. Normal:}
	Student's $t$-distribution with $\nu$ degrees of freedom has a probability density function whose variance for $\nu > 2$ is rigorously given by the second central moment relationship:
	\begin{align}
		\Var(T) = \frac{\nu}{\nu - 2}.
	\end{align}
	\begin{enumerate}
		\item Analytically prove that for all finite $\nu > 2$, the strict variance inequality $\Var(T) > \Var(Z) = 1$ holds.
		\item Explain from the conceptual perspective of sampling and uncertainty in estimating the parameter $S^2 \approx \sigma^2$, why the $t$-distribution requires heavier tails (higher probability in the extreme tails away from zero or leptokurtosis) than the standard normal distribution.
		\item Explicitly evaluate the limit $\displaystyle \lim_{\nu \to \infty} \Var(T)$ and prove its consistency with the distributional convergence in probability of Student's $t$-statistic toward the standard normal $Z$.
	\end{enumerate}
\end{problema}

\subsubsection*{1. Fundamental Level (Simple / Direct)}

\begin{solproblema}
	\begin{enumerate}
		\item Left-tailed one-sided statistical hypotheses:
		\begin{align}
			H_0: \mu &= 500 \text{ hours} \\
			H_a: \mu &< 500 \text{ hours}
		\end{align}
		
		\item $Z$ test statistic (since $\sigma = 40$ is known for the population):
		\begin{align}
			Z = \frac{\bar{X} - \mu_0}{\sigma / \sqrt{n}} = \frac{490 - 500}{40 / \sqrt{64}} = \frac{-10}{40 / 8} = \frac{-10}{5} = -2.0.
		\end{align}
		
		\item $p$-value (area under the standard normal curve to the left of $-2.0$):
		\begin{align}
			p\text{-value} = P(Z < -2.0) = \Phi(-2.0) \approx 0.0228 \text{ (or 2.28\%).}
		\end{align}
		
		\item Since the $p\text{-value} = 0.0228 < 0.05 = \alpha$, \textbf{we reject the null hypothesis ($H_0$)}.
		
		\textbf{Conclusion:} There is sufficient statistical evidence in the sample to reject the manufacturer's claim at the 5\% level and conclude that the average lifespan of the batteries is significantly less than 500 hours.
	\end{enumerate}
\end{solproblema}

		\end{align}

\subsubsection*{3. Analytical Level (Advanced / Connection)}

\begin{solproblema}
	\begin{enumerate}
		\item Exact arithmetic calculations for the sample ($n=9$ cables):
		\begin{align}
			\sum X_i &= 98 + 102 + 99 + 101 + 97 + 103 + 100 + 98 + 102 = 900. \\
			\bar{X} &= \frac{900}{9} = 100 \text{ ohms.}
		\end{align}
		We calculate the unbiased sample variance using $S_{xx} = \sum (X_i - \bar{X})^2$:
		\begin{align}
			\sum (X_i - 100)^2 &= (-2)^2 + 2^2 + (-1)^2 + 1^2 + (-3)^2 + 3^2 + 0^2 + (-2)^2 + 2^2 \\
			&= 4 + 4 + 1 + 1 + 9 + 9 + 0 + 4 + 4 = 36. \\
			s^2 &= \frac{\sum (X_i - \bar{X})^2}{n - 1} = \frac{36}{9 - 1} = \frac{36}{8} = 4.5. \\
			s &= \sqrt{4.5} \approx 2.1213 \text{ ohms.}
		\end{align}
		
		\item Two-tailed hypotheses regarding the resistance parameter:
		\begin{align}
			H_0: \mu &= 100 \text{ ohms} \\
			H_a: \mu &\neq 100 \text{ ohms}

		\begin{align}
			t = \frac{\bar{X} - \mu_0}{s / \sqrt{n}} = \frac{100 - 100}{2.1213 / \sqrt{9}} = \frac{0}{0.7071} = 0.

\begin{solproblema}
	\begin{enumerate}
		\item By the linearity of the expectation operator and since the observations $X_i$ are independent and identically distributed with $E[X_i] = \mu$:
		\begin{align}
			E[\bar{X}] = E\left[ \frac{1}{n} \sum_{i=1}^n X_i \right] = \frac{1}{n} \sum_{i=1}^n E[X_i] = \frac{1}{n} (n \mu) = \mu.
		\end{align}
		By independence of the observations, variances are directly additive ($\Var[\sum X_i] = \sum \Var[X_i]$):
		\begin{align}
			\Var[\bar{X}] = \Var\left[ \frac{1}{n} \sum_{i=1}^n X_i \right] = \frac{1}{n^2} \sum_{i=1}^n \Var[X_i] = \frac{1}{n^2} (n \sigma^2) = \frac{\sigma^2}{n}.
		\end{align}
		
		\item Let $Z_n = \dfrac{\bar{X} - \mu}{\sigma / \sqrt{n}}$. Applying the properties of central moments to a linearly standardized variable:
		\begin{align}
			E[Z_n] &= E\left[ \frac{\bar{X} - \mu}{\sigma/\sqrt{n}} \right] = \frac{E[\bar{X}] - \mu}{\sigma/\sqrt{n}} = \frac{\mu - \mu}{\sigma/\sqrt{n}} = 0. \\
			\Var[Z_n] &= \Var\left[ \frac{\bar{X} - \mu}{\sigma/\sqrt{n}} \right] = \left( \frac{1}{\sigma/\sqrt{n}} \right)^2 \Var[\bar{X} - \mu] = \frac{n}{\sigma^2} \Var[\bar{X}] = \frac{n}{\sigma^2} \left( \frac{\sigma^2}{n} \right) = 1.
		\end{align}
		
		\item \textbf{Asymptotic validity of the Central Limit Theorem:}
		The Central Limit Theorem fundamentally establishes that, under general regularity conditions and finite variance, the cumulative distribution function of $Z_n$ converges uniformly in law to the standard normal:
		\begin{align}
			\lim_{n \to \infty} P(Z_n \le z) = \Phi(z) = \frac{1}{\sqrt{2\pi}} \int_{-\infty}^z e^{-u^2/2} du, \quad \text{for all } z \in \mathbb{R}.
		\end{align}
		Therefore, when the sample size is moderately large ($n \ge 30$), the distribution of the sample mean $\bar{X}$ loses the specific skewness and kurtosis of the original distribution, allowing accurate calculation of $p$-values using the standard normal $Z$ table.
	\end{enumerate}
\end{solproblema}

\subsubsection*{4. Challenging Level (Experts / Deep Deduction)}

\begin{solproblema}
	We substitute into the fundamental ratio $T = \dfrac{Z}{\sqrt{V / \nu}}$ the exact analytical expressions relating the sample quantities to the population parameters under normality:
	\begin{align}
		Z &= \frac{\bar{X} - \mu}{\sigma / \sqrt{n}} \sim N(0,1), \\
		V &= \frac{(n - 1)S^2}{\sigma^2} \sim \chi^2(n-1), \quad \text{with degrees of freedom } \nu = n - 1.
	\end{align}
	Substituting both terms into the radical in the denominator of the $T$ statistic:
	\begin{align}
		T = \frac{\dfrac{\bar{X} - \mu}{\sigma / \sqrt{n}}}{\sqrt{\dfrac{\dfrac{(n - 1)S^2}{\sigma^2}}{n - 1}}}.
	\end{align}
	Simplifying the $(n - 1)$ factor under the square root in the denominator:
	\begin{align}
		\sqrt{\frac{(n - 1)S^2}{\sigma^2 (n - 1)}} = \sqrt{\frac{S^2}{\sigma^2}} = \frac{S}{\sigma}.
	\end{align}
	Rewriting the complete fraction of Student's statistic:
	\begin{align}
		T = \frac{\dfrac{\bar{X} - \mu}{\sigma / \sqrt{n}}}{\dfrac{S}{\sigma}} = \frac{\bar{X} - \mu}{\sigma / \sqrt{n}} \times \frac{\sigma}{S}.
	\end{align}
	Notice how the population dispersion parameter $\sigma$ in the numerator and denominator cancels out arithmetically:
	\begin{align}
		T = \frac{\bar{X} - \mu}{S / \sqrt{n}} \sim t_{n-1}.
	\end{align}
	\textbf{Structural theoretical implication:} This exact cancellation of $\sigma$ is what empowers Student's inference. It allows the $T$ statistic to be computed entirely from the sample data alone ($\bar{X}, S, n$) and the hypothesized value ($\mu_0$), operating under a mathematical reference distribution ($t_{n-1}$) that is 100\% independent of the true value of $\sigma$.
\end{solproblema}

\begin{solproblema}
	\begin{enumerate}
		\item \textbf{Analytical proof of greater variance:}
		For any number of degrees of freedom $\nu > 2$, the variance of the $t$-distribution is given by:
		\begin{align}
			\Var(T) = \frac{\nu}{\nu - 2}.
		\end{align}
		Since $\nu > 2 \implies \nu - 2 > 0$, we can rewrite the numerator by splitting $\nu = (\nu - 2) + 2$:
		\begin{align}
			\Var(T) = \frac{(\nu - 2) + 2}{\nu - 2} = 1 + \frac{2}{\nu - 2}.
		\end{align}
		Because the fractional remainder $\dfrac{2}{\nu - 2}$ is strictly positive for any finite $\nu > 2$, it is formally proven that:
		\begin{align}
			\Var(T) = 1 + \frac{2}{\nu - 2} > 1 = \Var(Z).
		\end{align}
		
		\item \textbf{Explanation of scale parameter uncertainty and leptokurtosis:}
		When the true population variance $\sigma^2$ is known, the only source of variability in standardizing $\bar{X}$ comes from the sampling error of the mean itself ($Z$). However, when we replace $\sigma$ with the sample estimator $S$, we introduce a \textbf{second source of random variability or uncertainty}: the estimation error of the sample standard deviation $S$.
		
		In small samples where $S$ accidentally underestimates $\sigma$ ($S \ll \sigma$), the ratio $T = \dfrac{\bar{X} - \mu}{S/\sqrt{n}}$ experiences significant numerical inflation in the denominator, driving the value of $T$ out into extreme tails. To mathematically capture this dual random variability without generating an excess of false positives in hypothesis tests, the Student's $t$ probability density function must be leptokurtic: its tails are strictly thicker than the normal distribution ($P(|T| > k) > P(|Z| > k)$), providing an exact inferential safety margin against the unknown $\sigma$.
		
		\item \textbf{Asymptotic limit for infinite samples:}
		Evaluating the variance behavior as $\nu \to \infty$:
		\begin{align}
			\lim_{\nu \to \infty} \Var(T) = \lim_{\nu \to \infty} \left( 1 + \frac{2}{\nu - 2} \right) = 1 + 0 = 1.
		\end{align}
		This limit is perfectly consistent and aligned with Student's convergence theorem: as the sample size increases ($n \to \infty$), the estimator $S^2$ converges in probability to the population constant $\sigma^2$ by the Law of Large Numbers. With the uncertainty in the denominator eliminated, the heavy-tailed distribution $t_\nu$ converges uniquely and indistinguishably in distribution toward the standard normal $Z \sim N(0,1)$.
	\end{enumerate}
\end{solproblema}
