% Recordar
\begin{problema}
	\label{prob:1d85679}
	Sea $X_1, X_2, \dots, X_n$ una muestra aleatoria independiente extraída de una población cuya variable aleatoria satisface $E[X^2] = \theta$, donde $\theta > 0$ es el segundo momento poblacional respecto al origen. Se propone el estadístico muestral:
	\begin{align}
		\hat{\theta} = \frac{1}{n} \sum_{i=1}^n X_i^2.
	\end{align}
	Demuestre analíticamente mediante linealidad del operador esperanza que $\hat{\theta}$ es un estimador estrictamente insesgado para $\theta$.
\end{problema}

\begin{problema}
	\label{prob:6f639cf}
	El tiempo entre fallas de los servidores de un centro de datos en la nube se modela mediante una variable aleatoria continua $X$ con distribución Exponencial de parámetro $\lambda > 0$, cuya función de densidad es:
	\begin{align}
		f(x \mid \lambda) = \lambda e^{-\lambda x}, \quad x > 0.
	\end{align}
	\begin{enumerate}
		\item Obtenga el estimador del parámetro $\lambda$ utilizando el Método de Momentos ($\hat{\lambda}_{MoM}$).
		\item Obtenga el estimador de máxima verosimilitud $\hat{\lambda}_{MLE}$. ¿Coinciden ambos métodos en este modelo? Explique bajo qué condición general el Método de Momentos y la Máxima Verosimilitud coinciden.
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:0363afd}
	En una campaña publicitaria en redes sociales, el resultado de hacer clic en un anuncio por parte del $i$-ésimo usuario se modela mediante una variable aleatoria Bernoulli $X_i \sim \text{Bernoulli}(p)$, donde $p \in (0,1)$ es la verdadera tasa de clics (CTR) desconocida y $n$ es el número total de impresiones independientes.
	\begin{enumerate}
		\item Deduzca el estimador de máxima verosimilitud $\hat{p}_{MLE}$ a partir de la función de verosimilitud de ensayos binarios.
		\item Calcule la varianza exacta $\Var(\hat{p}_{MLE})$ en función de $p$ y $n$, e indique la expresión del error estándar estimado $\text{SE}(\hat{p}_{MLE})$.
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:977b8dd}
	\textbf{Eficiencia absoluta y Cota Inferior de Cramér-Rao (CRLB):} sea $X_1,\dots,X_n$ una muestra de una distribución Poisson con tasa $\lambda>0$.
	\begin{enumerate}
		\item Calcule la Información de Fisher $I(\lambda)$ contenida en una sola observación $X \sim \text{Poisson}(\lambda)$ derivando el logaritmo de su densidad y calculando su segunda derivada esperada.
		\item Determine la Cota Inferior de Cramér-Rao para la varianza de cualquier estimador insesgado de $\lambda$ basado en una muestra de tamaño $n$.
		\item Demuestre que el estimador $\hat{\lambda}_{MLE} = \bar{X}$ alcanza exactamente dicha cota, concluyendo que es el Estimador Insesgado de Mínima Varianza Uniformemente (UMVUE).
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:23bb182}
	Sea $X_1, X_2, \ldots, X_n$ una muestra aleatoria de una variable con distribución Uniforme en el intervalo $[0, \theta]$, donde $\theta > 0$ es un parámetro desconocido.
	\begin{enumerate}
		\item Encuentre el estimador del Método de Momentos $\hat{\theta}_{MoM}$ y el estimador de máxima verosimilitud $\hat{\theta}_{MLE} = X_{(n)} = \max(X_1,\dots,X_n)$ (justifique brevemente por qué el procedimiento habitual de derivar e igualar a cero falla aquí).
		\item Evalúe comparativamente el sesgo de $\hat{\theta}_{MoM}$ y $\hat{\theta}_{MLE}$, y construya a partir de este último un estimador modificado $\hat{\theta}^*$ que sea completamente insesgado. Juzgue cuál de los tres estimadores ($\hat{\theta}_{MoM}$, $\hat{\theta}_{MLE}$, $\hat{\theta}^*$) recomendaría en la práctica y por qué.
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:0675127}
	Diseñe un problema original que ilustre la conexión entre la estimación puntual clásica y el entrenamiento de modelos en ciencia de datos: para una única observación binaria $y \in \{0,1\}$ con probabilidad predicha $p$ por un modelo de clasificación, demuestre que minimizar la pérdida de entropía cruzada $\mathcal{L}(p) = -[y\ln p + (1-y)\ln(1-p)]$ respecto a $p$ es matemáticamente equivalente a maximizar la log-verosimilitud de un modelo Bernoulli, y por lo tanto produce el mismo estimador que la máxima verosimilitud clásica.
\end{problema}

\subsection*{Soluciones detalladas}

% Recordar
\begin{solproblema}[prob:1d85679]
	Por definición, un estimador $\hat{\theta}$ es insesgado para $\theta$ si $E[\hat{\theta}] = \theta$. Aplicamos el operador de esperanza al estadístico propuesto:
	\begin{align}
		E[\hat{\theta}] = E\left[ \frac{1}{n} \sum_{i=1}^n X_i^2 \right] = \frac{1}{n} \sum_{i=1}^n E\left[ X_i^2 \right],
	\end{align}
	por linealidad de la esperanza. Dado que las variables son i.i.d., comparten el mismo segundo momento $E[X_i^2] = \theta$ para todo $i$. Sustituyendo:
	\begin{align}
		E[\hat{\theta}] = \frac{1}{n} \sum_{i=1}^n \theta = \frac{1}{n} (n \theta) = \theta.
	\end{align}
	Como $E[\hat{\theta}] = \theta$, el estimador es insesgado.
\end{solproblema}

% Comprender
\begin{solproblema}[prob:6f639cf]
	\begin{enumerate}
		\item El primer momento teórico de la exponencial es $\mu_1 = E[X] = 1/\lambda$. Igualando con el momento muestral $m_1=\bar{X}$: $\hat{\lambda}_{MoM} = 1/\bar{X}$.
		\item La log-verosimilitud es $\ell(\lambda) = n\ln\lambda - \lambda\sum x_i$. Derivando e igualando a cero: $\dfrac{n}{\lambda}-\sum x_i=0 \implies \hat{\lambda}_{MLE}=1/\bar{X}$.

		\textbf{Conclusión:} ambos métodos coinciden en la exponencial, resultando en $1/\bar{X}$. En general, MoM y MLE coinciden cuando el modelo pertenece a la familia exponencial con un solo parámetro y el estadístico suficiente coincide con el momento muestral usado en la ecuación de momentos —como ocurre aquí, donde tanto el primer momento poblacional como la verosimilitud dependen de los datos únicamente a través de $\bar{X}$.
	\end{enumerate}
\end{solproblema}

% Aplicar
\begin{solproblema}[prob:0363afd]
	\begin{enumerate}
		\item La verosimilitud es $L(p) = p^{\sum x_i}(1-p)^{n-\sum x_i}$, con log-verosimilitud $\ell(p) = (\sum x_i)\ln p + (n-\sum x_i)\ln(1-p)$. Derivando e igualando a cero:
		\begin{align}
			\frac{\sum x_i}{p} - \frac{n-\sum x_i}{1-p} = 0 \implies \hat{p}_{MLE} = \frac{1}{n}\sum_{i=1}^n X_i = \bar{X}.
		\end{align}
		\item Como $\Var(X_i)=p(1-p)$: $\Var(\hat{p}_{MLE}) = \Var(\bar{X}) = \dfrac{p(1-p)}{n}$. El error estándar estimado, reemplazando $p$ por $\hat{p}$, es $\text{SE}(\hat{p}_{MLE}) = \sqrt{\hat{p}(1-\hat{p})/n}$.
	\end{enumerate}
\end{solproblema}

% Analizar
\begin{solproblema}[prob:977b8dd]
	\begin{enumerate}
		\item $\ln f(X\mid\lambda) = X\ln\lambda-\lambda-\ln(X!)$. Derivando dos veces: $\frac{\partial}{\partial\lambda}\ln f = X/\lambda-1$, $\frac{\partial^2}{\partial\lambda^2}\ln f = -X/\lambda^2$. La Información de Fisher es:
		\begin{align}
			I(\lambda) = -E\left[-\frac{X}{\lambda^2}\right] = \frac{E[X]}{\lambda^2} = \frac{\lambda}{\lambda^2} = \frac{1}{\lambda}.
		\end{align}
		\item Para $n$ observaciones: $\text{CRLB} = \dfrac{1}{nI(\lambda)} = \dfrac{\lambda}{n}$.
		\item $\Var(\hat{\lambda}_{MLE}) = \Var(\bar{X}) = \Var(X_i)/n = \lambda/n \equiv \text{CRLB}$. Como $\hat{\lambda}_{MLE}$ es insesgado y alcanza exactamente la cota de Cramér-Rao, es el UMVUE.
	\end{enumerate}
\end{solproblema}

% Evaluar
\begin{solproblema}[prob:23bb182]
	\begin{enumerate}
		\item $\hat{\theta}_{MoM}=2\bar{X}$ (pues $\mu_1=\theta/2$). Derivar $\ell(\theta)=-n\ln\theta$ e igualar a cero da $-n/\theta=0$, sin solución: el soporte de la densidad depende de $\theta$, violando las condiciones de regularidad. La verosimilitud $L(\theta)=\theta^{-n}$ (para $\theta\ge X_{(n)}$) es estrictamente decreciente, así que se maximiza en el valor más pequeño admisible: $\hat{\theta}_{MLE}=X_{(n)}$.
		\item $E[\hat{\theta}_{MoM}]=\theta$ (insesgado). Para el MLE, con $f_{X_{(n)}}(y)=ny^{n-1}/\theta^n$: $E[\hat{\theta}_{MLE}] = \frac{n}{n+1}\theta$, así que $\text{Sesgo}(\hat{\theta}_{MLE})=-\theta/(n+1)$ (subestima sistemáticamente). El estimador corregido es $\hat{\theta}^* = \frac{n+1}{n}X_{(n)}$, insesgado por construcción.

		\textbf{Juicio:} en la práctica se recomienda $\hat{\theta}^*$: conserva la eficiencia del MLE (que usa toda la información relevante vía el estadístico suficiente $X_{(n)}$, con varianza mucho menor que $\hat{\theta}_{MoM}=2\bar{X}$) mientras corrige exactamente su sesgo sistemático, dominando a los otros dos en error cuadrático medio para casi cualquier $n$.
	\end{enumerate}
\end{solproblema}

% Crear
\begin{solproblema}[prob:0675127]
	La log-verosimilitud de una única observación Bernoulli$(p)$ es $\ell(p) = y\ln p + (1-y)\ln(1-p)$ (caso $n=1$ de la deducción general de $\ell(p)$ para la muestra completa). La pérdida de entropía cruzada se define como:
	\begin{align}
		\mathcal{L}(p) = -[y\ln p + (1-y)\ln(1-p)] = -\ell(p).
	\end{align}
	Por lo tanto, $\mathcal{L}(p) = -\ell(p)$: la pérdida de entropía cruzada es exactamente el negativo de la log-verosimilitud. Minimizar $\mathcal{L}(p)$ respecto a $p$ es entonces idéntico a maximizar $\ell(p)$:
	\begin{align}
		\arg\min_p \mathcal{L}(p) = \arg\min_p [-\ell(p)] = \arg\max_p \ell(p).
	\end{align}
	En consecuencia, el valor de $p$ que minimiza la entropía cruzada es exactamente el estimador de máxima verosimilitud del modelo Bernoulli. Esto explica por qué entrenar una red neuronal o un modelo de regresión logística minimizando \emph{cross-entropy loss} sobre datos etiquetados binarios es, matemáticamente, el mismo procedimiento que ajustar los parámetros de un modelo Bernoulli (o Binomial/Multinomial, en el caso multiclase) por máxima verosimilitud.
\end{solproblema}
