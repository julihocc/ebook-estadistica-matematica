\section*{Problemas}

\subsection*{Enunciados de los problemas}

% Recordar
\begin{problema}
	\label{prob:c7c2f12}
	Un analista financiero ajusta un modelo de regresión de series de tiempo con $n = 10$ observaciones consecutivas y obtiene los siguientes residuos ordenados temporalmente:
	\begin{align}
		e = \{2.1,\ 1.8,\ -0.5,\ -1.2,\ -2.0,\ -1.5,\ 0.8,\ 1.9,\ 2.2,\ 1.0\}.
	\end{align}
	Calcule el estadístico de Durbin-Watson $DW = \dfrac{\sum_{i=2}^n (e_i - e_{i-1})^2}{\sum_{i=1}^n e_i^2}$ y determine si existe autocorrelación positiva, negativa, o ausencia de autocorrelación en los residuos.
\end{problema}

\begin{problema}
	\label{prob:90810fe}
	En un modelo de regresión sobre gasto en publicidad ($X$) y ventas ($Y$), se grafican los residuos contra los valores ajustados y se observa el siguiente patrón: para valores ajustados bajos ($\hat{Y} < 50$) los residuos oscilan en $\pm 2$; para valores ajustados intermedios ($50 \leq \hat{Y} < 150$) oscilan en $\pm 8$; para valores ajustados altos ($\hat{Y} \geq 150$) oscilan en $\pm 20$.
	\begin{enumerate}
		\item Identifique el patrón geométrico descrito y el supuesto clásico que se viola.
		\item Proponga una transformación correctiva y justifique por qué sería efectiva en este escenario.
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:46ca545}
	Un analista de crédito construye un modelo con $p = 3$ predictores financieros potencialmente correlacionados. Al regredir cada predictor contra los otros dos, obtiene los siguientes coeficientes de determinación auxiliares: $R_1^2 = 0.85$, $R_2^2 = 0.55$, $R_3^2 = 0.30$.
	\begin{enumerate}
		\item Calcule el Factor de Inflación de Varianza $\text{VIF}_j = 1/(1-R_j^2)$ para cada predictor.
		\item Usando el umbral crítico convencional $\text{VIF} > 5$, determine cuál(es) predictor(es) ameritan intervención.
		\item Proponga qué remedio (Ridge, Lasso, o eliminación de variables) aplicaría para el predictor más problemático, y justifique su elección.
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:269eee0}
	\textbf{Deducción de la fórmula computacional de la Distancia de Cook a partir de su definición matricial:} partiendo de la definición general
	\begin{align}
		D_i = \frac{(\hat{\boldsymbol\beta}-\hat{\boldsymbol\beta}_{(-i)})^T (\mathbf{X}^T\mathbf{X}) (\hat{\boldsymbol\beta}-\hat{\boldsymbol\beta}_{(-i)})}{(p+1)\hat\sigma^2},
	\end{align}
	y usando el Teorema de Allen-PRESS (que expresa $\hat{\boldsymbol\beta}-\hat{\boldsymbol\beta}_{(-i)}$ en términos del residuo ordinario $e_i$, el apalancamiento $h_{ii}$, y la fila $\mathbf{x}_i$ de la matriz de diseño), deduzca la fórmula computacional simplificada
	\begin{align}
		D_i = \frac{r_i^2}{p+1}\cdot\frac{h_{ii}}{1-h_{ii}},
	\end{align}
	donde $r_i$ es el residuo estudentizado internamente. Demuestre por qué observaciones con apalancamiento $h_{ii}\to 1$ producen distancias de Cook explosivas independientemente de la magnitud del residuo ordinario $e_i$.
\end{problema}

\begin{problema}
	\label{prob:dc2ca7b}
	En un modelo con $n = 50$ observaciones y $p = 4$ predictores (5 parámetros incluyendo el intercepto), se identifican tres observaciones sospechosas con los siguientes valores de apalancamiento ($h_{ii}$) y residuo estudentizado internamente ($r_i$):
	\begin{align}
		\text{Obs. A: } h_{ii}=0.15,\ r_i=1.2; \qquad \text{Obs. B: } h_{ii}=0.45,\ r_i=0.8; \qquad \text{Obs. C: } h_{ii}=0.08,\ r_i=3.5.
	\end{align}
	\begin{enumerate}
		\item Calcule la Distancia de Cook para cada observación usando $D_i = \dfrac{r_i^2}{p+1}\cdot\dfrac{h_{ii}}{1-h_{ii}}$.
		\item Usando el umbral convencional $4/n$, determine cuáles observaciones son influyentes.
		\item Evalúe el mecanismo distinto por el cual las Observaciones B y C resultan influyentes (alto apalancamiento vs. residuo grande), a pesar de que la Observación A no lo es pese a tener valores moderados en ambas cantidades.
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:f0b580a}
	Diseñe un problema original de diagnóstico de heterocedasticidad (distinto del caso de publicidad y ventas ya visto): un modelo de consumo energético industrial ($Y$) en función del tamaño de la planta y otras $p=4$ variables de proceso, ajustado sobre $n=80$ observaciones. Al estimar el modelo auxiliar de Breusch-Pagan (regresión de los residuos al cuadrado sobre los predictores originales), se obtiene $R_{\text{aux}}^2 = 0.22$. Calcule el estadístico $LM = n\cdot R_{\text{aux}}^2$, compárelo contra el cuantil crítico $\chi^2_{0.05,4} = 9.488$, y decida si existe evidencia de heterocedasticidad al nivel $\alpha=0.05$.
\end{problema}

\subsection*{Soluciones detalladas}

% Recordar
\begin{solproblema}[prob:c7c2f12]
	\textbf{Numerador:} sumando las 9 diferencias sucesivas al cuadrado, $\sum_{i=2}^{10}(e_i-e_{i-1})^2 = 14.79$.

	\textbf{Denominador:} $\sum_{i=1}^{10} e_i^2 = 2.1^2+1.8^2+(-0.5)^2+\cdots+1.0^2 = 25.68$.

	\textbf{Estadístico:} $DW = 14.79/25.68 \approx 0.576$.

	Como $DW\approx0.576\ll2$, existe evidencia fuerte de \textbf{autocorrelación positiva} en los residuos: observaciones consecutivas tienden a tener el mismo signo, violando el supuesto de independencia.
\end{solproblema}

% Comprender
\begin{solproblema}[prob:90810fe]
	\begin{enumerate}
		\item La amplitud de los residuos crece sistemáticamente con el valor ajustado ($\pm2\to\pm8\to\pm20$), generando la clásica \textbf{forma de embudo} en la gráfica de residuos vs. ajustados. Esto viola el supuesto de \textbf{homocedasticidad}: la varianza del error no es constante, sino que crece con el nivel de la respuesta.
		\item La heterocedasticidad con varianza aproximadamente proporcional a la media (o a su cuadrado) se corrige típicamente aplicando $\log(Y)$ a la variable respuesta. Esta transformación comprime las diferencias absolutas proporcionalmente más en la región de valores grandes que en la de valores pequeños, estabilizando la dispersión relativa de los residuos.
	\end{enumerate}
\end{solproblema}

% Aplicar
\begin{solproblema}[prob:46ca545]
	\begin{enumerate}
		\item $\text{VIF}_1 = 1/(1-0.85) \approx 6.667$, $\text{VIF}_2 = 1/(1-0.55) \approx 2.222$, $\text{VIF}_3 = 1/(1-0.30) \approx 1.429$.
		\item Solo el predictor 1 supera el umbral ($\text{VIF}_1\approx6.667>5$), indicando multicolinealidad severa: su varianza está inflada en un factor de $\approx6.67$ respecto a la ortogonalidad perfecta. Los predictores 2 y 3 son tolerables.
		\item Como no se desea necesariamente eliminar el predictor 1 (podría tener relevancia teórica), la \textbf{regresión Ridge} es la opción más adecuada: penaliza la norma $L_2$ de todos los coeficientes de forma continua, estabilizando $(\mathbf{X}^T\mathbf{X}+\lambda\mathbf{I})^{-1}$ sin descartar variables.
	\end{enumerate}
\end{solproblema}

% Analizar
\begin{solproblema}[prob:269eee0]
	Por el Teorema de Allen-PRESS, el cambio en el vector de coeficientes al eliminar la $i$-ésima observación es:
	\begin{align}
		\hat{\boldsymbol\beta}-\hat{\boldsymbol\beta}_{(-i)} = \frac{(\mathbf{X}^T\mathbf{X})^{-1}\mathbf{x}_i \, e_i}{1-h_{ii}}.
	\end{align}
	Sustituyendo en la forma cuadrática de $D_i$:
	\begin{align}
		D_i = \frac{1}{(p+1)\hat\sigma^2}\left[\frac{(\mathbf{X}^T\mathbf{X})^{-1}\mathbf{x}_i \, e_i}{1-h_{ii}}\right]^T (\mathbf{X}^T\mathbf{X}) \left[\frac{(\mathbf{X}^T\mathbf{X})^{-1}\mathbf{x}_i \, e_i}{1-h_{ii}}\right] = \frac{e_i^2}{(p+1)\hat\sigma^2}\cdot\frac{h_{ii}}{(1-h_{ii})^2},
	\end{align}
	usando $\mathbf{x}_i^T(\mathbf{X}^T\mathbf{X})^{-1}\mathbf{x}_i = h_{ii}$. Con el residuo estudentizado $r_i = e_i/(\hat\sigma\sqrt{1-h_{ii}})$, de modo que $e_i^2/\hat\sigma^2 = r_i^2(1-h_{ii})$:
	\begin{align}
		D_i = \frac{r_i^2(1-h_{ii})}{p+1}\cdot\frac{h_{ii}}{(1-h_{ii})^2} = \frac{r_i^2}{p+1}\cdot\frac{h_{ii}}{1-h_{ii}}.
	\end{align}
	El factor $h_{ii}/(1-h_{ii})$ diverge a infinito cuando $h_{ii}\to1$, sin importar $r_i$: una observación con apalancamiento cercano a $1$ ocupa una posición tan extrema que la recta ajustada casi pasa por ella, ocultando (vía el mismo término $(1-h_{ii})\to0$ en el denominador del residuo estudentizado) la verdadera magnitud del desajuste que se revelaría al eliminarla.
\end{solproblema}

% Evaluar
\begin{solproblema}[prob:dc2ca7b]
	Con $p+1=5$:
	\begin{align}
		D_A = \frac{1.44}{5}(0.1765) \approx 0.0508, \qquad D_B = \frac{0.64}{5}(0.8182) \approx 0.1047, \qquad D_C = \frac{12.25}{5}(0.0870) \approx 0.2130.
	\end{align}
	Con umbral $4/50=0.08$: $D_A<0.08$ (no influyente); $D_B, D_C>0.08$ (influyentes).

	La Observación B es influyente por su \textbf{alto apalancamiento} ($h_{ii}=0.45$), a pesar de un residuo moderado: su posición inusual en el espacio de predictores le da gran capacidad de "halar" la recta hacia sí. La Observación C es influyente por tener un \textbf{residuo grande} ($r_i=3.5$, posible atípico) pese a un apalancamiento bajo: su posición en $X$ es típica, pero su valor de $Y$ se aparta enormemente del patrón general. La Observación A no es influyente porque no presenta ninguno de los dos mecanismos en grado severo.
\end{solproblema}

% Crear
\begin{solproblema}[prob:f0b580a]
	\begin{align}
		LM = n \cdot R_{\text{aux}}^2 = 80 \times 0.22 = 17.6.
	\end{align}
	Como $LM = 17.6 > \chi^2_{0.05,4} = 9.488$, \textbf{se rechaza $H_0$}: existe evidencia estadística significativa de \textbf{heterocedasticidad} en los errores del modelo de consumo energético al nivel del 5\%. Los estimadores de mínimos cuadrados siguen siendo insesgados, pero dejan de ser eficientes, y los errores estándar reportados podrían ser incorrectos; se recomienda usar errores estándar robustos (Huber-White) o transformar la variable respuesta.
\end{solproblema}
