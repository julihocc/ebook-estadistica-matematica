\section{Problemas resueltos de regresión lineal múltiple}

\subsection{Enunciados de los problemas}

\subsubsection*{1. Nivel Fundamental (Sencillos / Directos)}

\begin{problema}
	\label{prob:reg_mult_combustible}
	Se estudia el rendimiento medio de combustible ($Y$, en km/l) de automóviles urbanos en función de dos variables predictoras continuas: el peso bruto del vehículo ($X_1$, en toneladas) y la cilindrada de su motor ($X_2$, en litros de capacidad). Tras recopilar una muestra aleatoria de $n = 30$ automóviles en condiciones de prueba estandarizadas, se ajusta el siguiente modelo de regresión lineal múltiple por mínimos cuadrados ordinarios:
	\begin{align}
		\hat{Y} = 25.0 - 3.0X_1 - 1.5X_2, \quad R^2 = 0.780, \quad R_{\text{adj}}^2 = 0.763.
	\end{align}
	\begin{enumerate}
		\item Interprete el significado físico y geométrico de los coeficientes parciales de regresión $\hat{\beta}_1 = -3.0$ y $\hat{\beta}_2 = -1.5$, especificando explícitamente la condición \emph{ceteris paribus}.
		\item Calcule la predicción puntual del rendimiento esperable para un sedán compacto con peso de $1.5$ toneladas y un motor de $2.0$ litros.
		\item Si la varianza muestral total de los rendimientos observados en los 30 vehículos fue $S_Y^2 = 12.0$, calcule la Suma de Cuadrados de la Regresión ($\text{SCR}$) y la Suma de Cuadrados Residual ($\text{SCE}$).
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:reg_mult_pruebas_t}
	En el modelo del Problema \ref{prob:reg_mult_combustible} sobre el rendimiento vehicular con $n = 30$ observaciones ($p = 2$ regresores y 1 intercepto), el software analítico reporta que los errores estándar estimados para cada parámetro de regresión son:
	\begin{align}
		SE(\hat{\beta}_0) = 2.50, \quad SE(\hat{\beta}_1) = 0.80, \quad SE(\hat{\beta}_2) = 0.60.
	\end{align}
	\begin{enumerate}
		\item Formule formalmente el juego de hipótesis nula y alternativa para someter a prueba docimásica individual al peso del vehículo ($X_1$) y a la cilindrada ($X_2$).
		\item Calcule el estadístico de prueba $t$ de Student para cada coeficiente parcial ($t_j = \dfrac{\hat{\beta}_j}{SE(\hat{\beta}_j)}$).
		\item Al nivel de significancia bilateral $\alpha = 0.05$ (con cuantil crítico $t_{0.025, \, 27} = 2.052$), determine si ambos predictores contribuyen de forma individual y estadísticamente probada al modelo.
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:reg_mult_r2_adj_basico}
	Una firma de consultoría en marketing analiza el impacto de la diversificación de canales publicitarios sobre las ventas mensuales de corporaciones ($Y$, en miles de dólares). En una muestra histórica de $n = 40$ trimestres evaluados con un modelo de tres variables independientes ($p = 3$: gasto en TV $X_1$, gasto en radio $X_2$ y número de ejecutivos de venta $X_3$), la varianza explicada arrojó un coeficiente nominal de determinación de $R^2 = 0.780$.
	\begin{enumerate}
		\item Calcule algebraicamente el Coeficiente de Determinación Ajustado ($R_{\text{adj}}^2$) mediante la fórmula de corrección por grados de libertad:
		\begin{align}
			R_{\text{adj}}^2 = 1 - \left[ \frac{n - 1}{n - p - 1} (1 - R^2) \right].
		\end{align}
		\item Explique por qué en análisis multivariable el indicador $R_{\text{adj}}^2$ es estrictamente menor al $R^2$ nominal ($R_{\text{adj}}^2 < R^2$) siempre que se incorporan dos o más predictores ($p \geq 2$).
		\item Si un analista principiante propone añadir al modelo un cuarto predictor $X_4$ que resulta ser el código postal numérico de la sede del corporativo (con una correlación marginal nula respecto a las ventas, de modo que el $R^2$ apenas sube de $0.780$ a $0.781$), determine cuantitativamente qué le ocurriría al nuevo $R_{\text{adj}}^2$ del modelo con $p = 4$.
	\end{enumerate}
\end{problema}

\subsubsection*{2. Nivel Operativo (Intermedios / De Desarrollo)}

\begin{problema}
	\label{prob:reg_mult_ventas_anidados}
	Para optimizar las proyecciones de ventas trimestrales ($Y$, en miles de dólares) del Problema \ref{prob:reg_mult_r2_adj_basico} ($n = 40$), la dirección de la empresa contrasta dos especificaciones de modelos de regresión anidados:
	\begin{itemize}
		\item \textbf{Modelo 1 (Restringido, $p_1 = 2$):} $\hat{Y} = 50.0 + 2.5X_1 + 1.8X_2$, con $R_1^2 = 0.750$.
		\item \textbf{Modelo 2 (Completo, $p_2 = 3$):} $\hat{Y} = 45.0 + 2.3X_1 + 1.6X_2 + 0.80X_3$, con $R_2^2 = 0.780$.
	\end{itemize}
	\begin{enumerate}
		\item Calcule los respectivos $R_{\text{adj}}^2$ del Modelo 1 y del Modelo 2.
		\item Realice una \textbf{Prueba $F$ Parcial o Incremental} ($H_0: \beta_3 = 0$) al nivel $\alpha = 0.05$ para evaluar con rigor formal si la adición del número de vendedores ($X_3$) mejora significativamente el ajuste del modelo, calculando el estadístico:
		\begin{align}
			F_{\text{parcial}} = \frac{(R_2^2 - R_1^2) / (p_2 - p_1)}{(1 - R_2^2) / (n - p_2 - 1)},
		\end{align}
		y comparándolo contra el cuantil crítico tabulado de Fisher $F_{0.05, \, 1, \, 36} = 4.113$.
		\item Argumente integralmente qué modelo es técnica y gerencialmente recomendable.
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:reg_mult_vif_viviendas}
	Se calibra un modelo de regresión múltiple para predecir el valor de tasación de viviendas residenciales ($Y$, en miles de dólares) en función de $p = 4$ características estructurales y de localización: área construida ($X_1$, en $\text{m}^2$), número de habitaciones ($X_2$), antigüedad construida ($X_3$, en años) y distancia radial al centro urbano ($X_4$, en km). En una muestra de $n = 50$ propiedades se obtuvo:
	\begin{align}
		\hat{Y} = 50.0 + 0.80X_1 + 15.0X_2 - 2.0X_3 - 5.0X_4,
	\end{align}
	y los Factores de Inflación de Varianza ($\text{VIF}_j = \dfrac{1}{1 - R_j^2}$, donde $R_j^2$ es la correlación múltiple de regredir la variable $X_j$ contra los otros $p-1$ predictores) son:
	\begin{align}
		\text{VIF}(X_1) = 4.2, \quad \text{VIF}(X_2) = 3.8, \quad \text{VIF}(X_3) = 1.5, \quad \text{VIF}(X_4) = 1.2.
	\end{align}
	\begin{enumerate}
		\item Interprete en términos económicos el significado de los coeficientes de área ($\hat{\beta}_1 = 0.80$) y distancia ($X_4$ con $\hat{\beta}_4 = -5.0$).
		\item Evalúe los diagnósticos de colinealidad mediante las magnitudes de los $\text{VIF}_j$. ¿Existe un problema severo de multicolinealidad que anule el modelo o es un grado moderado admisible?
		\item Calcule el valor proyectado de tasación de una residencia con $120$ $\text{m}^2$, $3$ habitaciones, $10$ años de antigüedad y situada exactamente a $8$ km del centro comercial de la ciudad.
		\item ¿Qué estrategia de modelación (por ejemplo, reducción de dimensionalidad, estandarización o selección de variables) aplicaría si el $\text{VIF}(X_1)$ superara el umbral crítico de 10?
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:reg_mult_dummy_salarios}
	Un científico de datos en un departamento de capital humano modela la remuneración salarial anual ($Y$, en miles de dólares) en función de los años de experiencia profesional ($X_1$) y del nivel educativo. En lugar de cometer el error de codificar el nivel educativo como una variable ordinal falsa ($1, 2, 3$), crea un sistema de \textbf{variables indicadoras o mudas (\emph{dummy variables})}, utilizando a los profesionales con \emph{Bachillerato / Educación Media} como el grupo o nivel de referencia base ($D_1 = D_2 = 0$). Se definen:
	\begin{align}
		D_1 = \begin{cases} 1 & \text{si tiene Licenciatura / Grado Universitario} \\ 0 & \text{en otro caso} \end{cases}, \qquad D_2 = \begin{cases} 1 & \text{si tiene Maestría / Posgrado Avanzado} \\ 0 & \text{en otro caso} \end{cases}.
	\end{align}
	Con una muestra aleatoria de $n = 100$ empleados, se ajusta el modelo lineal paramétrico:
	\begin{align}
		\hat{Y} = 22.0 + 2.5X_1 + 12.0D_1 + 25.0D_2, \quad R^2 = 0.760.
	\end{align}
	\begin{enumerate}
		\item Interprete el valor numérico del intercepto $\hat{\beta}_0 = 22.0$ y de los coeficientes categóricos $\hat{\beta}_2 = 12.0$ y $\hat{\beta}_3 = 25.0$ sobre los salarios de la empresa.
		\item Escriba las rectas de regresión de salario en función de la experiencia ($X_1$) separadas explícitamente para cada uno de los tres subgrupos educativos (Bachillerato, Licenciatura y Maestría).
		\item Prediga el salario esperado para un gerente con $8$ años de experiencia que posee un título de Maestría.
		\item Explique con rigor en qué consiste la \textbf{Trampa de la Variable Muda (\emph{Dummy Variable Trap})} en álgebra matricial lineal y por qué habría sido fatal si el analista hubiera incluido una tercera variable muda $D_0 = 1$ para el grupo de Bachillerato junto con el intercepto $\beta_0$.
	\end{enumerate}
\end{problema}

\subsubsection*{3. Nivel Analítico (Avanzados / De Conexión)}

\begin{problema}
	\label{prob:reg_mult_matricial_exacto}
	En un estudio econométrico con $n = 5$ observaciones temporales sobre la rentabilidad porcentual del capital ($Y$) en función de dos índices predictivos ($X_1$: índice de liquidez, $X_2$: apalancamiento financiero), se especifica el modelo en notación matricial general de mínimos cuadrados ordinarios:
	\begin{align}
		\mathbf{Y} = \mathbf{X}\boldsymbol{\beta} + \boldsymbol{\varepsilon}.
	\end{align}
	Tras el procesamiento de los datos brutos, se calculan con exactitud las matrices de productos cruzados $\mathbf{X}^T \mathbf{X}$ y el vector proyectivo $\mathbf{X}^T \mathbf{Y}$:
	\begin{align}
		\mathbf{X}^T \mathbf{X} = \begin{pmatrix} 5 & 10 & 20 \\ 10 & 30 & 40 \\ 20 & 40 & 90 \end{pmatrix}, \qquad \mathbf{X}^T \mathbf{Y} = \begin{pmatrix} 50 \\ 120 \\ 230 \end{pmatrix}.
	\end{align}
	(La primera columna de $\mathbf{X}$ es el vector de unos $\mathbf{1}_5$ correspondiente al intercepto $\beta_0$).
	\begin{enumerate}
		\item Sabiendo que la inversa de la matriz normal simétrica es:
		\begin{align}
			(\mathbf{X}^T \mathbf{X})^{-1} = \frac{1}{50} \begin{pmatrix} 110 & -10 & -20 \\ -10 & 5 & 0 \\ -20 & 0 & 5 \end{pmatrix} = \begin{pmatrix} 2.2 & -0.2 & -0.4 \\ -0.2 & 0.1 & 0.0 \\ -0.4 & 0.0 & 0.1 \end{pmatrix},
		\end{align}
		calcule el vector solución exacto de estimadores de mínimos cuadrados $\hat{\boldsymbol{\beta}} = (\mathbf{X}^T \mathbf{X})^{-1} \mathbf{X}^T \mathbf{Y} = (\hat{\beta}_0, \hat{\beta}_1, \hat{\beta}_2)^T$.
		\item Escriba la ecuación escalar del modelo predictivo resultante.
		\item Defina formalmente la \textbf{Matriz Proyectora o Matriz Sombrero (\emph{Hat Matrix})} $\mathbf{H} = \mathbf{X}(\mathbf{X}^T \mathbf{X})^{-1} \mathbf{X}^T$ y demuestre algebraicamente sus dos propiedades axiomáticas: simetría ($\mathbf{H}^T = \mathbf{H}$) e idempotencia ($\mathbf{H}^2 = \mathbf{H}$). Explique cómo el operador $\mathbf{H}$ transforma el vector de respuestas brutas $\mathbf{Y}$ en el vector de valores predichos $\hat{\mathbf{Y}}$.
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:reg_mult_varianzas_ic_global}
	Para el modelo econométrico del Problema \ref{prob:reg_mult_matricial_exacto} ($n = 5, p = 2$), suponga que una muestra extendida de $n = 25$ empresas operando con la misma estructura simétrica normalizada de covarianzas $(\mathbf{X}^T \mathbf{X})^{-1}$ arrojó una Suma de Cuadrados Residual $\text{SCE} = 44.0$ y un vector de coeficientes estimado idéntico $\hat{\boldsymbol{\beta}} = (1.2, \; 2.6, \; 1.8)^T$.
	\begin{enumerate}
		\item Calcule la varianza residual insesgada del sistema multivariable $\hat{\sigma}^2 = \text{CME} = \dfrac{\text{SCE}}{n - p - 1}$.
		\item Calcule la \textbf{Matriz de Varianzas y Covarianzas Estimada de los Parámetros} $\widehat{\text{Cov}}(\hat{\boldsymbol{\beta}}) = \hat{\sigma}^2 (\mathbf{X}^T \mathbf{X})^{-1}$ y extraiga los errores estándar $SE(\hat{\beta}_0)$, $SE(\hat{\beta}_1)$ y $SE(\hat{\beta}_2)$.
		\item Construya el intervalo de confianza del $95\%$ para el parámetro real del índice de liquidez $\beta_1$ ($t_{0.025, \, 22} = 2.074$).
		\item Si la Suma de Cuadrados Total de rentabilidades en esa muestra fue $\text{SCT} = 264.0$, efectúe una \textbf{Prueba $F$ Global de Significancia de la Regresión} ($H_0: \beta_1 = \beta_2 = 0$ vs $H_a$: al menos un $\beta_j \neq 0$) con $\alpha = 0.05$ ($F_{0.05, \, 2, \, 22} = 3.443$), calculando la tabla ANOVA del modelo múltiple.
	\end{enumerate}
\end{problema}

\subsubsection*{4. Nivel Desafiante (Expertos / Deducción Profunda)}

\begin{problema}
	\label{prob:reg_mult_deduccion_matricial_mco}
	\textbf{Deducción analítica de MCO vectorial-matricial e insesgo poblacional en el modelo general de $p$ regresores:}
	Considere el modelo de regresión lineal múltiple en el espacio vectorial $\mathbb{R}^n$:
	\begin{align}
		\mathbf{Y} = \mathbf{X}\boldsymbol{\beta} + \boldsymbol{\varepsilon},
	\end{align}
	donde $\mathbf{Y}$ es el vector de respuestas de dimensión $n \times 1$, $\mathbf{X}$ es la matriz de diseño no aleatoria de dimensión $n \times (p+1)$ con rango completo columna ($\text{rango}(\mathbf{X}) = p+1 \leq n$), $\boldsymbol{\beta}$ es el vector de parámetros poblacionales de dimensión $(p+1) \times 1$, y $\boldsymbol{\varepsilon}$ es el vector de errores con $E[\boldsymbol{\varepsilon}] = \mathbf{0}$ y $\text{Var}(\boldsymbol{\varepsilon}) = E[\boldsymbol{\varepsilon}\boldsymbol{\varepsilon}^T] = \sigma^2 \mathbf{I}_n$.
	\begin{enumerate}
		\item Partiendo de la forma cuadrática escalar del error de suma de cuadrados $S(\boldsymbol{\beta}) = \boldsymbol{\varepsilon}^T \boldsymbol{\varepsilon} = (\mathbf{Y} - \mathbf{X}\boldsymbol{\beta})^T (\mathbf{Y} - \mathbf{X}\boldsymbol{\beta})$, aplique las reglas de derivación vectorial matricial sobre formas cuadráticas respecto al vector $\boldsymbol{\beta}$ para deducir de forma rigurosa las ecuaciones normales $\mathbf{X}^T \mathbf{X} \hat{\boldsymbol{\beta}} = \mathbf{X}^T \mathbf{Y}$ y justifique la unicidad de la solución analítica $\hat{\boldsymbol{\beta}} = (\mathbf{X}^T \mathbf{X})^{-1} \mathbf{X}^T \mathbf{Y}$.
		\item Demuestre que el vector de residuos empíricos $\hat{\boldsymbol{\varepsilon}} = \mathbf{Y} - \hat{\mathbf{Y}} = (\mathbf{I}_n - \mathbf{H})\mathbf{Y}$ es ortogonal al espacio columna completo generado por la matriz de diseño $\mathbf{X}$ ($\mathbf{X}^T \hat{\boldsymbol{\varepsilon}} = \mathbf{0}$) y pruebe que la suma de productos cruzados entre las predicciones y los residuos se anula idénticamente ($\hat{\mathbf{Y}}^T \hat{\boldsymbol{\varepsilon}} = 0$).
		\item Demuestre aplicando el operador esperanza $E[\cdot]$ que el vector $\hat{\boldsymbol{\beta}}$ es un estimador multivariable \textbf{insesgado} ($E[\hat{\boldsymbol{\beta}}] = \boldsymbol{\beta}$) y que su matriz teórica de covarianzas equivale exactamente a $\text{Var}(\hat{\boldsymbol{\beta}}) = \sigma^2 (\mathbf{X}^T \mathbf{X})^{-1}$.
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:reg_mult_gauss_markov_meli}
	\textbf{Demostración del Teorema de Gauss-Markov (Teorema MELI/BLUE - \emph{Best Linear Unbiased Estimator}) y penalización del $R_{\text{adj}}^2$:}
	En el marco del modelo lineal matricial $\mathbf{Y} = \mathbf{X}\boldsymbol{\beta} + \boldsymbol{\varepsilon}$ con los supuestos de Gauss-Markov verificados ($E[\boldsymbol{\varepsilon}] = \mathbf{0}$, $\text{Var}(\boldsymbol{\varepsilon}) = \sigma^2 \mathbf{I}_n$):
	\begin{enumerate}
		\item Sea $\tilde{\boldsymbol{\beta}} = \mathbf{C}\mathbf{Y}$ un estimador lineal competidor arbitrario que se exige que sea insesgado para el vector $\boldsymbol{\beta}$ ($E[\tilde{\boldsymbol{\beta}}] = \boldsymbol{\beta}$ para todo $\boldsymbol{\beta}$). Expresemos la matriz de ponderación como una perturbación de MCO: $\mathbf{C} = (\mathbf{X}^T \mathbf{X})^{-1}\mathbf{X}^T + \mathbf{D}$. Demuestre paso a paso que para que $\tilde{\boldsymbol{\beta}}$ sea insesgado, la matriz perturbadora debe satisfacer obligatoriamente la ortogonalidad $\mathbf{D}\mathbf{X} = \mathbf{0}$.
		\item Con la condición de insesgo comprobada, demuestre algebraicamente que la matriz de varianzas-covarianzas de $\tilde{\boldsymbol{\beta}}$ se descompone aditivamente de forma exacta en:
		\begin{align}
			\text{Var}(\tilde{\boldsymbol{\beta}}) = \text{Var}(\hat{\boldsymbol{\beta}}) + \sigma^2 \mathbf{D}\mathbf{D}^T.
		\end{align}
		Concluya a partir de la semidefinición positiva de la matriz cuadrática $\mathbf{D}\mathbf{D}^T \succeq \mathbf{0}$ que el estimador MCO posee varianza mínima en el sentido matricial de Loewner ($\text{Var}(\tilde{\boldsymbol{\beta}}) - \text{Var}(\hat{\boldsymbol{\beta}}) \succeq \mathbf{0}$), probando el Teorema de Gauss-Markov.
		\item Demuestre que si en un modelo que contiene $p$ variables explicativas reales se decide incorporar un predictor adicional ($X_{p+1}$) cuya pendiente verdadera en la población es nula ($\beta_{p+1} = 0$), el coeficiente nominal muestral $R^2$ cumple la relación de dominancia empírica $R_{p+1}^2 \geq R_p^2$, mientras que el Coeficiente de Determinación Ajustado $R_{\text{adj}, p+1}^2$ es estrictamente inferior al previo $R_{\text{adj}, p}^2$ a menos que el estadístico $t$ de Student de ese nuevo regresor cumpla con el umbral de magnitud $|t_{p+1}| > 1$.
	\end{enumerate}
\end{problema}


\subsection{Soluciones detalladas}

\subsubsection*{1. Nivel Fundamental (Sencillos / Directos)}

\begin{solucion}
	\begin{enumerate}
		\item \textbf{Interpretación física de los coeficientes parciales:}
		En regresión múltiple, cada parámetro representa la tasa de cambio del valor esperado de $Y$ respecto a una variable, \emph{manteniendo fijas todas las demás explicativas} (condición \emph{ceteris paribus}):
		\begin{itemize}
			\item $\hat{\beta}_1 = -3.0$: Para automóviles que poseen la \textbf{misma cilindrada de motor} ($X_2$ constante), por cada tonelada adicional en el peso bruto del vehículo ($X_1$), el rendimiento de combustible se reduce en promedio en $3.0$ km/l.
			\item $\hat{\beta}_2 = -1.5$: Para automóviles con un \textbf{peso vehicular idéntico} ($X_1$ constante), por cada litro adicional de capacidad o cilindrada del motor ($X_2$), el rendimiento promedio de combustible disminuye en $1.5$ km/l.
		\end{itemize}
		
		\item Cálculo del rendimiento proyectado para un automóvil con $X_1 = 1.5$ toneladas y $X_2 = 2.0$ litros:
		\begin{align}
			\hat{Y} = 25.0 - 3.0(1.5) - 1.5(2.0) = 25.0 - 4.5 - 3.0 = 17.5 \text{ km/l}.
		\end{align}
		Se estima un rendimiento en condiciones urbanas de $17.5$ kilómetros por litro.
		
		\item Con $n = 30$ observaciones totales y una varianza muestral observada $S_Y^2 = 12.0$, la Suma de Cuadrados Total del rendimiento resulta:
		\begin{align}
			\text{SCT} = (n - 1) S_Y^2 = (30 - 1)(12.0) = 29 \times 12.0 = 348.0 \text{ (km/l)}^2.
		\end{align}
		Con el coeficiente de determinación nominal $R^2 = 0.780$, obtenemos la Suma de Cuadrados de la Regresión ($\text{SCR}$):
		\begin{align}
			\text{SCR} = R^2 \times \text{SCT} = (0.780)(348.0) = 271.44 \text{ (km/l)}^2.
		\end{align}
		Por el principio algebraico de partición ortogonal del error en MCO, la variabilidad no explicada es:
		\begin{align}
			\text{SCE} = \text{SCT} - \text{SCR} = 348.0 - 271.44 = 76.56 \text{ (km/l)}^2.
		\end{align}
	\end{enumerate}
\end{solucion}

\begin{solucion}
	\begin{enumerate}
		\item \textbf{Formulación formal de hipótesis docimásicas parciales:}
		Para evaluar si cada variable explicativa aporta información predictiva estadísticamente no redundante al modelo (dada la presencia de la otra):
		\begin{itemize}
			\item Para el peso del vehículo ($X_1$):
			\begin{align}
				H_{0,1}: \beta_1 = 0 \quad (\text{El peso no aporta significancia predictiva al rendimiento una vez controlada la cilindrada}). \\
				H_{a,1}: \beta_1 \neq 0 \quad (\text{El peso del vehículo influye significativamente en el rendimiento}).
			\end{align}
			\item Para la cilindrada del motor ($X_2$):
			\begin{align}
				H_{0,2}: \beta_2 = 0 \quad (\text{La cilindrada es irrelevante para predecir el consumo, controlado el peso del auto}). \\
				H_{a,2}: \beta_2 \neq 0 \quad (\text{La cilindrada tiene un impacto significativo y directo sobre el rendimiento}).
			\end{align}
		\end{itemize}
		
		\item Cálculo del estadístico $t$ de Student individual bajo las hipótesis nulas ($H_{0,j}: \beta_j = 0$):
		Los grados de libertad residuales del modelo con $n=30$ y $p=2$ predictores son $\nu_E = n - p - 1 = 30 - 2 - 1 = 27$.
		\begin{align}
			t_1 &= \frac{\hat{\beta}_1}{SE(\hat{\beta}_1)} = \frac{-3.0}{0.80} = -3.750, \\
			t_2 &= \frac{\hat{\beta}_2}{SE(\hat{\beta}_2)} = \frac{-1.5}{0.60} = -2.500.
		\end{align}
		
		\item Regla de decisión y conclusión:
		El cuantil crítico en la distribución $t$ de Student bilateral con $\nu_E = 27$ al $\alpha = 0.05$ arrojó el valor absoluto de corte $t_{0.025, \, 27} = 2.052$.
		\begin{itemize}
			\item Para el peso ($X_1$): Como $|t_1| = 3.750 > 2.052$ ($p$-valor $\approx 0.0008$), \textbf{se rechaza $H_{0,1}$}. El peso neto es un determinante probada y de alta robustez de la economía de combustible.
			\item Para la cilindrada ($X_2$): Como $|t_2| = 2.500 > 2.052$ ($p$-valor $\approx 0.0188$), \textbf{se rechaza $H_{0,2}$}. La capacidad volumétrica del motor contribuye significativamente a la predicción del consumo, más allá y por encima del peso físico del automóvil.
		\end{itemize}
	\end{enumerate}
\end{solucion}

\begin{solucion}
	\begin{enumerate}
		\item Cálculo riguroso del Coeficiente de Determinación Ajustado ($R_{\text{adj}}^2$):
		Para $n = 40$ trimestres, $p = 3$ predictores en el modelo y un coeficiente no ajustado de $R^2 = 0.780$:
		\begin{align}

		\item \textbf{Justificación teórica y penalización dimensional:} En cualquier ajuste de regresión por mínimos cuadrados ordinarios, la adición de una nueva variable predictora (incluso si se tratara de secuencias de números aleatorios puros sin ninguna relación geométrica con las ventas) invariablemente reduce la Suma de Cuadrados Residual o, en el peor de los casos, la mantiene idéntica ($\text{SCE}_{p+1} \leq \text{SCE}_p$). En consecuencia, el \textbf{$R^2$ nominal nunca puede disminuir} al agregar variables, lo cual crea un peligroso espejismo analítico que incita a sobreajustar modelos complejos con ruido (\emph{overfitting}).

		\begin{align}
			R_{\text{adj}}^2 = 1 - \frac{\text{CME}}{\text{CMT}} = 1 - \frac{\text{SCE} / (n - p - 1)}{\text{SCT} / (n - 1)}.
		\end{align}
		Cada vez que añadimos un predictor al modelo ($p \to p+1$), el término $(n - p - 1)$ en el denominador de la varianza del error \emph{disminuye exactamente en una unidad}, lo cual penaliza al indicador al hacer crecer el cociente $\text{CME}/\text{CMT}$. Para que el $R_{\text{adj}}^2$ neto logre subir, la variable recién ingresada debe absorber una cantidad de varianza verdadera lo suficientemente grande para superar y compensar la pérdida de ese grado de libertad en el error.

			\mathbf{D}_0 + \mathbf{D}_1 + \mathbf{D}_2 = \mathbf{1}_n = \mathbf{X}_0.

		\item \textbf{Definición y demostración algebraica de las propiedades de la Matriz Sombrero ($\mathbf{H}$):}
		La matriz $\mathbf{H} = \mathbf{X}(\mathbf{X}^T \mathbf{X})^{-1} \mathbf{X}^T$, de dimensión $n \times n$, se denomina \textbf{Matriz Proyectora o \emph{Hat Matrix}} porque actúa como el operador lineal que le ``coloca el sombrero'' (la estimación $\hat{\cdot}$) al vector de observaciones respuesta brutas $\mathbf{Y}$, proyectándolo ortogonalmente sobre el subespacio columna de la matriz $\mathbf{X}$:
		\begin{align}
			\hat{\mathbf{Y}} = \mathbf{X}\hat{\boldsymbol{\beta}} = \mathbf{X}(\mathbf{X}^T \mathbf{X})^{-1}\mathbf{X}^T \mathbf{Y} = \mathbf{H}\mathbf{Y}.
		\end{align}
		
		Demostración de sus dos propiedades estructurales fundamentales:
		\begin{itemize}
			\item \textbf{Simetría ($\mathbf{H}^T = \mathbf{H}$):} Puesto que la traspuesta de una multiplicación de matrices invierte el orden de los factores, y dado que la matriz normal de productos cruzados $\mathbf{X}^T \mathbf{X}$ (y por ende su matriz inversa $(\mathbf{X}^T \mathbf{X})^{-1}$) es simétrica por construcción:
			\begin{align}
				\mathbf{H}^T &= \left[ \mathbf{X}(\mathbf{X}^T \mathbf{X})^{-1} \mathbf{X}^T \right]^T = \left(\mathbf{X}^T\right)^T \left[ (\mathbf{X}^T \mathbf{X})^{-1} \right]^T \mathbf{X}^T \\
				&= \mathbf{X} (\mathbf{X}^T \mathbf{X})^{-1} \mathbf{X}^T = \mathbf{H}. \quad (\text{Q.E.D.})
			\end{align}
			\item \textbf{Idempotencia ($\mathbf{H}^2 = \mathbf{H}$):} Al elevar al cuadrado el operador proyectivo y asociar las matrices intermedias:
			\begin{align}
				\mathbf{H}^2 = \mathbf{H} \cdot \mathbf{H} &= \left[ \mathbf{X}(\mathbf{X}^T \mathbf{X})^{-1} \mathbf{X}^T \right] \left[ \mathbf{X}(\mathbf{X}^T \mathbf{X})^{-1} \mathbf{X}^T \right] \\
				&= \mathbf{X}(\mathbf{X}^T \mathbf{X})^{-1} \left[ \mathbf{X}^T \mathbf{X} \right] (\mathbf{X}^T \mathbf{X})^{-1} \mathbf{X}^T.
			\end{align}
			El producto de la matriz de productos cruzados por su propia inversa se colapsa en la matriz identidad de dimensión $(p+1) \times (p+1)$ ($\mathbf{X}^T \mathbf{X}(\mathbf{X}^T \mathbf{X})^{-1} = \mathbf{I}_{p+1}$):
			\begin{align}
				\mathbf{H}^2 = \mathbf{X} \cdot \mathbf{I}_{p+1} \cdot (\mathbf{X}^T \mathbf{X})^{-1} \mathbf{X}^T = \mathbf{X}(\mathbf{X}^T \mathbf{X})^{-1} \mathbf{X}^T = \mathbf{H}. \quad (\text{Q.E.D.})
			\end{align}
		\end{itemize}
		La idempotencia garantiza físicamente que proyectar dos veces consecutivas los datos sobre el plano de regresión no altera las estimaciones ($\mathbf{H}\hat{\mathbf{Y}} = \hat{\mathbf{Y}}$), pues los valores ajustados ya viven dentro del espacio explicado del modelo.
	\end{enumerate}
\end{solucion}

\subsubsection*{4. Nivel Desafiante (Expertos / Deducción Profunda)}

\begin{solucion}
	\begin{enumerate}
		\item \textbf{Deducción analítica de MCO matricial:}
		Sea la función objetivo de suma de cuadrados de los errores expresada mediante el producto interno vectorial euclidiano:
		\begin{align}
			S(\boldsymbol{\beta}) &= (\mathbf{Y} - \mathbf{X}\boldsymbol{\beta})^T (\mathbf{Y} - \mathbf{X}\boldsymbol{\beta}) \\
			&= (\mathbf{Y}^T - \boldsymbol{\beta}^T \mathbf{X}^T)(\mathbf{Y} - \mathbf{X}\boldsymbol{\beta}) = \mathbf{Y}^T \mathbf{Y} - \mathbf{Y}^T \mathbf{X}\boldsymbol{\beta} - \boldsymbol{\beta}^T \mathbf{X}^T \mathbf{Y} + \boldsymbol{\beta}^T \mathbf{X}^T \mathbf{X}\boldsymbol{\beta}.
		\end{align}
		Puesto que el término intermedio $\mathbf{Y}^T \mathbf{X}\boldsymbol{\beta}$ es un escalar ($1 \times 1$), coincide con su propia traspuesta matricial ($\mathbf{Y}^T \mathbf{X}\boldsymbol{\beta} = (\mathbf{Y}^T \mathbf{X}\boldsymbol{\beta})^T = \boldsymbol{\beta}^T \mathbf{X}^T \mathbf{Y}$). Podemos simplificar la función a:
		\begin{align}
			S(\boldsymbol{\beta}) = \mathbf{Y}^T \mathbf{Y} - 2\boldsymbol{\beta}^T \mathbf{X}^T \mathbf{Y} + \boldsymbol{\beta}^T \mathbf{X}^T \mathbf{X}\boldsymbol{\beta}.
		\end{align}
		Para hallar el mínimo vectorial, aplicamos el gradiente con respecto al vector de incógnitas $\boldsymbol{\beta}$, utilizando las identidades de cálculo matricial diferencial $\dfrac{\partial (\boldsymbol{\beta}^T \mathbf{a})}{\partial \boldsymbol{\beta}} = \mathbf{a}$ y $\dfrac{\partial (\boldsymbol{\beta}^T \mathbf{A} \boldsymbol{\beta})}{\partial \boldsymbol{\beta}} = 2\mathbf{A}\boldsymbol{\beta}$ para matrices simétricas:
		\begin{align}
			\frac{\partial S(\boldsymbol{\beta})}{\partial \boldsymbol{\beta}} = -2\mathbf{X}^T \mathbf{Y} + 2\mathbf{X}^T \mathbf{X}\boldsymbol{\beta} = \mathbf{0}.
		\end{align}
		Dividiendo ambos lados de la ecuación vectorial por la constante $+2$, obtenemos el sistema general de \textbf{Ecuaciones Normales Matriciales de Gauss}:
		\begin{align}
			\mathbf{X}^T \mathbf{X} \hat{\boldsymbol{\beta}} = \mathbf{X}^T \mathbf{Y}.
		\end{align}
		Por hipótesis, la matriz de diseño $\mathbf{X}$ tiene rango columna completo ($r(\mathbf{X}) = p+1$), lo cual garantiza por el álgebra lineal real que la matriz cuadrada de productos cruzados $\mathbf{X}^T \mathbf{X}$, de dimensión $(p+1) \times (p+1)$, es \textbf{simétrica definida positiva y no singular} ($\det(\mathbf{X}^T \mathbf{X}) > 0$). Por tanto, existe una inversa única $(\mathbf{X}^T \mathbf{X})^{-1}$. Pre-multiplicando el sistema normal por dicha inversa, aislamos con unicidad absoluta al vector de estimadores:
		\begin{align}
			(\mathbf{X}^T \mathbf{X})^{-1} \mathbf{X}^T \mathbf{X} \hat{\boldsymbol{\beta}} = (\mathbf{X}^T \mathbf{X})^{-1} \mathbf{X}^T \mathbf{Y} \implies \hat{\boldsymbol{\beta}} = (\mathbf{X}^T \mathbf{X})^{-1} \mathbf{X}^T \mathbf{Y}.
		\end{align}
		
		\item \textbf{Ortogonalidad matricial entre residuos empíricos y predictores ($\mathbf{X}^T \hat{\boldsymbol{\varepsilon}} = \mathbf{0}$):}
		El vector de residuos de MCO es $\hat{\boldsymbol{\varepsilon}} = \mathbf{Y} - \hat{\mathbf{Y}} = \mathbf{Y} - \mathbf{X}\hat{\boldsymbol{\beta}}$. Sustituyendo $\hat{\boldsymbol{\beta}} = (\mathbf{X}^T \mathbf{X})^{-1}\mathbf{X}^T \mathbf{Y}$ e incorporando la matriz sombrero $\mathbf{H}$:
		\begin{align}
			\hat{\boldsymbol{\varepsilon}} = \mathbf{Y} - \mathbf{H}\mathbf{Y} = (\mathbf{I}_n - \mathbf{H})\mathbf{Y}.
		\end{align}
		Para demostrar la ortogonalidad geométrica al espacio columna de $\mathbf{X}$, pre-multiplicamos el vector residual por la traspuesta del diseño:
		\begin{align}
			\mathbf{X}^T \hat{\boldsymbol{\varepsilon}} = \mathbf{X}^T (\mathbf{Y} - \mathbf{X}\hat{\boldsymbol{\beta}}) = \mathbf{X}^T \mathbf{Y} - \mathbf{X}^T \mathbf{X} \hat{\boldsymbol{\beta}}.
		\end{align}
		Invocando la identidad exacta de las Ecuaciones Normales ($\mathbf{X}^T \mathbf{X}\hat{\boldsymbol{\beta}} = \mathbf{X}^T \mathbf{Y}$), se demuestra ineludiblemente:
		\begin{align}
			\mathbf{X}^T \hat{\boldsymbol{\varepsilon}} = \mathbf{X}^T \mathbf{Y} - \mathbf{X}^T \mathbf{Y} = \mathbf{0}. \quad (\text{Ortogonalidad al subespacio spanned de } \mathbf{X}).
		\end{align}
		Como el vector de predicciones $\hat{\mathbf{Y}} = \mathbf{X}\hat{\boldsymbol{\beta}}$ pertenece al espacio columna de $\mathbf{X}$, el producto interno entre predicciones y residuos es idénticamente nulo:
		\begin{align}
			\hat{\mathbf{Y}}^T \hat{\boldsymbol{\varepsilon}} = (\mathbf{X}\hat{\boldsymbol{\beta}})^T \hat{\boldsymbol{\varepsilon}} = \hat{\boldsymbol{\beta}}^T (\mathbf{X}^T \hat{\boldsymbol{\varepsilon}}) = \hat{\boldsymbol{\beta}}^T [\mathbf{0}] = 0. \quad (\text{Ortogonalidad empírica completa}).
		\end{align}
		
		\item \textbf{Demostración de insesgo y deducción del tensor de varianzas matricial:}
		Expresamos a $\hat{\boldsymbol{\beta}}$ sustituyendo la respuesta poblacional real $\mathbf{Y} = \mathbf{X}\boldsymbol{\beta} + \boldsymbol{\varepsilon}$:
		\begin{align}
			\hat{\boldsymbol{\beta}} &= (\mathbf{X}^T \mathbf{X})^{-1} \mathbf{X}^T (\mathbf{X}\boldsymbol{\beta} + \boldsymbol{\varepsilon}) \\
			&= (\mathbf{X}^T \mathbf{X})^{-1} \mathbf{X}^T \mathbf{X} \boldsymbol{\beta} + (\mathbf{X}^T \mathbf{X})^{-1} \mathbf{X}^T \boldsymbol{\varepsilon} = \boldsymbol{\beta} + (\mathbf{X}^T \mathbf{X})^{-1} \mathbf{X}^T \boldsymbol{\varepsilon}.
		\end{align}
		Tomando la esperanza matemática matricial incondicional bajo la no aleatoriedad de $\mathbf{X}$:
		\begin{align}
			E[\hat{\boldsymbol{\beta}}] = E\left[ \boldsymbol{\beta} + (\mathbf{X}^T \mathbf{X})^{-1} \mathbf{X}^T \boldsymbol{\varepsilon} \right] = \boldsymbol{\beta} + (\mathbf{X}^T \mathbf{X})^{-1} \mathbf{X}^T E[\boldsymbol{\varepsilon}].
		\end{align}
		Por el supuesto central probabilístico de Gauss-Markov ($E[\boldsymbol{\varepsilon}] = \mathbf{0}$):
		\begin{align}
			E[\hat{\boldsymbol{\beta}}] = \boldsymbol{\beta} + \mathbf{0} = \boldsymbol{\beta}. \quad (\text{Estimador multivariable insesgado}).
		\end{align}
		Para derivar la matriz de covarianzas exacta $\text{Var}(\hat{\boldsymbol{\beta}}) = E[(\hat{\boldsymbol{\beta}} - \boldsymbol{\beta})(\hat{\boldsymbol{\beta}} - \boldsymbol{\beta})^T]$, utilizamos que la desviación respecto a la media es $\hat{\boldsymbol{\beta}} - \boldsymbol{\beta} = (\mathbf{X}^T \mathbf{X})^{-1} \mathbf{X}^T \boldsymbol{\varepsilon}$:
		\begin{align}
			\text{Var}(\hat{\boldsymbol{\beta}}) &= E\left[ \left( (\mathbf{X}^T \mathbf{X})^{-1} \mathbf{X}^T \boldsymbol{\varepsilon} \right) \left( (\mathbf{X}^T \mathbf{X})^{-1} \mathbf{X}^T \boldsymbol{\varepsilon} \right)^T \right] \\
			&= E\left[ (\mathbf{X}^T \mathbf{X})^{-1} \mathbf{X}^T \boldsymbol{\varepsilon} \boldsymbol{\varepsilon}^T \mathbf{X} (\mathbf{X}^T \mathbf{X})^{-1} \right].
		\end{align}
		Extrayendo las matrices deterministas fuera del operador de esperanza y reconociendo que por hipótesis de homoscedasticidad no correlacionada $E[\boldsymbol{\varepsilon}\boldsymbol{\varepsilon}^T] = \sigma^2 \mathbf{I}_n$:
		\begin{align}
			\text{Var}(\hat{\boldsymbol{\beta}}) &= (\mathbf{X}^T \mathbf{X})^{-1} \mathbf{X}^T E[\boldsymbol{\varepsilon}\boldsymbol{\varepsilon}^T] \mathbf{X} (\mathbf{X}^T \mathbf{X})^{-1} \\
			&= (\mathbf{X}^T \mathbf{X})^{-1} \mathbf{X}^T (\sigma^2 \mathbf{I}_n) \mathbf{X} (\mathbf{X}^T \mathbf{X})^{-1} = \sigma^2 (\mathbf{X}^T \mathbf{X})^{-1} \left[ \mathbf{X}^T \mathbf{X} \right] (\mathbf{X}^T \mathbf{X})^{-1}.
		\end{align}
		El producto en el centro de la cadena cancela al penúltimo término, colapsando el tensor de manera limpia y rotunda a:
		\begin{align}
			\text{Var}(\hat{\boldsymbol{\beta}}) = \sigma^2 (\mathbf{X}^T \mathbf{X})^{-1}. \quad (\text{Q.E.D.})
		\end{align}
	\end{enumerate}
\end{solucion}

\begin{solucion}
	\begin{enumerate}
		\item \textbf{Demostración de la condición de insesgo sobre la matriz perturbadora $\mathbf{D}$:}
		Sea el estimador lineal alternativo $\tilde{\boldsymbol{\beta}} = \mathbf{C}\mathbf{Y}$ donde $\mathbf{C} = (\mathbf{X}^T \mathbf{X})^{-1}\mathbf{X}^T + \mathbf{D}$.
		Sustituyendo el modelo poblacional $\mathbf{Y} = \mathbf{X}\boldsymbol{\beta} + \boldsymbol{\varepsilon}$:
		\begin{align}
			\tilde{\boldsymbol{\beta}} &= \left[ (\mathbf{X}^T \mathbf{X})^{-1}\mathbf{X}^T + \mathbf{D} \right] (\mathbf{X}\boldsymbol{\beta} + \boldsymbol{\varepsilon}) \\
			&= (\mathbf{X}^T \mathbf{X})^{-1}\mathbf{X}^T \mathbf{X}\boldsymbol{\beta} + (\mathbf{X}^T \mathbf{X})^{-1}\mathbf{X}^T \boldsymbol{\varepsilon} + \mathbf{D}\mathbf{X}\boldsymbol{\beta} + \mathbf{D}\boldsymbol{\varepsilon} \\
			&= \boldsymbol{\beta} + \mathbf{D}\mathbf{X}\boldsymbol{\beta} + \left[ (\mathbf{X}^T \mathbf{X})^{-1}\mathbf{X}^T + \mathbf{D} \right] \boldsymbol{\varepsilon}.
		\end{align}
		Tomando la esperanza y aplicando que $E[\boldsymbol{\varepsilon}] = \mathbf{0}$:
		\begin{align}
			E[\tilde{\boldsymbol{\beta}}] = \boldsymbol{\beta} + \mathbf{D}\mathbf{X}\boldsymbol{\beta} + \mathbf{0} = \left( \mathbf{I}_{p+1} + \mathbf{D}\mathbf{X} \right) \boldsymbol{\beta}.
		\end{align}
		Para que el estimador sea estrictamente insesgado de manera universal ($E[\tilde{\boldsymbol{\beta}}] = \boldsymbol{\beta}$ para cualquier vector concebible de parámetros verdaderos $\boldsymbol{\beta} \in \mathbb{R}^{p+1}$), el término que multiplica a $\boldsymbol{\beta}$ debe ser idéntico a la matriz identidad, lo que impone la condición obligatoria de ortogonalidad del diseño:
		\begin{align}
			\mathbf{I}_{p+1} + \mathbf{D}\mathbf{X} = \mathbf{I}_{p+1} \iff \mathbf{D}\mathbf{X} = \mathbf{0}.
		\end{align}
		
		\item \textbf{Demostración algebraica de optimalidad y varianza mínima (Teorema MELI/BLUE):}
		Con la condición $\mathbf{D}\mathbf{X} = \mathbf{0}$ garantizada, la desviación de $\tilde{\boldsymbol{\beta}}$ respecto a la media poblacional se reduce a:
		\begin{align}
			\tilde{\boldsymbol{\beta}} - \boldsymbol{\beta} = \left[ (\mathbf{X}^T \mathbf{X})^{-1}\mathbf{X}^T + \mathbf{D} \right] \boldsymbol{\varepsilon} = \mathbf{C}\boldsymbol{\varepsilon}.
		\end{align}
		La matriz teórica de varianzas de este estimador general es:
		\begin{align}
			\text{Var}(\tilde{\boldsymbol{\beta}}) &= E\left[ (\tilde{\boldsymbol{\beta}} - \boldsymbol{\beta})(\tilde{\boldsymbol{\beta}} - \boldsymbol{\beta})^T \right] = E\left[ \mathbf{C}\boldsymbol{\varepsilon} \boldsymbol{\varepsilon}^T \mathbf{C}^T \right] = \mathbf{C} E[\boldsymbol{\varepsilon}\boldsymbol{\varepsilon}^T] \mathbf{C}^T = \sigma^2 \mathbf{C}\mathbf{C}^T.
		\end{align}
		Sustituyendo $\mathbf{C} = (\mathbf{X}^T \mathbf{X})^{-1}\mathbf{X}^T + \mathbf{D}$ en el producto cuadrático:
		\begin{align}
			\text{Var}(\tilde{\boldsymbol{\beta}}) &= \sigma^2 \left[ (\mathbf{X}^T \mathbf{X})^{-1}\mathbf{X}^T + \mathbf{D} \right] \left[ \mathbf{X}(\mathbf{X}^T \mathbf{X})^{-1} + \mathbf{D}^T \right] \\
			&= \sigma^2 \left[ (\mathbf{X}^T \mathbf{X})^{-1}\mathbf{X}^T \mathbf{X}(\mathbf{X}^T \mathbf{X})^{-1} + (\mathbf{X}^T \mathbf{X})^{-1}\mathbf{X}^T \mathbf{D}^T + \mathbf{D}\mathbf{X}(\mathbf{X}^T \mathbf{X})^{-1} + \mathbf{D}\mathbf{D}^T \right].
		\end{align}
		Analicemos los términos cruzados: el tercer término contiene al producto $\mathbf{D}\mathbf{X}$, que por el inciso 1 demostramos que es la matriz nula ($\mathbf{D}\mathbf{X} = \mathbf{0}$). Su traspuesta en el segundo término también se anula en consecuencia ($\mathbf{X}^T \mathbf{D}^T = (\mathbf{D}\mathbf{X})^T = \mathbf{0}^T = \mathbf{0}$).
		Al cancelarse por completo las interferencias cruzadas, la matriz de varianza del estimador competidor se expresa en forma aditiva pura:
		\begin{align}
			\text{Var}(\tilde{\boldsymbol{\beta}}) = \sigma^2 (\mathbf{X}^T \mathbf{X})^{-1} + \sigma^2 \mathbf{D}\mathbf{D}^T = \text{Var}(\hat{\boldsymbol{\beta}}) + \sigma^2 \mathbf{D}\mathbf{D}^T.
		\end{align}
		Para cualquier matriz real $\mathbf{D}$, el producto exterior $\mathbf{D}\mathbf{D}^T$ es siempre y bajo toda circunstancia una matriz \textbf{semidefinida positiva ($\mathbf{D}\mathbf{D}^T \succeq \mathbf{0}$)}, es decir, para todo vector no nulo $\mathbf{v} \in \mathbb{R}^{p+1}$ se cumple que $\mathbf{v}^T (\mathbf{D}\mathbf{D}^T)\mathbf{v} = \|\mathbf{D}^T \mathbf{v}\|^2 \geq 0$.
		En conclusión, la diferencia de varianzas entre cualquier estimador lineal insesgado y el estimador por MCO es siempre semidefinida positiva:
		\begin{align}
			\text{Var}(\tilde{\boldsymbol{\beta}}) - \text{Var}(\hat{\boldsymbol{\beta}}) = \sigma^2 \mathbf{D}\mathbf{D}^T \succeq \mathbf{0} \implies \text{Var}(\tilde{\beta}_j) \geq \text{Var}(\hat{\beta}_j) \quad \forall j.
		\end{align}
		Esto demuestra con elegancia e indiscutible rigor matemático el \textbf{Teorema de Gauss-Markov}: el estimador de mínimos cuadrados ordinarios ($\hat{\boldsymbol{\beta}}$) es el Estimador Lineal Insesgado de Varianza Mínima (\textbf{MELI / BLUE}).
		
		\item \textbf{Demostración de la penalización analítica del $R_{\text{adj}}^2$ al añadir predictores superfluos:}
		Sea $R_p^2$ el coeficiente nominal con $p$ regresores y $R_{p+1}^2$ el coeficiente con $p+1$ regresores tras incluir la variable $X_{p+1}$.
		Por la teoría del incremento de suma de cuadrados ($\text{SCR}_{p+1} = \text{SCR}_p + \text{SCR}(X_{p+1} | X_1,\dots,X_p)$), como toda suma de cuadrados de una proyección ortogonal real es no negativa ($\text{SCR}_{\text{extra}} \geq 0$), la varianza explicada nominal obedece la ley de monotonia no decreciente:
		\begin{align}
			R_{p+1}^2 \geq R_p^2.
		\end{align}
		Relacionemos ahora el coeficiente ajustado $R_{\text{adj}, p}^2 = 1 - \dfrac{\text{CME}_p}{\text{CMT}}$ al añadir un regresor: para que el nuevo modelo supere al anterior ($R_{\text{adj}, p+1}^2 > R_{\text{adj}, p}^2$), es condición necesaria y suficiente que el nuevo Cuadrado Medio del Error sea estrictamente inferior al previo ($\text{CME}_{p+1} < \text{CME}_p$).
		La relación algebraica exacta que vincula las dos varianzas residuales muestrales tras introducir la variable $X_{p+1}$ con un estadístico $t$ de Student parcial ($t_{p+1}$) resulta de la identidad de partición del ANOVA:
		\begin{align}
			\text{SCE}_p = \text{SCE}_{p+1} + \frac{\text{SCE}_{p+1}}{n - p - 2} t_{p+1}^2 = \text{SCE}_{p+1} \left( 1 + \frac{t_{p+1}^2}{n - p - 2} \right).
		\end{align}
		Dividiendo ambos miembros entre sus respectivos grados de libertad residuales ($\nu_p = n - p - 1$ y $\nu_{p+1} = n - p - 2$):
		\begin{align}
			(n - p - 1) \text{CME}_p = (n - p - 2) \text{CME}_{p+1} \left( 1 + \frac{t_{p+1}^2}{n - p - 2} \right).
		\end{align}
		Despejando la razón de varianzas residuales:
		\begin{align}
			\frac{\text{CME}_p}{\text{CME}_{p+1}} = \frac{n - p - 2 + t_{p+1}^2}{n - p - 1}.
		\end{align}
		Para que el nuevo modelo mejore la varianza del error ($\text{CME}_{p+1} < \text{CME}_p \iff \dfrac{\text{CME}_p}{\text{CME}_{p+1}} > 1$), exigimos que el numerador sea estrictamente mayor al denominador:
		\begin{align}
			n - p - 2 + t_{p+1}^2 > n - p - 1 \implies t_{p+1}^2 > 1 \iff |t_{p+1}| > 1.
		\end{align}
		\textbf{Conclusión analítica fundamental:} Si incorporamos al modelo una variable predictora redundante o espuria ($X_{p+1}$) cuyo estadístico de prueba empírico parcial sea menor a la unidad en valor absoluto ($|t_{p+1}| < 1$, lo que en muestras grandes equivale a un $p$-valor mayor a aproximadamente $0.32$), el \textbf{Coeficiente de Determinación Ajustado ($R_{\text{adj}}^2$) disminuirá de manera inexorable} ($R_{\text{adj}, p+1}^2 < R_{\text{adj}, p}^2$). Esta penalización matemática intrínseca actúa como una barrera o filtro objetivo de parsimonia que protege al modelo contra el sobreajuste y la inflación artificial de variables no informativas.
	\end{enumerate}
\end{solucion}
