% Recordar
\begin{problema}
	\label{prob:0d6ff67}
	Se estudia el rendimiento medio de combustible ($Y$, en km/l) de automóviles urbanos en función de dos variables predictoras continuas: el peso bruto del vehículo ($X_1$, en toneladas) y la cilindrada de su motor ($X_2$, en litros de capacidad). Tras recopilar una muestra aleatoria de $n = 30$ automóviles en condiciones de prueba estandarizadas, se ajusta el siguiente modelo de regresión lineal múltiple por mínimos cuadrados ordinarios:
	\begin{align}
		\hat{Y} = 25.0 - 3.0X_1 - 1.5X_2, \quad R^2 = 0.780, \quad R_{\text{adj}}^2 = 0.763.
	\end{align}
	\begin{enumerate}
		\item Interprete el significado físico de los coeficientes parciales $\hat{\beta}_1 = -3.0$ y $\hat{\beta}_2 = -1.5$, especificando explícitamente la condición \emph{ceteris paribus}.
		\item Calcule la predicción puntual del rendimiento esperable para un sedán compacto con peso de $1.5$ toneladas y un motor de $2.0$ litros.
		\item Si la varianza muestral total de los rendimientos observados fue $S_Y^2 = 12.0$, calcule la Suma de Cuadrados de la Regresión ($\text{SCR}$) y la Suma de Cuadrados Residual ($\text{SCE}$).
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:9f8863e}
	Una firma de consultoría en marketing analiza el impacto de la diversificación de canales publicitarios sobre las ventas mensuales de corporaciones ($Y$, en miles de dólares). En una muestra histórica de $n = 40$ trimestres evaluados con un modelo de tres variables independientes ($p = 3$), la varianza explicada arrojó $R^2 = 0.780$.
	\begin{enumerate}
		\item Calcule el Coeficiente de Determinación Ajustado $R_{\text{adj}}^2 = 1 - \left[\dfrac{n-1}{n-p-1}(1-R^2)\right]$.
		\item Explique por qué en análisis multivariable $R_{\text{adj}}^2$ es estrictamente menor al $R^2$ nominal siempre que se incorporan dos o más predictores.
		\item Si un analista añade un cuarto predictor $X_4$ irrelevante (el código postal de la sede, con el $R^2$ apenas subiendo de $0.780$ a $0.781$), determine qué le ocurre al nuevo $R_{\text{adj}}^2$ con $p=4$.
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:1d085e7}
	Se calibra un modelo de regresión múltiple para predecir el valor de tasación de viviendas residenciales ($Y$, en miles de dólares) en función de $p = 4$ variables: área construida ($X_1$, en m$^2$), número de habitaciones ($X_2$), antigüedad ($X_3$, en años) y distancia al centro urbano ($X_4$, en km). Con $n = 50$ propiedades:
	\begin{align}
		\hat{Y} = 50.0 + 0.80X_1 + 15.0X_2 - 2.0X_3 - 5.0X_4,
	\end{align}
	con $\text{VIF}(X_1) = 4.2$, $\text{VIF}(X_2) = 3.8$, $\text{VIF}(X_3) = 1.5$, $\text{VIF}(X_4) = 1.2$.
	\begin{enumerate}
		\item Interprete en términos económicos los coeficientes de área ($\hat{\beta}_1$) y distancia ($\hat{\beta}_4$).
		\item Evalúe los diagnósticos de colinealidad mediante los $\text{VIF}_j$: ¿existe un problema severo de multicolinealidad o es un grado moderado admisible?
		\item Calcule el valor proyectado de tasación de una residencia con $120$ m$^2$, $3$ habitaciones, $10$ años de antigüedad y $8$ km del centro.
		\item ¿Qué estrategia de modelación aplicaría si el $\text{VIF}(X_1)$ superara el umbral crítico de 10?
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:15ced94}
	\textbf{Deducción analítica de MCO vectorial-matricial e insesgo poblacional en el modelo general de $p$ regresores:} considere $\mathbf{Y} = \mathbf{X}\boldsymbol{\beta} + \boldsymbol{\varepsilon}$, donde $\mathbf{X}$ es de rango columna completo, $E[\boldsymbol{\varepsilon}] = \mathbf{0}$ y $\text{Var}(\boldsymbol{\varepsilon}) = \sigma^2\mathbf{I}_n$.
	\begin{enumerate}
		\item Partiendo de $S(\boldsymbol{\beta}) = (\mathbf{Y}-\mathbf{X}\boldsymbol{\beta})^T(\mathbf{Y}-\mathbf{X}\boldsymbol{\beta})$, aplique cálculo diferencial matricial para deducir las \textbf{ecuaciones normales} $\mathbf{X}^T\mathbf{X}\hat{\boldsymbol{\beta}} = \mathbf{X}^T\mathbf{Y}$ y justifique la unicidad de $\hat{\boldsymbol{\beta}} = (\mathbf{X}^T\mathbf{X})^{-1}\mathbf{X}^T\mathbf{Y}$.
		\item Demuestre que el vector de residuos $\hat{\boldsymbol{\varepsilon}} = (\mathbf{I}_n-\mathbf{H})\mathbf{Y}$ es ortogonal al espacio columna de $\mathbf{X}$ ($\mathbf{X}^T\hat{\boldsymbol{\varepsilon}}=\mathbf{0}$) y que $\hat{\mathbf{Y}}^T\hat{\boldsymbol{\varepsilon}}=0$.
		\item Demuestre que $\hat{\boldsymbol{\beta}}$ es insesgado ($E[\hat{\boldsymbol{\beta}}]=\boldsymbol{\beta}$) y que $\text{Var}(\hat{\boldsymbol{\beta}}) = \sigma^2(\mathbf{X}^T\mathbf{X})^{-1}$.
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:ee59291}
	Un científico de datos modela la remuneración salarial anual ($Y$, en miles de dólares) en función de los años de experiencia ($X_1$) y variables indicadoras del nivel educativo, con Bachillerato como grupo de referencia ($D_1=D_2=0$): $D_1=1$ si tiene Licenciatura, $D_2=1$ si tiene Maestría. Con $n=100$ empleados: $\hat{Y} = 22.0 + 2.5X_1 + 12.0D_1 + 25.0D_2$, $R^2=0.760$.
	\begin{enumerate}
		\item Interprete el intercepto $\hat{\beta}_0=22.0$ y los coeficientes categóricos $\hat{\beta}_2=12.0$ y $\hat{\beta}_3=25.0$.
		\item Escriba las tres rectas de salario vs. experiencia separadas por subgrupo educativo, y prediga el salario de un gerente con 8 años de experiencia y Maestría.
		\item Evalúe en qué consiste la \textbf{Trampa de la Variable Muda} (\emph{Dummy Variable Trap}) y por qué habría sido fatal para el álgebra matricial del modelo incluir una tercera variable muda $D_0=1$ para Bachillerato junto con el intercepto $\beta_0$.
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:a6a21c3}
	Diseñe un problema original de regresión múltiple (distinto de automóviles, marketing, viviendas o salarios ya vistos): un hospital modela el costo total de una hospitalización ($Y$, en miles de dólares) en función de la edad del paciente ($X_1$, en años) y el número de días de estancia ($X_2$). Con $n=45$ pacientes se ajustó $\hat{Y} = 2.0 + 0.05X_1 + 1.2X_2$, $R^2=0.71$, $SE(\hat{\beta}_1)=0.02$, $SE(\hat{\beta}_2)=0.15$. Formule y resuelva las pruebas $t$ individuales para ambos predictores ($\nu=42$, $t_{0.025,42}=2.018$), interprete el modelo, y prediga el costo para un paciente de $65$ años con $5$ días de estancia.
\end{problema}

\subsection*{Soluciones detalladas}

% Recordar
\begin{solproblema}[prob:0d6ff67]
	\begin{enumerate}
		\item Manteniendo $X_2$ constante, cada tonelada adicional de peso ($X_1$) reduce el rendimiento en $3.0$ km/l. Manteniendo $X_1$ constante, cada litro adicional de cilindrada ($X_2$) reduce el rendimiento en $1.5$ km/l.
		\item $\hat{Y} = 25.0-3.0(1.5)-1.5(2.0) = 25.0-4.5-3.0 = 17.5$ km/l.
		\item $\text{SCT}=(n-1)S_Y^2=(29)(12.0)=348.0$. $\text{SCR}=R^2\times\text{SCT}=(0.780)(348.0)=271.44$. $\text{SCE}=348.0-271.44=76.56$.
	\end{enumerate}
\end{solproblema}

% Comprender
\begin{solproblema}[prob:9f8863e]
	\begin{enumerate}
		\item $R_{\text{adj}}^2 = 1-\left[\frac{39}{36}(0.220)\right] = 1-0.23833 \approx 0.7617$ (76.17\%).
		\item Añadir cualquier predictor nunca puede aumentar $\text{SCE}$ ($\text{SCE}_{p+1}\le\text{SCE}_p$), así que el $R^2$ nominal nunca disminuye, incentivando el sobreajuste. $R_{\text{adj}}^2 = 1-\text{CME}/\text{CMT}$ corrige esto: al añadir un predictor, $(n-p-1)$ disminuye en una unidad, penalizando el cociente $\text{CME}/\text{CMT}$ salvo que la nueva variable absorba suficiente varianza real para compensarlo.
		\item Con $R^2_{\text{nuevo}}=0.781$, $p=4$: $R_{\text{adj,nuevo}}^2 = 1-\left[\frac{39}{35}(0.219)\right] \approx 0.7560$. Aunque el $R^2$ subió superficialmente, el $R_{\text{adj}}^2$ \textbf{cayó} de $0.7617$ a $0.7560$, descartando la utilidad del código postal.
	\end{enumerate}
\end{solproblema}

\begin{solproblema}[prob:1d085e7]
	\begin{enumerate}
		\item $\hat{\beta}_1=0.80$: controlando habitaciones, antigüedad y distancia, cada m$^2$ adicional incrementa el valor en \$800. $\hat{\beta}_4=-5.0$: controlando el resto, cada km adicional de distancia al centro reduce el valor en \$5,000.
		\item Escala diagnóstica: $\text{VIF}<5$ es bajo/moderado, $5\le\text{VIF}<10$ es preocupante, $\text{VIF}\ge10$ es severo. Como el mayor VIF es $4.2<5.0$, \textbf{no existe multicolinealidad severa}; el solapamiento moderado entre área y habitaciones refleja su correlación geométrica natural, pero hay suficiente variabilidad ortogonal para estimar ambos efectos con precisión.
		\item $\hat{Y} = 50.0+0.80(120)+15.0(3)-2.0(10)-5.0(8) = 50+96+45-20-40 = 131.0$ miles de dólares.
		\item Si el VIF superara 10: fusionar variables colineales en un índice sintético, eliminar la variable con el $t$ menos significativo, aplicar regresión en componentes principales, o usar regresión Ridge ($\hat{\boldsymbol{\beta}}_{\text{Ridge}}=(\mathbf{X}^T\mathbf{X}+\lambda\mathbf{I})^{-1}\mathbf{X}^T\mathbf{Y}$) para estabilizar la inversión matricial.
	\end{enumerate}
\end{solproblema}

% Analizar
\begin{solproblema}[prob:15ced94]
	\begin{enumerate}
		\item Expandiendo $S(\boldsymbol{\beta}) = \mathbf{Y}^T\mathbf{Y} - 2\boldsymbol{\beta}^T\mathbf{X}^T\mathbf{Y} + \boldsymbol{\beta}^T\mathbf{X}^T\mathbf{X}\boldsymbol{\beta}$ y derivando respecto a $\boldsymbol{\beta}$:
		\begin{align}
			\frac{\partial S(\boldsymbol{\beta})}{\partial\boldsymbol{\beta}} = -2\mathbf{X}^T\mathbf{Y}+2\mathbf{X}^T\mathbf{X}\boldsymbol{\beta} = \mathbf{0} \implies \mathbf{X}^T\mathbf{X}\hat{\boldsymbol{\beta}} = \mathbf{X}^T\mathbf{Y}.
		\end{align}
		Como $\mathbf{X}$ tiene rango completo, $\mathbf{X}^T\mathbf{X}$ es simétrica definida positiva y no singular, garantizando la solución única $\hat{\boldsymbol{\beta}} = (\mathbf{X}^T\mathbf{X})^{-1}\mathbf{X}^T\mathbf{Y}$.
		\item Con $\hat{\boldsymbol{\varepsilon}} = \mathbf{Y}-\mathbf{X}\hat{\boldsymbol{\beta}}$: $\mathbf{X}^T\hat{\boldsymbol{\varepsilon}} = \mathbf{X}^T\mathbf{Y}-\mathbf{X}^T\mathbf{X}\hat{\boldsymbol{\beta}} = \mathbf{X}^T\mathbf{Y}-\mathbf{X}^T\mathbf{Y} = \mathbf{0}$ (por las ecuaciones normales). Como $\hat{\mathbf{Y}}=\mathbf{X}\hat{\boldsymbol{\beta}}$ vive en el espacio columna de $\mathbf{X}$: $\hat{\mathbf{Y}}^T\hat{\boldsymbol{\varepsilon}} = \hat{\boldsymbol{\beta}}^T\mathbf{X}^T\hat{\boldsymbol{\varepsilon}} = \hat{\boldsymbol{\beta}}^T\mathbf{0} = 0$.
		\item Sustituyendo $\mathbf{Y}=\mathbf{X}\boldsymbol{\beta}+\boldsymbol{\varepsilon}$: $\hat{\boldsymbol{\beta}} = \boldsymbol{\beta}+(\mathbf{X}^T\mathbf{X})^{-1}\mathbf{X}^T\boldsymbol{\varepsilon}$. Tomando esperanza con $E[\boldsymbol{\varepsilon}]=\mathbf{0}$: $E[\hat{\boldsymbol{\beta}}]=\boldsymbol{\beta}$. Para la varianza:
		\begin{align}
			\text{Var}(\hat{\boldsymbol{\beta}}) = (\mathbf{X}^T\mathbf{X})^{-1}\mathbf{X}^T E[\boldsymbol{\varepsilon}\boldsymbol{\varepsilon}^T]\mathbf{X}(\mathbf{X}^T\mathbf{X})^{-1} = \sigma^2(\mathbf{X}^T\mathbf{X})^{-1}[\mathbf{X}^T\mathbf{X}](\mathbf{X}^T\mathbf{X})^{-1} = \sigma^2(\mathbf{X}^T\mathbf{X})^{-1}.
		\end{align}
	\end{enumerate}
\end{solproblema}

% Evaluar
\begin{solproblema}[prob:ee59291]
	\begin{enumerate}
		\item $\hat{\beta}_0=22.0$: salario medio estimado del grupo base (Bachillerato) con $X_1=0$. $\hat{\beta}_2=12.0$: a igual experiencia, un profesional con Licenciatura gana en promedio \$12,000 más que uno con Bachillerato. $\hat{\beta}_3=25.0$: a igual experiencia, un profesional con Maestría gana \$25,000 más que uno con Bachillerato.
		\item Bachillerato: $\hat{Y}=22.0+2.5X_1$. Licenciatura: $\hat{Y}=34.0+2.5X_1$. Maestría: $\hat{Y}=47.0+2.5X_1$. Para $X_1=8$ y Maestría: $\hat{Y}=47.0+2.5(8)=67.0$ miles de dólares.
		\item Si se incluyeran las tres variables mudas ($D_0,D_1,D_2$) junto con el intercepto, se cumpliría $D_0+D_1+D_2=\mathbf{1}_n$ para toda fila, haciendo las columnas de $\mathbf{X}$ linealmente dependientes ($\text{rango}(\mathbf{X})<p+1$). Esto anula el determinante $\det(\mathbf{X}^T\mathbf{X})=0$, volviendo la matriz normal singular y no invertible, imposibilitando calcular $\hat{\boldsymbol{\beta}}=(\mathbf{X}^T\mathbf{X})^{-1}\mathbf{X}^T\mathbf{Y}$. Por eso siempre debe excluirse una categoría como referencia basal.
	\end{enumerate}
\end{solproblema}

\begin{solproblema}[prob:a6a21c3]
	$H_{0,1}:\beta_1=0$ vs. $H_{a,1}:\beta_1\neq0$; $H_{0,2}:\beta_2=0$ vs. $H_{a,2}:\beta_2\neq0$. Los estadísticos son:
	\begin{align}
		t_1 = \frac{0.05}{0.02} = 2.5, \qquad t_2 = \frac{1.2}{0.15} = 8.0.
	\end{align}
	Como $|t_1|=2.5>2.018$ y $|t_2|=8.0>2.018$, \textbf{se rechazan ambas} $H_0$: tanto la edad como los días de estancia contribuyen significativamente al costo, controlando por la otra variable. Interpretación: a igual número de días de estancia, cada año adicional de edad del paciente incrementa el costo en \$50; a igual edad, cada día adicional de estancia incrementa el costo en \$1,200. La predicción para un paciente de 65 años con 5 días de estancia es:
	\begin{align}
		\hat{Y} = 2.0+0.05(65)+1.2(5) = 2.0+3.25+6.0 = 11.25 \text{ miles de dólares.}
	\end{align}
\end{solproblema}
