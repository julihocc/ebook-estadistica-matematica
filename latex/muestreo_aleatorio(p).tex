\section*{Problemas}

\subsection{Enunciados de los problemas}

\subsubsection*{1. Nivel Fundamental (Sencillos / Directos)}

\begin{problema}
	\label{prob:2.6.1}
	Una población finita consta de $N = 6$ elementos distintos $\{a, b, c, d, e, f\}$. Se desea extraer una muestra aleatoria de tamaño $n = 3$.
	\begin{enumerate}
		\item Determina el número total de muestras posibles si la extracción se realiza \textbf{con reemplazo}.
		\item Determina el número total de muestras posibles si la extracción se realiza \textbf{sin reemplazo} y sin importar el orden (muestras no ordenadas).
		\item Explica formalmente por qué, a medida que el tamaño poblacional $N$ crece hacia el infinito, la diferencia probabilística entre muestreo con y sin reemplazo tiende a cero.
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:2.6.2}
	Sea $X_1, X_2, \ldots, X_n$ una muestra aleatoria simple (variables aleatorias independientes e idénticamente distribuidas) extraída de una población con media teórica $\mu$ y varianza $\sigma^2 < \infty$. Demuestra algebraicamente, aplicando las propiedades del operador esperanza matemática, que la media muestral $\bar{X} = \frac{1}{n} \sum_{i=1}^n X_i$ es un \textbf{estimador insesgado} del parámetro poblacional $\mu$, es decir, que $E(\bar{X}) = \mu$.
\end{problema}

\begin{problema}
	\label{prob:2.6.3}
	El tiempo de ensamblaje de un componente mecánico sigue una distribución con media poblacional $\mu = 45$ minutos y desviación estándar $\sigma = 12$ minutos. Si se toma una muestra aleatoria de tamaño $n$, calcula el error estándar de la media muestral, denotado por $\text{DE}(\bar{X}) = \sigma_{\bar{X}}$, para cada uno de los siguientes tamaños de muestra y comenta sobre la velocidad de reducción del error:
	\begin{enumerate}
		\item $n = 9$
		\item $n = 36$
		\item $n = 144$
		\item $n = 576$
	\end{enumerate}
\end{problema}


\subsubsection*{2. Nivel Operativo (Intermedios / Aplicados)}

\begin{problema}
	\label{prob:2.6.4}
	Una auditoría gubernamental evalúa los ingresos mensuales de un gremio comercial que cuenta con exactamente $N = 400$ empresas registradas. La desviación estándar poblacional se conoce históricamente y es de $\sigma = \$15,000$ MXN. Un equipo de auditores selecciona una muestra aleatoria de $n = 64$ empresas.
	\begin{enumerate}
		\item Calcula la varianza de la media muestral $\text{Var}(\bar{X})$ bajo el supuesto de muestreo \textbf{con reemplazo} (población infinita o muestreo multinomial).
		\item Calcula la varianza de la media muestral $\text{Var}(\bar{X})$ bajo muestreo \textbf{sin reemplazo}, aplicando el factor de corrección por población finita (FPC) dado por $\frac{N-n}{N-1}$.
		\item ¿Qué porcentaje de reducción en la varianza muestral se logra gracias a la corrección por población finita en este escenario auditor?
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:2.6.5}
	Un dado equilibrado de 6 caras se lanza $n = 36$ veces de manera independiente. Se sabe que para un solo lanzamiento, la media poblacional es $\mu = 3.5$ y la varianza es $\sigma^2 = \frac{35}{12} \approx 2.9167$. Sea $\bar{X}_{36}$ el promedio aritmético de los 36 lanzamientos.
	Utilizando el Teorema del Límite Central (TLC), calcula la probabilidad aproximada de que el promedio muestral obtenido se encuentre estrictamente entre $3.1$ y $3.9$, es decir, $P(3.1 < \bar{X}_{36} < 3.9)$.
\end{problema}

\begin{problema}
	\label{prob:2.6.6}
	Una embotelladora de agua mineral desea calibrar su línea de producción. La cantidad vertida en cada botella es una variable aleatoria con desviación estándar conocida de $\sigma = 8$ mililitros.
	¿Cuál es el tamaño de muestra mínimo $n$ que un ingeniero de calidad debe inspeccionar para tener una certeza probabilística de al menos $95\%$ de que la media muestral $\bar{X}$ no difiera de la verdadera media poblacional de llenado $\mu$ en más de $1.5$ mililitros?
\end{problema}


\subsubsection*{3. Nivel Analítico (Avanzados / Demostraciones)}

\begin{problema}
	\label{prob:2.6.7}
	Demuestra la \textbf{Ley Débil de los Grandes Números (WLLN)} utilizando la desigualdad de Chebyshev. En concreto, prueba que si $X_1, X_2, \ldots, X_n$ son variables aleatorias i.i.d. con media $\mu$ y varianza $\sigma^2 < \infty$, entonces para cualquier tolerancia arbitraria $\epsilon > 0$, la probabilidad de que la media muestral $\bar{X}_n$ diste de $\mu$ en $\epsilon$ o más decrece a cero cuando $n \to \infty$:
	\begin{align}
		\lim_{n \to \infty} P(|\bar{X}_n - \mu| \ge \epsilon) = 0
	\end{align}
\end{problema}

\begin{problema}
	\label{prob:2.6.8}
	Demuestra algebraicamente por qué el estimador de la varianza muestral requiere dividir entre $n-1$ (y no entre $n$) para ser insesgado. Sea $S^2 = \frac{1}{n-1} \sum_{i=1}^n (X_i - \bar{X})^2$ la varianza muestral de una MAS obtenida de una población con media $\mu$ y varianza $\sigma^2$. Demuestra paso a paso que $E(S^2) = \sigma^2$.
	\emph{Sugerencia:} Suma y resta $\mu$ dentro del término cuadrático $(X_i - \bar{X})^2 = [(X_i - \mu) - (\bar{X} - \mu)]^2$.
\end{problema}


\subsubsection*{4. Nivel Desafiante (Expertos / Paradojas)}

\begin{problema}
	\label{prob:2.6.9}
	El Teorema del Límite Central postula que la suma de variables independientes convergerá a la distribución normal siempre que el momento de segundo orden ($\sigma^2$) sea finito. Analiza qué sucede cuando se extraen muestras de una población con colas extremadamente pesadas: la distribución de Cauchy estándar, cuya densidad es $f(x) = \frac{1}{\pi(1+x^2)}$ para $x \in (-\infty, \infty)$.
	\begin{enumerate}
		\item Demuestra o argumenta formalmente por qué la integral que define la esperanza teórica $E(X)$ no converge absolutamente en la distribución de Cauchy.
		\item Si se extrae una muestra aleatoria de tamaño $n$ de una población Cauchy, ¿cuál es la función de densidad o comportamiento de la media muestral $\bar{X}_n$? ¿Ocurre convergencia hacia la normal conforme $n \to \infty$?
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:2.6.10}
	En la práctica computacional, el Teorema del Límite Central proporciona una aproximación asintótica. Para evaluar la velocidad y calidad exacta de esta convergencia en muestras finitas $n$, se emplea el \textbf{Teorema de Berry-Esseen}.
	Sea $X_1, X_2, \ldots, X_n$ una muestra i.i.d. con $E(X_i) = \mu$, $\text{Var}(X_i) = \sigma^2 > 0$ y tercer momento absoluto central finito $\rho_3 = E[|X_i - \mu|^3] < \infty$. La cota de Berry-Esseen establece que el máximo error absoluto de aproximación entre la función de distribución empírica estandarizada $F_n(z)$ y la distribución normal estándar $\Phi(z)$ satisface:
	\begin{align}
		\sup_{z \in \mathbb{R}} |F_n(z) - \Phi(z)| \le \frac{C \cdot \rho_3}{\sigma^3 \sqrt{n}}
	\end{align}
	donde $C < 0.4784$ es la constante universal de Berry-Esseen.
	Si una población exponencial con parámetro $\lambda = 1$ ($\mu=1, \sigma=1, \rho_3 \approx 3.6$) es muestreada con $n = 100$, calcula la cota superior del error absoluto en probabilidad que garantiza el Teorema de Berry-Esseen e interpreta su significado para la toma de decisiones en ingeniería.
\end{problema}

\begin{solproblema}
	\begin{enumerate}
		\item \textbf{Muestreo con reemplazo:} En cada una de las $n = 3$ extracciones hay exactamente $N = 6$ opciones disponibles, dado que el elemento extraído se devuelve a la población antes de la siguiente selección. Por el principio multiplicativo del conteo, el número total de muestras posibles ordenadas es:
		\begin{align}
			N^n = 6^3 = 216 \text{ muestras.}
		\end{align}
		
		\item \textbf{Muestreo sin reemplazo (no ordenado):} Dado que cada elemento solo puede aparecer una vez por muestra y el orden en que se extraen no altera el conjunto final, el número total de subconjuntos posibles es dado por el coeficiente binomial:
		\begin{align}
			\binom{N}{n} = \binom{6}{3} = \frac{6!}{3!(6-3)!} = \frac{6 \times 5 \times 4}{3 \times 2 \times 1} = 20 \text{ muestras.}
		\end{align}
		
		\item \textbf{Equivalencia asintótica cuando $N \to \infty$:} Al extraer una muestra fija $n$ sin reemplazo, la probabilidad de seleccionar un elemento específico en la primera extracción es $1/N$, en la segunda es $1/(N-1)$, y en la $k$-ésima es $1/(N-k+1)$. A medida que $N \to \infty$ con $n$ constante, las razones $\frac{N-k}{N} \to 1$. Por lo tanto, el agotamiento poblacional es despreciable y el muestreo sin reemplazo converge en distribución al muestreo con reemplazo (independiente).
	\end{enumerate}
\end{solproblema}

\begin{solproblema}
	Por hipótesis, $X_1, \ldots, X_n$ son independientes e idénticamente distribuidas, de modo que para todo $i \in \{1, \ldots, n\}$ se cumple que $E(X_i) = \mu$. Aplicando la linealidad del operador esperanza (la esperanza de una suma es la suma de las esperanzas, y las constantes multiplicativas salen de la esperanza):
	\begin{align}
		E(\bar{X}) &= E\left( \frac{1}{n} \sum_{i=1}^n X_i \right) \\
		&= \frac{1}{n} \sum_{i=1}^n E(X_i) \\
		&= \frac{1}{n} \sum_{i=1}^n \mu \\
		&= \frac{1}{n} (n \cdot \mu) = \mu.
	\end{align}
	Como $E(\bar{X}) = \mu$, queda demostrado algebraicamente que el promedio muestral es un estimador insesgado de la media poblacional.
\end{solproblema}

\begin{solproblema}
	El error estándar de la media ($\sigma_{\bar{X}}$) mide la dispersión de los promedios muestrales alrededor de $\mu$ y está dado por $\sigma_{\bar{X}} = \frac{\sigma}{\sqrt{n}}$. Con $\sigma = 12$:
	\begin{enumerate}
		\item Para $n = 9$: $\sigma_{\bar{X}} = \frac{12}{\sqrt{9}} = \frac{12}{3} = 4.00 \text{ min.}$
		\item Para $n = 36$: $\sigma_{\bar{X}} = \frac{12}{\sqrt{36}} = \frac{12}{6} = 2.00 \text{ min.}$
		\item Para $n = 144$: $\sigma_{\bar{X}} = \frac{12}{\sqrt{144}} = \frac{12}{12} = 1.00 \text{ min.}$
		\item Para $n = 576$: $\sigma_{\bar{X}} = \frac{12}{\sqrt{576}} = \frac{12}{24} = 0.50 \text{ min.}$
	\end{enumerate}
	\textbf{Comentario sobre la velocidad de reducción:} La dispersión decrece de manera estrictamente inversamente proporcional a la raíz cuadrada del tamaño muestral ($\sqrt{n}$). Para reducir el error estándar a la mitad (por ejemplo, de 4.0 a 2.0, o de 2.0 a 1.0), es obligatorio cuadruplicar el tamaño de la muestra inspeccionada ($9 \to 36 \to 144$).
\end{solproblema}


\subsubsection*{2. Nivel Operativo (Intermedios / Aplicados)}

\begin{solproblema}
	Datos conocidos: $N = 400$, $\sigma = 15,000$ MXN, y $n = 64$.
	\begin{enumerate}
		\item \textbf{Varianza bajo muestreo con reemplazo (o población infinita):}
		\begin{align}
			\text{Var}_{C/R}(\bar{X}) = \frac{\sigma^2}{n} = \frac{(15,000)^2}{64} = \frac{225,000,000}{64} = 3,515,625 \text{ MXN}^2.
		\end{align}
		El error estándar con reemplazo sería $\sigma_{\bar{X}} = \sqrt{3,515,625} = \$1,875 \text{ MXN}$.
		
		\item \textbf{Varianza bajo muestreo sin reemplazo (población finita):}
		Calculamos primero el factor de corrección por población finita (FPC):
		\begin{align}
			\text{FPC} = \frac{N - n}{N - 1} = \frac{400 - 64}{400 - 1} = \frac{336}{399} \approx 0.842105.
		\end{align}
		Multiplicamos la varianza muestral por el FPC:
		\begin{align}
			\text{Var}_{S/R}(\bar{X}) = \frac{\sigma^2}{n} \left( \frac{N-n}{N-1} \right) = 3,515,625 \times 0.842105 \approx 2,960,156.25 \text{ MXN}^2.
		\end{align}
		El error estándar real sin reemplazo es $\sigma_{\bar{X}, S/R} = \sqrt{2,960,156.25} \approx \$1,720.51 \text{ MXN}$.
		
		\item \textbf{Porcentaje de reducción de varianza:}
		La reducción relativa en varianza es exactamente igual a $1 - \text{FPC}$:
		\begin{align}
			\text{Reducción} = 1 - \frac{336}{399} = \frac{63}{399} \approx 0.15789 \text{ (15.79\%).}
		\end{align}
		Al muestrear el $16\%$ de la población ($64/400$), la ganancia de precisión al no repetir empresas reduce la varianza en un $15.79\%$.
	\end{enumerate}
\end{solproblema}

\begin{solproblema}
	Para el dado justo: $\mu = 3.5$, $\sigma^2 = \frac{35}{12} \approx 2.9167$, $n = 36$.
	El error estándar del promedio de 36 lanzamientos es:
	\begin{align}
		\sigma_{\bar{X}} = \frac{\sigma}{\sqrt{n}} = \sqrt{\frac{35/12}{36}} = \sqrt{\frac{35}{432}} \approx \sqrt{0.0810185} \approx 0.284637.
	\end{align}
	Queremos calcular $P(3.1 < \bar{X}_{36} < 3.9)$. Estandarizamos los límites del intervalo mediante la variable $Z = \frac{\bar{X} - \mu}{\sigma_{\bar{X}}}$:
	\begin{align}
		z_1 &= \frac{3.1 - 3.5}{0.284637} = \frac{-0.40}{0.284637} \approx -1.4053 \\
		z_2 &= \frac{3.9 - 3.5}{0.284637} = \frac{+0.40}{0.284637} \approx +1.4053
	\end{align}

	\begin{align}
		P(3.1 < \bar{X}_{36} < 3.9) &\approx P(-1.405 < Z < +1.405) \\
		&= \Phi(1.405) - \Phi(-1.405) = 2\Phi(1.405) - 1.

	\begin{align}
		P(3.1 < \bar{X}_{36} < 3.9) \approx 2(0.9200) - 1 = 0.8400 \text{ (84.00\%).}
	\end{align}

\subsubsection*{3. Nivel Analítico (Avanzados / Demostraciones)}

\begin{solproblema}
	Sea $X_1, \ldots, X_n$ una secuencia de variables aleatorias i.i.d. con $E(X_i) = \mu$ y $\text{Var}(X_i) = \sigma^2$.
	Sabemos por las propiedades de la esperanza y varianza que para la media muestral $\bar{X}_n = \frac{1}{n}\sum_{i=1}^n X_i$:
	\begin{align}
		E(\bar{X}_n) &= \mu, \\
		\text{Var}(\bar{X}_n) &= \frac{\sigma^2}{n}.
	\end{align}
	La **Desigualdad de Chebyshev** establece que para cualquier variable aleatoria $Y$ con media $\mu_Y$ y varianza $\sigma_Y^2$, y para todo constante $k > 0$ o tolerancia $\epsilon > 0$:
	\begin{align}
		P(|Y - \mu_Y| \ge \epsilon) \le \frac{\text{Var}(Y)}{\epsilon^2}.
	\end{align}
	Sustituyendo $Y = \bar{X}_n$, $\mu_Y = \mu$ y $\text{Var}(Y) = \frac{\sigma^2}{n}$ en la desigualdad:
	\begin{align}
		0 \le P(|\bar{X}_n - \mu| \ge \epsilon) \le \frac{\sigma^2/n}{\epsilon^2} = \frac{\sigma^2}{n \epsilon^2}.
	\end{align}
	Tomando el límite en ambos extremos cuando el tamaño muestral $n \to \infty$ (considerando $\sigma^2$ y $\epsilon > 0$ constantes fijas):
	\begin{align}
		\lim_{n \to \infty} \frac{\sigma^2}{n \epsilon^2} = 0.
	\end{align}
	Por el Teorema del Sándwich de límites algebraicos, concluimos rigurosamente que:
	\begin{align}
		\lim_{n \to \infty} P(|\bar{X}_n - \mu| \ge \epsilon) = 0,
	\end{align}
	lo cual demuestra que la media muestral $\bar{X}_n$ converge en probabilidad a la media verdadera $\mu$ ($\bar{X}_n \xrightarrow{p} \mu$).
\end{solproblema}

\begin{solproblema}
	Queremos encontrar $E(S^2)$ donde $S^2 = \frac{1}{n-1} \sum_{i=1}^n (X_i - \bar{X})^2$.
	Primero, analizamos la suma de cuadrados de las desviaciones respecto a la media muestral $\sum_{i=1}^n (X_i - \bar{X})^2$. Sumamos y restamos la verdadera media poblacional $\mu$ dentro del paréntesis:
	\begin{align}
		\sum_{i=1}^n (X_i - \bar{X})^2 &= \sum_{i=1}^n [(X_i - \mu) - (\bar{X} - \mu)]^2 \\
		&= \sum_{i=1}^n \left[ (X_i - \mu)^2 - 2(X_i - \mu)(\bar{X} - \mu) + (\bar{X} - \mu)^2 \right] \\
		&= \sum_{i=1}^n (X_i - \mu)^2 - 2(\bar{X} - \mu) \sum_{i=1}^n (X_i - \mu) + n(\bar{X} - \mu)^2.
	\end{align}
	Notemos que la suma lineal respecto a $\mu$ se puede simplificar: $\sum_{i=1}^n (X_i - \mu) = \sum_{i=1}^n X_i - n\mu = n\bar{X} - n\mu = n(\bar{X} - \mu)$.
	Sustituyendo esto en el término central:
	\begin{align}
		\sum_{i=1}^n (X_i - \bar{X})^2 &= \sum_{i=1}^n (X_i - \mu)^2 - 2(\bar{X} - \mu)[n(\bar{X} - \mu)] + n(\bar{X} - \mu)^2 \\
		&= \sum_{i=1}^n (X_i - \mu)^2 - 2n(\bar{X} - \mu)^2 + n(\bar{X} - \mu)^2 \\
		&= \sum_{i=1}^n (X_i - \mu)^2 - n(\bar{X} - \mu)^2.
	\end{align}
	Ahora aplicamos el operador esperanza $E(\cdot)$ a ambos lados de esta identidad:
	\begin{align}
		E\left[ \sum_{i=1}^n (X_i - \bar{X})^2 \right] &= \sum_{i=1}^n E[(X_i - \mu)^2] - n \cdot E[(\bar{X} - \mu)^2].
	\end{align}
	Por definición de varianza teórica poblacional y varianza de la media muestral:
	\begin{itemize}
		\item $E[(X_i - \mu)^2] = \text{Var}(X_i) = \sigma^2$ para cada uno de los $n$ términos.
		\item $E[(\bar{X} - \mu)^2] = \text{Var}(\bar{X}) = \frac{\sigma^2}{n}$.
	\end{itemize}
	Sustituyendo estas varianzas exactas:
	\begin{align}
		E\left[ \sum_{i=1}^n (X_i - \bar{X})^2 \right] &= \sum_{i=1}^n \sigma^2 - n \left( \frac{\sigma^2}{n} \right) \\
		&= n\sigma^2 - \sigma^2 = (n - 1)\sigma^2.
	\end{align}
	Finalmente, al dividir por la constante de normalización $(n-1)$:
	\begin{align}
		E(S^2) = E\left[ \frac{1}{n-1} \sum_{i=1}^n (X_i - \bar{X})^2 \right] = \frac{1}{n-1} E\left[ \sum_{i=1}^n (X_i - \bar{X})^2 \right] = \frac{(n-1)\sigma^2}{n-1} = \sigma^2.
	\end{align}
	\textbf{Conclusión:} Si se dividiera entre $n$, el estimador tendría una esperanza de $\frac{n-1}{n}\sigma^2$, subestimando sistemáticamente la varianza poblacional. Dividir entre los $n-1$ grados de libertad corrige este sesgo exactamente.
\end{solproblema}


\subsubsection*{4. Nivel Desafiante (Expertos / Paradojas)}

\begin{solproblema}
	\begin{enumerate}
		\item \textbf{Inexistencia de momentos (Esperanza no convergente):}
		La esperanza de una variable aleatoria continua con densidad $f(x) = \frac{1}{\pi(1+x^2)}$ está dada por:
		\begin{align}
			E(X) = \int_{-\infty}^{\infty} x \cdot \frac{1}{\pi(1+x^2)} dx = \frac{1}{\pi} \int_{-\infty}^{\infty} \frac{x}{1+x^2} dx.
		\end{align}
		Para que una integral impropia en ambos extremos converja en el sentido de Lebesgue, las integrales de las partes positiva ($x \ge 0$) y negativa ($x < 0$) deben ser absolutamente finitas por separado:
		\begin{align}
			\int_0^{\infty} \frac{x}{1+x^2} dx = \lim_{b \to \infty} \left[ \frac{1}{2} \ln(1+x^2) \right]_0^b = \lim_{b \to \infty} \frac{1}{2} \ln(1+b^2) = \infty.
		\end{align}
		Como la integral sobre $[0, \infty)$ diverge a $+\infty$ (y análogamente sobre $(-\infty, 0]$ diverge a $-\infty$), la integral impropia total es una indeterminación de la forma $\infty - \infty$. Por tanto, el momento de primer orden $E(X)$ (y con mayor razón el momento cuadrático $\sigma^2$) **no existe**.
		
		\item \textbf{Distribución de la media muestral en Cauchy:}
		Sea $\bar{X}_n = \frac{1}{n} \sum_{i=1}^n X_i$ donde $X_i \sim \text{Cauchy}(0,1)$ i.i.d. Utilizando funciones características (transformada de Fourier), la función característica de un Cauchy estándar es $\varphi_X(t) = E(e^{itX}) = e^{-|t|}$.
		La función característica del promedio muestral $\bar{X}_n$ es:
		\begin{align}
			\varphi_{\bar{X}_n}(t) &= E\left[ \exp\left( it \frac{1}{n} \sum_{k=1}^n X_k \right) \right] = \prod_{k=1}^n \varphi_X\left( \frac{t}{n} \right) \\
			&= \prod_{k=1}^n \exp\left( -\left| \frac{t}{n} \right| \right) = \exp\left( -n \frac{|t|}{n} \right) = e^{-|t|}.
		\end{align}
		¡Sorprendentemente, la función característica $\varphi_{\bar{X}_n}(t) = e^{-|t|}$ es idéntica a la función característica de una sola observación Cauchy!
		Por el teorema de unicidad de la inversión de Fourier, la distribución del promedio de $n$ variables de Cauchy es exactamente la misma distribución de Cauchy original:
		\begin{align}
			\bar{X}_n \sim \text{Cauchy}(0,1).
		\end{align}
		\textbf{Interpretación estadística:} En poblaciones de Cauchy (o Pareto con colas extremas $\alpha \le 1$), **no ocurre ninguna convergencia hacia la normal** ni reducción de varianza. Tomar el promedio de un millón de datos Cauchy tiene exactamente la misma dispersión, probabilidad de valores extremos y error de estimación que tomar un solo dato al azar. El Teorema del Límite Central se fractura por completo al violarse la hipótesis $\sigma^2 < \infty$.
	\end{enumerate}
\end{solproblema}

\begin{solproblema}
	Para una variable $X \sim \text{Exp}(\lambda=1)$: la densidad es $f(x) = e^{-x}$ para $x \ge 0$.
	\begin{itemize}
		\item Media poblacional: $\mu = E(X) = 1$.
		\item Desviación estándar: $\sigma = \sqrt{\text{Var}(X)} = 1$.
		\item Tercer momento central absoluto: $\rho_3 = E[|X - \mu|^3] = \int_0^{\infty} |x - 1|^3 e^{-x} dx \approx 3.60$.
		\item Constante universal de Berry-Esseen: $C < 0.4784$.
	\end{itemize}
	Evaluando la cota de Berry-Esseen para una muestra de tamaño $n = 100$:
	\begin{align}
		\sup_{z \in \mathbb{R}} |F_{100}(z) - \Phi(z)| \le \frac{C \cdot \rho_3}{\sigma^3 \sqrt{n}} = \frac{0.4784 \times 3.60}{1^3 \times \sqrt{100}} = \frac{1.72224}{10} \approx 0.1722.
	\end{align}
	\textbf{Interpretación para la ingeniería y ciencia de datos:}
	La cota garantiza rigurosamente que al usar la tabla normal estándar $\Phi(z)$ para aproximar probabilidades de la media muestral con $n=100$ de una población tan asimétrica como la exponencial, el máximo error posible en cualquier punto de la curva es del **$17.22\%$** (o menos de $0.1722$ en probabilidad absoluta).
	Esto demuestra por qué la clásica "regla empírica del $n \ge 30$" es insuficiente para poblaciones fuertemente sesgadas: para reducir el error máximo asintótico de aproximación del TLC a menos del $1\%$ ($0.01$), una distribución exponencial requeriría un tamaño muestral de al menos $n \ge \left(\frac{1.72224}{0.01}\right)^2 \approx 29,661$ observaciones.
\end{solproblema}
