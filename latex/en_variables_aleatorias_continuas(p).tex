\begin{problema}
	\label{prob:14b3125}
	Consider the function $f(x) = c \cdot e^{-x}$ for $x \ge 0$ and $f(x) = 0$ for $x < 0$, where $c$ is a positive constant.
	\begin{enumerate}
		\item Determine the value of $c$ such that $f(x)$ is a valid probability density function (PDF).
		\item Compute the probability $P(1 \le X \le 3)$ by integrating the PDF over the interval.
		\item Compute the cumulative distribution function (CDF) $F(x)$ for all $x \in \mathbb{R}$.
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:287c45c}
	Let $X$ be a continuous random variable with PDF $f(x) = 2x$ for $0 \le x \le 1$ and $f(x) = 0$ otherwise.
	\begin{enumerate}
		\item Compute the CDF $F(x) = P(X \le x)$ for all $x \in \mathbb{R}$.
		\item Verify the fundamental properties: $F(-\infty) = 0$, $F(+\infty) = 1$, and nondecreasing monotonicity.
		\item Compute $P(0.3 \le X \le 0.7)$ using the CDF.
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:61a9dd8}
	The waiting time in a bank line (in minutes) follows an exponential distribution with mean $\mu = 5$ minutes; that is, $X \sim \text{Exp}(\lambda = 0.2)$.
	\begin{enumerate}
		\item Compute $P(X \le 3)$ and $P(X \ge 10)$.
		\item Find the quantile $q_{0.90}$ (the time below which 90\% of customers wait).
		\item The bank wants to guarantee that 95\% of customers wait less than $t$ minutes. Determine $t$.
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:19f62fd}
	For a continuous random variable $X$ with PDF $f(x)$ and a function $g: \mathbb{R} \to \mathbb{R}$, prove the \emph{Continuous LOTUS Theorem} (Law of the Unconscious Statistician):
	$$\E[g(X)] = \int_{-\infty}^{\infty} g(x) \cdot f(x)\,dx$$
	\begin{enumerate}
		\item Prove the particular case $g(X) = X^{2}$.
		\item Derive the variance formula $\Var(X) = \E[X^{2}] - (\E[X])^{2}$ using LOTUS with $g(x) = (x - \mu)^{2}$.
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:8ec7e10}
	The Kolmogorov-Smirnov (KS) test compares the empirical CDF $F_{n}$ of a sample with a theoretical CDF $F_{0}$ through the statistic:
	$$D_{n} = \sup_{x \in \mathbb{R}} |F_{n}(x) - F_{0}(x)|$$
	For a sample of $n = 50$ observations from $\text{Uniform}(0, 1)$ with $D_{50} = 0.18$, evaluate whether the null hypothesis (the data come from $\text{Uniform}(0,1)$) is rejected at level $\alpha = 0.05$ (critical value $K_{0.95} \approx 1.358$, where $\sqrt{n} D_n \xrightarrow{d} K$ under $H_0$), and justify your decision.
\end{problema}

\begin{problema}
	\label{prob:0d490fc}
	Design an original piecewise probability density function (different from the exponential, uniform, or normal densities already seen in this section): the useful lifetime $X$ (in years, with a minimum of 1 year) of a low-cost electronic component has heavy-tailed density $f(x) = k/x^{2}$ for $x \ge 1$, and $f(x) = 0$ otherwise. Find the normalization constant $k$ and compute $P(1 \le X \le 2)$.
\end{problema}

\subsection*{Detailed solutions}

% Remember
\begin{solproblema}[prob:14b3125]
	\begin{enumerate}
		\item The normalization condition requires:
		\begin{align*}
			\int_{-\infty}^{\infty} f(x)\,dx = \int_{0}^{\infty} c \cdot e^{-x}\,dx = c \cdot [-e^{-x}]_{0}^{\infty} = c = 1.
		\end{align*}
		Therefore, $c = 1$.
		\item $P(1 \le X \le 3) = \int_{1}^{3} e^{-x}\,dx = [-e^{-x}]_{1}^{3} = e^{-1} - e^{-3} \approx 0.3679 - 0.0498 = 0.3181$.
		\item $F(x) = \int_{-\infty}^{x} f(t)\,dt = \begin{cases} 0 & \text{if } x < 0 \\ 1 - e^{-x} & \text{if } x \ge 0 \end{cases}$, the CDF of the exponential distribution with $\lambda=1$.
	\end{enumerate}
\end{solproblema}

% Understand
\begin{solproblema}[prob:287c45c]
	\begin{enumerate}
		\item $F(x) = \int_{-\infty}^{x} f(t)\,dt = \begin{cases} 0 & \text{if } x < 0 \\ x^{2} & \text{if } 0 \le x \le 1 \\ 1 & \text{if } x > 1 \end{cases}$.
		\item Verification: $F(-\infty)=0$, $F(+\infty)=1$, and $F$ is nondecreasing on $[0,1]$ (it grows from $0$ to $1$).
		\item $P(0.3 \le X \le 0.7) = F(0.7)-F(0.3) = 0.7^2-0.3^2 = 0.49-0.09=0.40$.
	\end{enumerate}
\end{solproblema}

% Apply
\begin{solproblema}[prob:61a9dd8]
	The CDF is $F(x) = 1 - e^{-0.2x}$ for $x \ge 0$.
	\begin{enumerate}
		\item $P(X \le 3) = 1-e^{-0.6}\approx1-0.5488=0.4512$. $P(X\ge10)=e^{-2}\approx0.1353$.
		\item $q_{0.90}$ satisfies $1-e^{-0.2q_{0.90}}=0.9$, that is $q_{0.90}=-5\ln(0.1)=5\ln10\approx11.51$ min.
		\item For $P(X\le t)=0.95$: $t=-5\ln(0.05)\approx14.98$ min.
	\end{enumerate}
\end{solproblema}

% Analyze
\begin{solproblema}[prob:19f62fd]
	\begin{enumerate}
		\item For $g(X)=X^2$, the operational definition of LOTUS directly gives $\E[X^2]=\int_{-\infty}^{\infty}x^2f(x)\,dx$, which is simply expectation with $g(x)=x^2$.
		\item For variance with $g(X)=(X-\mu)^2$ where $\mu=\E[X]$:
		\begin{align*}
			\Var(X) &= \E[(X-\mu)^2] = \int_{-\infty}^{\infty}(x-\mu)^2f(x)\,dx = \int_{-\infty}^{\infty}(x^2-2\mu x+\mu^2)f(x)\,dx \\
			&= \E[X^2]-2\mu\E[X]+\mu^2 = \E[X^2]-2\mu^2+\mu^2 = \E[X^2]-(\E[X])^2,
		\end{align*}
		using linearity of the integral and the constancy of $\mu$ under the integral.
	\end{enumerate}
\end{solproblema}

% Evaluate
\begin{solproblema}[prob:8ec7e10]
	Under $H_0$, by the Glivenko-Cantelli theorem, the empirical CDF converges uniformly to the theoretical CDF, and the standardized statistic $\sqrt{n}D_n$ converges in distribution to the Kolmogorov distribution $K$. For $n=50$ and $D_{50}=0.18$:
	\begin{align*}
		\sqrt{50}\cdot0.18 \approx 1.273 < 1.358 = K_{0.95}.
	\end{align*}
	Since the observed statistic ($1.273$) does not exceed the critical value ($1.358$), \textbf{we do not reject} $H_0$ at level $\alpha=0.05$: there is not enough statistical evidence to reject that the data come from $\text{Uniform}(0,1)$. It is important to note that "not rejecting" is not the same as "confirming" the null hypothesis: the KS test simply does not detect a sufficiently large discrepancy between the empirical and theoretical CDFs with this sample of size 50.
\end{solproblema}

% Create
\begin{solproblema}[prob:0d490fc]
	The normalization condition requires:
	\begin{align*}
		\int_{1}^{\infty} \frac{k}{x^2}\,dx = k\left[-\frac{1}{x}\right]_{1}^{\infty} = k(0-(-1)) = k = 1.
	\end{align*}
	Therefore $k=1$, and $f(x) = 1/x^2$ for $x\ge1$ (a continuous Pareto distribution with shape parameter $1$). The requested probability is:
	\begin{align*}
		P(1 \le X \le 2) = \int_{1}^{2} \frac{1}{x^2}\,dx = \left[-\frac{1}{x}\right]_{1}^{2} = -\frac{1}{2}+1 = \frac{1}{2}.
	\end{align*}
\end{solproblema}
