	In a group of statistics students, 60\% are women ($M$) and 40\% are men ($H$). It is known that 10\% of the women are left-handed ($Z$) and 15\% of the men are left-handed. \end{problema}

\begin{problema}

	A student is preparing for a general exam. If the student studies properly ($E$), the probability of passing ($A$) is 0.90. If the student does not study ($E'$), the probability of passing is only 0.30. The probability that the student studies is 0.70.
	\begin{enumerate}
		\item What is the total probability that the student passes the exam?
		\item If the student passed, what is the conditional probability that they studied?
		\item If the student did not pass, what is the probability that they did not study?
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:a7e87e4}
	Rigorously prove Bayes' theorem for a finite and exhaustive partition $\{B_1, B_2, \dots, B_k\}$ of the sample space $S$ given an observable event $A$ with $P(A)>0$:
	\begin{align}
		P(B_j | A) = \frac{P(A|B_j)P(B_j)}{\sum_{i=1}^k P(A|B_i)P(B_i)}, \quad \text{for every } j \in \{1, \dots, k\}.
	\end{align}
\end{problema}

% Evaluate
\begin{problema}
	\label{prob:898898e}
	\textbf{The Prosecutor's Fallacy and odds ratio:} In a forensic investigation, a biological sample is recovered at the crime scene. The extracted DNA profile has a random-match probability in the population of $\alpha = 10^{-6}$ (that is, $P(\text{match} | \text{innocent}) = 10^{-6}$). A suspect is identified through a mass search and their DNA matches the sample perfectly ($P(\text{match} | \text{guilty}) = 1$). The prosecutor tells the jury that, because the random-match probability is only one in a million, the probability that the suspect is innocent is $0.0001\%$.

	Using the odds-ratio formulation of Bayes' theorem, evaluate why the prosecutor's reasoning is wrong. If $N = 5,000,000$ people in the region could have had access to the scene and there is no other evidence against the suspect, compute the true posterior probability of guilt $P(\text{guilty} | \text{match})$.
\end{problema}

% Create
\begin{problema}
	\label{prob:3bf9f42}
	Design an original credit-risk classification problem: a bank classifies its loan applicants into three credit-history categories---Excellent, Regular, and Poor---which represent $50\%$, $35\%$, and $15\%$ of applicants, respectively. Historical default rates are $2\%$, $8\%$, and $25\%$ in each category. Compute the full vector of posterior probabilities given that a loan entered default, and identify which category is the most likely source of defaults in absolute terms, despite not being the largest category.
\end{problema}

\subsection*{Detailed solutions}

% Remember
\begin{solproblema}[prob:054ed26]
	\begin{enumerate}
		\item \textbf{No information:} the complete sample space has 6 equiprobable events $S=\{1,2,3,4,5,6\}$. The outcomes strictly less than 4 are $A=\{1,2,3\}$, so $P(A) = \frac{3}{6} = 0.50$.
		\item \textbf{With odd-number information:} the conditioned space is $I=\{1,3,5\}$. The elements of $A$ that are also in $I$ are $A \cap I = \{1,3\}$. Applying the definition of conditional probability:
		\begin{align}
			P(A|I) = \frac{P(A \cap I)}{P(I)} = \frac{2/6}{3/6} = \frac{2}{3} \approx 0.6667.
		\end{align}
	\end{enumerate}
\end{solproblema}

% Understand
\begin{solproblema}[prob:bde7784]
	Let $M$ be the event woman, with $P(M)=0.60$, and $H$ the event man, with $P(H)=0.40$. The left-handedness rates are $P(Z|M)=0.10$ and $P(Z|H)=0.15$.

	By the total probability rule, the probability that a randomly selected person is left-handed is:
	\begin{align}
		P(Z) = P(Z|M)P(M) + P(Z|H)P(H) = (0.10)(0.60) + (0.15)(0.40) = 0.060 + 0.060 = 0.120.
	\end{align}
	By Bayes' theorem:
	\begin{align}
		P(M|Z) = \frac{P(Z|M)P(M)}{P(Z)} = \frac{0.060}{0.120} = 0.50 \text{ (50\%).}
	\end{align}
	This result ($50\%$) is lower than the original proportion of women in the group ($60\%$) because \emph{observing that the person is left-handed} is evidence that relatively favors men: although there are more women overall, men have a higher left-handedness rate ($15\%$ versus $10\%$), so conditioning on "being left-handed" shifts probability toward the group with greater relative propensity for that trait, not merely toward the larger group.
\end{solproblema}

% Apply
\begin{solproblema}[prob:e14faa0]
	Let $A$ be the event of passing and $E$ the event of studying ($P(E)=0.70$, $P(E')=0.30$). The conditional exam probabilities are $P(A|E)=0.90$ and $P(A|E')=0.30$.
	\begin{enumerate}
		\item Total marginal probability of passing:
		\begin{align}
			P(A) = P(A|E)P(E) + P(A|E')P(E') = (0.90)(0.70) + (0.30)(0.30) = 0.63 + 0.09 = 0.72 \text{ (72\%).}
		\end{align}
		\item By Bayes' theorem, if the student passed, the probability that they studied is:
		\begin{align}
			P(E|A) = \frac{P(A|E)P(E)}{P(A)} = \frac{0.63}{0.72} = 0.875 \text{ (87.5\%).}
		\end{align}
		\item For $A'$: $P(A') = 1-0.72=0.28$; $P(A'|E)=0.10$, $P(A'|E')=0.70$. By Bayes over the complements:
		\begin{align}
			P(E'|A') = \frac{P(A'|E')P(E')}{P(A')} = \frac{(0.70)(0.30)}{0.28} = \frac{0.21}{0.28} = 0.75 \text{ (75\%).}
		\end{align}
	\end{enumerate}
\end{solproblema}

% Analyze
\begin{solproblema}[prob:a7e87e4]
	Let $\{B_1, B_2, \dots, B_k\}$ be an exhaustive ($\bigcup_{i=1}^k B_i = S$) and mutually exclusive partition. By the product rule of conditional probability, for any index $j$:
	\begin{align}
		P(B_j \cap A) = P(B_j | A) P(A) = P(A | B_j) P(B_j).
	\end{align}
	Solving for $P(B_j|A)$, since $P(A)>0$:
	\begin{align}
		P(B_j | A) = \frac{P(A | B_j) P(B_j)}{P(A)}. \label{eq:bayes_num_tdb}
	\end{align}
	Because the events $B_i$ form a partition of $S$, express $A$ as the disjoint union of its intersections with each cell:
	\begin{align}
		A = A \cap S = A \cap \left( \bigcup_{i=1}^k B_i \right) = \bigcup_{i=1}^k (A \cap B_i).
	\end{align}
	By Kolmogorov's third axiom (additivity for disjoint events):
	\begin{align}
		P(A) = \sum_{i=1}^k P(A \cap B_i) = \sum_{i=1}^k P(A | B_i) P(B_i). \label{eq:prob_total_k_tdb}
	\end{align}
	Substituting (\ref{eq:prob_total_k_tdb}) into (\ref{eq:bayes_num_tdb}) gives the general formula for Bayes' theorem with $k$ states.
\end{solproblema}

% Evaluate
\begin{solproblema}[prob:898898e]
	Let $I$ be the event of being the true culprit, and $E$ the DNA-profile match. The prosecutor commits a conditional fallacy by wrongly equating $P(E|I') = 10^{-6}$ with the inverse probability $P(I'|E)$.

	Evaluate the formal truth using Bayes' theorem in odds form:
	\begin{align}
		\frac{P(I|E)}{P(I'|E)} = \underbrace{\frac{P(E|I)}{P(E|I')}}_{\text{Likelihood Ratio (LR)}} \times \underbrace{\frac{P(I)}{P(I')}}_{\text{prior odds}}.
	\end{align}
	With no evidence beyond geographic location in a population of $N = 5,000,000$ inhabitants, the prior probability that an arbitrary individual is guilty is $P(I) = \frac{1}{5,000,000} = 2 \times 10^{-7}$, with $P(I') \approx 1$. The likelihood ratio is:
	\begin{align}
		\text{LR} = \frac{P(E|I)}{P(E|I')} = \frac{1}{10^{-6}} = 1,000,000.
	\end{align}
	The posterior odds are:
	\begin{align}
		\text{Odds}(I|E) = 1,000,000 \times \frac{1/5,000,000}{1} = \frac{1}{5} = 0.20.
	\end{align}
	Converting to an exact probability:
	\begin{align}
		P(I|E) = \frac{0.20}{1+0.20} = \frac{1}{6} \approx 0.1667 \text{ (16.67\%).}
	\end{align}
	The probability that the suspect is guilty is only $16.67\%$ (not $99.9999\%$ as the prosecutor claimed). The error lies in failing to consider that, when evaluating a profile in a population of 5 million where the random-match frequency is 1 in 1 million, about 5 purely random false positives are statistically expected in the population---the suspect is just one among those 5 equally compatible candidates, plus the true culprit.
\end{solproblema}

% Create
\begin{solproblema}[prob:3bf9f42]
	Let the categories be $Exc$, $Reg$, and $Def$ with $P(Exc)=0.50$, $P(Reg)=0.35$, $P(Def)=0.15$, and default rates $P(D|Exc)=0.02$, $P(D|Reg)=0.08$, and $P(D|Def)=0.25$.

	The marginal probability of default is:
	\begin{align}
		P(D) = (0.02)(0.50)+(0.08)(0.35)+(0.25)(0.15) = 0.010+0.028+0.0375 = 0.0755.
	\end{align}
	The vector of posterior probabilities is:
	\begin{align}
		P(Exc|D) &= \frac{0.010}{0.0755} \approx 0.1325 \text{ (13.25\%)} \\
		P(Reg|D) &= \frac{0.028}{0.0755} \approx 0.3709 \text{ (37.09\%)} \\
		P(Def|D) &= \frac{0.0375}{0.0755} \approx 0.4967 \text{ (49.67\%)}
	\end{align}
	The \textbf{Poor} category is the most likely source of defaults in absolute terms ($49.67\%$), despite representing only $15\%$ of applicants: its default rate ($25\%$) is so disproportionately high that it dominates the calculation, analogously to how a small but highly defective production line can produce more defective parts in absolute terms than a large but reliable line.
\end{solproblema}
