\begin{problema}
	\label{prob:2c27c8e}
	Una variable aleatoria discreta $X$ tiene como soporte el conjunto $\set{1, 2, 3, 4}$ y su función de masa de probabilidad está dada por
	\begin{align*}
		f(x) = P(X = x) = c\, x^2, \quad \text{para } x \in \set{1, 2, 3, 4},
	\end{align*}
	donde $c$ es una constante real.
	\begin{enumerate}
		\item Determine el valor exacto de la constante de normalización $c$ para que $f(x)$ sea una función de masa de probabilidad legítima.
		\item Calcule las probabilidades condicionales y marginales $P(X \ge 3)$ y $P(X \text{ es par} \mid X \ge 2)$.
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:9efeaae}
	En un proceso automático de manufactura de circuitos integrados, el número de soldaduras defectuosas por oblea, $X$, presenta la función de masa de probabilidad:
	\begin{align*}
		p_X(x) = \begin{cases}
			0.40 & \text{si } x = 0, \\
			0.30 & \text{si } x = 1, \\
			0.20 & \text{si } x = 2, \\
			0.10 & \text{si } x = 3, \\
			0 & \text{en otro caso.}
		\end{cases}
	\end{align*}
	\begin{enumerate}
		\item Escriba la función de distribución acumulada $F_X(x)$ en forma de partes (función a trozos) para todo $x \in \mathbb{R}$.
		\item Si una oblea es rechazada cuando presenta $2$ o más defectos, calcule la probabilidad de rechazo utilizando exclusivamente los valores de $F_X(x)$.
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:c324c4f}
	Sea $X$ una variable aleatoria discreta que modela el número de transacciones en error durante una hora pico en una base de datos bancaria, con soporte $\mathcal{S}_X = \set{0, 1, 2, 3, 4}$ y función de masa de probabilidad explícita dada por:
	\begin{center}
		\begin{tabular}{c|ccccc}
			\hline
			$x$ & $0$ & $1$ & $2$ & $3$ & $4$ \\
			\hline
			$f_X(x)$ & $0.35$ & $0.30$ & $0.20$ & $0.10$ & $0.05$ \\
			\hline
		\end{tabular}
	\end{center}
	\begin{enumerate}
		\item Calcule formalmente la esperanza matemática $\mu = \mathbb{E}[X]$ utilizando la definición analítica fundamental.
		\item Calcule el segundo momento no central $\mathbb{E}[X^2]$.
		\item Evalúe la varianza operacional de $X$ utilizando la identidad de König-Huygens ($\text{Var}(X) = \mathbb{E}[X^2] - \mu^2$).
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:cb2247c}
	Sea $N \in \mathbb{N}$ un entero positivo fijo. Considere la familia de variables aleatorias discretas $X_N$ con soporte $\mathcal{S} = \set{1, 2, \dots, N}$ cuya función de masa de probabilidad adopta la forma
	\begin{align*}
		f_N(x) = \frac{c_N}{x(x+1)}, \quad x \in \set{1, 2, \dots, N}.
	\end{align*}
	Demuestre formalmente, utilizando la técnica de descomposición en fracciones parciales y sumas telescópicas, que la constante de normalización requerida es exactamente $c_N = \dfrac{N+1}{N}$.
\end{problema}

\begin{problema}
	\label{prob:d332420}
	Una corporación de ingeniería civil e infraestructura evalúa la licitación de un megaproyecto minero cuya rentabilidad neta en millones de dólares es una variable aleatoria discreta $R$ con la siguiente distribución condicionada por los escenarios económicos globales:
	\begin{center}
		\begin{tabular}{l|cc}
			\hline
			\textbf{Escenario Económico} & \textbf{Probabilidad ($p_k$)} & \textbf{Rentabilidad Neta ($r_k$)} \\
			\hline
			Expansión acelerada & $0.25$ & $+120$ M USD \\
			Estabilidad moderada & $0.55$ & $+40$ M USD \\
			Recesión severa & $0.20$ & $-50$ M USD \\
			\hline
		\end{tabular}
	\end{center}
	\begin{enumerate}
		\item Determine el valor esperado del proyecto $\mathbb{E}[R]$ y la desviación estándar del retorno $\sigma_R$, interpretando esta última como medida de volatilidad o riesgo de capital.
		\item Evalúe, mediante la métrica de seguridad financiera $U = \mathbb{E}[R] - 1.25\sigma_R$, si la corporación debe aceptar la licitación cuando el umbral mínimo de aprobación es $U \ge 0$.
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:e538b1b}
	Diseñe un problema original que involucre la función de distribución acumulada (CDF) de una variable aleatoria discreta de su elección (distinto de dados, manufactura de circuitos o proyectos mineros ya vistos en esta sección). Defina la función de masa de probabilidad de su variable, construya la CDF correspondiente, y calcule el cuantil de orden $0.75$ (tercer cuartil) de su distribución.
\end{problema}

\subsection*{Soluciones detalladas}

% Recordar
\begin{solproblema}[prob:2c27c8e]
	\begin{enumerate}
		\item Toda función de masa de probabilidad debe cumplir con el axioma de normalización $\sum_{x \in \mathcal{S}_X} f(x) = 1$. Al sustituir el soporte $\mathcal{S}_X = \set{1, 2, 3, 4}$:
		\begin{align*}
			\sum_{x=1}^{4} c\, x^2 = c(1 + 4 + 9 + 16) = 30c.
		\end{align*}
		Al igualar la suma a $1$: $c = \dfrac{1}{30}$, por lo que $f(x) = \dfrac{x^2}{30}$.
		\item $P(X \ge 3) = f(3) + f(4) = \dfrac{9+16}{30} = \dfrac{5}{6}$. El evento condicionante es $B=\{X\ge2\}=\{2,3,4\}$ con $P(X\ge2)=\dfrac{4+9+16}{30}=\dfrac{29}{30}$. La intersección de $\{X \text{ par}\}$ con $B$ es $\{2,4\}$, con probabilidad $\dfrac{4+16}{30}=\dfrac{2}{3}$. Por lo tanto:
		\begin{align*}
			P(X \text{ es par} \mid X \ge 2) = \frac{20/30}{29/30} = \frac{20}{29} \approx 0.6897.
		\end{align*}
	\end{enumerate}
\end{solproblema}

\begin{solproblema}[prob:9efeaae]
	\begin{enumerate}
		\item Acumulando progresivamente las probabilidades sobre el soporte $\mathcal{S}_X = \set{0,1,2,3}$:
		\begin{align*}
			F_X(x) = \begin{cases}
				0 & \text{si } x < 0, \\
				0.40 & \text{si } 0 \le x < 1, \\
				0.70 & \text{si } 1 \le x < 2, \\
				0.90 & \text{si } 2 \le x < 3, \\
				1.00 & \text{si } x \ge 3.
			\end{cases}
		\end{align*}
		\item La probabilidad de rechazo corresponde a $P(X \ge 2) = 1 - P(X < 2) = 1 - F_X(1) = 1 - 0.70 = 0.30$. Existe un $30\%$ de probabilidad de que una oblea sea descartada por fallas de soldadura.
	\end{enumerate}
\end{solproblema}

\begin{solproblema}[prob:c324c4f]
	\begin{enumerate}
		\item $\mu = \mathbb{E}[X] = \sum x f_X(x) = 0(0.35)+1(0.30)+2(0.20)+3(0.10)+4(0.05) = 0+0.30+0.40+0.30+0.20 = 1.20$.
		\item $\mathbb{E}[X^2] = \sum x^2 f_X(x) = 0(0.35)+1(0.30)+4(0.20)+9(0.10)+16(0.05) = 0+0.30+0.80+0.90+0.80 = 2.80$.
		\item Por König-Huygens: $\text{Var}(X) = \mathbb{E}[X^2] - \mu^2 = 2.80 - 1.20^2 = 2.80 - 1.44 = 1.36$.
	\end{enumerate}
\end{solproblema}

% Analizar
\begin{solproblema}[prob:cb2247c]
	La condición de normalización exige $\sum_{x=1}^{N} f_N(x) = c_N \sum_{x=1}^{N} \dfrac{1}{x(x+1)} = 1$. Descomponemos el término general mediante fracciones parciales:
	\begin{align*}
		\frac{1}{x(x+1)} = \frac{1}{x} - \frac{1}{x+1}.
	\end{align*}
	Al sustituir en la suma, se revela una estructura telescópica donde los términos intermedios se cancelan sistemáticamente:
	\begin{align*}
		\sum_{x=1}^{N} \left( \frac{1}{x} - \frac{1}{x+1} \right) = \left( 1 - \frac{1}{2} \right) + \left( \frac{1}{2} - \frac{1}{3} \right) + \dots + \left( \frac{1}{N} - \frac{1}{N+1} \right) = 1 - \frac{1}{N+1} = \frac{N}{N+1}.
	\end{align*}
	Multiplicando por $c_N$ e igualando a $1$: $c_N \left(\dfrac{N}{N+1}\right) = 1 \implies c_N = \dfrac{N+1}{N}$, quedando demostrada la identidad.
\end{solproblema}

% Evaluar
\begin{solproblema}[prob:d332420]
	\begin{enumerate}
		\item El valor esperado es:
		\begin{align*}
			\mathbb{E}[R] = 0.25(120) + 0.55(40) + 0.20(-50) = 30 + 22 - 10 = 42 \text{ M USD}.
		\end{align*}
		El segundo momento es $\mathbb{E}[R^2] = 0.25(120^2)+0.55(40^2)+0.20(-50^2) = 0.25(14400)+0.55(1600)+0.20(2500) = 3600+880+500=4980$. Por König-Huygens:
		\begin{align*}
			\text{Var}(R) = 4980 - 42^2 = 4980 - 1764 = 3216 \implies \sigma_R = \sqrt{3216} \approx 56.71 \text{ M USD}.
		\end{align*}
		Esta desviación estándar mide qué tanto puede desviarse el retorno real respecto al valor esperado: una volatilidad de casi $57$ M USD frente a una media de $42$ M USD indica un riesgo considerable.
		\item Evaluamos el umbral de seguridad financiera:
		\begin{align*}
			U = \mathbb{E}[R] - 1.25\sigma_R = 42 - 1.25(56.71) \approx 42 - 70.89 = -28.89 \text{ M USD}.
		\end{align*}
		Como $U < 0$, \textbf{la corporación no debe aceptar la licitación}: aunque el retorno esperado es positivo, la volatilidad es tan alta que, ajustado por riesgo, el proyecto no cumple el umbral mínimo de aprobación.
	\end{enumerate}
\end{solproblema}

\begin{solproblema}[prob:e538b1b]
	\textbf{Ejemplo:} sea $X$ el número de tickets de soporte técnico abiertos por hora en un centro de atención al cliente, con soporte $\mathcal{S}_X=\{0,1,2,3,4\}$ y función de masa $f_X(0)=0.10$, $f_X(1)=0.25$, $f_X(2)=0.30$, $f_X(3)=0.25$, $f_X(4)=0.10$.

	La CDF acumulada es:
	\begin{align*}
		F_X(x) = \begin{cases}
			0 & \text{si } x<0 \\
			0.10 & \text{si } 0\le x<1 \\
			0.35 & \text{si } 1\le x<2 \\
			0.65 & \text{si } 2\le x<3 \\
			0.90 & \text{si } 3\le x<4 \\
			1.00 & \text{si } x\ge4
		\end{cases}
	\end{align*}
	El cuantil de orden $0.75$ es $q_{0.75} = \inf\{x \in \mathcal{S}_X : F_X(x)\ge0.75\}$. Como $F_X(2)=0.65<0.75$ y $F_X(3)=0.90\ge0.75$, se tiene $q_{0.75}=3$ tickets por hora.
\end{solproblema}
