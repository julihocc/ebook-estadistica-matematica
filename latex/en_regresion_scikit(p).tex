\section*{Problems}

\subsection*{Problem Statements}

% Remember
\begin{problema}
\label{prob:eaaa432}
A data scientist fits a model with \texttt{scikit-learn}:
\begin{align}
\texttt{lm = LinearRegression(); lm.fit(X\_train, y\_train)}
\end{align}
Explain in one sentence each what the following methods calculate and return: \texttt{lm.fit(X,y)}, \texttt{lm.predict(X\_new)}, and \texttt{lm.score(X,y)}. Specify which method modifies \texttt{lm}'s internal state.
\end{problema}

% Understand
\begin{problema}
\label{prob:ab54576}
An analyst has $n=250$ observations and runs \texttt{train\_test\_split(X, y, test\_size=0.2, random\_state=42)}. Calculate the exact number of training and test observations, and explain why fixing \texttt{random\_state} guarantees exact reproducibility of the split across executions.
\end{problema}

% Apply
\begin{problema}
\label{prob:7554cf7}
On a test set of $n=5$ observations, a model predicts $\hat Y=\{18.2,9.5,8.1,14.0,15.9\}$ against actual values $Y=\{17.9,10.4,7.8,14.2,15.6\}$. Calculate $R^2$ manually using $R^2=1-\text{SSE}/\text{SST}$, and explain why it must exactly match \texttt{lm.score(X\_test, y\_test)}.
\end{problema}

% Analyze
\begin{problema}
\label{prob:011e008}
\texttt{scikit-learn} does not solve \texttt{LinearRegression} by directly inverting $\mathbf X^T\mathbf X$; instead it uses a singular-value decomposition (SVD) of $\mathbf X$. Explain why both methods give the same result when $\mathbf X$ has full column rank, and explain SVD's numerical advantage when $\mathbf X^T\mathbf X$ is nearly singular because of severe multicollinearity.
\end{problema}

% Evaluate
\begin{problema}
\label{prob:5ad88f8}
\textbf{Critical analysis of the minimum-norm solution under perfect collinearity:} suppose two predictors are perfectly collinear ($X_2=2X_1$ exactly), so $\mathbf X^T\mathbf X$ is singular and the normal equations have no unique solution. Explain why \texttt{scikit-learn} (using SVD) still returns a coefficient vector without error, what that particular minimum-norm solution represents, and why it must not be interpreted as if each coefficient had an isolated causal meaning.
\end{problema}

% Create
\begin{problema}
\label{prob:0abfd8d}
Design an original cross-validation problem with \texttt{scikit-learn} (different from \texttt{fit}/\texttt{predict}/\texttt{score} and perfect collinearity): an analyst runs \texttt{cross\_val\_score(lm, X, y, cv=5, scoring='r2')} and obtains scores $[0.75,0.71,0.77,0.73,0.74]$. Calculate their mean and sample standard deviation, and explain why reporting both is more informative than the mean alone for communicating reliability to a nontechnical team.
\end{problema}

\subsection*{Detailed Solutions}

% Remember
\begin{solproblema}[prob:eaaa432]
\texttt{lm.fit(X,y)} calculates optimal least-squares coefficients and stores them in \texttt{lm.intercept\_} and \texttt{lm.coef\_}, changing the object's state; \texttt{lm.predict(X\_new)} uses those stored coefficients to calculate $\hat Y$ without changing state; \texttt{lm.score(X,y)} calculates the fitted model's $R^2$ on $(X,y)$ and also leaves state unchanged. Only \texttt{fit} modifies \texttt{lm}.
\end{solproblema}

% Understand
\begin{solproblema}[prob:ab54576]
With $n=250$ and \texttt{test\_size}=0.2, $n_{\text{test}}=50$ and $n_{\text{train}}=200$. Fixing \texttt{random\_state} initializes the pseudorandom generator with a deterministic seed, so the internal shuffling and partitioning algorithm assigns exactly the same observations to each subset every time with that seed.
\end{solproblema}

% Apply
\begin{solproblema}[prob:7554cf7]
$\bar Y=(17.9+10.4+7.8+14.2+15.6)/5=13.18$. The residual sum is $\text{SSE}=0.09+0.81+0.09+0.04+0.09=1.12$ and $\text{SST}=\sum(Y_i-\bar Y)^2\approx65.85$:
\begin{align}
R^2=1-\frac{\text{SSE}}{\text{SST}}=1-\frac{1.12}{65.85}\approx0.983.
\end{align}
This matches \texttt{lm.score(X\_test,y\_test)} because \texttt{score()} implements exactly this formula.
\end{solproblema}

% Analyze
\begin{solproblema}[prob:011e008]
When $\mathbf X$ has full column rank, $\mathbf X^T\mathbf X$ is invertible and direct inversion and SVD solve the same normal-equation system, giving identical results up to floating-point rounding. When $\mathbf X^T\mathbf X$ is nearly singular, its condition number is approximately the \textbf{square} of $\mathbf X$'s condition number, so direct inversion greatly amplifies rounding error. SVD works directly on $\mathbf X$, avoids that squaring, and is much more stable under severe multicollinearity.
\end{solproblema}

% Evaluate
\begin{solproblema}[prob:5ad88f8]
When $X_2=2X_1$, the design columns are linearly dependent, at least one singular value is zero, and $\mathbf X^T\mathbf X$ is singular. Infinitely many coefficient combinations give the same fitted values. \texttt{scikit-learn} does not fail because the Moore--Penrose pseudoinverse $\mathbf X^+$ is defined even with zero singular values; the zero terms are omitted. It returns the solution with minimum Euclidean norm among all solutions that minimize $\|\mathbf Y-\mathbf X\boldsymbol\beta\|^2$. This is numerically valid and reproducible, but individual coefficients have no isolated causal interpretation because $X_1$ and $X_2$ never vary independently.
\end{solproblema}

% Create
\begin{solproblema}[prob:0abfd8d]
\begin{align}
\bar R_{\text{CV}}^2=\frac{0.75+0.71+0.77+0.73+0.74}{5}=\frac{3.70}{5}=0.740.
\end{align}
Squared deviations from the mean are $0.0001+0.0009+0.0009+0.0001+0.0000=0.0020$. Dividing by $k-1=4$,
\begin{align}
s_{R^2}^2=\frac{0.0020}{4}=0.0005\implies s_{R^2}=\sqrt{0.0005}\approx0.02236.
\end{align}
Reporting only the mean hides consistency across data partitions: two models may share mean $R^2$ while one has highly dispersed scores and the other nearly identical scores. Reporting $s_{R^2}$ (or $\bar R_{\text{CV}}^2\pm s_{R^2}$) communicates both average performance and confidence that it will recur on new data.
\end{solproblema}
