\begin{problema}\label{prob:39c2251}
A random sample of $n=36$ students has sample mean $\bar X=78$ points on a mathematics exam. Suppose the population standard deviation is known and exactly $\sigma=12$ points.
\begin{enumerate}[label=(\alph*)]
	\item Construct a $95\%$ confidence interval for the population mean $\mu$.
	\item Construct a $99\%$ confidence interval for $\mu$.
	\item Which interval is wider? Explain the theoretical reason why width changes with confidence level.
\end{enumerate}
\end{problema}

\begin{problema}\label{prob:8fd99c8}
An LED-bulb manufacturer wants to estimate the average lifetime of a new product line. A random sample of $n=16$ bulbs gives $\bar X=1{,}200$ hours and sample standard deviation $s=80$ hours.
\begin{enumerate}[label=(\alph*)]
	\item Should the normal $Z$ distribution or Student's $t$ distribution be used? Justify your answer.
	\item Construct a $95\%$ confidence interval for the true average lifetime.
	\item Interpret the resulting interval in context.
\end{enumerate}
\end{problema}

\begin{problema}\label{prob:6af5954}
A telecommunications firm wants to determine the true average duration of calls made by corporate subscribers. From an audit of $n=100$ calls, it obtains $\bar X=4.5$ minutes and $s=2.0$ minutes.
\begin{enumerate}[label=(\alph*)]
	\item Construct the $95\%$ confidence interval for the mean duration $\mu$.
	\item Determine the current margin of error $E$.
	\item If management requires a maximum margin of error $E=0.2$ minutes at the same $95\%$ confidence level, derive and compute the minimum required sample size.
\end{enumerate}
\end{problema}

\begin{problema}\label{prob:0257f5f}
\textbf{Analytical derivation of the conservative sample-size bound for a Bernoulli proportion:} when designing a survey to estimate a population proportion $p\in(0,1)$ with maximum margin of error $E$ and confidence $(1-\alpha)$, the half-width of the asymptotic normal interval is
\[
	E=Z_{\alpha/2}\sqrt{\frac{p(1-p)}{n}}.
\]
\begin{enumerate}[label=(\alph*)]
	\item Prove with differential calculus that $g(p)=p(1-p)$ reaches its absolute maximum on $[0,1]$ at $p=0.5$.
	\item Prove that, when no prior information about $p$ is available, the formula that guarantees the margin of error will not exceed $E$ is
	\[
		n=\left\lceil\frac{Z_{\alpha/2}^2}{4E^2}\right\rceil.
	\]
	\item A data-science consulting firm needs to estimate support for a public-health initiative with maximum margin of error $E=0.03$ and $95\%$ confidence. Compute the minimum number of valid surveys required.
\end{enumerate}
\end{problema}

\begin{problema}\label{prob:a3ef454}
\textbf{Epistemological foundation of confidence intervals:} consider the frequentist normal interval
\[
	[L(X),U(X)]=\left[\bar X-Z_{\alpha/2}\frac{\sigma}{\sqrt n},\,
	\bar X+Z_{\alpha/2}\frac{\sigma}{\sqrt n}\right].
\]
Suppose that after taking the real sample of $36$ students from Problem \ref{prob:39c2251}, the numerical interval $[74.08,81.92]$ is published.
\begin{enumerate}[label=(\alph*)]
	\item Evaluate why, under strict frequentist reasoning, it is theoretically incorrect to say that ``the probability that the true mean $\mu$ lies inside $[74.08,81.92]$ is $95\%$.''
	\item Define rigorously what ``$95\%$ confidence'' means from the perspective of infinite repeated sampling, and contrast it with a Bayesian credible interval, where assigning a direct probability to the parameter given the data is valid.
\end{enumerate}
\end{problema}

\begin{problema}\label{prob:18d6b96}
Design an original problem illustrating the duality between confidence intervals and hypothesis tests, different from the grade, bulb, and call-duration examples above. A pharmaceutical laboratory claims its tablets contain exactly $500$ mg of active ingredient. An external quality-control sample of $n=20$ tablets gives $\bar X=496$ mg and $s=9$ mg. Build the $95\%$ confidence interval for the true mean content, determine whether $500$ mg is contained in it, and explain the conclusion as an implicit two-sided hypothesis test at the same significance level.
\end{problema}

\begin{solproblema}[prob:39c2251]
\begin{enumerate}[label=(\alph*)]
	\item With $Z_{0.025}=1.96$ and $\operatorname{SE}=12/\sqrt{36}=2$,
	\[
		IC_{95\%}=78\pm1.96(2)=[74.08,\ 81.92].
	\]
	\item With $Z_{0.005}=2.576$,
	\[
		IC_{99\%}=78\pm2.576(2)=[72.85,\ 83.15].
	\]
	\item The $99\%$ interval is wider. Higher confidence requires capturing more tail probability under the reference distribution, so the critical value and the interval width increase.
\end{enumerate}
\end{solproblema}

\begin{solproblema}[prob:8fd99c8]
\begin{enumerate}[label=(\alph*)]
	\item Use Student's $t$ distribution because $n=16$ is small and the population standard deviation $\sigma$ is unknown.
	\item With $df=15$ and $t_{0.025,15}=2.131$,
	\[
		IC_{95\%}=1200\pm2.131\left(\frac{80}{\sqrt{16}}\right)
		=1200\pm42.62=[1157.38,\ 1242.62].
	\]
	\item In repeated samples of $16$ bulbs, about $95\%$ of intervals constructed by this method would contain the true mean lifetime.
\end{enumerate}
\end{solproblema}

\begin{solproblema}[prob:6af5954]
\begin{enumerate}[label=(\alph*)]
	\item With $n=100$ and $Z_{0.025}=1.96$,
	\[
		IC_{95\%}=4.5\pm1.96\left(\frac{2.0}{\sqrt{100}}\right)
		=4.5\pm0.392=[4.108,\ 4.892].
	\]
	\item The current margin of error is $E=0.392$ minutes.
	\item From $E=Z_{\alpha/2}s/\sqrt n$,
	\[
		n=\left(\frac{1.96(2.0)}{0.2}\right)^2=384.16,
	\]
	so the minimum integer sample size is $385$ calls.
\end{enumerate}
\end{solproblema}

\begin{solproblema}[prob:0257f5f]
\begin{enumerate}[label=(\alph*)]
	\item For $g(p)=p-p^2$, $g'(p)=1-2p$, so the critical point is $p=0.5$. Since $g''(p)=-2<0$, this is a maximum, and $g(0.5)=0.25$.
	\item Since $p(1-p)\le0.25$,
	\[
		Z_{\alpha/2}\sqrt{\frac{p(1-p)}n}\le
		Z_{\alpha/2}\sqrt{\frac{0.25}{n}}.
	\]
	Requiring the right-hand side to be at most $E$ gives
	\[
		n\ge\frac{Z_{\alpha/2}^2}{4E^2},
	\]
	so use the ceiling of that value.
	\item With $Z_{0.025}=1.96$ and $E=0.03$,
	\[
		n=\frac{1.96^2}{4(0.03)^2}\approx1067.11,
	\]
	so at least $1068$ valid surveys are required.
\end{enumerate}
\end{solproblema}

\begin{solproblema}[prob:a3ef454]
\begin{enumerate}[label=(\alph*)]
	\item In frequentist statistics, $\mu$ is fixed, not random. Once the sample has been observed, $[74.08,81.92]$ is a fixed numerical interval. The event $\mu$ belongs to it is already either true or false, so it does not have probability $0.95$.
	\item The $95\%$ confidence level is a property of the interval-generating procedure: over infinitely many repeated samples, $95\%$ of the resulting random intervals would contain the fixed true $\mu$. In a Bayesian analysis, the parameter is treated probabilistically through a posterior distribution, so a credible interval may directly satisfy a statement such as $P(A\le\theta\le B\mid X=x)=0.95$.
\end{enumerate}
\end{solproblema}

\begin{solproblema}[prob:18d6b96]
Use a $t$ interval with $df=19$. With $t_{0.025,19}\approx2.093$,
\[
	SE=\frac9{\sqrt{20}}\approx2.012,\qquad
	E=2.093(2.012)\approx4.21.
\]
Therefore,
\[
	IC_{95\%}(\mu)=496\pm4.21=[491.79,\ 500.21].
\]
The claimed value $500$ mg is inside the interval, so the corresponding two-sided test of $H_0:\mu=500$ at $\alpha=0.05$ would not reject.
\end{solproblema}
