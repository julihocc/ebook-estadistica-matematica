\section*{Problemas}

\subsection*{Enunciados de los problemas}

% Recordar
\begin{problema}
	\label{prob:ddb67d0}
	Un fabricante de baterías afirma que la vida media de su producto es de $\mu_0=500$ horas. Un laboratorio independiente sospecha que la vida media real es \emph{menor}. Formule las hipótesis nula ($H_0$) y alternativa ($H_a$), e identifique si la prueba es de cola izquierda, derecha o de dos colas.
\end{problema}

% Comprender
\begin{problema}
	\label{prob:78013c6}
	Un examen de detección de una enfermedad rara se diseña con $H_0:$ ``el paciente está sano'' contra $H_a:$ ``el paciente está enfermo''. Describa en palabras qué representa el Error Tipo I y el Error Tipo II en este contexto médico específico, y razone cuál de los dos errores tiene consecuencias prácticas más graves.
\end{problema}

% Aplicar
\begin{problema}
	\label{prob:74dc84f}
	Una cadena de restaurantes afirma que el tiempo promedio de espera es de $\mu_0=12$ minutos. Una muestra de $n=49$ clientes reporta $\bar x=12.9$ minutos con desviación estándar poblacional conocida $\sigma=2.8$ minutos. Formule las hipótesis para una prueba de dos colas, calcule el estadístico $Z$, obtenga el valor $p$ correspondiente, y decida al nivel $\alpha=0.05$.
\end{problema}

% Analizar
\begin{problema}
	\label{prob:fad6841}
	Para la prueba $H_0:\mu=\mu_0$ contra $H_a:\mu>\mu_0$ con $\sigma$ conocida y tamaño de muestra $n$ fijo, deduzca la expresión general de la función de potencia $\text{Pot}(\mu_a)=1-\beta(\mu_a)$ como función del verdadero valor $\mu_a$, expresándola en términos de $\Phi(\cdot)$. Analice el comportamiento de $\text{Pot}(\mu_a)$ en los límites $\mu_a\to\mu_0$ y $\mu_a\to\infty$.
\end{problema}

% Evaluar
\begin{problema}
	\label{prob:66368eb}
	Un científico de datos ejecuta $m=20$ pruebas de hipótesis estadísticamente independientes sobre distintas hipótesis nulas verdaderas, cada una al nivel individual $\alpha=0.05$, sin aplicar ninguna corrección. Evalúe esta práctica calculando la Tasa de Error Familiar (FWER) resultante, y proponga y justifique el nivel de significación individual corregido $\alpha^*$ (corrección de Bonferroni) necesario para mantener el FWER global en $0.05$.
\end{problema}

% Crear
\begin{problema}
	\label{prob:d04b6da}
	Diseñe un ejemplo original (contexto, $H_0$, $H_a$, y un tipo de error más costoso) de una prueba de hipótesis, distinto de los ejemplos de baterías, restaurantes o exámenes médicos ya vistos en esta sección. Formule ambas hipótesis, identifique la dirección de la prueba, y explique cuál error (Tipo I o Tipo II) sería más grave en su contexto y por qué.
\end{problema}

\subsection*{Soluciones detalladas}

% Recordar
\begin{solproblema}[prob:ddb67d0]
	El valor afirmado por el fabricante se toma como hipótesis nula:
	\begin{align*}
		H_0: \mu=500\text{ horas}, \qquad H_a: \mu<500\text{ horas}.
	\end{align*}
	Como $H_a$ apunta hacia valores menores, es una \textbf{prueba de cola izquierda}.
\end{solproblema}

% Comprender
\begin{solproblema}[prob:78013c6]
	\textbf{Error Tipo I} (rechazar $H_0$ siendo verdadera): el examen declara \emph{enfermo} a un paciente que en realidad está \emph{sano} (falso positivo). \textbf{Error Tipo II} (no rechazar $H_0$ siendo falsa): el examen declara \emph{sano} a un paciente que en realidad está \emph{enfermo} (falso negativo). En el contexto de una enfermedad, el Error Tipo II suele ser el más grave: un paciente enfermo no recibe tratamiento oportuno, con consecuencias potencialmente irreversibles, mientras que un falso positivo usualmente solo implica pruebas de confirmación adicionales.
\end{solproblema}

% Aplicar
\begin{solproblema}[prob:74dc84f]
	$H_0:\mu=12$, $H_a:\mu\neq12$. El estadístico es $Z = \dfrac{12.9-12}{2.8/\sqrt{49}} = \dfrac{0.9}{0.4} = 2.25$. El valor $p$ (dos colas) es $p=2(1-\Phi(2.25))=2(0.01222)=0.02444$. Como $p=0.02444<\alpha=0.05$, \textbf{se rechaza} $H_0$: hay evidencia de que el tiempo promedio de espera difiere de 12 minutos.
\end{solproblema}

% Analizar
\begin{solproblema}[prob:fad6841]
	La región de rechazo es $\bar X>c$, con $c=\mu_0+Z_\alpha\sigma/\sqrt n$ bajo $H_0$. La función de potencia es:
	\begin{align*}
		\text{Pot}(\mu_a) = P(\bar X>c\mid\mu=\mu_a) = 1-\Phi\left(\frac{c-\mu_a}{\sigma/\sqrt n}\right) = 1-\Phi\left(Z_\alpha+\frac{\mu_0-\mu_a}{\sigma/\sqrt n}\right).
	\end{align*}
	Cuando $\mu_a\to\mu_0$: el argumento tiende a $Z_\alpha$, así $\text{Pot}(\mu_0)\to1-\Phi(Z_\alpha)=\alpha$ (sin efecto real, la única forma de "rechazar" es cometer un Error Tipo I). Cuando $\mu_a\to\infty$: el argumento tiende a $-\infty$, así $\text{Pot}(\mu_a)\to1$ (ante efectos arbitrariamente grandes, la prueba detecta la desviación con certeza asintótica).
\end{solproblema}

% Evaluar
\begin{solproblema}[prob:66368eb]
	Bajo independencia entre las $m$ pruebas, la probabilidad de que ninguna cometa un Error Tipo I es $(1-\alpha)^m$, así:
	\begin{align*}
		\text{FWER} = 1-(1-0.05)^{20} = 1-(0.95)^{20} \approx1-0.3585 = 0.6415.
	\end{align*}
	Aunque cada prueba controla el riesgo individual en apenas $5\%$, la probabilidad de \emph{al menos un} falso positivo en el conjunto es un alarmante $64.15\%$: la práctica sin corrección es \textbf{inadecuada}. La corrección de Bonferroni exige $\alpha^* = \alpha/m = 0.05/20 = 0.0025$ como umbral individual, lo cual acota el FWER combinado en el nivel global deseado ($\text{FWER}\le m\alpha^*=20(0.0025)=0.05$).
\end{solproblema}

% Crear
\begin{solproblema}[prob:d04b6da]
	\textbf{Ejemplo:} un sistema de detección de fraude en tarjetas de crédito se diseña con $H_0:$ ``la transacción es legítima'' contra $H_a:$ ``la transacción es fraudulenta'' (prueba de una cola, orientada a detectar el evento raro). El Error Tipo I (bloquear una transacción legítima) genera una molestia al cliente y posible pérdida de confianza en el servicio; el Error Tipo II (no detectar una transacción fraudulenta) genera una pérdida financiera directa para el banco o el cliente. En este contexto, el Error Tipo II suele considerarse más grave financieramente, aunque un exceso de Errores Tipo I (demasiados bloqueos falsos) también daña la experiencia del cliente — el diseño del sistema debe balancear ambos costos explícitamente, no minimizar solo uno de ellos.
\end{solproblema}
