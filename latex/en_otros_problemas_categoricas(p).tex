\section*{Problems}

\subsection*{Problem Statements}

% Remember
\begin{problema}
\label{prob:9556f7f}
A regression model with dummy variable \texttt{Sex\_Male} (1 for a male customer, 0 for a female customer), fitted to Monthly Income, reports
\begin{align}
\hat{\texttt{Total\_Spend}}=3570.91+0.146\,\texttt{Monthly\_Income}+231.41\,\texttt{Sex\_Male}.
\end{align}
Interpret the \texttt{Sex\_Male} coefficient and predict total spending for a male and female customer, both with monthly income \$10,000.
\end{problema}

% Understand
\begin{problema}
\label{prob:f520b9b}
A model includes an interaction between dummy variable $D$ and continuous predictor $X$: $\hat Y=\hat\beta_0+\hat\beta_1X+\hat\beta_2D+\hat\beta_3(X\times D)$. Derive the two reduced equations for $D=0$ and $D=1$, and interpret $\hat\beta_3$ specifically as the difference in slopes between groups.
\end{problema}

% Apply
\begin{problema}
\label{prob:5652f5a}
An analyst compares nested models over $n=150$ observations. Model 1 (restricted, without city dummy variables) has $R_1^2=0.68$ with $p_1=1$ predictor; Model 2 (full, with two \texttt{City Tier} dummies) has $R_2^2=0.74$ with $p_2=3$ predictors. Perform the partial $F$ test ($H_0$: both dummy coefficients are zero) at $\alpha=0.05$ ($F_{0.05,2,146}\approx3.06$), using $F_{\text{partial}}=\dfrac{(R_2^2-R_1^2)/(p_2-p_1)}{(1-R_2^2)/(n-p_2-1)}$.
\end{problema}

% Analyze
\begin{problema}
\label{prob:43c3e82}
\textbf{Formal proof of the Dummy Variable Trap:} let $\mathbf X$ be the design matrix of a model with an intercept and all $k$ dummy variables ($D_1,\ldots,D_k$) for a $k$-level categorical variable, with no reference level omitted. Prove algebraically that the columns satisfy an exact linear dependence ($\sum_{j=1}^kD_j=\mathbf1$ for every observation), and conclude why $\mathbf X^T\mathbf X$ is exactly singular and the normal equations have no unique solution.
\end{problema}

% Evaluate
\begin{problema}
\label{prob:a61b69d}
\textbf{Formal equivalence between one-way ANOVA and dummy regression:} consider one-way ANOVA with $k$ treatments, $Y_{ij}=\mu+\tau_i+\epsilon_{ij}$, and its multiple-regression form with $k-1$ dummies, $Y=\beta_0+\beta_1D_1+\cdots+\beta_{k-1}D_{k-1}+\epsilon$, where treatment $k$ is the reference. Evaluate and prove that (a) $\beta_0=\mu_k$, (b) $\beta_j=\mu_j-\mu_k$ for $j=1,\ldots,k-1$, and (c) the joint partial $F$ test of the $k-1$ dummy coefficients is algebraically identical to the global one-way ANOVA $F$ test.
\end{problema}

% Create
\begin{problema}
\label{prob:a9bca82}
Design an original two-level dummy-variable problem (different from sex, region, or city): a model predicts annual performance bonus ($Y$, thousands of dollars) from years of experience ($X$) and employee department, with Marketing as reference and $D_{\text{Eng}}=1$ for Engineering: $\hat Y=5.0+0.8X+2.5D_{\text{Eng}}$. Interpret the dummy coefficient and predict the bonus for an Engineering employee with 6 years of experience and a Marketing employee with the same experience.
\end{problema}

\subsection*{Detailed Solutions}

% Remember
\begin{solproblema}[prob:9556f7f]
The coefficient 231.41 means that, holding monthly income constant, a male customer spends on average \$231.41 more than a female customer (the reference level). At income \$10,000:
\begin{align}
\text{Male: }\hat Y=3570.91+0.146(10000)+231.41=5262.32,\qquad \text{Female: }\hat Y=3570.91+0.146(10000)=5030.91.
\end{align}
The difference is exactly the dummy coefficient, \$231.41.
\end{solproblema}

% Understand
\begin{solproblema}[prob:f520b9b]
For $D=0$, $\hat Y=\hat\beta_0+\hat\beta_1X$. For $D=1$, $\hat Y=(\hat\beta_0+\hat\beta_2)+(\hat\beta_1+\hat\beta_3)X$. The $D=0$ slope is $\hat\beta_1$ and the $D=1$ slope is $\hat\beta_1+\hat\beta_3$, so $\hat\beta_3$ is exactly the slope difference. If $\hat\beta_3=0$, the groups have parallel lines; if $\hat\beta_3\ne0$, the effect of $X$ differs by group.
\end{solproblema}

% Apply
\begin{solproblema}[prob:5652f5a]
\begin{align}
F_{\text{partial}}=\frac{(0.74-0.68)/(3-1)}{(1-0.74)/(150-3-1)}=\frac{0.03}{0.001781}\approx16.85.
\end{align}
Since $16.85>3.06$, \textbf{reject $H_0$}: the City Tier dummy variables are jointly significant; adding city category significantly improves the model.
\end{solproblema}

% Analyze
\begin{solproblema}[prob:43c3e82]
Each dummy $D_j$ equals 1 exactly when an observation belongs to level $j$ and is zero otherwise. Since levels are mutually exclusive and collectively exhaustive, every row has exactly one dummy equal to 1:
\begin{align}
\sum_{j=1}^kD_{ij}=1\quad\text{for every observation }i\implies\sum_{j=1}^k\mathbf D_j=\mathbf1_n.
\end{align}
Thus the intercept column is exactly the sum of the $k$ dummy columns in $\mathbf X=[\mathbf1_n\ \mathbf D_1\cdots\mathbf D_k]$. Therefore $\text{rank}(\mathbf X)\le k<k+1$, and the $(k+1)\times(k+1)$ matrix $\mathbf X^T\mathbf X$ is singular. The normal equations have infinitely many coefficient solutions producing the same fitted values.
\end{solproblema}

% Evaluate
\begin{solproblema}[prob:a61b69d]
For a reference-treatment observation $k$, all dummies are zero and $E[Y]=\beta_0$; under ANOVA, $E[Y_{kj}]=\mu_k$, so $\beta_0=\mu_k$. For treatment $j<k$, $E[Y]=\beta_0+\beta_j=\mu_j$, hence $\beta_j=\mu_j-\mu_k$. The hypothesis $H_0:\beta_1=\cdots=\beta_{k-1}=0$ is therefore exactly $H_0:\mu_1=\cdots=\mu_k$, the global ANOVA null. The regression explained sum of squares from the dummies equals ANOVA's treatment sum of squares, and degrees of freedom are $k-1$ and $N-k$ in both formulations. Consequently, $F_{\text{partial}}$ is exactly the same number as $F=\text{MSTR}/\text{MSE}$.
\end{solproblema}

% Create
\begin{solproblema}[prob:a9bca82]
The coefficient $\hat\beta_2=2.5$ means that, holding experience fixed, an Engineering employee receives on average \$2,500 more annual bonus than a Marketing employee (the reference group). At $X=6$:
\begin{align}
\text{Engineering: }\hat Y=5.0+0.8(6)+2.5=12.3\text{ thousand},\\
\text{Marketing: }\hat Y=5.0+0.8(6)=9.8\text{ thousand}.
\end{align}
The difference, $12.3-9.8=2.5$, is exactly the dummy coefficient.
\end{solproblema}
