\section{Solved Problems in Model Validation}

\subsection{Problem Statements}

\subsubsection*{1. Fundamental Level (Simple / Direct)}

\begin{problema}
	\label{prob:val_mod_overfitting_basico}
	A data science team fits a parametric multiple linear regression model to estimate the market value of residential properties ($Y$, in thousands of dollars) using built area ($X_1$) and built age ($X_2$) as explanatory variables. They have a historical dataset of $n = 200$ observations, which is partitioned into an $80\%$ training subset ($n_{\text{train}} = 160$) and a $20\%$ holdout testing subset ($n_{\text{test}} = 40$).
	The analytical processing yielded the following computational goodness-of-fit metrics:
	\begin{itemize}
		\item \textbf{Training set ($n_{\text{train}} = 160$):} $R_{\text{train}}^2 = 0.760$, Residual Standard Error $RSE_{\text{train}} = 24.50$ thousand dollars.
		\item \textbf{Holdout testing set ($n_{\text{test}} = 40$):} $R_{\text{test}}^2 = 0.720$, Residual Standard Error $RSE_{\text{test}} = 27.80$ thousand dollars.
	\end{itemize}
	\begin{enumerate}
		\item Empirically evaluate whether there is evidence of overfitting by comparing the gap or percentage degradation between the in-sample and out-of-sample metrics.
		\item If the average market value across the testing subset is $\bar{Y}_{\text{test}} = 250.0$ thousand dollars, compute the mean relative residual prediction error percentage with respect to the market average.
		\item Formulate an argument explaining why a simple two-way split (\emph{Holdout Method}) can be sensitive or unstable when the total sample size is moderate or small, and propose a superior technical alternative.
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:val_mod_metricas_mse_mae}
	To validate the forecasting accuracy of residential electricity demand at an urban substation ($Y$, in megawatt-hours), an engineer examines a previously calibrated regression model on an external validation sample comprising $n_{\text{test}} = 4$ completely independent random days. The true observed consumption values and the model predictions are:
	\begin{center}
		\begin{tabular}{ccc}
			\hline
			\textbf{Day ($i$)} & \textbf{True Consumption ($Y_i$)} & \textbf{Model Prediction ($\hat{Y}_i$)} \\ \hline
			1 & 120.0 & 115.0 \\
			2 & 135.0 & 138.0 \\
			3 & 110.0 & 112.0 \\
			4 & 145.0 & 141.0 \\ \hline
		\end{tabular}
	\end{center}
	\begin{enumerate}
		\item Compute the \textbf{Test Mean Squared Error (\emph{Test MSE})}:
		\begin{align}
			\text{MSE}_{\text{test}} = \frac{1}{n_{\text{test}}} \sum_{i=1}^{n_{\text{test}}} (Y_i - \hat{Y}_i)^2.
		\end{align}
		\item Compute the \textbf{Root Mean Squared Error (\emph{Test RMSE})} and the \textbf{Mean Absolute Error (\emph{Test MAE})}:
		\begin{align}
			\text{RMSE}_{\text{test}} = \sqrt{\text{MSE}_{\text{test}}}, \qquad \text{MAE}_{\text{test}} = \frac{1}{n_{\text{test}}} \sum_{i=1}^{n_{\text{test}}} |Y_i - \hat{Y}_i|.
		\end{align}
		\item Explain why the indicator $\text{RMSE}_{\text{test}}$ is invariably greater than or equal to $\text{MAE}_{\text{test}}$, and discuss what geometric penalty the former imposes when large forecasting errors occur.
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:val_mod_kfold_basico}
	An analytical firm evaluates the generalization stability of a credit risk forecasting algorithm using the \textbf{$k$-Fold Cross-Validation} methodology with $k = 5$ equal partitions across a base sample of $n = 100$ client profiles. The independent coefficient of determination ($R_{\text{CV}, j}^2$) computed on the validation fold as each fold $j$ is rotated was:
	\begin{center}
		\begin{tabular}{|c|c|c|c|c|c|}
			\hline
			\textbf{Fold ($j$)} & 1 & 2 & 3 & 4 & 5 \\ \hline
			$R_{\text{CV}, j}^2$ & 0.680 & 0.720 & 0.650 & 0.700 & 0.690 \\ \hline
		\end{tabular}
	\end{center}
	\begin{enumerate}
		\item Compute the Mean Cross-Validation Coefficient of Determination ($\bar{R}_{\text{CV}}^2$) and the unbiased sample standard deviation across the iterations ($s_{R^2}$).
		\item Construct an approximate $95\%$ confidence interval for the expected generalization performance on new data, using the critical quantile of Student's $t$ distribution with $\nu = k - 1 = 4$ degrees of freedom ($t_{0.025, \, 4} = 2.776$):
		\begin{align}
			IC_{95\%}\left( R_{\text{population}}^2 \right) = \bar{R}_{\text{CV}}^2 \pm t_{\alpha/2, \, k-1} \left( \frac{s_{R^2}}{\sqrt{k}} \right).
		\end{align}
		\item Interpret what the magnitude of the coefficient of variation ($CV = s_{R^2} / \bar{R}_{\text{CV}}^2$) reveals regarding the homogeneity of the population sample across the folds.
	\end{enumerate}
\end{problema}

\subsubsection*{2. Operational Level (Intermediate / Development)}

\begin{problema}
	\label{prob:val_mod_comparacion_10fold}
	To optimize a retail sales prediction system, two linear regression architectures compete under a rigorous $10$-Fold Cross-Validation protocol:
	\begin{itemize}
		\item \textbf{Model A (Parsimonious, $p_A = 3$ predictors):} $\bar{R}_{\text{CV}, A}^2 = 0.780 \pm 0.040$ ($\text{mean} \pm \text{standard deviation across 10 folds}$).
		\item \textbf{Model B (Complex, $p_B = 5$ predictors):} $\bar{R}_{\text{CV}, B}^2 = 0.800 \pm 0.060$ ($\text{mean} \pm \text{standard deviation across 10 folds}$).
	\end{itemize}
	\begin{enumerate}
		\item Determine which specification exhibits greater mean explanatory power and which demonstrates superior structural robustness or stability via its percentage Coefficient of Variation.
		\item If both models were to be deployed in production across a finite population of $n = 150$ stores, compute the expected Adjusted Coefficient of Determination ($R_{\text{adj}}^2$) for each when accounting for degrees of freedom penalization.
		\item Applying the \textbf{Principle of Parsimony (Occam's Razor)} and the law of diminishing returns in statistical modeling, conclusively justify which model management should adopt.
	\end{enumerate}
\end{problema}

	\begin{center}

			Residual Standard Error ($RSE$) & 15.20 & 28.70 \\
			Mean Absolute Error ($MAE$) & 12.10 & 22.50 \\ \hline
		\end{tabular}
	\end{center}
	\begin{enumerate}

		\item Perform formal differential diagnosis: does the model suffer from a pathology of \textbf{Severe Overfitting (high variance)} or \textbf{Underfitting (high bias)}? Justify according to industry diagnostic benchmarks.

	\end{enumerate}

\subsubsection*{3. Analytical Level (Advanced / Synthesis)}

\begin{problema}
	\label{prob:val_mod_aic_bic_teoria}
	To model the price-demand elasticity of public transportation in a metropolitan study comprising $n = 100$ evaluated routes, three candidate multivariable models with different levels of structural complexity were calibrated ($p$ is the total number of estimated parameters, including intercept $\beta_0$ and error variance $\sigma^2$).
	To avoid the optimistic bias of $R^2$ without resorting to sample-reducing partitions, information theorists proposed the \textbf{Akaike Information Criterion (AIC)} and the \textbf{Bayesian (Schwarz) Information Criterion (BIC)}:
	\begin{align}
		\text{AIC} = n \ln\left( \frac{\text{SSD}}{n} \right) + 2p, \qquad \text{BIC} = n \ln\left( \frac{\text{SSD}}{n} \right) + p \ln(n),
	\end{align}
	where $\text{SSD} = \sum e_i^2$ is the Residual Sum of Squares from the OLS fit.
	The optimization engine yielded the following results:
	\begin{center}
		\begin{tabular}{|l|c|c|c|}
			\hline
			\textbf{Candidate Model} & \textbf{Parameters ($p$)} & \textbf{SSD ($\sum e_i^2$)} & \textbf{$\text{SSD}/n$} \\ \hline
			Model A (Basic) & 3 & 450.00 & 4.500 \\
			Model B (Intermediate) & 5 & 380.00 & 3.800 \\
			Model C (Saturated) & 8 & 350.00 & 3.500 \\ \hline
		\end{tabular}
	\end{center}
	\begin{enumerate}
		\item Manually compute, step by step and with logarithmic precision ($\ln(4.50) = 1.5041, \ln(3.80) = 1.3350, \ln(3.50) = 1.2528, \ln(100) = 4.6052$), the exact numerical values of $\text{AIC}$ and $\text{BIC}$ for all three models.
		\item Determine which model is indisputably selected as optimal under the $\text{AIC}$ criterion and which under the $\text{BIC}$ criterion.
		\item Explain theoretically why the $\text{BIC}$ criterion applies a \textbf{much more severe asymptotic penalty} on complexity ($p$) than $\text{AIC}$ whenever sample size satisfies $n \geq 8$, and discuss what implications this has for selecting parsimonious models in Big Data applications.
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:val_mod_cp_mallows_press}
	In a labor econometrics investigation ($n = 30$ firms in the manufacturing sector), researchers seek to select the best subset of variables to predict hourly productivity ($Y$). A full master model with $p_{\text{max}} = 6$ parameters is formulated (whose unbiased fit yielded an unbiased residual variance of $\hat{\sigma}^2 = \text{MSE}_{\text{full}} = 4.00$).
	A parsimonious candidate sub-model retaining only $p = 3$ parameters and achieving an observed Residual Sum of Squares of $\text{SSD}_p = 13.00$ is evaluated.
	\begin{enumerate}
		\item Compute \textbf{Mallows' $C_p$ Statistic} for the $p=3$ parameter sub-model via the formula:
		\begin{align}
			C_p = \frac{\text{SSD}_p}{\hat{\sigma}^2} - n + 2p.
		\end{align}
		Interpret whether the evaluated model suffers from omitted variable bias by comparing the value of $C_p$ against the theoretical unbiased boundary ($C_p \approx p$).
		\item If executing leave-one-out cross-validation on this sub-model yields a prediction residual sum of squares of $\text{PRESS} = \sum_{i=1}^n (Y_i - \hat{Y}_{(-i)})^2 = 18.00$, and the sample Total Sum of Squares was $\text{SST} = 150.00$, compute the \textbf{Predictive Cross-Validation Coefficient ($Q^2$ or $R_{\text{pred}}^2$)}:
		\begin{align}
			Q^2 = 1 - \frac{\text{PRESS}}{\text{SST}}.
		\end{align}
		\item Numerically prove why, in the presence of points with high geometric leverage in the design (where the diagonal element of the hat matrix $H_{ii}$ approaches unity), the leave-one-out prediction residual violently exceeds the in-sample residual ($|Y_i - \hat{Y}_{(-i)}| \gg |Y_i - \hat{Y}_i|$), causing the indicator $Q^2$ to be drastically lower than ordinary $R^2$ ($Q^2 \ll R^2$).
	\end{enumerate}
\end{problema}

\subsubsection*{4. Challenging Level (Expert / Deep Deduction)}

\begin{problema}
	\label{prob:val_mod_deduccion_press_allen}
	\textbf{Rigorous Analytical Derivation of the Closed-Form Allen-PRESS Identity in Matrix LOOCV:}
	Consider the general linear regression model in matrix form $\mathbf{Y} = \mathbf{X}\boldsymbol{\beta} + \boldsymbol{\varepsilon}$, where $\mathbf{X}$ is the non-singular $n \times (p+1)$ design matrix. Let $\hat{\boldsymbol{\beta}}$ be the OLS estimator vector using all $n$ observations, and let $\hat{\boldsymbol{\beta}}_{(-i)}$ be the estimator vector computed after removing the $i$-th observation $(\mathbf{x}_i^T, Y_i)$ from the normal equations.
	\begin{enumerate}
		\item Knowing that the cross-product matrix without the $i$-th observation obeys rank-1 dyadic subtraction:
		\begin{align}
			\mathbf{X}_{(-i)}^T \mathbf{X}_{(-i)} = \mathbf{X}^T \mathbf{X} - \mathbf{x}_i \mathbf{x}_i^T,
		\end{align}
		apply the Sherman-Morrison-Woodbury matrix inversion formula:
		\begin{align}
			\left(\mathbf{A} - \mathbf{u}\mathbf{v}^T\right)^{-1} = \mathbf{A}^{-1} + \frac{\mathbf{A}^{-1}\mathbf{u}\mathbf{v}^T \mathbf{A}^{-1}}{1 - \mathbf{v}^T \mathbf{A}^{-1}\mathbf{u}},
		\end{align}
		to prove analytically that the inverse of the incomplete design is expressed as:
		\begin{align}
			\left(\mathbf{X}_{(-i)}^T \mathbf{X}_{(-i)}\right)^{-1} = (\mathbf{X}^T \mathbf{X})^{-1} + \frac{(\mathbf{X}^T \mathbf{X})^{-1}\mathbf{x}_i \mathbf{x}_i^T (\mathbf{X}^T \mathbf{X})^{-1}}{1 - H_{ii}},
		\end{align}
		where $H_{ii} = \mathbf{x}_i^T (\mathbf{X}^T \mathbf{X})^{-1}\mathbf{x}_i$ is the $i$-th element on the main diagonal of hat matrix $\mathbf{H}$.
		\item Starting from projection relation $\mathbf{X}_{(-i)}^T \mathbf{Y}_{(-i)} = \mathbf{X}^T \mathbf{Y} - \mathbf{x}_i Y_i$, pre-multiply this expression by the Sherman-Morrison inverse derived in part 1 to prove algebraically that the leave-one-out estimator for observation $i$ satisfies the vector identity:
		\begin{align}
			\hat{\boldsymbol{\beta}}_{(-i)} = \hat{\boldsymbol{\beta}} - \frac{(\mathbf{X}^T \mathbf{X})^{-1}\mathbf{x}_i e_i}{1 - H_{ii}},
		\end{align}
		where $e_i = Y_i - \mathbf{x}_i^T \hat{\boldsymbol{\beta}}$ is the ordinary least squares residual from the full model.
		\item By taking the inner product of the difference vector with $\mathbf{x}_i^T$, prove the \textbf{Allen-PRESS Theorem}: the leave-one-out cross-validation residual ($\hat{\varepsilon}_{(-i)} = Y_i - \hat{Y}_{(-i)}$) can be computed exactly without re-fitting any auxiliary regression as:
		\begin{align}
			\hat{\varepsilon}_{(-i)} = \frac{e_i}{1 - H_{ii}} \implies \text{PRESS} = \sum_{i=1}^n \left( \frac{e_i}{1 - H_{ii}} \right)^2.
		\end{align}
		Explain why, if a design point is a pathological geometric leverage extreme such that $H_{ii} \to 1$, its contribution to the $\text{PRESS}$ statistic surges to infinity, destroying the predictive generalization of the model.
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:val_mod_sesgo_varianza_ridge}
	\textbf{Exact Theoretical Decomposition of Generalized Mean Squared Error into Bias, Variance, and Irreducible Noise, and Optimization via Ridge Regularization:}
	Let the true stochastic process in a bi-dimensional universe be generated by the unobservable true parametric function:
	\begin{align}
		Y = f(\mathbf{x}) + \varepsilon,
	\end{align}
	where $\varepsilon$ is population measurement random noise with zero mean ($E[\varepsilon] = 0$) and constant variance $\text{Var}(\varepsilon) = E[\varepsilon^2] = \sigma_{\varepsilon}^2$, independent of variables $\mathbf{x}$.
	Let $\hat{f}(\mathbf{x})$ be a parametric regression estimator trained on a random data sample $\mathcal{D} = \{(\mathbf{x}_i, Y_i)\}_{i=1}^n$ drawn from the same generating process.
	\begin{enumerate}
		\item Prove analytically, step by step via algebraic expansion of unconditional mathematical expectation over the training set ($E_{\mathcal{D}}[\cdot]$) and over a new independent test observation $(\mathbf{x}_0, Y_0)$, that the Expected Prediction Mean Squared Error at point $\mathbf{x}_0$ decomposes orthogonally into the \textbf{Fundamental Trilogy of Statistical Learning (Bias-Variance-Noise Decomposition)}:
		\begin{align}
			E_{\mathcal{D}, \, \varepsilon}\left[ \left( Y_0 - \hat{f}(\mathbf{x}_0) \right)^2 \right] = \left( \text{Bias}\left[\hat{f}(\mathbf{x}_0)\right] \right)^2 + \text{Var}\left( \hat{f}(\mathbf{x}_0) \right) + \sigma_{\varepsilon}^2,
		\end{align}
		where the Bias and Variance of the estimated projection are defined rigorously as:
		\begin{align}
			\text{Bias}\left[\hat{f}(\mathbf{x}_0)\right] &= E_{\mathcal{D}}\left[\hat{f}(\mathbf{x}_0)\right] - f(\mathbf{x}_0), \\
			\text{Var}\left( \hat{f}(\mathbf{x}_0) \right) &= E_{\mathcal{D}}\left[ \left( \hat{f}(\mathbf{x}_0) - E_{\mathcal{D}}\left[\hat{f}(\mathbf{x}_0)\right] \right)^2 \right].
		\end{align}
		\item Consider the pure multiple linear model $f(\mathbf{x}) = \mathbf{x}^T \boldsymbol{\beta}$. Knowing that the classical OLS estimator ($\hat{\boldsymbol{\beta}}_{\text{OLS}}$) is strictly unbiased ($\text{Bias} = 0$), but that in the presence of multicollinearity or high dimensionality its variance tensor $\sigma^2 (\mathbf{X}^T \mathbf{X})^{-1}$ experiences a spectral explosion in its inverse eigenvalues, demonstrate how the \textbf{Ridge (Tikhonov) Regression estimator ($\hat{\boldsymbol{\beta}}_{\text{Ridge}} = (\mathbf{X}^T \mathbf{X} + \lambda \mathbf{I})^{-1}\mathbf{X}^T \mathbf{Y}$)} voluntarily introduces bounded parametric bias in exchange for collapsing projection variance.
		Prove algebraically why there axiomatically exists across the continuous space of the regularization hyperparameter a strict real value $\lambda^* > 0$ where the net reduction in sample variance far outweighs the square of the induced bias:
		\begin{align}
			\frac{\partial}{\partial \lambda} \left[ \text{Var}\left(\hat{Y}_{\text{Ridge}}\right) + \text{Bias}^2\left(\hat{Y}_{\text{Ridge}}\right) \right]_{\lambda = 0^+} < 0,
		\end{align}
		proving how cross-validation-validated regularization minimizes real-world generalization error.
	\end{enumerate}
\end{problema}


\subsection{Detailed Solutions}

\subsubsection*{1. Fundamental Level (Simple / Direct)}

\begin{solucion}
	\begin{enumerate}
		\item \textbf{Empirical Overfitting Diagnosis:}
		Overfitting occurs when a predictive model learns idiosyncratic random patterns or measurement noise present exclusively in the specific calibration sample, losing generality on new data from the same market. Let us evaluate the observed sample gaps:
		\begin{itemize}
			\item \textbf{Gap in Coefficient of Determination ($\Delta R^2$):}
			\begin{align}
				\Delta R^2 = R_{\text{train}}^2 - R_{\text{test}}^2 = 0.760 - 0.720 = +0.040 \quad (4.0\text{ percentage points}).
			\end{align}
			The model loses just $4.0\%$ of explained variance capacity when evaluated on out-of-sample data.
			\item \textbf{Gap in Residual Standard Error ($\Delta RSE$):}
			\begin{align}
				\text{Relative error increase} = \frac{RSE_{\text{test}} - RSE_{\text{train}}}{RSE_{\text{train}}} \times 100 = \frac{27.80 - 24.50}{24.50} \times 100 = \frac{3.30}{24.50} \times 100 \approx +13.47\%.
			\end{align}
		\end{itemize}
		In applied statistical modeling, the following diagnostic scale is standard:
		\begin{itemize}
			\item $\Delta R^2 \leq 0.05$ and $RSE$ increase $\leq 15\%$: \textbf{Mild / acceptable overfitting}, expected from normal sampling fluctuation across finite testing samples ($n_{\text{test}} = 40$).
			\item $\Delta R^2 > 0.15$ or $RSE$ increase $> 40\%$: \textbf{Severe / pathological overfitting}, indicating statistical memorization and lack of parsimony.
		\end{itemize}
		Therefore, we conclude that \textbf{no pathological overfitting exists}. The model generalizes reasonably and consistently to the global real estate market.
		
		\item Computation of mean relative predictive error percentage on the holdout testing subset:
		For an average testing property price of $\bar{Y}_{\text{test}} = 250.0$ thousand dollars and a residual dispersion of $RSE_{\text{test}} = 27.80$ thousand:
		\begin{align}
			\text{Relative Forecast Error \%} = \frac{RSE_{\text{test}}}{\bar{Y}_{\text{test}}} \times 100 = \frac{27.80}{250.00} \times 100 = 11.12\%.
		\end{align}
		Model predictions on new housing units incur an average standard deviation of just $11.12\%$ of the property's true appraisal value.
		
		\item \textbf{Structural limitations of the Holdout Method (simple 80/20 split) and a superior alternative:}
		When total sample size is moderate (such as $n = 200$), reserving a single fixed testing block of $n_{\text{test}} = 40$ creates two severe vulnerabilities:
		\begin{enumerate}
			\item \textbf{High variance or sensitivity to random partitioning:} Out-of-sample error estimation depends heavily on whether those particular 40 randomly selected properties happened to be atypical, excessively expensive, or unusually homogeneous. Another random seed could yield substantially different $R_{\text{test}}^2$ metrics.
			\item \textbf{Loss of sample power during estimation:} By withholding $20\%$ of the data, parameter vector $\boldsymbol{\beta}$ is calibrated with less information ($160$ instead of $200$), inducing a conservative bias that underestimates the true predictive capability of the model had it been trained on all available data.
		\end{enumerate}
		\textbf{Superior Alternative:} Implement a \textbf{$k$-Fold Cross-Validation protocol ($k=5$ or $k=10$)}. This method systematically rotates the sample into 5 or 10 partitions, ensuring that $100\%$ of the original data is used exactly once for holdout testing and that errors are averaged, definitively stabilizing inference regarding generalization capacity.
	\end{enumerate}
\end{solucion}

\begin{solucion}
	\begin{enumerate}
		\item \textbf{Exact computation of Test Mean Squared Error (\emph{Test MSE}):}
		We tabulate the residual errors and their squares across the $n_{\text{test}} = 4$ forecasting observations:
		\begin{center}
			\begin{tabular}{ccccc}
				\hline
				\textbf{Day ($i$)} & \textbf{True ($Y_i$)} & \textbf{Prediction ($\hat{Y}_i$)} & \textbf{Residual ($e_i = Y_i - \hat{Y}_i$)} & \textbf{Squared Residual ($e_i^2$)} \\ \hline
				1 & 120.0 & 115.0 & $+5.0$ & $25.00$ \\
				2 & 135.0 & 138.0 & $-3.0$ & $9.00$ \\
				3 & 110.0 & 112.0 & $-2.0$ & $4.00$ \\
				4 & 145.0 & 141.0 & $+4.0$ & $16.00$ \\ \hline
				\textbf{Sum} & & & & $\sum e_i^2 = \mathbf{54.00}$ \\ \hline
			\end{tabular}
		\end{center}
		Applying the mean squared error definition:
		\begin{align}
			\text{MSE}_{\text{test}} = \frac{1}{n_{\text{test}}} \sum_{i=1}^{4} e_i^2 = \frac{54.00}{4} = 13.50\text{ (MWh)}^2.
		\end{align}
		
		\item \textbf{Computation of Root Mean Squared Error (\emph{RMSE}) and Mean Absolute Error (\emph{MAE}):}
		\begin{itemize}
			\item For $\text{RMSE}_{\text{test}}$ (expressed in the exact same physical units as response $Y$, megawatt-hours):
			\begin{align}
				\text{RMSE}_{\text{test}} = \sqrt{\text{MSE}_{\text{test}}} = \sqrt{13.50} \approx 3.6742\text{ MWh}.
			\end{align}
			\item For Mean Absolute Error ($\text{MAE}_{\text{test}}$), we sum the absolute values of the residuals ($|e_1| = 5, |e_2| = 3, |e_3| = 2, |e_4| = 4 \implies \sum |e_i| = 14.00$):
			\begin{align}
				\text{MAE}_{\text{test}} = \frac{1}{n_{\text{test}}} \sum_{i=1}^{4} |e_i| = \frac{14.00}{4} = 3.5000\text{ MWh}.
			\end{align}
		\end{itemize}
		
		\item \textbf{Invariant algebraic dominance ($\text{RMSE} \geq \text{MAE}$) and geometric penalty:}
		By the classical \textbf{Cauchy-Schwarz Inequality for generalized means (or $L_p$ norms)} in Euclidean vector spaces, the quadratic ($L_2$) root-mean-square norm of any set of deviations is strictly greater than the absolute ($L_1$) mean norm, achieving equality only in the trivial pathological case where all absolute residuals are numerically constant ($|e_1| = |e_2| = \dots = |e_n|$):
		\begin{align}
			\text{RMSE}_{\text{test}} \geq \text{MAE}_{\text{test}} \quad (3.6742 > 3.5000).
		\end{align}
		
		\textbf{Geometric Penalty of RMSE:} By squaring each residual error ($e_i^2$) before taking the average, the $\text{MSE}$ metric (and consequently $\text{RMSE}$) assigns an exponentially aggressive, disproportionate weight to \textbf{extreme predictive deviations or large outliers}. For example, a single forecast error of $10\text{ MWh}$ contributes $100$ units to the sum of squares (equal to the total penalty of one hundred individual small errors of $1\text{ MWh}$).
		For this reason, $\text{RMSE}$ is used as the standard validation benchmark in engineering and finance when committing a severe forecasting error generates catastrophic operational costs, whereas $\text{MAE}$ is preferred when assessing average robust performance in distributions prone to spurious outliers.

			\left(\mathbf{A} - \mathbf{u}\mathbf{v}^T\right)^{-1} = \mathbf{A}^{-1} + \frac{\mathbf{A}^{-1}\mathbf{u}\mathbf{v}^T \mathbf{A}^{-1}}{1 - \mathbf{v}^T \mathbf{A}^{-1}\mathbf{u}}.

			\left(\mathbf{X}_{(-i)}^T \mathbf{X}_{(-i)}\right)^{-1} &= (\mathbf{X}^T \mathbf{X})^{-1} + \frac{(\mathbf{X}^T \mathbf{X})^{-1}\mathbf{x}_i \mathbf{x}_i^T (\mathbf{X}^T \mathbf{X})^{-1}}{1 - \mathbf{x}_i^T (\mathbf{X}^T \mathbf{X})^{-1}\mathbf{x}_i}.
		\end{align}
		Recognizing by definition that the scalar quadratic form in the denominator is precisely the diagonal element of the hat matrix $H_{ii} = \mathbf{x}_i^T (\mathbf{X}^T \mathbf{X})^{-1}\mathbf{x}_i$, we conclude in a single rigorous step:
		\begin{align}
			\left(\mathbf{X}_{(-i)}^T \mathbf{X}_{(-i)}\right)^{-1} = (\mathbf{X}^T \mathbf{X})^{-1} + \frac{(\mathbf{X}^T \mathbf{X})^{-1}\mathbf{x}_i \mathbf{x}_i^T (\mathbf{X}^T \mathbf{X})^{-1}}{1 - H_{ii}}. \quad (\text{Q.E.D.})
		\end{align}
		
		\item \textbf{Vector proof of the leave-one-out estimator $\hat{\boldsymbol{\beta}}_{(-i)}$:}
		We pre-multiply the derived inverse by the excluded projection vector $\mathbf{X}_{(-i)}^T \mathbf{Y}_{(-i)} = \mathbf{X}^T \mathbf{Y} - \mathbf{x}_i Y_i$:
		\begin{align}
			\hat{\boldsymbol{\beta}}_{(-i)} &= \left(\mathbf{X}_{(-i)}^T \mathbf{X}_{(-i)}\right)^{-1} \mathbf{X}_{(-i)}^T \mathbf{Y}_{(-i)} \\
			&= \left[ (\mathbf{X}^T \mathbf{X})^{-1} + \frac{(\mathbf{X}^T \mathbf{X})^{-1}\mathbf{x}_i \mathbf{x}_i^T (\mathbf{X}^T \mathbf{X})^{-1}}{1 - H_{ii}} \right] \left( \mathbf{X}^T \mathbf{Y} - \mathbf{x}_i Y_i \right).
		\end{align}
		We expand the four scalar products algebraically and factor out common vector $(\mathbf{X}^T \mathbf{X})^{-1}\mathbf{x}_i$:
		\begin{align}
			\hat{\boldsymbol{\beta}}_{(-i)} &= (\mathbf{X}^T \mathbf{X})^{-1}\mathbf{X}^T \mathbf{Y} - (\mathbf{X}^T \mathbf{X})^{-1}\mathbf{x}_i Y_i + \frac{(\mathbf{X}^T \mathbf{X})^{-1}\mathbf{x}_i \mathbf{x}_i^T (\mathbf{X}^T \mathbf{X})^{-1}\mathbf{X}^T \mathbf{Y}}{1 - H_{ii}} - \frac{(\mathbf{X}^T \mathbf{X})^{-1}\mathbf{x}_i \mathbf{x}_i^T (\mathbf{X}^T \mathbf{X})^{-1}\mathbf{x}_i Y_i}{1 - H_{ii}}.
		\end{align}
		We recognize that the first term is exactly the full-sample OLS estimator $\hat{\boldsymbol{\beta}} = (\mathbf{X}^T \mathbf{X})^{-1}\mathbf{X}^T \mathbf{Y}$. Moreover, in the third term the inner scalar product is the in-sample fitted value $\mathbf{x}_i^T \hat{\boldsymbol{\beta}} = \hat{Y}_i$, and in the fourth term we once again identify quadratic form $H_{ii}$:
		\begin{align}
			\hat{\boldsymbol{\beta}}_{(-i)} = \hat{\boldsymbol{\beta}} - (\mathbf{X}^T \mathbf{X})^{-1}\mathbf{x}_i Y_i + \frac{(\mathbf{X}^T \mathbf{X})^{-1}\mathbf{x}_i \hat{Y}_i}{1 - H_{ii}} - \frac{(\mathbf{X}^T \mathbf{X})^{-1}\mathbf{x}_i H_{ii} Y_i}{1 - H_{ii}}.
		\end{align}
		Grouping under a common denominator $(1 - H_{ii})$ all terms containing projection vector $(\mathbf{X}^T \mathbf{X})^{-1}\mathbf{x}_i$:
		\begin{align}
			\hat{\boldsymbol{\beta}}_{(-i)} &= \hat{\boldsymbol{\beta}} + (\mathbf{X}^T \mathbf{X})^{-1}\mathbf{x}_i \left[ -Y_i + \frac{\hat{Y}_i}{1 - H_{ii}} - \frac{H_{ii} Y_i}{1 - H_{ii}} \right] \\
			&= \hat{\boldsymbol{\beta}} + (\mathbf{X}^T \mathbf{X})^{-1}\mathbf{x}_i \left[ \frac{-Y_i(1 - H_{ii}) + \hat{Y}_i - H_{ii} Y_i}{1 - H_{ii}} \right] \\
			&= \hat{\boldsymbol{\beta}} + (\mathbf{X}^T \mathbf{X})^{-1}\mathbf{x}_i \left[ \frac{-Y_i + Y_i H_{ii} + \hat{Y}_i - H_{ii} Y_i}{1 - H_{ii}} \right] = \hat{\boldsymbol{\beta}} - \frac{(\mathbf{X}^T \mathbf{X})^{-1}\mathbf{x}_i (Y_i - \hat{Y}_i)}{1 - H_{ii}}.
		\end{align}
		Since the ordinary residual is $e_i = Y_i - \hat{Y}_i$, the parametric leave-one-out updating identity is proven:
		\begin{align}
			\hat{\boldsymbol{\beta}}_{(-i)} = \hat{\boldsymbol{\beta}} - \frac{(\mathbf{X}^T \mathbf{X})^{-1}\mathbf{x}_i e_i}{1 - H_{ii}}. \quad (\text{Q.E.D.})
		\end{align}
		
		\item \textbf{Proof of the Allen-PRESS Theorem:}
		To compute the forecast for excluded observation $i$, we pre-multiply the leave-one-out estimator by its design vector $\mathbf{x}_i^T$:
		\begin{align}
			\hat{Y}_{(-i)} = \mathbf{x}_i^T \hat{\boldsymbol{\beta}}_{(-i)} = \mathbf{x}_i^T \hat{\boldsymbol{\beta}} - \frac{\mathbf{x}_i^T (\mathbf{X}^T \mathbf{X})^{-1}\mathbf{x}_i e_i}{1 - H_{ii}} = \hat{Y}_i - \frac{H_{ii} e_i}{1 - H_{ii}}.
		\end{align}
		The leave-one-out cross-validation forecasting residual is obtained by subtracting this projected value from true response $Y_i$:
		\begin{align}
			\hat{\varepsilon}_{(-i)} = Y_i - \hat{Y}_{(-i)} &= Y_i - \left[ \hat{Y}_i - \frac{H_{ii} e_i}{1 - H_{ii}} \right] = (Y_i - \hat{Y}_i) + \frac{H_{ii} e_i}{1 - H_{ii}} \\
			&= e_i + \frac{H_{ii} e_i}{1 - H_{ii}} = e_i \left( 1 + \frac{H_{ii}}{1 - H_{ii}} \right) = e_i \left( \frac{1 - H_{ii} + H_{ii}}{1 - H_{ii}} \right) = \frac{e_i}{1 - H_{ii}}.
		\end{align}
		Squaring and summing across all rows $i=1,\dots,n$, we complete the analytical proof of the PRESS statistic:
		\begin{align}
			\text{PRESS} = \sum_{i=1}^n \hat{\varepsilon}_{(-i)}^2 = \sum_{i=1}^n \left( \frac{e_i}{1 - H_{ii}} \right)^2. \quad (\text{Allen Identity proven}).
		\end{align}
		
		\textbf{Catastrophic Instability from Extreme Leverage:} Geometrically, $H_{ii}$ represents Mahalanobis distance in explanatory space $\mathbf{X}$ measuring how isolated point $i$ is from the multivariable centroid of the sample. When an observation is a hyper-leverage extreme ($H_{ii} \to 1^-$), the denominator collapses to zero ($1 - H_{ii} \to 0^+$).
		This causes any non-zero elementary residual at that position ($e_i \neq 0$) to be divided by an infinitesimal quantity, inflating the predictive residual to infinity ($\hat{\varepsilon}_{(-i)} \to \infty$). Physically, because that single anomalous row is solely responsible for anchoring the tilt of the regression hyperplane in that region of space, removing it during LOOCV causes the regression plane to collapse or rotate abruptly, yielding a catastrophic generalization error at that point.
	\end{enumerate}
\end{solucion}

\begin{solucion}
	\begin{enumerate}
		\item \textbf{Analytical derivation of the Bias-Variance-Noise Decomposition:}
		For a specific point in design space $\mathbf{x}_0$, let the true generating process value be $Y_0 = f(\mathbf{x}_0) + \varepsilon_0$, where $\varepsilon_0$ is random noise with zero mean and variance $\sigma_{\varepsilon}^2$, statistically independent of calibration sample $\mathcal{D}$.
		The Expected Prediction Mean Squared Error when predicting $Y_0$ evaluated across all noise realizations and all random training sets of size $n$ is formulated as:
		\begin{align}
			E_{\mathcal{D}, \, \varepsilon_0}\left[ \left( Y_0 - \hat{f}(\mathbf{x}_0) \right)^2 \right] = E_{\mathcal{D}, \, \varepsilon_0}\left[ \left( f(\mathbf{x}_0) + \varepsilon_0 - \hat{f}(\mathbf{x}_0) \right)^2 \right].
		\end{align}
		We artificially add and subtract inside the bracket the mathematical expectation of the projection across the training set, $\bar{f}(\mathbf{x}_0) = E_{\mathcal{D}}\left[\hat{f}(\mathbf{x}_0)\right]$:
		\begin{align}
			Y_0 - \hat{f}(\mathbf{x}_0) = \left[ f(\mathbf{x}_0) - \bar{f}(\mathbf{x}_0) \right] + \left[ \bar{f}(\mathbf{x}_0) - \hat{f}(\mathbf{x}_0) \right] + \varepsilon_0.
		\end{align}
		Squaring the trinomial and applying linear expectation operator $E_{\mathcal{D}, \, \varepsilon_0}[\cdot]$, we obtain the sum of the three pure squares plus the three cross-product covariance terms:
		\begin{align}
			E_{\mathcal{D}, \, \varepsilon_0}\left[ \left( Y_0 - \hat{f}(\mathbf{x}_0) \right)^2 \right] &= E_{\mathcal{D}\left[ \left( \bar{f}(\mathbf{x}_0) - f(\mathbf{x}_0) \right)^2 \right]} + E_{\mathcal{D}}\left[ \left( \hat{f}(\mathbf{x}_0) - \bar{f}(\mathbf{x}_0) \right)^2 \right] + E_{\varepsilon_0}[\varepsilon_0^2] + \\
			&\quad 2\left( \bar{f}(\mathbf{x}_0) - f(\mathbf{x}_0) \right) E_{\mathcal{D}}\left[ \bar{f}(\mathbf{x}_0) - \hat{f}(\mathbf{x}_0) \right] + \\
			&\quad 2\left( \bar{f}(\mathbf{x}_0) - f(\mathbf{x}_0) \right) E_{\varepsilon_0}[\varepsilon_0] + 2 E_{\mathcal{D}}\left[ \bar{f}(\mathbf{x}_0) - \hat{f}(\mathbf{x}_0) \right] E_{\varepsilon_0}[\varepsilon_0].
		\end{align}
		Let us verify with absolute rigor the orthogonal cancellation of all three cross-product terms:
		\begin{itemize}
			\item By definition of conditional mean, the average deviation of an estimator from its own expectation is identically zero: $E_{\mathcal{D}}\left[ \bar{f}(\mathbf{x}_0) - \hat{f}(\mathbf{x}_0) \right] = \bar{f}(\mathbf{x}_0) - E_{\mathcal{D}}\left[\hat{f}(\mathbf{x}_0)\right] = \bar{f}(\mathbf{x}_0) - \bar{f}(\mathbf{x}_0) = 0$. Thus, the first two cross-products vanish completely.
			\item By the central hypothesis of the generating process, random error is unbiased: $E_{\varepsilon_0}[\varepsilon_0] = 0$. This cancels the third cross-product.
		\end{itemize}
		The three surviving quadratic terms represent exactly the canonical components:
		\begin{align}
			E_{\mathcal{D}, \, \varepsilon_0}\left[ \left( Y_0 - \hat{f}(\mathbf{x}_0) \right)^2 \right] &= \left[ E_{\mathcal{D}}\left[\hat{f}(\mathbf{x}_0)\right] - f(\mathbf{x}_0) \right]^2 + E_{\mathcal{D}}\left[ \left( \hat{f}(\mathbf{x}_0) - E_{\mathcal{D}}\left[\hat{f}(\mathbf{x}_0)\right] \right)^2 \right] + \sigma_{\varepsilon}^2 \\
			&= \left( \text{Bias}\left[\hat{f}(\mathbf{x}_0)\right] \right)^2 + \text{Var}\left( \hat{f}(\mathbf{x}_0) \right) + \sigma_{\varepsilon}^2. \quad (\text{Q.E.D.})
		\end{align}
		
		\item \textbf{Optimization of the Bias-Variance trade-off via Ridge Regression ($\lambda^* > 0$):}
		In OLS linear regression, estimator $\hat{\boldsymbol{\beta}}_{\text{OLS}} = (\mathbf{X}^T \mathbf{X})^{-1}\mathbf{X}^T \mathbf{Y}$ has $\text{Bias} = 0$, meaning its population predictive error consists purely of its variance plus irreducible noise. However, if design matrix $\mathbf{X}^T \mathbf{X}$ is ill-conditioned (multicollinearity), its small eigenvalues ($\gamma_{\min} \to 0^+$) cause the trace of the spatial variance tensor to explode:
		\begin{align}
			\text{Var}\left( \hat{Y}_{\text{OLS}}(\mathbf{x}_0) \right) = \sigma^2 \mathbf{x}_0^T (\mathbf{X}^T \mathbf{X})^{-1}\mathbf{x}_0 = \sigma^2 \sum_{j=1}^{p+1} \frac{(\mathbf{x}_0^T \mathbf{v}_j)^2}{\gamma_j} \to \infty.
		\end{align}
		The regularized Ridge estimator introduces a Tikhonov penalty with weight $\lambda \geq 0$:
		\begin{align}
			\hat{\boldsymbol{\beta}}_{\text{Ridge}}(\lambda) = (\mathbf{X}^T \mathbf{X} + \lambda \mathbf{I})^{-1}\mathbf{X}^T \mathbf{Y}.
		\end{align}
		Let us evaluate in matrix form the analytical behavior of squared Bias and Variance as functions of parameter $\lambda$:
		\begin{itemize}
			\item \textbf{Decreasing Variance:} The variance matrix of the regularized Ridge estimator is $\text{Var}(\hat{\boldsymbol{\beta}}_{\text{Ridge}}) = \sigma^2 (\mathbf{X}^T \mathbf{X} + \lambda \mathbf{I})^{-1} \mathbf{X}^T \mathbf{X} (\mathbf{X}^T \mathbf{X} + \lambda \mathbf{I})^{-1}$. Its derivative with respect to $\lambda$ evaluated at origin $\lambda = 0^+$ yields a strictly negative slope proportional to $-2\sigma^2 (\mathbf{X}^T \mathbf{X})^{-2}$, proving that \textbf{variance drops at an accelerated linear rate} with even a tiny injection of $\lambda$.
			\item \textbf{Increasing Quadratic Bias:} The expected value of the Ridge estimator is $E[\hat{\boldsymbol{\beta}}_{\text{Ridge}}] = (\mathbf{X}^T \mathbf{X} + \lambda \mathbf{I})^{-1}\mathbf{X}^T \mathbf{X}\boldsymbol{\beta}$, meaning vector bias is $\text{Bias}[\hat{\boldsymbol{\beta}}_{\text{Ridge}}] = -\lambda (\mathbf{X}^T \mathbf{X} + \lambda \mathbf{I})^{-1}\boldsymbol{\beta}$. When evaluating squared bias at critical point $\lambda = 0^+$, since bias itself is of order $O(\lambda)$, its square is of higher infinitesimal order $O(\lambda^2)$. Consequently, the first derivative of squared bias with respect to $\lambda$ at the origin is \textbf{exactly zero}:
			\begin{align}
				\frac{\partial}{\partial \lambda} \left[ \text{Bias}^2\left( \hat{Y}_{\text{Ridge}} \right) \right]_{\lambda = 0^+} = 0.
			\end{align}
		\end{itemize}
		Summing the two rates of change in the infinitesimal neighborhood of $\lambda = 0^+$:
		\begin{align}
			\frac{\partial}{\partial \lambda} \left[ \text{Expected Prediction MSE} \right]_{\lambda = 0^+} = \frac{\partial \text{Var}}{\partial \lambda}\Bigg|_{0^+} + \frac{\partial \text{Bias}^2}{\partial \lambda}\Bigg|_{0^+} = (\text{Strictly Negative Value}) + 0 < 0.
		\end{align}
		This demonstration proves indisputably the \textbf{Optimal Regularization Hyperparameter Existence Theorem ($\exists \lambda^* > 0$)}: for any imperfectly orthogonal linear model, introducing a controlled artificial bias via Ridge regularization always reduces sample variance by a greater amount than the added bias, minimizing the generalization loss function. $k$-Fold cross-validation serves as the empirical mechanism that explores this convex bias-variance curve to locate precisely that optimal valley $\lambda^*$.
	\end{enumerate}
\end{solucion}
