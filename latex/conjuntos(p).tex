\section*{Problemas}

\subsection{Enunciados de los problemas}

\subsubsection*{1. Nivel Fundamental (Sencillos / Directos)}

\begin{problema}
	\label{prob:2.1.1}
	En un grupo de 50 estudiantes, 35 practican fútbol, 25 practican básquetbol y 15 practican ambos deportes. ¿Cuántos estudiantes no practican ningún deporte?
\end{problema}

\begin{problema}
	\label{prob:2.1.2}
	De un grupo de 100 personas entrevistadas en un congreso internacional:
	\begin{itemize}
		\item 60 hablan inglés ($I$)
		\item 45 hablan francés ($F$)
		\item 30 hablan ambos idiomas ($I \cap F$)
	\end{itemize}
	Encuentra las siguientes cardinalidades:
	\begin{enumerate}
		\item $\card{I \cup F}$
		\item $\card{I \backslash F}$
		\item $\card{F \backslash I}$
		\item $\card{(I \cup F)'}$
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:2.1.3}
	Sea el espacio muestral $S$ y dos subconjuntos $A, B \subset S$. Enumera explícitamente los $2^2 = 4$ elementos disjuntos que conforman la partición fundamental $\particion(A,B)$ del espacio $S$.
\end{problema}

\subsubsection*{2. Nivel Operativo (Intermedios / De Desarrollo)}

\begin{problema}
	\label{prob:2.1.4}
	Extiende la construcción de una partición al caso de tres subconjuntos $A, B, C \subset S$. ¿Cuántos elementos tiene la partición $\particion(A,B,C)$ y cuáles son explícitamente estos subconjuntos elementales disjuntos?
\end{problema}

\begin{problema}
	\label{prob:2.1.5}
	En una encuesta realizada a 200 estudiantes universitarios sobre sus asignaturas preferidas, se encontró que:
	\begin{itemize}
		\item 120 estudian matemáticas
		\item 90 estudian física
		\item 70 estudian química
		\item 40 estudian matemáticas y física
		\item 30 estudian matemáticas y química
		\item 20 estudian física y química
		\item 10 estudian las tres materias simultáneamente
	\end{itemize}
	Usando la notación formal de particiones, encuentra:
	\begin{enumerate}
		\item El número de estudiantes que estudian solo matemáticas.
		\item El número de estudiantes que no estudian ninguna de las tres materias.
		\item El número de estudiantes que estudian exactamente dos materias.
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:2.1.6}
	En un club deportivo con 120 miembros inscritos, 65 practican tenis, 50 practican natación y 35 no practican ninguno de estos dos deportes. Mediante el planteamiento algebraico sobre la partición $\particion(T,N)$, determina:
	\begin{enumerate}
		\item El número de miembros que practican tanto tenis como natación.
		\item El número de miembros que practican exactamente un deporte.
	\end{enumerate}
\end{problema}

\subsubsection*{3. Nivel Analítico (Avanzados / De Conexión)}

\begin{problema}
	\label{prob:2.1.7}
	Demuestra rigurosamente que para cualesquiera dos conjuntos $A, B \subset S$:
	\begin{align}
		\card{A \cup B} = \card{A} + \card{B} - \card{A \cap B}
	\end{align}
	\emph{Sugerencia:} Expresa cada uno de los conjuntos algebraicamente en términos de las variables de la partición $\particion(A,B)$.
\end{problema}

\begin{problema}
	\label{prob:2.1.8}
	Utilizando el principio de inclusión-exclusión o la teoría de particiones, demuestra la identidad de cardinalidad para la unión de tres conjuntos $A, B, C \subset S$:
	\begin{align}
		\card{A \cup B \cup C} = \card{A} + \card{B} + \card{C} - \card{A \cap B} - \card{A \cap C} - \card{B \cap C} + \card{A \cap B \cap C}.
	\end{align}
\end{problema}

\subsubsection*{4. Nivel Desafiante (Expertos / Deducción Profunda)}

\begin{problema}
	\label{prob:2.1.9}
	Sea una colección finita de $n$ subconjuntos $A_1, A_2, \dots, A_n \subset S$.
	\begin{enumerate}
		\item Demuestra por inducción matemática o por combinatoria de indicadores que el número máximo de elementos elementales disjuntos en la partición $\particion(A_1, A_2, \dots, A_n)$ es exactamente $2^n$.
		\item Justifica formalmente bajo qué condiciones lógicas uno o más de los elementos de esta partición podrían ser el conjunto vacío ($\emptyset$) al modelar un fenómeno empírico real.
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:2.1.10}
	Un equipo de analistas en un ensayo clínico epidemiológico con un total de $\card{S} = 500$ pacientes evaluó la presencia de tres síntomas de una patología nueva ($S_1, S_2, S_3$). El informe preliminar presenta la siguiente estadística censal:
	\begin{itemize}
		\item $\card{S_1} = 300, \quad \card{S_2} = 250, \quad \card{S_3} = 200$
		\item $\card{S_1 \cap S_2} = 180, \quad \card{S_1 \cap S_3} = 150, \quad \card{S_2 \cap S_3} = 130$
		\item $\card{S_1 \cap S_2 \cap S_3} = 30$
	\end{itemize}
	Mediante el desglose algebraico en los 8 elementos de la partición $\particion(S_1, S_2, S_3)$, demuestra rigurosamente por qué el informe contiene un error lógico inevitable que imposibilita que estos datos correspondan a un censo real. Identifica específicamente cuál elemento de la partición arroja una cardinalidad matemáticamente absurda.
\end{problema}


\subsection{Soluciones detalladas}

\subsubsection*{1. Nivel Fundamental (Sencillos / Directos)}

\begin{solucion}
	Sea $F$ la colección de estudiantes que practican fútbol y $B$ el conjunto que practica básquetbol.
	A partir de la partición del espacio muestral en cuatro regiones disjuntas, definimos las cardinalidades elementales:
	\begin{align}
		x_3 &= \card{F \cap B} = 15 \\
		x_2 &= \card{F \backslash B} = \card{F} - \card{F \cap B} = 35 - 15 = 20 \\
		x_1 &= \card{B \backslash F} = \card{B} - \card{F \cap B} = 25 - 15 = 10
	\end{align}
	El total de estudiantes que practican al menos un deporte es la unión $\card{F \cup B} = x_1 + x_2 + x_3 = 10 + 20 + 15 = 45$.
	
	Por complementariedad respecto a los 50 estudiantes del grupo:
	\begin{align}
		x_0 &= \card{(F \cup B)'} = \card{S} - \card{F \cup B} = 50 - 45 = 5.
	\end{align}
	Exactamente 5 estudiantes no practican ningún deporte.
\end{solucion}

\begin{solucion}
	Usando las variables de la partición fundamental $\particion(I,F)$ donde $x_3 = \card{I \cap F}$, $x_2 = \card{I \backslash F}$, $x_1 = \card{F \backslash I}$ y $x_0 = \card{(I \cup F)'}$:
	\begin{align}
		x_3 &= \card{I \cap F} = 30 \\
		x_2 &= \card{I \backslash F} = \card{I} - x_3 = 60 - 30 = 30 \\
		x_1 &= \card{F \backslash I} = \card{F} - x_3 = 45 - 30 = 15 \\
		x_0 &= \card{S} - (x_1 + x_2 + x_3) = 100 - (15 + 30 + 30) = 25
	\end{align}
	Por consiguiente, respondemos a los incisos:
	\begin{enumerate}
		\item $\card{I \cup F} = x_1 + x_2 + x_3 = 15 + 30 + 30 = 75$ personas.
		\item $\card{I \backslash F} = x_2 = 30$ personas (hablan inglés pero no francés).
		\item $\card{F \backslash I} = x_1 = 15$ personas (hablan francés pero no inglés).
		\item $\card{(I \cup F)'} = x_0 = 25$ personas (no hablan ninguno de los dos idiomas).
	\end{enumerate}
\end{solucion}

\begin{solucion}
	Para dos subconjuntos $A, B \subset S$, cada elemento del espacio $S$ puede pertenecer o no pertenecer (complemento) a cada uno de los dos conjuntos. Esto genera $2 \times 2 = 4$ intersecciones elementales mutuamente disjuntas cuya unión totaliza al espacio $S$:
	\begin{itemize}
		\item $E_0 = A' \cap B'$ (elementos exteriores a ambos conjuntos)
		\item $E_1 = A' \cap B$ (elementos exclusivos de $B$)
		\item $E_2 = A \cap B'$ (elementos exclusivos de $A$)
		\item $E_3 = A \cap B$ (elementos en la intersección común de ambos)
	\end{itemize}
	Por construcción, se verifica que $E_i \cap E_j = \emptyset$ para todo $i \neq j$ y que $\bigcup_{k=0}^3 E_k = S$.
\end{solucion}

\subsubsection*{2. Nivel Operativo (Intermedios / De Desarrollo)}

\begin{solucion}
	Para tres subconjuntos $A, B, C \subset S$, al incluir la membresía dual o binaria respecto a un tercer conjunto $C$ o su complemento $C'$, el número de regiones se duplica respecto al caso de dos conjuntos: $2^3 = 8$ elementos elementales disjuntos.
	
	Estos 8 subconjuntos, ordenados sistemáticamente en código binario (donde pertenecer al conjunto activa un bit y pertenecer al complemento lo apaga), son:
	\begin{itemize}
		\item $E_0 = A' \cap B' \cap C'$ (ninguno de los tres conjuntos)
		\item $E_1 = A' \cap B' \cap C$ (sólamente en $C$)
		\item $E_2 = A' \cap B \cap C'$ (sólamente en $B$)
		\item $E_3 = A' \cap B \cap C$ (simultáneamente en $B$ y $C$, pero no en $A$)
		\item $E_4 = A \cap B' \cap C'$ (sólamente en $A$)
		\item $E_5 = A \cap B' \cap C$ (simultáneamente en $A$ y $C$, pero no en $B$)
		\item $E_6 = A \cap B \cap C'$ (simultáneamente en $A$ y $B$, pero no en $C$)
		\item $E_7 = A \cap B \cap C$ (en la intersección simultánea de los tres conjuntos)
	\end{itemize}
\end{solucion}

\begin{solucion}
	Sean las variables de membresía para las tres materias: $M$ (matemáticas), $F$ (física) y $Q$ (química). Representando cada intersección elemental de la partición $\particion(M,F,Q)$ por su cardinalidad $x_k = \card{E_k}$ ($k=0,1,\dots,7$):
	\begin{align}
		x_7 &= \card{M \cap F \cap Q} = 10 \\
		x_6 &= \card{M \cap F \cap Q'} = \card{M \cap F} - x_7 = 40 - 10 = 30 \\
		x_5 &= \card{M \cap F' \cap Q} = \card{M \cap Q} - x_7 = 30 - 10 = 20 \\
		x_3 &= \card{M' \cap F \cap Q} = \card{F \cap Q} - x_7 = 20 - 10 = 10 \\
		x_4 &= \card{M \cap F' \cap Q'} = \card{M} - (x_7 + x_6 + x_5) = 120 - (10 + 30 + 20) = 60 \\
		x_2 &= \card{M' \cap F \cap Q'} = \card{F} - (x_7 + x_6 + x_3) = 90 - (10 + 30 + 10) = 40 \\
		x_1 &= \card{M' \cap F' \cap Q} = \card{Q} - (x_7 + x_5 + x_3) = 70 - (10 + 20 + 10) = 30 \\
		x_0 &= \card{S} - \sum_{k=1}^7 x_k = 200 - (60 + 40 + 30 + 30 + 20 + 10 + 10) = 200 - 200 = 0
	\end{align}
	Por lo tanto:
	\begin{enumerate}
		\item Solo estudian matemáticas: $x_4 = 60$ estudiantes.
		\item No estudian ninguna de las tres materias: $x_0 = 0$ estudiantes.
		\item Estudian exactamente dos materias: $x_6 + x_5 + x_3 = 30 + 20 + 10 = 60$ estudiantes.
	\end{enumerate}
\end{solucion}

\begin{solucion}
	Sea el conjunto $T$ (tenis) y $N$ (natación), con un total de $\card{S} = 120$ miembros en el club.
	Sabemos que $x_0 = \card{(T \cup N)'} = 35$. Por tanto, el número de miembros en la unión es:
	\begin{align}
		\card{T \cup N} = \card{S} - x_0 = 120 - 35 = 85.
	\end{align}
	Usando la relación fundamental de cardinalidad para dos conjuntos:
	\begin{align}
		\card{T \cup N} &= \card{T} + \card{N} - \card{T \cap N} \\
		85 &= 65 + 50 - \card{T \cap N} \implies 85 = 115 - \card{T \cap N} \\
		\card{T \cap N} &= 115 - 85 = 30.
	\end{align}
	Las variables de la partición quedan entonces como:
	\begin{align}
		x_3 &= \card{T \cap N} = 30 \\
		x_2 &= \card{T \backslash N} = 65 - 30 = 35 \\
		x_1 &= \card{N \backslash T} = 50 - 30 = 20
	\end{align}
	\begin{enumerate}
		\item Practican tanto tenis como natación ($x_3$): 30 miembros.
		\item Practican exactamente un deporte ($x_2 + x_1$): $35 + 20 = 55$ miembros.
	\end{enumerate}
\end{solucion}

\subsubsection*{3. Nivel Analítico (Avanzados / De Conexión)}

\begin{solucion}
	Consideremos las cardinalidades de los elementos de la partición $\particion(A,B)$:
	\begin{itemize}
		\item $x_1 = \card{A' \cap B}$
		\item $x_2 = \card{A \cap B'}$
		\item $x_3 = \card{A \cap B}$
	\end{itemize}
	Dado que los conjuntos elementales en una partición son mutuamente disjuntos por construcción algebraica, la cardinalidad de cualquier unión es la suma aritmética directa de las cardinalidades de sus regiones componentes:
	\begin{align}
		\card{A} &= x_2 + x_3 \\
		\card{B} &= x_1 + x_3 \\
		\card{A \cap B} &= x_3 \\
		\card{A \cup B} &= x_1 + x_2 + x_3
	\end{align}
	Evaluando la expresión del lado derecho de la identidad que queremos demostrar:
	\begin{align}
		\card{A} + \card{B} - \card{A \cap B} &= (x_2 + x_3) + (x_1 + x_3) - x_3 \\
		&= x_1 + x_2 + 2x_3 - x_3 \\
		&= x_1 + x_2 + x_3 = \card{A \cup B}.
	\end{align}
	Lo cual demuestra de manera exacta y sin ambigüedad geométrica el teorema de adición.
\end{solucion}

\begin{solucion}
	Para tres conjuntos $A, B, C$, expresamos la unión agrupando dos términos y aplicando el teorema demostrado en el Problema \ref{prob:2.1.7}:
	\begin{align}
		\card{A \cup B \cup C} &= \card{(A \cup B) \cup C} \\
		&= \card{A \cup B} + \card{C} - \card{(A \cup B) \cap C}.
	\end{align}
	Por la ley distributiva de la intersección respecto a la unión, $(A \cup B) \cap C = (A \cap C) \cup (B \cap C)$. Aplicando nuevamente la fórmula de dos conjuntos sobre esta intersección distribuida:
	\begin{align}
		\card{(A \cup B) \cap C} &= \card{(A \cap C) \cup (B \cap C)} \\
		&= \card{A \cap C} + \card{B \cap C} - \card{(A \cap C) \cap (B \cap C)} \\
		&= \card{A \cap C} + \card{B \cap C} - \card{A \cap B \cap C}.
	\end{align}
	Sustituyendo esto y la expansión de $\card{A \cup B} = \card{A} + \card{B} - \card{A \cap B}$ en la primera ecuación principal:
	\begin{align}
		\card{A \cup B \cup C} &= [\card{A} + \card{B} - \card{A \cap B}] + \card{C} - [\card{A \cap C} + \card{B \cap C} - \card{A \cap B \cap C}] \\
		&= \card{A} + \card{B} + \card{C} - \card{A \cap B} - \card{A \cap C} - \card{B \cap C} + \card{A \cap B \cap C}.
	\end{align}
	Quedando demostrada algebraicamente la fórmula del principio de inclusión-exclusión para orden tres.
\end{solucion}

\subsubsection*{4. Nivel Desafiante (Expertos / Deducción Profunda)}

\begin{solucion}
	\begin{enumerate}
		\item \textbf{Demostración combinatorial:}
		Cada subconjunto elemental de una partición de $n$ conjuntos $A_1, A_2, \dots, A_n$ tiene la forma de intersección simultánea:
		\begin{align}
			E = X_1 \cap X_2 \cap \dots \cap X_n, \quad \text{donde cada } X_i \in \{A_i, A_i'\}.
		\end{align}
		Como para cada índice $i = 1, 2, \dots, n$ existen exactamente 2 opciones excluyentes ($A_i$ o su complemento $A_i'$), por el principio fundamental del conteo (regla del producto), el número total de combinaciones lógicas distintas es:
		\begin{align}
			\underbrace{2 \times 2 \times \cdots \times 2}_{n \text{ factores}} = 2^n.
		\end{align}
		
		\item \textbf{Justificación empírica de regiones vacías ($\emptyset$):}
		Desde un punto de vista puramente formal, $2^n$ es la cota superior combinatoria del número de regiones disjuntas. Sin embargo, al modelar variables observacionales concretas del mundo real, muchas de estas regiones resultan matemáticamente vacías ($\card{E_k} = 0$) por incompatibilidad estructural o determinística.
		Por ejemplo, si en una muestra médica $A_1$ es el conjunto de pacientes masculinos y $A_2$ el de pacientes embarazadas, la región intersección $A_1 \cap A_2$ es biológicamente imposible, por lo que su celda elemental en la partición es vacía ($\emptyset$).
	\end{enumerate}
\end{solucion}

\begin{solucion}
	Para verificar la coherencia lógica y matemática del informe, debemos calcular las variables $x_k = \card{E_k}$ ($k=0,1,\dots,7$) para la partición generada por los tres síntomas ($S_1, S_2, S_3$):
	\begin{align}
		x_7 &= \card{S_1 \cap S_2 \cap S_3} = 30 \\
		x_6 &= \card{S_1 \cap S_2 \cap S_3'} = \card{S_1 \cap S_2} - x_7 = 180 - 30 = 150 \\
		x_5 &= \card{S_1 \cap S_2' \cap S_3} = \card{S_1 \cap S_3} - x_7 = 150 - 30 = 120 \\
		x_3 &= \card{S_1' \cap S_2 \cap S_3} = \card{S_2 \cap S_3} - x_7 = 130 - 30 = 100
	\end{align}
	Ahora, deduzcamos el número de pacientes que poseen exclusivamente el primer síntoma ($x_4 = \card{S_1 \cap S_2' \cap S_3'}$):
	\begin{align}
		\card{S_1} &= x_7 + x_6 + x_5 + x_4 \\
		300 &= 30 + 150 + 120 + x_4 \\
		300 &= 300 + x_4 \implies x_4 = 0 \text{ (hasta aquí no hay contradicción).}
	\end{align}
	Deducimos ahora el número de pacientes con exclusivamente el segundo síntoma ($x_2 = \card{S_1' \cap S_2 \cap S_3'}$):
	\begin{align}
		\card{S_2} &= x_7 + x_6 + x_3 + x_2 \\
		250 &= 30 + 150 + 100 + x_2 \\
		250 &= 280 + x_2 \implies x_2 = -30.
	\end{align}
	\textbf{Conclusión:} El cálculo arroja una cardinalidad negativa ($x_2 = -30$) para la región de pacientes que sólo manifiestan el síntoma $S_2$. En la axiomática del conteo y la teoría de conjuntos de Kolmogorov, la función de cardinalidad $\card{\cdot} \ge 0$ no admite valores negativos. Esto demuestra sin lugar a dudas que las cifras publicadas en el informe preliminar son inconsistentes entre sí y reflejan un error en la recolección o tabulación censal de las intersecciones.
\end{solucion}
