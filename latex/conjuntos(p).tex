\section*{Problemas}

\subsection{Enunciados de los problemas}

\begin{problema}
	Extiende la construcción de una partición al caso de tres subconjuntos $A, B, C \subset S$. ¿Cuántos elementos tiene la partición $\particion(A,B,C)$?
\end{problema}

\begin{problema}
	En una encuesta a 200 estudiantes, se encontró que:
	\begin{itemize}
		\item 120 estudian matemáticas
		\item 90 estudian física
		\item 70 estudian química
		\item 40 estudian matemáticas y física
		\item 30 estudian matemáticas y química
		\item 20 estudian física y química
		\item 10 estudian las tres materias
	\end{itemize}
	
	Usando la notación de particiones, encuentra:
	\begin{enumerate}
		\item El número de estudiantes que estudian solo matemáticas
		\item El número de estudiantes que no estudian ninguna de las tres materias
		\item El número de estudiantes que estudian exactamente dos materias
	\end{enumerate}
\end{problema}

\begin{problema}
	De un grupo de 100 personas:
	\begin{itemize}
		\item 60 hablan inglés
		\item 45 hablan francés
		\item 30 hablan ambos idiomas
	\end{itemize}
	
	Encuentra:
	\begin{enumerate}
		\item $\card{I \cup F}$
		\item $\card{I \backslash F}$
		\item $\card{F \backslash I}$
		\item $\card{(I \cup F)'}$
	\end{enumerate}
\end{problema}

\begin{problema}
	Demuestra que para cualesquiera dos conjuntos $A, B \subset S$:
	\begin{align}
		\card{A \cup B} = \card{A} + \card{B} - \card{A \cap B}
	\end{align}
	
	\emph{Sugerencia:} Expresa cada conjunto en términos de la partición $\particion(A,B)$.
\end{problema}

\begin{problema}
	En un grupo de 50 estudiantes, 35 practican fútbol, 25 practican básquetbol y 15 practican ambos deportes. ¿Cuántos estudiantes no practican ningún deporte?
\end{problema}


\subsection{Soluciones detalladas}

\begin{solucion}
	Para tres conjuntos $A, B, C$, la partición tiene $2^3 = 8$ elementos:
	\begin{itemize}
		\item $E_0 = A' \cap B' \cap C'$
		\item $E_1 = A' \cap B' \cap C$
		\item $E_2 = A' \cap B \cap C'$
		\item $E_3 = A' \cap B \cap C$
		\item $E_4 = A \cap B' \cap C'$
		\item $E_5 = A \cap B' \cap C$
		\item $E_6 = A \cap B \cap C'$
		\item $E_7 = A \cap B \cap C$
	\end{itemize}
\end{solucion}

\begin{solucion}
	Sea $M$ = matemáticas, $F$ = física, $Q$ = química. Usando la partición de 8 elementos:
	
	\begin{align}
		x_7 &= \card{M \cap F \cap Q} = 10 \\
		x_6 &= \card{M \cap F \cap Q'} = 40 - 10 = 30 \\
		x_5 &= \card{M \cap F' \cap Q} = 30 - 10 = 20 \\
		x_3 &= \card{M' \cap F \cap Q} = 20 - 10 = 10 \\
		x_4 &= \card{M \cap F' \cap Q'} = 120 - 30 - 20 - 10 = 60 \\
		x_2 &= \card{M' \cap F \cap Q'} = 90 - 30 - 10 - 10 = 40 \\
		x_1 &= \card{M' \cap F' \cap Q} = 70 - 20 - 10 - 10 = 30 \\
		x_0 &= \card{M' \cap F' \cap Q'} = 200 - (60+30+20+10+40+30+10) = 0
	\end{align}
	
	\begin{enumerate}
		\item Solo matemáticas: $x_4 = 60$
		\item Ninguna materia: $x_0 = 0$
		\item Exactamente dos materias: $x_6 + x_5 + x_3 = 30 + 20 + 10 = 60$
	\end{enumerate}
\end{solucion}

\begin{solucion}
	Usando la partición $\particion(I,F)$ con $x_0, x_1, x_2, x_3$:
	
	\begin{align}
		x_3 &= \card{I \cap F} = 30 \\
		x_2 &= \card{I \backslash F} = 60 - 30 = 30 \\
		x_1 &= \card{F \backslash I} = 45 - 30 = 15 \\
		x_0 &= 100 - (30 + 30 + 15) = 25
	\end{align}
	
	\begin{enumerate}
		\item $\card{I \cup F} = x_1 + x_2 + x_3 = 15 + 30 + 30 = 75$
		\item $\card{I \backslash F} = x_2 = 30$
		\item $\card{F \backslash I} = x_1 = 15$
		\item $\card{(I \cup F)'} = x_0 = 25$
	\end{enumerate}
\end{solucion}

\begin{solucion}
	Usando la partición:
	\begin{align}
		\card{A} &= x_2 + x_3 \\
		\card{B} &= x_1 + x_3 \\
		\card{A \cap B} &= x_3 \\
		\card{A \cup B} &= x_1 + x_2 + x_3
	\end{align}
	
	Entonces:
	\begin{align}
		\card{A} + \card{B} - \card{A \cap B} &= (x_2 + x_3) + (x_1 + x_3) - x_3 \\
		&= x_1 + x_2 + x_3 = \card{A \cup B}
	\end{align}
\end{solucion}

\begin{solucion}
	Sea $F$ = fútbol, $B$ = básquetbol.
	
	\begin{align}
		x_3 &= \card{F \cap B} = 15 \\
		x_2 &= \card{F \backslash B} = 35 - 15 = 20 \\
		x_1 &= \card{B \backslash F} = 25 - 15 = 10 \\
		x_0 &= 50 - (20 + 10 + 15) = 5
	\end{align}
	
	Por lo tanto, 5 estudiantes no practican ningún deporte.
\end{solucion}
