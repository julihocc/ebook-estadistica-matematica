\begin{problema}
	\label{prob:293fd20}
	The Central Limit Theorem states that the sum of independent variables will converge to the normal distribution whenever the second-order moment ($\sigma^2$) is finite. Evaluate what happens when samples are drawn from a population with extremely heavy tails: the standard Cauchy distribution, whose density is $f(x) = \frac{1}{\pi(1+x^2)}$ for $x \in (-\infty, \infty)$.
	\begin{enumerate}
		\item Prove or formally argue why the integral defining the theoretical expectation $E(X)$ does not converge absolutely for the Cauchy distribution.
		\item If a random sample of size $n$ is drawn from a Cauchy population, what is the density function or behavior of the sample mean $\bar{X}_n$? Does convergence toward the normal occur as $n \to \infty$? Judge the practical implication for an analyst who blindly assumes the CLT on any data set.
	\end{enumerate}
\end{problema}

\subsection*{Detailed solutions}

% Remember
\begin{solproblema}[prob:c0cebb4]
	The standard error of the mean ($\sigma_{\bar{X}}$) measures the dispersion of sample means around $\mu$ and is given by $\sigma_{\bar{X}} = \frac{\sigma}{\sqrt{n}}$. With $\sigma = 12$:
	\begin{enumerate}
		\item For $n = 9$: $\sigma_{\bar{X}} = \frac{12}{\sqrt{9}} = 4.00$ min.
		\item For $n = 36$: $\sigma_{\bar{X}} = \frac{12}{\sqrt{36}} = 2.00$ min.
		\item For $n = 144$: $\sigma_{\bar{X}} = \frac{12}{\sqrt{144}} = 1.00$ min.
		\item For $n = 576$: $\sigma_{\bar{X}} = \frac{12}{\sqrt{576}} = 0.50$ min.
	\end{enumerate}
	\textbf{Comment on the speed of reduction:} the dispersion decreases strictly in inverse proportion to the square root of the sample size ($\sqrt{n}$). To reduce the standard error by half (for example, from 4.0 to 2.0, or from 2.0 to 1.0), it is necessary to quadruple the inspected sample size ($9 \to 36 \to 144$).
\end{solproblema}

% Understand
\begin{solproblema}[prob:116017b]
	\begin{enumerate}
		\item \textbf{Sampling with replacement:} in each of the $n = 3$ draws there are exactly $N = 6$ available options, since the selected element is returned to the population before the next selection. By the multiplicative counting principle:
		\begin{align}
			N^n = 6^3 = 216 \text{ samples.}
		\end{align}
		\item \textbf{Sampling without replacement (unordered):} the total number of possible subsets is the binomial coefficient:
		\begin{align}
			\binom{N}{n} = \binom{6}{3} = \frac{6!}{3!(6-3)!} = 20 \text{ samples.}
		\end{align}
		\item \textbf{Asymptotic equivalence when $N \to \infty$:} when drawing a fixed sample size $n$ without replacement, the probability of selecting a specific element on the first draw is $1/N$, on the second draw is $1/(N-1)$, and on the $k$-th draw is $1/(N-k+1)$. As $N \to \infty$ with $n$ constant, the ratios $\frac{N-k}{N} \to 1$. Therefore, population depletion becomes negligible and sampling without replacement converges in distribution to sampling with replacement (independent sampling).
	\end{enumerate}
\end{solproblema}

% Apply
\begin{solproblema}[prob:2da8cc3]
	Data: $\sigma = 8$ ml, maximum tolerated margin of error $E = 1.5$ ml, $95\%$ confidence level ($\alpha = 0.05$, $z_{\alpha/2}=1.96$). The margin of error for estimating the mean is $E = z_{\alpha/2} \cdot \sigma/\sqrt{n}$. Solving for the minimum sample size:
	\begin{align}
		n = \left( \frac{z_{\alpha/2} \cdot \sigma}{E} \right)^2 = \left( \frac{1.96 \times 8}{1.5} \right)^2 \approx (10.4533)^2 \approx 109.27.
	\end{align}
	Since $n$ must be an integer and at least $95\%$ certainty is required, always round up to the next integer:
	\begin{align}
		n_{\text{min}} = 110 \text{ bottles.}
	\end{align}
\end{solproblema}

% Analyze
\begin{solproblema}[prob:0c980d4]
	We want to find $E(S^2)$ where $S^2 = \frac{1}{n-1} \sum_{i=1}^n (X_i - \bar{X})^2$. Add and subtract the true population mean $\mu$ inside the parentheses:
	\begin{align}
		\sum_{i=1}^n (X_i - \bar{X})^2 &= \sum_{i=1}^n [(X_i - \mu) - (\bar{X} - \mu)]^2 \\
		&= \sum_{i=1}^n (X_i - \mu)^2 - 2(\bar{X} - \mu) \sum_{i=1}^n (X_i - \mu) + n(\bar{X} - \mu)^2.
	\end{align}
	Since $\sum_{i=1}^n (X_i - \mu) = n\bar{X} - n\mu = n(\bar{X} - \mu)$, substituting into the middle term gives:
	\begin{align}
		\sum_{i=1}^n (X_i - \bar{X})^2 &= \sum_{i=1}^n (X_i - \mu)^2 - 2n(\bar{X} - \mu)^2 + n(\bar{X} - \mu)^2 = \sum_{i=1}^n (X_i - \mu)^2 - n(\bar{X} - \mu)^2.
	\end{align}
	Applying expectation to both sides, with $E[(X_i - \mu)^2] = \sigma^2$ and $E[(\bar{X} - \mu)^2] = \sigma^2/n$:
	\begin{align}
		E\left[ \sum_{i=1}^n (X_i - \bar{X})^2 \right] = n\sigma^2 - n\left(\frac{\sigma^2}{n}\right) = n\sigma^2 - \sigma^2 = (n-1)\sigma^2.
	\end{align}
	Finally, dividing by $(n-1)$:
	\begin{align}
		E(S^2) = \frac{(n-1)\sigma^2}{n-1} = \sigma^2.
	\end{align}
	\textbf{Conclusion:} if one divided by $n$, the estimator would have expectation $\frac{n-1}{n}\sigma^2$, systematically underestimating the population variance. Dividing by the $n-1$ degrees of freedom corrects this bias exactly.
\end{solproblema}

% Evaluate
\begin{solproblema}[prob:293fd20]
	\begin{enumerate}
		\item \textbf{Nonexistence of moments:} the expectation of $X$ with density $f(x) = \frac{1}{\pi(1+x^2)}$ is given by $E(X) = \frac{1}{\pi}\int_{-\infty}^{\infty} \frac{x}{1+x^2}dx$. For the integral to converge in the Lebesgue sense, the positive and negative parts must be absolutely finite separately:
		\begin{align}
			\int_0^{\infty} \frac{x}{1+x^2} dx = \lim_{b \to \infty} \frac{1}{2}\ln(1+b^2) = \infty.
		\end{align}
		Since the integral over $[0,\infty)$ diverges (and analogously over $(-\infty,0]$), the total improper integral is an indeterminate form $\infty - \infty$. Therefore, $E(X)$ (and even more so $\sigma^2$) \textbf{does not exist}.

		\item \textbf{Distribution of the sample mean:} the characteristic function of a standard Cauchy is $\varphi_X(t) = e^{-|t|}$. For $\bar{X}_n = \frac{1}{n}\sum X_k$:
		\begin{align}
			\varphi_{\bar{X}_n}(t) = \prod_{k=1}^n \varphi_X\left(\frac{t}{n}\right) = \exp\left(-n\frac{|t|}{n}\right) = e^{-|t|}.
		\end{align}
		This characteristic function is identical to that of a single observation, so $\bar{X}_n \sim \text{Cauchy}(0,1)$ for every $n$: \textbf{there is no convergence toward the normal} and no variance reduction. An analyst who blindly assumes the CLT (for example, applying the "$n\ge30$ rule" without verifying that $\sigma^2<\infty$) to extremely heavy-tailed data---common in financial returns or network latencies---would construct completely invalid confidence intervals and hypothesis tests, since averaging more data does not reduce the estimation error at all.
	\end{enumerate}
\end{solproblema}

% Create
\begin{solproblema}[prob:b7567ec]
	\textbf{Example:} the diameter of screws produced by a CNC machine follows a distribution with population mean $\mu = 10$ mm and standard deviation $\sigma = 0.5$ mm. A random sample of $n = 64$ screws is taken, and we want to compute $P(9.9 < \bar{X} < 10.1)$.

	The standard error of the mean is $\sigma_{\bar{X}} = \sigma/\sqrt{n} = 0.5/\sqrt{64} = 0.0625$ mm. Standardizing the limits:
	\begin{align}
		z_1 = \frac{9.9-10}{0.0625} = -1.6, \qquad z_2 = \frac{10.1-10}{0.0625} = 1.6.
	\end{align}
	By the CLT, $\bar{X}$ is approximately normal, so:
	\begin{align}
		P(9.9 < \bar{X} < 10.1) \approx P(-1.6 < Z < 1.6) = 2\Phi(1.6) - 1 = 2(0.9452) - 1 = 0.8904 \text{ (89.04\%).}
	\end{align}
\end{solproblema}
