\subsection{Soluciones detalladas}

\subsubsection*{1. Nivel Fundamental (Sencillos / Directos)}

\begin{solproblema}[prob:1.2.1]
	Desarrollando término a término cada sumatoria según los índices indicados:
	\begin{enumerate}
		\item $\displaystyle\sum_{j=0}^{6}X_{j} = X_{0} + X_{1} + X_{2} + X_{3} + X_{4} + X_{5} + X_{6}$
		\item $\displaystyle\sum_{k=1}^{4}\left( Y_{k}-3 \right)^{2} = (Y_{1}-3)^{2} + (Y_{2}-3)^{2} + (Y_{3}-3)^{2} + (Y_{4}-3)^{2}$
		\item $\displaystyle\sum_{k=1}^{N}a = \underbrace{a + a + \cdots + a}_{N \text{ veces}} = N \cdot a$
		\item $\displaystyle\sum_{n=2}^{5}{f_{n}}X_{n} = f_{2}X_{2} + f_{3}X_{3} + f_{4}X_{4} + f_{5}X_{5}$
		\item $\displaystyle\sum_{m=0}^{3}\left( X_{m}-a \right) = (X_{0}-a) + (X_{1}-a) + (X_{2}-a) + (X_{3}-a)$
	\end{enumerate}
\end{solproblema}

\begin{solproblema}[prob:1.2.2]
	Los restantes números son $100 - (20 + 40 + 30) = 10$ números con valor $7$.
	Utilizando la fórmula de la media ponderada por frecuencias:
	\begin{align}
		\bar{X} &= \dfrac{\sum f_{i}X_{i}}{N} = \dfrac{20(4) + 40(5) + 30(6) + 10(7)}{100} \\
		&= \dfrac{80 + 200 + 180 + 70}{100} = \dfrac{530}{100} = 5.30
	\end{align}
	La media aritmética de los 100 números es $5.30$.
\end{solproblema}

\begin{solproblema}[prob:1.2.3]
	El peso medio combinado de varios grupos se obtiene mediante la media ponderada, donde los pesos de ponderación son los tamaños de los grupos ($n_{i}$):
	\begin{align}
		\bar{X} &= \dfrac{\sum n_{i}\bar{X}_{i}}{\sum n_{i}} = \dfrac{15(162) + 20(148) + 10(153) + 18(140)}{15 + 20 + 10 + 18} \\
		&= \dfrac{2430 + 2960 + 1530 + 2520}{63} = \dfrac{9440}{63} \approx 149.84 \text{ libras.}
	\end{align}
\end{solproblema}

\subsubsection*{2. Nivel Operativo (Intermedios / De Desarrollo)}

\begin{solproblema}[prob:1.2.4]
	A partir de la tabla \ref{tab:1.2.1}, determinamos el punto medio o marca de clase ($X_{i}$) para cada intervalo de estatura (en pulgadas) y multiplicamos por su frecuencia $f_{i}$:
	\begin{itemize}
		\item Intervalo 60--62: $X_{1} = 61$, $f_{1} = 5 \implies f_{1}X_{1} = 305$
		\item Intervalo 63--65: $X_{2} = 64$, $f_{2} = 18 \implies f_{2}X_{2} = 1152$
		\item Intervalo 66--68: $X_{3} = 67$, $f_{3} = 42 \implies f_{3}X_{3} = 2814$
		\item Intervalo 69--71: $X_{4} = 70$, $f_{4} = 27 \implies f_{4}X_{4} = 1890$
		\item Intervalo 72--74: $X_{5} = 73$, $f_{5} = 8 \implies f_{5}X_{5} = 584$
	\end{itemize}
	Sumando las frecuencias y los productos:
	\begin{align}
		N &= \sum f_{i} = 5 + 18 + 42 + 27 + 8 = 100 \\
		\sum f_{i}X_{i} &= 305 + 1152 + 2814 + 1890 + 584 = 6745
	\end{align}
	Por lo tanto, la estatura media es:
	\begin{align}
		\bar{X} = \dfrac{6745}{100} = 67.45 \text{ pulgadas.}
	\end{align}
\end{solproblema}

\begin{solproblema}[prob:1.2.7]
	Por definición de media aritmética para la variable $Z$:
	\begin{align}
		\bar{Z} &= \dfrac{1}{N}\sum_{i=1}^{N} Z_{i} = \dfrac{1}{N}\sum_{i=1}^{N} (X_{i} + Y_{i}) \\
		&= \dfrac{1}{N}\left( \sum_{i=1}^{N}X_{i} + \sum_{i=1}^{N}Y_{i} \right) \\
		&= \dfrac{1}{N}\sum_{i=1}^{N}X_{i} + \dfrac{1}{N}\sum_{i=1}^{N}Y_{i} = \bar{X} + \bar{Y}.
	\end{align}
	Esto demuestra que la media aritmética del operador suma es lineal.
\end{solproblema}

\begin{solproblema}[prob:1.2.8]
	Los $N=8$ datos son: $5, 8, 11, 9, 12, 6, 14, 10$.
	
	a) Con pivote $P=9$, calculamos las desviaciones $d_{i} = X_{i} - 9$:
	\begin{align}
		d_{1} &= 5-9 = -4, & d_{2} &= 8-9 = -1, & d_{3} &= 11-9 = 2, & d_{4} &= 9-9 = 0 \\
		d_{5} &= 12-9 = 3, & d_{6} &= 6-9 = -3, & d_{7} &= 14-9 = 5, & d_{8} &= 10-9 = 1
	\end{align}
	La suma de desviaciones es $\sum d_{i} = -4 - 1 + 2 + 0 + 3 - 3 + 5 + 1 = 3$.
	Aplicando la fórmula con pivote:
	\begin{align}
		\bar{X} = P + \dfrac{\sum d_{i}}{N} = 9 + \dfrac{3}{8} = 9 + 0.375 = 9.375.
	\end{align}

	b) Con pivote $P=20$, las desviaciones $d_{i} = X_{i} - 20$ son:
	\begin{align}
		-15, -12, -9, -11, -8, -14, -6, -10 \implies \sum d_{i} = -85.
	\end{align}
	Aplicando la fórmula:
	\begin{align}
		\bar{X} = 20 + \dfrac{-85}{8} = 20 - 10.625 = 9.375.
	\end{align}
	El resultado es invariante respecto a la elección del pivote.
\end{solproblema}

\subsubsection*{3. Nivel Analítico (Avanzados / De Conexión)}

\begin{solproblema}[prob:1.2.5]
	Partimos de la definición de desviación respecto al pivote: $d_{i} = X_{i} - P$. Despejando la variable original tenemos $X_{i} = P + d_{i}$.
	
	Sumando desde $i=1$ hasta $N$ en ambos lados de la igualdad:
	\begin{align}
		\sum_{i=1}^{N} X_{i} &= \sum_{i=1}^{N} (P + d_{i}) = \sum_{i=1}^{N} P + \sum_{i=1}^{N} d_{i} = N \cdot P + \sum_{i=1}^{N} d_{i}.
	\end{align}
	Dividiendo toda la expresión entre $N$:
	\begin{align}
		\dfrac{\sum_{i=1}^{N} X_{i}}{N} &= \dfrac{N \cdot P}{N} + \dfrac{\sum_{i=1}^{N} d_{i}}{N} \implies \bar{X} = P + \dfrac{\sum_{i=1}^{N} d_{i}}{N}.
	\end{align}
	Lo cual completa la demostración.
\end{solproblema}

\begin{solproblema}[prob:1.2.6]
	Sea $d_{i} = X_{i} - \bar{X}$. Sumando para todos los $N$ elementos del conjunto:
	\begin{align}
		\sum_{i=1}^{N} d_{i} &= \sum_{i=1}^{N} (X_{i} - \bar{X}) = \sum_{i=1}^{N} X_{i} - \sum_{i=1}^{N} \bar{X}.
	\end{align}
	Como $\bar{X}$ es una constante respecto al índice de sumatoria $i$, $\sum_{i=1}^{N} \bar{X} = N\bar{X}$.
	Además, por definición de media aritmética, $\sum_{i=1}^{N} X_{i} = N\bar{X}$.
	
	Sustituyendo ambos términos:
	\begin{align}
		\sum_{i=1}^{N} d_{i} = N\bar{X} - N\bar{X} = 0.
	\end{align}
	Esta propiedad demuestra que la media aritmética es el centro de gravedad de la distribución de datos.
\end{solproblema}

\subsubsection*{4. Nivel Desafiante (Expertos / Deducción Profunda)}

\begin{solproblema}[prob:1.2.9]
	La marca de clase del intervalo central (66--68) en la tabla \ref{tab:1.2.1} es $P = 67$ pulgadas. Como las clases tienen un ancho constante $c = 3$ pulgadas, podemos usar la variable codificada $u_{i} = \dfrac{X_{i} - P}{c}$:
	\begin{itemize}
		\item Cl. 60--62 ($X_{1}=61$): $u_{1} = \frac{61-67}{3} = -2 \implies f_{1}u_{1} = 5(-2) = -10$
		\item Cl. 63--65 ($X_{2}=64$): $u_{2} = \frac{64-67}{3} = -1 \implies f_{2}u_{2} = 18(-1) = -18$
		\item Cl. 66--68 ($X_{3}=67$): $u_{3} = \frac{67-67}{3} = 0 \implies f_{3}u_{3} = 42(0) = 0$
		\item Cl. 69--71 ($X_{4}=70$): $u_{4} = \frac{70-67}{3} = 1 \implies f_{4}u_{4} = 27(1) = 27$
		\item Cl. 72--74 ($X_{5}=73$): $u_{5} = \frac{73-67}{3} = 2 \implies f_{5}u_{5} = 8(2) = 16$
	\end{itemize}
	Sumando las frecuencias por el código:
	\begin{align}
		\sum f_{i}u_{i} = -10 - 18 + 0 + 27 + 16 = 15.
	\end{align}
	Aplicando la fórmula abreviada de la media con cambio de escala y origen:
	\begin{align}
		\bar{X} = P + c\left( \dfrac{\sum f_{i}u_{i}}{N} \right) = 67 + 3\left( \dfrac{15}{100} \right) = 67 + 3(0.15) = 67 + 0.45 = 67.45 \text{ pulgadas.}
	\end{align}
	El resultado coincide exactamente con el cálculo directo del Problema \ref{prob:1.2.4}, reduciendo significativamente la complejidad aritmética.
\end{solproblema}

\begin{solproblema}[prob:1.2.10]
	A partir de la tabla \ref{tab:1.2.2} de pesos (en libras) de los $N=100$ estudiantes, determinamos primero las frecuencias acumuladas ($F_{i}$) para ubicar la clase mediana (donde se alcanza la posición $N/2 = 50$):
	\begin{itemize}
		\item 118--126: $f_{1} = 3 \implies F_{1} = 3$
		\item 127--135: $f_{2} = 5 \implies F_{2} = 8$
		\item 136--144: $f_{3} = 9 \implies F_{3} = 17$
		\item 145--153: $f_{4} = 12 \implies F_{4} = 29$
		\item 154--162: $f_{5} = 5 \implies F_{5} = 34$ (Nota: analizando la tabla completa del intervalo mediano, ubicamos la clase donde $F_{i} \ge 50$).
	\end{itemize}
	Supongamos de la distribución que el intervalo mediano es aquel que contiene a la observación número 50. Sea $L_{m}$ el límite real inferior de la clase mediana, $F_{m-1}$ la frecuencia acumulada anterior a la clase mediana, $f_{m}$ la frecuencia de la clase mediana y $c$ la amplitud del intervalo.
	
	La fórmula de interpolación lineal para la mediana ($Med$) en datos agrupados es:
	\begin{align}
		Med = L_{m} + \left( \dfrac{\dfrac{N}{2} - F_{m-1}}{f_{m}} \right) c.
	\end{align}
	Al sustituir los valores exactos de la tabla \ref{tab:1.2.2}, se obtiene el peso mediano interpolado, el cual divide a la población escolar en dos mitades equitativas (el 50\% de los estudiantes pesa menos de este valor y el 50\% pesa más).
\end{solproblema}
