\begin{problema}
	\label{prob:2c27c8e}
	A discrete random variable $X$ has support $\set{1, 2, 3, 4}$ and probability mass function
	\begin{align*}
		f(x) = P(X = x) = c\, x^2, \quad \text{for } x \in \set{1, 2, 3, 4},
	\end{align*}
	where $c$ is a real constant.
	\begin{enumerate}
		\item Determine the exact value of the normalization constant $c$ so that $f(x)$ is a legitimate probability mass function.
		\item Compute the marginal and conditional probabilities $P(X \ge 3)$ and $P(X \text{ is even} \mid X \ge 2)$.
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:9efeaae}
	In an automatic integrated-circuit manufacturing process, the number of defective solder joints per wafer, $X$, has probability mass function:
	\begin{align*}
		p_X(x) = \begin{cases}
			0.40 & \text{if } x = 0, \\
			0.30 & \text{if } x = 1, \\
			0.20 & \text{if } x = 2, \\
			0.10 & \text{if } x = 3, \\
			0 & \text{otherwise.}
		\end{cases}
	\end{align*}
	\begin{enumerate}
		\item Write the cumulative distribution function $F_X(x)$ in piecewise form for all $x \in \mathbb{R}$.
		\item If a wafer is rejected when it has $2$ or more defects, compute the rejection probability using only the values of $F_X(x)$.
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:c324c4f}
	Let $X$ be a discrete random variable that models the number of erroneous transactions during a peak hour in a banking database, with support $\mathcal{S}_X = \set{0, 1, 2, 3, 4}$ and explicit probability mass function:
	\begin{center}
		\begin{tabular}{c|ccccc}
			\hline
			$x$ & $0$ & $1$ & $2$ & $3$ & $4$ \\
			\hline
			$f_X(x)$ & $0.35$ & $0.30$ & $0.20$ & $0.10$ & $0.05$ \\
			\hline
		\end{tabular}
	\end{center}
	\begin{enumerate}
		\item Formally compute the mathematical expectation $\mu = \mathbb{E}[X]$ using the fundamental analytical definition.
		\item Compute the second raw moment $\mathbb{E}[X^2]$.
		\item Evaluate the operational variance of $X$ using the Konig-Huygens identity ($\text{Var}(X) = \mathbb{E}[X^2] - \mu^2$).
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:cb2247c}
	Let $N \in \mathbb{N}$ be a fixed positive integer. Consider the family of discrete random variables $X_N$ with support $\mathcal{S} = \set{1, 2, \dots, N}$ whose probability mass function has the form
	\begin{align*}
		f_N(x) = \frac{c_N}{x(x+1)}, \quad x \in \set{1, 2, \dots, N}.
	\end{align*}
	Prove formally, using partial fractions and telescoping sums, that the required normalization constant is exactly $c_N = \dfrac{N+1}{N}$.
\end{problema}

\begin{problema}
	\label{prob:d332420}
	A civil-engineering and infrastructure corporation is evaluating a bid for a mining megaproject whose net profitability in millions of dollars is a discrete random variable $R$ with the following distribution conditioned on global economic scenarios:
	\begin{center}
		\begin{tabular}{l|cc}
			\hline
			\textbf{Economic Scenario} & \textbf{Probability ($p_k$)} & \textbf{Net Profitability ($r_k$)} \\
			\hline
			Accelerated expansion & $0.25$ & $+120$ M USD \\
			Moderate stability & $0.55$ & $+40$ M USD \\
			Severe recession & $0.20$ & $-50$ M USD \\
			\hline
		\end{tabular}
	\end{center}
	\begin{enumerate}
		\item Determine the expected value of the project $\mathbb{E}[R]$ and the standard deviation of return $\sigma_R$, interpreting the latter as a measure of volatility or capital risk.
		\item Evaluate, using the financial safety metric $U = \mathbb{E}[R] - 1.25\sigma_R$, whether the corporation should accept the bid when the minimum approval threshold is $U \ge 0$.
	\end{enumerate}
\end{problema}

\subsection*{Detailed solutions}

% Remember
\begin{solproblema}[prob:2c27c8e]
	\begin{enumerate}
		\item Every probability mass function must satisfy the normalization axiom $\sum_{x \in \mathcal{S}_X} f(x) = 1$. Substituting the support $\mathcal{S}_X = \set{1, 2, 3, 4}$:
		\begin{align*}
			\sum_{x=1}^{4} c\, x^2 = c(1 + 4 + 9 + 16) = 30c.
		\end{align*}
		Equating the sum to $1$: $c = \dfrac{1}{30}$, so $f(x) = \dfrac{x^2}{30}$.
		\item $P(X \ge 3) = f(3) + f(4) = \dfrac{9+16}{30} = \dfrac{5}{6}$. The conditioning event is $B=\{X\ge2\}=\{2,3,4\}$ with $P(X\ge2)=\dfrac{4+9+16}{30}=\dfrac{29}{30}$. The intersection of $\{X \text{ even}\}$ with $B$ is $\{2,4\}$, with probability $\dfrac{4+16}{30}=\dfrac{2}{3}$. Therefore:
		\begin{align*}
			P(X \text{ is even} \mid X \ge 2) = \frac{20/30}{29/30} = \frac{20}{29} \approx 0.6897.
		\end{align*}
	\end{enumerate}
\end{solproblema}

\begin{solproblema}[prob:9efeaae]
	\begin{enumerate}
		\item Accumulating the probabilities progressively over the support $\mathcal{S}_X = \set{0,1,2,3}$:
		\begin{align*}
			F_X(x) = \begin{cases}
				0 & \text{if } x < 0, \\
				0.40 & \text{if } 0 \le x < 1, \\
				0.70 & \text{if } 1 \le x < 2, \\
				0.90 & \text{if } 2 \le x < 3, \\
				1.00 & \text{if } x \ge 3.
			\end{cases}
		\end{align*}
		\item The rejection probability is $P(X \ge 2) = 1 - P(X < 2) = 1 - F_X(1) = 1 - 0.70 = 0.30$. There is a $30\%$ probability that a wafer is discarded because of soldering failures.
	\end{enumerate}
\end{solproblema}

% Apply
\begin{solproblema}[prob:c324c4f]
	\begin{enumerate}
		\item $\mu = \mathbb{E}[X] = \sum x f_X(x) = 0(0.35)+1(0.30)+2(0.20)+3(0.10)+4(0.05) = 0+0.30+0.40+0.30+0.20 = 1.20$.
		\item $\mathbb{E}[X^2] = \sum x^2 f_X(x) = 0(0.35)+1(0.30)+4(0.20)+9(0.10)+16(0.05) = 0+0.30+0.80+0.90+0.80 = 2.80$.
		\item By Konig-Huygens: $\text{Var}(X) = \mathbb{E}[X^2] - \mu^2 = 2.80 - 1.20^2 = 2.80 - 1.44 = 1.36$.
	\end{enumerate}
\end{solproblema}

% Analyze
\begin{solproblema}[prob:cb2247c]
	The normalization condition requires $\sum_{x=1}^{N} f_N(x) = c_N \sum_{x=1}^{N} \dfrac{1}{x(x+1)} = 1$. Decompose the general term using partial fractions:
	\begin{align*}
		\frac{1}{x(x+1)} = \frac{1}{x} - \frac{1}{x+1}.
	\end{align*}
	Substitution in the sum reveals a telescoping structure where the intermediate terms cancel systematically:
	\begin{align*}
		\sum_{x=1}^{N} \left( \frac{1}{x} - \frac{1}{x+1} \right) = \left( 1 - \frac{1}{2} \right) + \left( \frac{1}{2} - \frac{1}{3} \right) + \dots + \left( \frac{1}{N} - \frac{1}{N+1} \right) = 1 - \frac{1}{N+1} = \frac{N}{N+1}.
	\end{align*}
	Multiplying by $c_N$ and setting equal to $1$: $c_N \left(\dfrac{N}{N+1}\right) = 1 \implies c_N = \dfrac{N+1}{N}$, proving the identity.
\end{solproblema}

% Evaluate
\begin{solproblema}[prob:d332420]
	\begin{enumerate}
		\item The expected value is:
		\begin{align*}
			\mathbb{E}[R] = 0.25(120) + 0.55(40) + 0.20(-50) = 30 + 22 - 10 = 42 \text{ M USD}.
		\end{align*}
		The second moment is $\mathbb{E}[R^2] = 0.25(120^2)+0.55(40^2)+0.20(-50^2) = 0.25(14400)+0.55(1600)+0.20(2500) = 3600+880+500=4980$. By Konig-Huygens:
		\begin{align*}
			\text{Var}(R) = 4980 - 42^2 = 4980 - 1764 = 3216 \implies \sigma_R = \sqrt{3216} \approx 56.71 \text{ M USD}.
		\end{align*}
		This standard deviation measures how much the actual return may deviate from the expected value: volatility of almost $57$ M USD against a mean of $42$ M USD indicates substantial risk.
		\item Evaluate the financial safety threshold:
		\begin{align*}
			U = \mathbb{E}[R] - 1.25\sigma_R = 42 - 1.25(56.71) \approx 42 - 70.89 = -28.89 \text{ M USD}.
		\end{align*}
		Since $U < 0$, \textbf{the corporation should not accept the bid}: although the expected return is positive, volatility is so high that, after risk adjustment, the project does not meet the minimum approval threshold.
	\end{enumerate}
\end{solproblema}

\begin{solproblema}[prob:e538b1b]
	\textbf{Example:} let $X$ be the number of technical-support tickets opened per hour in a customer-service center, with support $\mathcal{S}_X=\{0,1,2,3,4\}$ and mass function $f_X(0)=0.10$, $f_X(1)=0.25$, $f_X(2)=0.30$, $f_X(3)=0.25$, $f_X(4)=0.10$.

	The cumulative CDF is:
	\begin{align*}
		F_X(x) = \begin{cases}
			0 & \text{if } x<0 \\
			0.10 & \text{if } 0\le x<1 \\
			0.35 & \text{if } 1\le x<2 \\
			0.65 & \text{if } 2\le x<3 \\
			0.90 & \text{if } 3\le x<4 \\
			1.00 & \text{if } x\ge4
		\end{cases}
	\end{align*}
	The order-$0.75$ quantile is $q_{0.75} = \inf\{x \in \mathcal{S}_X : F_X(x)\ge0.75\}$. Since $F_X(2)=0.65<0.75$ and $F_X(3)=0.90\ge0.75$, we have $q_{0.75}=3$ tickets per hour.
\end{solproblema}
