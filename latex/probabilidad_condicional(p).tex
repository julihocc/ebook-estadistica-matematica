\section*{Problemas}

\subsection{Enunciados de los problemas}

\subsubsection*{1. Nivel Fundamental (Sencillos / Directos)}

\begin{problema}
	\label{prob:2.4.1}
	Se lanzan dos dados balanceados simultáneamente. Sea $A$ el evento de que la suma sea al menos 9, y sea $B$ el evento de que el primer dado muestre un número par.
	\begin{enumerate}
		\item Calcula $P(A)$
		\item Calcula $P(B)$
		\item Calcula $P(A|B)$
		\item ¿Son $A$ y $B$ eventos independientes?
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:2.4.2}
	Una urna contiene 5 bolas rojas y 3 bolas azules. Se extraen dos bolas sucesivamente sin reemplazo. Sea $A$ = \texttt{``la primera bola es roja''} y $B$ = \texttt{``la segunda bola es roja''}.
	\begin{enumerate}
		\item Calcula $P(A)$
		\item Calcula $P(B|A)$
		\item Calcula $P(B|A')$
		\item Calcula $P(B)$ usando la regla de la probabilidad total.
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:2.4.3}
	Sean dos eventos $A, B \subset S$ tales que $P(A) = 0.60$, $P(B) = 0.50$ y la probabilidad conjunta es $P(A \cap B) = 0.30$. Calcula directamente:
	\begin{enumerate}
		\item $P(A|B)$ y $P(B|A)$
		\item $P(A \cap B')$
		\item $P(A|B')$
	\end{enumerate}
\end{problema}

\subsubsection*{2. Nivel Operativo (Intermedios / De Desarrollo)}

\begin{problema}
	\label{prob:2.4.4}
	En una planta automatizada, el 70\% de las piezas procesadas provienen de la línea de producción A y el 30\% de la línea B. La línea A tiene una tasa de defectos del 5\%, mientras que la línea B tiene una tasa de defectos del 10\%. Si el equipo de control de calidad selecciona una pieza al azar:
	\begin{enumerate}
		\item ¿Cuál es la probabilidad total de que la pieza sea defectuosa?
		\item Si la pieza seleccionada resulta defectuosa, ¿cuál es la probabilidad condicional de que provenga de la línea A?
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:2.4.5}
	Un proveedor de hardware tecnológico distribuye microcontroladores de dos plantas: tipo A con probabilidad 0.60 y tipo B con probabilidad 0.40. Los componentes tipo A fallan en las pruebas térmicas con probabilidad del 2\%, mientras que los del tipo B fallan con el 5\%. Si un ingeniero selecciona un microcontrolador al azar y falla la prueba térmica, ¿cuál es la probabilidad de que haya sido ensamblado en la planta A?
\end{problema}

\begin{problema}
	\label{prob:2.4.6}
	En un equipo de ciencia de datos, el 60\% de los scripts de análisis están desarrollados en \texttt{Python}, el 30\% en \texttt{R} y el 10\% en \texttt{C++}. La probabilidad de que un script pase todas las pruebas de integración en su primera ejecución es de 0.85 si fue escrito en \texttt{Python}, de 0.75 si fue escrito en \texttt{R} y de 0.95 si fue escrito en \texttt{C++}. Si se audita un script al azar y se comprueba que superó las pruebas de integración a la primera, ¿cuál es la probabilidad de que haya sido escrito en \texttt{Python}?
\end{problema}

\subsubsection*{3. Nivel Analítico (Avanzados / De Conexión)}

\begin{problema}
	\label{prob:2.4.7}
	Se sabe que el 80\% de los estudiantes del departamento de estadística aprueban el examen general de conocimientos. De los estudiantes que aprueban, el 90\% asistió regularmente al seminario de preparación. De los que no aprueban, sólo el 40\% asistió regularmente a dicho seminario.
	\begin{enumerate}
		\item Si se selecciona un estudiante al azar y aprobó el examen, ¿cuál es la probabilidad condicional de que haya asistido regularmente?
		\item Si un estudiante confiesa no haber asistido regularmente, ¿cuál es la probabilidad de que haya aprobado el examen?
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:2.4.8}
	\begin{enumerate}
		\item Demuestra que si $P(B|A) = P(B)$, entonces los eventos $A$ y $B$ son estadísticamente independientes.
		\item Demuestra que si $A$ y $B$ son independientes y $0 < P(A) < 1$, entonces sus eventos complementarios $A'$ y $B'$ también son mutuamente independientes.
	\end{enumerate}
\end{problema}

\subsubsection*{4. Nivel Desafiante (Expertos / Deducción Profunda)}

\begin{problema}
	\label{prob:2.4.9}
	\begin{enumerate}
		\item Demuestra formalmente la Regla General de la Cadena (o de la Multiplicación) para una secuencia finita de $n$ eventos con $P(A_1 \cap \dots \cap A_{n-1}) > 0$:
		\begin{align}
			P(A_1 \cap A_2 \cap \dots \cap A_n) = P(A_1)P(A_2|A_1)P(A_3|A_1 \cap A_2) \cdots P(A_n|A_1 \cap \dots \cap A_{n-1}).
		\end{align}
		\item Utiliza esta regla para calcular la probabilidad exacta de que al extraer 4 cartas sucesivamente y sin reemplazo de una baraja inglesa de 52 cartas, las 4 pertenezcan a palos diferentes.
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:2.4.10}
	\textbf{La Paradoja de Monty Hall:} En un programa de concursos, un participante se encuentra ante tres puertas cerradas ($P_1, P_2, P_3$). Detrás de exactamente una puerta está el premio principal y detrás de las otras dos hay cabras. El participante elige una puerta al azar (supongamos, sin pérdida de generalidad, la Puerta 1). El presentador, quien conoce con certeza qué hay detrás de cada puerta y está obligado por las reglas a abrir siempre una puerta con cabra distinta a la elegida por el participante (si tiene dos opciones disponibles, elige cada una con probabilidad $0.5$), abre la Puerta 3 mostrando una cabra.
	
	Utilizando la axiomática de probabilidad condicional formal, demuestra matemáticamente por qué la probabilidad de ganar si el participante cambia su elección a la Puerta 2 es exactamente $\dfrac{2}{3}$, mientras que si mantiene su elección en la Puerta 1 la probabilidad es de sólo $\dfrac{1}{3}$.
\end{problema}

\subsubsection*{1. Nivel Fundamental (Sencillos / Directos)}

\begin{solucion}
	El espacio muestral del lanzamiento de dos dados tiene $36$ puntos.
	\begin{enumerate}
		\item $A = \{(3,6), (4,5), (4,6), (5,4), (5,5), (5,6), (6,3), (6,4), (6,5), (6,6)\}$, entonces $\card{A} = 10$ y $P(A) = \frac{10}{36} = \frac{5}{18}$.
		\item $B$ = primer dado par = $\{2,4,6\} \times \{1,2,3,4,5,6\}$, por lo que $\card{B} = 18$ y $P(B) = \frac{18}{36} = \frac{1}{2}$.
		\item La intersección $A \cap B = \{(4,5), (4,6), (6,3), (6,4), (6,5), (6,6)\}$ tiene $\card{A \cap B} = 6$. Por definición de probabilidad condicional:
		\begin{align}
			P(A|B) = \frac{P(A \cap B)}{P(B)} = \frac{6/36}{18/36} = \frac{6}{18} = \frac{1}{3}.
		\end{align}
		\item Como $P(A|B) = \frac{1}{3} \neq \frac{5}{18} = P(A)$, los eventos $A$ y $B$ \textbf{no son independientes}.
	\end{enumerate}
\end{solucion}

\begin{solucion}
	La urna contiene un total de $5 + 3 = 8$ bolas.
	\begin{enumerate}
		\item La probabilidad inicial para la primera bola es $P(A) = \frac{5}{8}$.
		\item Si la primera bola fue roja, quedan en la urna 4 rojas y 3 azules ($7$ en total):
		\begin{align}
			P(B|A) = \frac{4}{7}.
		\end{align}
		\item Si la primera bola fue azul ($A'$), quedan 5 rojas y 2 azules ($7$ en total):
		\begin{align}
			P(B|A') = \frac{5}{7}.
		\end{align}
		\item Por la regla de la probabilidad total (o partición en $A$ y $A'$):
		\begin{align}
			P(B) &= P(B|A)P(A) + P(B|A')P(A') = \left(\frac{4}{7} \cdot \frac{5}{8}\right) + \left(\frac{5}{7} \cdot \frac{3}{8}\right) \\
			&= \frac{20}{56} + \frac{15}{56} = \frac{35}{56} = \frac{5}{8}.
		\end{align}
	\end{enumerate}
\end{solucion}

\begin{solucion}
	Conociendo $P(A)=0.60$, $P(B)=0.50$ y $P(A \cap B)=0.30$:
	\begin{enumerate}
		\item Probabilidades condicionales directas:
		\begin{align}
			P(A|B) &= \frac{P(A \cap B)}{P(B)} = \frac{0.30}{0.50} = 0.60 \\
			P(B|A) &= \frac{P(A \cap B)}{P(A)} = \frac{0.30}{0.60} = 0.50
		\end{align}
		(Obsérvese que aquí $P(A|B)=P(A)$, reflejando independencia en este ejercicio particular).
		
		\item La probabilidad de $A$ y no $B$ es:
		\begin{align}
			P(A \cap B') = P(A) - P(A \cap B) = 0.60 - 0.30 = 0.30.
		\end{align}
		
		\item Dado que $P(B') = 1 - P(B) = 1 - 0.50 = 0.50$:
		\begin{align}
			P(A|B') = \frac{P(A \cap B')}{P(B')} = \frac{0.30}{0.50} = 0.60.
		\end{align}
	\end{enumerate}
\end{solucion}

\subsubsection*{2. Nivel Operativo (Intermedios / De Desarrollo)}

\begin{solucion}
	Sea $D$ el evento de pieza defectuosa, $A$ el evento de línea A y $B$ el de línea B.
	\begin{enumerate}
		\item Por la regla de la probabilidad total:
		\begin{align}
			P(D) &= P(D|A)P(A) + P(D|B)P(B) = (0.05)(0.70) + (0.10)(0.30) \\
			&= 0.035 + 0.030 = 0.065 \text{ (o 6.5\%).}
		\end{align}
		
		\item Por el teorema de Bayes, dado que ocurrió $D$:
		\begin{align}
			P(A|D) = \frac{P(D|A)P(A)}{P(D)} = \frac{0.035}{0.065} = \frac{35}{65} = \frac{7}{13} \approx 0.5385 \text{ (53.85\%).}
		\end{align}
	\end{enumerate}
\end{solucion}

\begin{solucion}
	Sea $F$ el evento de fallo térmico, con $P(A)=0.60$ y $P(B)=0.40$. Las tasas de fallo son $P(F|A)=0.02$ y $P(F|B)=0.05$.
	La probabilidad total de que una pieza falle es:
	\begin{align}
		P(F) = P(F|A)P(A) + P(F|B)P(B) = (0.02)(0.60) + (0.05)(0.40) = 0.012 + 0.020 = 0.032.
	\end{align}
	Aplicando el teorema de Bayes para obtener la probabilidad a posteriori de la planta A:
	\begin{align}
		P(A|F) = \frac{P(F|A)P(A)}{P(F)} = \frac{0.012}{0.032} = \frac{12}{32} = \frac{3}{8} = 0.375 \text{ (37.5\%).}
	\end{align}
\end{solucion}

\begin{solucion}
	Sea $E$ el éxito en pruebas (pasa a la primera), y sean los lenguajes $P_y$ (\texttt{Python}), $R$ (\texttt{R}) y $C$ (\texttt{C++}).
	Los datos a priori son: $P(P_y)=0.60$, $P(R)=0.30$, $P(C)=0.10$.
	Las probabilidades condicionales de éxito son: $P(E|P_y)=0.85$, $P(E|R)=0.75$, $P(E|C)=0.95$.
	
	La probabilidad marginal total de que un script pase las pruebas es:
	\begin{align}
		P(E) &= P(E|P_y)P(P_y) + P(E|R)P(R) + P(E|C)P(C) \\
		&= (0.85)(0.60) + (0.75)(0.30) + (0.95)(0.10) \\

	\end{align}
	La probabilidad condicional de que el script exitoso provenga de \texttt{Python} es:
	\begin{align}
		P(P_y|E) = \frac{P(E|P_y)P(P_y)}{P(E)} = \frac{0.510}{0.830} = \frac{51}{83} \approx 0.6145 \text{ (61.45\%).}

\subsubsection*{3. Nivel Analítico (Avanzados / De Conexión)}

\begin{solucion}
	Sea $A$ el evento de aprobar ($P(A)=0.80$, luego $P(A')=0.20$) y $R$ el de asistir regularmente.
	Sabemos por el enunciado que: $P(R|A)=0.90$ y $P(R|A')=0.40$.
	
	\begin{enumerate}
		\item El primer inciso se da directamente por la información condicional del enunciado:
		\begin{align}
			P(R|A) = 0.90 \text{ (90\%).}
		\end{align}
		
		\item Para el segundo inciso, necesitamos encontrar $P(A|R')$. Usamos el teorema de Bayes y la partición sobre el evento de no asistencia $R'$:
		\begin{align}
			P(A|R') = \frac{P(R'|A)P(A)}{P(R')}.
		\end{align}
		Calculamos las probabilidades condicionales complementarias de no asistir:
		\begin{align}
			P(R'|A) &= 1 - P(R|A) = 1 - 0.90 = 0.10 \\
			P(R'|A') &= 1 - P(R|A') = 1 - 0.40 = 0.60
		\end{align}
		La probabilidad total de no asistir al seminario es:
		\begin{align}
			P(R') &= P(R'|A)P(A) + P(R'|A')P(A') = (0.10)(0.80) + (0.60)(0.20) \\
			&= 0.08 + 0.12 = 0.20.
		\end{align}
		Por lo tanto, la probabilidad de haber aprobado habiendo faltado regularmente es:
		\begin{align}
			P(A|R') = \frac{(0.10)(0.80)}{0.20} = \frac{0.08}{0.20} = 0.40 \text{ (40\%).}
		\end{align}
	\end{enumerate}
\end{solucion}

\begin{solucion}
	\begin{enumerate}
		\item Por definición de probabilidad condicional para $P(A)>0$:
		\begin{align}
			P(B|A) = \frac{P(A \cap B)}{P(A)}.
		\end{align}
		Si esto es igual a $P(B)$, multiplicando ambos lados por $P(A)$ obtenemos:
		\begin{align}
			P(A \cap B) = P(A)P(B),
		\end{align}
		que constituye la definición de independencia.
		
		\item Supongamos que $A$ y $B$ son independientes ($P(A \cap B) = P(A)P(B)$). Queremos demostrar que $P(A' \cap B') = P(A')P(B')$.
		Por la Ley de De Morgan y la regla del complemento:
		\begin{align}
			P(A' \cap B') &= P((A \cup B)') = 1 - P(A \cup B).
		\end{align}
		Sustituyendo la expansión de la unión y la propiedad de independencia:
		\begin{align}
			P(A' \cap B') &= 1 - [P(A) + P(B) - P(A \cap B)] \\
			&= 1 - P(A) - P(B) + P(A)P(B) \\
			&= [1 - P(A)] - P(B)[1 - P(A)] \\
			&= [1 - P(A)][1 - P(B)] = P(A')P(B').
		\end{align}
		Lo cual demuestra de manera general que el complemento de eventos independientes preserva la independencia estructural.
	\end{enumerate}
\end{solucion}

\subsubsection*{4. Nivel Desafiante (Expertos / Deducción Profunda)}

\begin{solucion}
	\begin{enumerate}
		\item \textbf{Demostración por inducción de la Regla de la Cadena:}
		Para $n=2$, la fórmula se reduce a $P(A_1 \cap A_2) = P(A_1)P(A_2|A_1)$, que es exactamente la definición de probabilidad condicional.
		Asumamos como hipótesis inductiva que la identidad se cumple para $k$ eventos simultáneos.
		Para $k+1$ eventos, agrupamos los primeros $k$ en un solo evento compuesto $E_k = A_1 \cap A_2 \cap \dots \cap A_k$:
		\begin{align}
			P(A_1 \cap \dots \cap A_k \cap A_{k+1}) &= P(E_k \cap A_{k+1}) = P(E_k) \cdot P(A_{k+1}|E_k).
		\end{align}
		Al sustituir por hipótesis inductiva la expansión de $P(E_k)$, obtenemos inmediatamente el producto del lado derecho con el término $(k+1)$-ésimo adjunto, probando la inducción para todo $n$.
		
		\item \textbf{Extracción de 4 cartas de palos distintos:}
		Sea $C_i$ el evento de que la $i$-ésima carta sea de un palo diferente a todas las anteriores ($i=1,2,3,4$).
		\begin{itemize}
			\item Para $C_1$: la primera carta puede ser cualquiera de las 52 de la baraja $\implies P(C_1) = \frac{52}{52} = 1$.
			\item Para $C_2|C_1$: de las 51 cartas restantes, 39 cartas corresponden a los 3 palos aún no elegidos $\implies P(C_2|C_1) = \frac{39}{51}$.
			\item Para $C_3|C_1 \cap C_2$: quedan 50 cartas y 2 palos inéditos ($2 \times 13 = 26$ cartas útiles) $\implies P(C_3|C_1 \cap C_2) = \frac{26}{50}$.
			\item Para $C_4|C_1 \cap C_2 \cap C_3$: quedan 49 cartas y 1 palo inédito (13 cartas útiles) $\implies P(C_4|\dots) = \frac{13}{49}$.
		\end{itemize}
		Por la Regla de la Cadena, la probabilidad de las 4 cartas de palos distintos es:
		\begin{align}
			P(C_1 \cap C_2 \cap C_3 \cap C_4) = 1 \times \frac{39}{51} \times \frac{26}{50} \times \frac{13}{49} = \frac{39 \times 26 \times 13}{51 \times 50 \times 49} = \frac{13182}{124950} \approx 0.1055 \text{ (10.55\%).}
		\end{align}
	\end{enumerate}
\end{solucion}

\begin{solucion}
	Formalicemos el espacio condicional del juego. Sean las variables:
	\begin{itemize}
		\item $C_i$: El premio (coche) está detrás de la Puerta $i$ ($i=1,2,3$), con probabilidad a priori $P(C_1)=P(C_2)=P(C_3)=\frac{1}{3}$.
		\item $A_j$: El presentador abre la Puerta $j$ mostrando una cabra ($j=1,2,3$).
	\end{itemize}
	Dado que el concursante eligió la Puerta 1, el presentador nunca abre la Puerta 1 ni la puerta con el premio. Evaluamos las probabilidades condicionales de la acción del presentador al abrir la Puerta 3 ($A_3$) para cada posible ubicación del coche:
	\begin{align}
		P(A_3 | C_1) &= \frac{1}{2} \quad \text{(Si el coche está en 1, el presentador elige al azar entre 2 y 3).} \\
		P(A_3 | C_2) &= 1 \quad \text{(Si el coche está en 2, el presentador obligatoriamente debe abrir la 3).} \\
		P(A_3 | C_3) &= 0 \quad \text{(Si el coche está en 3, el presentador jamás abre la 3 para no revelar el premio).}
	\end{align}
	Por la regla de la probabilidad total, la probabilidad marginal de que el presentador abra la Puerta 3 es:
	\begin{align}
		P(A_3) &= P(A_3|C_1)P(C_1) + P(A_3|C_2)P(C_2) + P(A_3|C_3)P(C_3) \\
		&= \left(\frac{1}{2} \cdot \frac{1}{3}\right) + \left(1 \cdot \frac{1}{3}\right) + \left(0 \cdot \frac{1}{3}\right) = \frac{1}{6} + \frac{1}{3} + 0 = \frac{3}{6} = \frac{1}{2}.
	\end{align}
	
	Ahora, aplicando el teorema de Bayes para hallar la probabilidad a posteriori de que el coche esté en la Puerta 1 (si no cambia):
	\begin{align}
		P(C_1 | A_3) = \frac{P(A_3 | C_1)P(C_1)}{P(A_3)} = \frac{(1/2)(1/3)}{1/2} = \frac{1}{3}.
	\end{align}
	Y la probabilidad de que el coche esté en la Puerta 2 (la opción al cambiar de decisión):
	\begin{align}
		P(C_2 | A_3) = \frac{P(A_3 | C_2)P(C_2)}{P(A_3)} = \frac{(1)(1/3)}{1/2} = \frac{2}{3}.
	\end{align}
	\textbf{Conclusión formal:} La intervención dependiente del presentador transfiere toda la probabilidad complementaria hacia la puerta restante. Cambiar de opción duplica las probabilidades de éxito ($2/3$ frente a $1/3$).
\end{solucion}
