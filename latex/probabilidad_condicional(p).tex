\begin{problema}
	\label{prob:8794f31}
	Sean dos eventos $A, B \subset S$ tales que $P(A) = 0.60$, $P(B) = 0.50$ y la probabilidad conjunta es $P(A \cap B) = 0.30$. Calcula directamente:
	\begin{enumerate}
		\item $P(A|B)$ y $P(B|A)$
		\item $P(A \cap B')$
		\item $P(A|B')$
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:cf929e4}
	Una urna contiene 5 bolas rojas y 3 bolas azules. Se extraen dos bolas sucesivamente sin reemplazo. Sea $A$ = \texttt{``la primera bola es roja''} y $B$ = \texttt{``la segunda bola es roja''}.
	\begin{enumerate}
		\item Calcula $P(A)$
		\item Calcula $P(B|A)$ y $P(B|A')$
		\item Calcula $P(B)$ usando la regla de la probabilidad total, y explica por qué $P(B)=P(A)$ a pesar de que la extracción es sin reemplazo.
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:52e63a3}
	En una planta automatizada, el 70\% de las piezas procesadas provienen de la línea de producción A y el 30\% de la línea B. La línea A tiene una tasa de defectos del 5\%, mientras que la línea B tiene una tasa de defectos del 10\%. Si el equipo de control de calidad selecciona una pieza al azar:
	\begin{enumerate}
		\item ¿Cuál es la probabilidad total de que la pieza sea defectuosa?
		\item Si la pieza seleccionada resulta defectuosa, ¿cuál es la probabilidad condicional de que provenga de la línea A?
	\end{enumerate}
\end{problema}

	\begin{enumerate}
		\item Demuestra formalmente la Regla General de la Cadena (o de la Multiplicación) para una secuencia finita de $n$ eventos con $P(A_1 \cap \dots \cap A_{n-1}) > 0$:
		\begin{align}
			P(A_1 \cap A_2 \cap \dots \cap A_n) = P(A_1)P(A_2|A_1)P(A_3|A_1 \cap A_2) \cdots P(A_n|A_1 \cap \dots \cap A_{n-1}).
		\end{align}
		\item Utiliza esta regla para calcular la probabilidad exacta de que al extraer 4 cartas sucesivamente y sin reemplazo de una baraja inglesa de 52 cartas, las 4 pertenezcan a palos diferentes.
	\end{enumerate}

\begin{problema}
	\label{prob:c840e8f}
	\textbf{La Paradoja de Monty Hall:} En un programa de concursos, un participante se encuentra ante tres puertas cerradas ($P_1, P_2, P_3$). Detrás de exactamente una puerta está el premio principal y detrás de las otras dos hay cabras. El participante elige una puerta al azar (supongamos, sin pérdida de generalidad, la Puerta 1). El presentador, quien conoce con certeza qué hay detrás de cada puerta y está obligado por las reglas a abrir siempre una puerta con cabra distinta a la elegida por el participante (si tiene dos opciones disponibles, elige cada una con probabilidad $0.5$), abre la Puerta 3 mostrando una cabra.

	Utilizando la axiomática de probabilidad condicional formal, evalúa y demuestra matemáticamente por qué la probabilidad de ganar si el participante cambia su elección a la Puerta 2 es exactamente $\dfrac{2}{3}$, mientras que si mantiene su elección en la Puerta 1 la probabilidad es de sólo $\dfrac{1}{3}$, y explica por qué la intuición de que "quedan dos puertas, así que es 50-50" es incorrecta.
\end{problema}

\begin{problema}
	\label{prob:4241cab}
	Diseñe un problema original de diagnóstico médico: una prueba rápida para detectar una enfermedad rara que afecta al $1\%$ de la población tiene sensibilidad del $95\%$ (detecta correctamente a los enfermos) y especificidad del $90\%$ (da negativo correctamente en las personas sanas). Calcule, mediante el teorema de Bayes, la probabilidad de que una persona con resultado positivo realmente tenga la enfermedad, y comente por qué este resultado —conocido como la \emph{falacia de la tasa base}— suele sorprender a quienes no tienen formación en probabilidad.
\end{problema}

\subsection*{Soluciones detalladas}

% Recordar
\begin{solproblema}[prob:8794f31]
	Conociendo $P(A)=0.60$, $P(B)=0.50$ y $P(A \cap B)=0.30$:
	\begin{enumerate}
		\item Probabilidades condicionales directas:
		\begin{align}
			P(A|B) &= \frac{P(A \cap B)}{P(B)} = \frac{0.30}{0.50} = 0.60 \\
			P(B|A) &= \frac{P(A \cap B)}{P(A)} = \frac{0.30}{0.60} = 0.50
		\end{align}
		(Obsérvese que aquí $P(A|B)=P(A)$, reflejando independencia en este ejercicio particular).

		\item La probabilidad de $A$ y no $B$ es:
		\begin{align}
			P(A \cap B') = P(A) - P(A \cap B) = 0.60 - 0.30 = 0.30.
		\end{align}

		\item Dado que $P(B') = 1 - P(B) = 1 - 0.50 = 0.50$:
		\begin{align}
			P(A|B') = \frac{P(A \cap B')}{P(B')} = \frac{0.30}{0.50} = 0.60.
		\end{align}
	\end{enumerate}
\end{solproblema}

\begin{solproblema}[prob:cf929e4]
	La urna contiene un total de $5 + 3 = 8$ bolas.
	\begin{enumerate}
		\item La probabilidad inicial para la primera bola es $P(A) = \frac{5}{8}$.
		\item Si la primera bola fue roja, quedan en la urna 4 rojas y 3 azules ($7$ en total): $P(B|A) = \frac{4}{7}$. Si la primera bola fue azul ($A'$), quedan 5 rojas y 2 azules ($7$ en total): $P(B|A') = \frac{5}{7}$.
		\item Por la regla de la probabilidad total (partición en $A$ y $A'$):
		\begin{align}
			P(B) &= P(B|A)P(A) + P(B|A')P(A') = \left(\frac{4}{7} \cdot \frac{5}{8}\right) + \left(\frac{5}{7} \cdot \frac{3}{8}\right) = \frac{20}{56} + \frac{15}{56} = \frac{35}{56} = \frac{5}{8}.
		\end{align}
		Que $P(B)=P(A)=5/8$ no es casualidad: antes de observar ningún resultado, no hay ninguna información que distinga a la segunda posición de la primera dentro de la secuencia de extracción —por simetría, cualquier bola de la urna tiene la misma probabilidad marginal de terminar siendo "la roja de la posición 2" que "la roja de la posición 1"—, aunque el resultado específico de la primera extracción sí afecte la probabilidad \emph{condicional} de la segunda.
	\end{enumerate}
\end{solproblema}

% Aplicar
\begin{solproblema}[prob:52e63a3]
	Sea $D$ el evento de pieza defectuosa, $A$ el evento de línea A y $B$ el de línea B.
	\begin{enumerate}
		\item Por la regla de la probabilidad total:
		\begin{align}
			P(D) &= P(D|A)P(A) + P(D|B)P(B) = (0.05)(0.70) + (0.10)(0.30) = 0.035 + 0.030 = 0.065 \text{ (6.5\%).}
		\end{align}
		\item Por el teorema de Bayes, dado que ocurrió $D$:
		\begin{align}
			P(A|D) = \frac{P(D|A)P(A)}{P(D)} = \frac{0.035}{0.065} = \frac{7}{13} \approx 0.5385 \text{ (53.85\%).}
		\end{align}
	\end{enumerate}
\end{solproblema}

% Analizar
\begin{solproblema}[prob:8a7282b]
	\begin{enumerate}
		\item \textbf{Demostración por inducción de la Regla de la Cadena:} para $n=2$, la fórmula se reduce a $P(A_1 \cap A_2) = P(A_1)P(A_2|A_1)$, que es exactamente la definición de probabilidad condicional. Asumamos como hipótesis inductiva que la identidad se cumple para $k$ eventos simultáneos. Para $k+1$ eventos, agrupamos los primeros $k$ en un solo evento compuesto $E_k = A_1 \cap A_2 \cap \dots \cap A_k$:
		\begin{align}
			P(A_1 \cap \dots \cap A_k \cap A_{k+1}) = P(E_k \cap A_{k+1}) = P(E_k) \cdot P(A_{k+1}|E_k).
		\end{align}
		Al sustituir por hipótesis inductiva la expansión de $P(E_k)$, obtenemos inmediatamente el producto del lado derecho con el término $(k+1)$-ésimo adjunto, probando la inducción para todo $n$.

		\item \textbf{Extracción de 4 cartas de palos distintos:} sea $C_i$ el evento de que la $i$-ésima carta sea de un palo diferente a todas las anteriores.
		\begin{itemize}
			\item Para $C_1$: la primera carta puede ser cualquiera $\implies P(C_1) = 1$.
			\item Para $C_2|C_1$: de las 51 restantes, 39 corresponden a los 3 palos aún no elegidos $\implies P(C_2|C_1) = \frac{39}{51}$.
			\item Para $C_3|C_1 \cap C_2$: quedan 50 cartas y 2 palos inéditos ($26$ cartas útiles) $\implies P(C_3|\dots) = \frac{26}{50}$.
			\item Para $C_4|C_1 \cap C_2 \cap C_3$: quedan 49 cartas y 1 palo inédito (13 cartas útiles) $\implies P(C_4|\dots) = \frac{13}{49}$.
		\end{itemize}
		Por la Regla de la Cadena:
		\begin{align}
			P(C_1 \cap C_2 \cap C_3 \cap C_4) = 1 \times \frac{39}{51} \times \frac{26}{50} \times \frac{13}{49} \approx 0.1055 \text{ (10.55\%).}
		\end{align}
	\end{enumerate}
\end{solproblema}

\begin{solproblema}[prob:c840e8f]
	Formalicemos el espacio condicional del juego. Sean $C_i$: el premio está detrás de la Puerta $i$, con $P(C_1)=P(C_2)=P(C_3)=\frac{1}{3}$; y $A_3$: el presentador abre la Puerta 3 mostrando una cabra. Evaluamos las probabilidades condicionales de la acción del presentador para cada posible ubicación del premio:
	\begin{align}
		P(A_3 | C_1) &= \frac{1}{2} \quad \text{(si el premio está en 1, el presentador elige al azar entre 2 y 3)} \\
		P(A_3 | C_2) &= 1 \quad \text{(si el premio está en 2, el presentador obligatoriamente abre la 3)} \\
		P(A_3 | C_3) &= 0 \quad \text{(si el premio está en 3, el presentador jamás la abre)}
	\end{align}
	Por la regla de la probabilidad total:
	\begin{align}
		P(A_3) = \left(\frac{1}{2} \cdot \frac{1}{3}\right) + \left(1 \cdot \frac{1}{3}\right) + \left(0 \cdot \frac{1}{3}\right) = \frac{1}{6} + \frac{1}{3} = \frac{1}{2}.
	\end{align}
	Aplicando Bayes: $P(C_1|A_3) = \dfrac{(1/2)(1/3)}{1/2} = \dfrac{1}{3}$ y $P(C_2|A_3) = \dfrac{(1)(1/3)}{1/2} = \dfrac{2}{3}$.

	La intuición "quedan dos puertas, así que es 50-50" es incorrecta porque ignora que la acción del presentador \textbf{no es aleatoria ni simétrica respecto a la información que aporta}: el presentador usa su conocimiento certero del contenido de las puertas para \emph{siempre} evitar abrir la del premio, lo cual transfiere toda la probabilidad que originalmente tenían las dos puertas no elegidas hacia la única puerta no elegida que permanece cerrada. La Puerta 1 nunca actualiza su probabilidad ($1/3$ antes y después), pero la Puerta 2 absorbe la probabilidad completa que antes compartía con la Puerta 3 ($1/3+1/3=2/3$).
\end{solproblema}

\begin{solproblema}[prob:4241cab]
	Sea $D$ el evento de estar enfermo, con $P(D)=0.01$ y $P(D')=0.99$. La sensibilidad da $P(+|D)=0.95$, y la especificidad del $90\%$ implica una tasa de falsos positivos $P(+|D')=1-0.90=0.10$.

	La probabilidad total de un resultado positivo es:
	\begin{align}
		P(+) = P(+|D)P(D) + P(+|D')P(D') = (0.95)(0.01) + (0.10)(0.99) = 0.0095 + 0.099 = 0.1085.
	\end{align}
	Por el teorema de Bayes:
	\begin{align}
		P(D|+) = \frac{P(+|D)P(D)}{P(+)} = \frac{0.0095}{0.1085} \approx 0.0876 \text{ (8.76\%).}
	\end{align}
	A pesar de que la prueba parece muy confiable (95\% de sensibilidad, 90\% de especificidad), solo el $8.76\%$ de quienes reciben un resultado positivo realmente tienen la enfermedad. Esto sorprende porque la intuición se enfoca en la precisión de la prueba en sí misma, ignorando que la enfermedad es tan rara ($1\%$) que el número absoluto de falsos positivos entre la enorme mayoría de personas sanas ($0.10 \times 99\%$ de la población) supera ampliamente al número de verdaderos positivos entre la pequeña minoría enferma ($0.95\times1\%$).
\end{solproblema}
