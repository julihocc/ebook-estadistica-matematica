\section*{Problemas}

\subsection{Enunciados de los problemas}

\subsubsection*{1. Nivel Fundamental (Sencillos / Directos)}

\begin{problema}
	\label{prob-z-1}
	Una fábrica de baterías afirma que la duración promedio de sus baterías es de 500 horas con una desviación estándar poblacional conocida de $\sigma = 40$ horas. Un inspector de calidad toma una muestra aleatoria de $n = 64$ baterías y encuentra una duración promedio muestral de $\bar{X} = 490$ horas.
	\begin{enumerate}
		\item Formule las hipótesis nula ($H_0$) y alternativa ($H_a$) para probar si la duración promedio es estadísticamente menor a 500 horas.
		\item Calcule el estadístico de prueba $Z$.
		\item Determine el valor $p$ (p-valor) asociado a la prueba.
		\item ¿Se debe rechazar la hipótesis nula al nivel de significación $\alpha = 0.05$?
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob-t-1}
	Un fabricante de neumáticos afirma que sus llantas duran en promedio $\mu_0 = 50,000$ km. Un consumidor toma una muestra aleatoria de $n = 16$ llantas y encuentra una media muestral $\bar{X} = 48,500$ km con una desviación estándar muestral $s = 3,000$ km.
	\begin{enumerate}
		\item ¿Qué distribución debe usar para la prueba de hipótesis ($Z$ o $t$ de Student)? Explique por qué.
		\item Formule las hipótesis para probar si la duración promedio poblacional es menor a 50,000 km.
		\item Calcule el estadístico $t$.
		\item Determine el valor $p$ aproximado usando $\nu = n-1 = 15$ grados de libertad.
		\item ¿Se debe rechazar $H_0$ al nivel de significación $\alpha = 0.05$?
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob-z-t-new1}
	Una muestra aleatoria de tamaño $n = 25$ tomada de una población normal arroja una desviación estándar muestral $s = 12$.
	\begin{enumerate}
		\item Calcule el error estándar de la media muestral $\text{SE}(\bar{X}) = \dfrac{s}{\sqrt{n}}$.
		\item Determine los valores críticos del estadístico $t$ de Student para una prueba bilateral a los niveles de significación $\alpha = 0.05$ y $\alpha = 0.01$ con $\nu = 24$ grados de libertad.
	\end{enumerate}
\end{problema}

\subsubsection*{2. Nivel Operativo (Intermedios / De Desarrollo)}

\begin{problema}
	\label{prob-z-2}
	Un investigador pedagógico afirma que el coeficiente intelectual (CI) promedio de los estudiantes del campus universitario es diferente del promedio nacional de $\mu_0 = 100$. Para verificarlo, toma una muestra aleatoria de $n = 100$ estudiantes y encuentra una media muestral $\bar{X} = 105$. Suponga que la desviación estándar poblacional es conocida e igual a $\sigma = 15$.
	\begin{enumerate}
		\item Formule las hipótesis para una prueba de dos colas (bilateral).
		\item Calcule el estadístico de prueba $Z$.
		\item Determine el valor $p$ exacto.
		\item ¿Se debe rechazar $H_0$ al nivel de significación exigente $\alpha = 0.01$?
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob-t-2}
	Un nutricionista desea comprobar si una nueva dieta hiperproteica induce un cambio en el peso corporal. Toma una muestra de $n = 25$ pacientes que siguen la dieta durante 3 meses y encuentra que el aumento de peso promedio es $\bar{X} = 2.5$ kg con una desviación estándar muestral $s = 4.0$ kg.
	\begin{enumerate}
		\item Formule las hipótesis para probar si la dieta produce un incremento estadísticamente significativo en el peso (prueba unilateral de cola derecha).
		\item Calcule el estadístico $t$.
		\item Determine el valor $p$ con $\nu = 24$ grados de libertad.
		\item ¿Se debe rechazar $H_0$ al nivel de significación exploratorio $\alpha = 0.10$?
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob-z-3}
	Una compañía de telecomunicaciones asegura que el tiempo promedio de espera de un usuario en línea de atención al cliente es $\mu_0 = 8$ minutos con una desviación estándar $\sigma = 2$ minutos. Una firma auditora toma una muestra de $n = 49$ llamadas y encuentra un promedio muestral $\bar{X} = 8.5$ minutos.
	\begin{enumerate}
		\item Calcule el estadístico $Z$ para probar la hipótesis de que el tiempo de espera real excede los 8 minutos.
		\item Determine el valor $p$ de la prueba unilateral.
		\item ¿Cuál es la decisión y conclusión estadística al nivel de significación $\alpha = 0.05$?
		\item ¿Cambia la decisión si la auditoría exige un nivel de significación más estricto de $\alpha = 0.01$? Explique la implicación en términos del riesgo de error.
	\end{enumerate}
\end{problema}

\subsubsection*{3. Nivel Analítico (Avanzados / De Conexión)}

\begin{problema}
	\label{prob-t-3}
	Un fabricante de conductores eléctricos afirma que la resistencia promedio de sus cables industriales es exactamente de $\mu_0 = 100$ ohmios. Un ingeniero de pruebas toma una muestra de $n = 9$ tramos de cable y obtiene las siguientes lecturas de resistencia (en ohmios):
	\begin{center}
		98, 102, 99, 101, 97, 103, 100, 98, 102
	\end{center}
	\begin{enumerate}
		\item Calcule algebraicamente la media muestral $\bar{X}$ y la cuasivarianza muestral $s^2$.
		\item Formule las hipótesis para una prueba bilateral sobre la resistencia poblacional $\mu$.
		\item Calcule el estadístico $t$ de Student e indique sus grados de libertad.
		\item Determine el valor $p$ y concluya sobre la afirmación del fabricante al nivel $\alpha = 0.05$.
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob-z-t-new2}
	\textbf{El Teorema del Límite Central y la estandarización:} Sea $X_1, X_2, \dots, X_n$ una muestra aleatoria simple extraída de una población cuya distribución de probabilidad no es normal, pero que posee media poblacional $\mu$ y varianza $\sigma^2$ finitas.
	\begin{enumerate}
		\item Demuestra a partir de los operadores lineales de esperanza $E[\cdot]$ y varianza $\Var[\cdot]$ que el valor esperado y la varianza de la media muestral $\bar{X}$ son $E[\bar{X}] = \mu$ y $\Var[\bar{X}] = \dfrac{\sigma^2}{n}$.
		\item Demuestra algebraicamente que al aplicar la estandarización lineal $Z_n = \dfrac{\bar{X} - E[\bar{X}]}{\sqrt{\Var[\bar{X}]}}$, el estadístico resultante tiene media exacta 0 y varianza exacta 1 para cualquier tamaño $n$.
		\item Explica formalmente por qué para un tamaño grande ($n \ge 30$) la prueba de hipótesis sobre $\mu$ utilizando $Z_n = \dfrac{\bar{X} - \mu_0}{\sigma/\sqrt{n}}$ es asintóticamente válida incluso ante asimetría poblacional en los datos originales.
	\end{enumerate}
\end{problema}

\subsubsection*{4. Nivel Desafiante (Expertos / Deducción Profunda)}

\begin{problema}
	\label{prob-z-t-new3}
	\textbf{Deducción teórica de invarianza de escala del estadístico $t$ de Student:}
	En la teoría de distribuciones exactas del muestreo normal, si $Z \sim N(0,1)$ y $V \sim \chi^2(\nu)$ son dos variables aleatorias independientes, el estadístico de Student se define como el cociente transformado:
	\begin{align}
		T = \frac{Z}{\sqrt{V / \nu}} \sim t_\nu.
	\end{align}
	Sea $X_1, \dots, X_n$ una muestra aleatoria normal $N(\mu, \sigma^2)$. Sabemos que el estadístico tipificado es $Z = \dfrac{\bar{X} - \mu}{\sigma/\sqrt{n}} \sim N(0,1)$ y que la suma de cuadrados muestrales normalizada satisface $V = \dfrac{(n-1)S^2}{\sigma^2} \sim \chi^2(n-1)$, siendo $\bar{X}$ y $S^2$ independientes por el Teorema de Cochran.
	
	Demuestra algebraicamente paso a paso que al sustituir estas expresiones en la definición del estadístico $T$, el parámetro de escala poblacional desconocido ($\sigma$) se cancela de manera exacta, demostrando formalmente por qué la variable pivotal $\displaystyle T = \frac{\bar{X} - \mu}{S/\sqrt{n}}$ permite realizar pruebas exactas sobre el parámetro de ubicación $\mu$ sin necesitar el conocimiento previo de la dispersión $\sigma$.
\end{problema}

\begin{problema}
	\label{prob-z-t-new4}
	\textbf{Propiedades de cola pesada de la distribución $t$ de Student vs. Normal:}
	La distribución $t$ de Student de $\nu$ grados de libertad tiene una función de densidad cuya varianza para $\nu > 2$ está dada rigurosamente por la relación de segundo momento no central:
	\begin{align}
		\Var(T) = \frac{\nu}{\nu - 2}.
	\end{align}
	\begin{enumerate}
		\item Demuestra analíticamente que para todo $\nu > 2$ finito, se cumple la desigualdad estricta de varianza $\Var(T) > \Var(Z) = 1$.
		\item Explica desde la perspectiva conceptual del muestreo y la incertidumbre en la estimación del parámetro $S^2 \approx \sigma^2$, por qué la distribución $t$ requiere poseer colas más pesadas (mayor probabilidad en los extremos alejados del cero o leptocurtosis) que la normal estándar.
		\item Evalúe explícitamente el límite $\displaystyle \lim_{\nu \to \infty} \Var(T)$ y demuestre su consistencia con la convergencia distribucional en probabilidad del estadístico $t$ de Student hacia la normal estándar $Z$.
	\end{enumerate}
\end{problema}


\subsection{Soluciones detalladas}

\subsubsection*{1. Nivel Fundamental (Sencillos / Directos)}

\begin{solproblema}[prob-z-1]
	\begin{enumerate}
		\item Hipótesis estadísticas unilaterales de cola izquierda:
		\begin{align}
			H_0: \mu &= 500 \text{ horas} \\
			H_a: \mu &< 500 \text{ horas}
		\end{align}
		
		\item Estadístico de prueba $Z$ (dado que $\sigma = 40$ es conocido poblacionalmente):
		\begin{align}
			Z = \frac{\bar{X} - \mu_0}{\sigma / \sqrt{n}} = \frac{490 - 500}{40 / \sqrt{64}} = \frac{-10}{40 / 8} = \frac{-10}{5} = -2.0.
		\end{align}
		
		\item Valor $p$ (área bajo la curva normal estándar a la izquierda de $-2.0$):
		\begin{align}
			p\text{-valor} = P(Z < -2.0) = \Phi(-2.0) \approx 0.0228 \text{ (o 2.28\%).}
		\end{align}
		
		\item Como el $p\text{-valor} = 0.0228 < 0.05 = \alpha$, \textbf{rechazamos la hipótesis nula ($H_0$)}.
		
		\textbf{Conclusión:} Existe suficiente evidencia estadística en la muestra para rechazar la afirmación de la fábrica al nivel del 5\% y afirmar que la duración promedio de las baterías es significativamente inferior a 500 horas.
	\end{enumerate}
\end{solproblema}

\begin{solproblema}[prob-t-1]
	\begin{enumerate}
		\item Debemos utilizar la \textbf{distribución $t$ de Student} basándonos en dos criterios conjuntos fundamentales:
		\begin{itemize}
			\item La desviación estándar poblacional $\sigma$ es desconocida, por lo que debe ser estimada con la cuasidesviación estándar muestral $s = 3,000$ km.
			\item El tamaño de la muestra es pequeño ($n = 16 < 30$), lo que impide invocar directamente el Teorema del Límite Central con normalidad asintótica.
		\end{itemize}
		
		\item Hipótesis unilaterales:
		\begin{align}
			H_0: \mu &= 50,000 \text{ km} \\
			H_a: \mu &< 50,000 \text{ km}
		\end{align}
		
		\item Estadístico de Student $t$:
		\begin{align}
			t = \frac{\bar{X} - \mu_0}{s / \sqrt{n}} = \frac{48,500 - 50,000}{3,000 / \sqrt{16}} = \frac{-1,500}{3,000 / 4} = \frac{-1,500}{750} = -2.0.
		\end{align}
		
		\item Grados de libertad $\nu = n - 1 = 16 - 1 = 15$. Consultando la tabla $t_{15}$ para la cola izquierda de $-2.0$:
		\begin{align}
			p\text{-valor} = P(t_{15} < -2.0) \approx 0.032 \text{ (3.2\%).}
		\end{align}
		
		\item Dado que $p\text{-valor} \approx 0.032 < 0.05 = \alpha$, \textbf{rechazamos $H_0$}.
		
		\textbf{Conclusión:} Con un 95\% de confianza, los datos demuestran que la vida útil media real de los neumáticos es inferior al promedio de 50,000 km publicitado por el fabricante.
	\end{enumerate}
\end{solproblema}

\begin{solproblema}[prob-z-t-new1]
	\begin{enumerate}
		\item Para una muestra $n=25$ con desviación muestral $s=12$, el error estándar de la media es la incertidumbre en la estimación del centro:
		\begin{align}
			\text{SE}(\bar{X}) = \frac{s}{\sqrt{n}} = \frac{12}{\sqrt{25}} = \frac{12}{5} = 2.4.
		\end{align}
		
		\item Los grados de libertad son $\nu = n - 1 = 25 - 1 = 24$.
		\begin{itemize}
			\item Para una prueba de dos colas al nivel $\alpha = 0.05$, el área en cada cola extrema es $\alpha/2 = 0.025$. El valor crítico en la tabla $t$ con $\nu = 24$ es:
			\begin{align}
				t_{\alpha/2, \, \nu} = t_{0.025, \, 24} = 2.064.
			\end{align}
			La región de rechazo está dada por $|t| > 2.064$.
			
			\item Para una prueba de dos colas al nivel $\alpha = 0.01$, el área por cola es $\alpha/2 = 0.005$. El valor crítico correspondiente es:
			\begin{align}
				t_{\alpha/2, \, \nu} = t_{0.005, \, 24} = 2.797.
			\end{align}
			La región de rechazo al 1\% es $|t| > 2.797$.
		\end{itemize}
	\end{enumerate}
\end{solproblema}

\subsubsection*{2. Nivel Operativo (Intermedios / De Desarrollo)}

\begin{solproblema}[prob-z-2]
	\begin{enumerate}
		\item Hipótesis bilaterales (de dos colas):
		\begin{align}
			H_0: \mu &= 100 \\
			H_a: \mu &\neq 100
		\end{align}
		
		\item Estadístico normal $Z$ (dado que $\sigma = 15$ es conocido):
		\begin{align}
			Z = \frac{\bar{X} - \mu_0}{\sigma / \sqrt{n}} = \frac{105 - 100}{15 / \sqrt{100}} = \frac{5}{15 / 10} = \frac{5}{1.5} = 3.333.
		\end{align}
		
		\item Para una prueba bilateral, el valor $p$ es el doble del área en la cola exterior del estadístico:
		\begin{align}
			p\text{-valor} = 2 \times P(Z > 3.333) = 2 \times [1 - \Phi(3.333)] \approx 2 \times 0.00043 = 0.00086.
		\end{align}
		
		\item Evaluando al nivel exigente $\alpha = 0.01$:
		\begin{align}
			p\text{-valor} = 0.00086 < 0.01 = \alpha \implies \textbf{Rechazamos } H_0.
		\end{align}
		
		\textbf{Conclusión:} La diferencia de 5 puntos en el promedio muestral es tan amplia en términos del error estándar ($\text{SE}=1.5$) que resulta muy altamente improbable de atribuir al azar muestral. Se concluye que el CI promedio estudiantil del campus difiere de 100.
	\end{enumerate}
\end{solproblema}

\begin{solproblema}[prob-t-2]
	\begin{enumerate}
		\item Hipótesis unilaterales para verificar si el cambio de peso es positivo:
		\begin{align}
			H_0: \mu &= 0 \text{ kg (dieta sin efecto en el peso)} \\
			H_a: \mu &> 0 \text{ kg (dieta aumenta el peso corporal)}
		\end{align}
		
		\item Estadístico $t$ con cuasidesviación estándar $s = 4.0$ kg y $n = 25$:
		\begin{align}
			t = \frac{\bar{X} - \mu_0}{s / \sqrt{n}} = \frac{2.5 - 0}{4.0 / \sqrt{25}} = \frac{2.5}{4.0 / 5} = \frac{2.5}{0.8} = 3.125.
		\end{align}
		
		\item Grados de libertad $\nu = 25 - 1 = 24$. Evaluando la cola derecha:
		\begin{align}
			p\text{-valor} = P(t_{24} > 3.125) \approx 0.0023 \text{ (0.23\%).}
		\end{align}
		
		\item Comparación al nivel exploratorio $\alpha = 0.10$:
		\begin{align}
			p\text{-valor} \approx 0.0023 < 0.10 = \alpha \implies \textbf{Rechazamos } H_0.
		\end{align}
		
		\textbf{Conclusión:} El estadístico obtenido supera ampliamente el umbral crítico, confirmando estadísticamente que la dieta produce una ganancia de peso promedio estadísticamente significativa en el periodo observado.
	\end{enumerate}
\end{solproblema}

\begin{solproblema}[prob-z-3]
	\begin{enumerate}
		\item Hipótesis: $H_0: \mu = 8$ vs. $H_a: \mu > 8$. Estadístico tipificado con $\sigma = 2$ y $n = 49$:
		\begin{align}
			Z = \frac{\bar{X} - \mu_0}{\sigma / \sqrt{n}} = \frac{8.5 - 8.0}{2 / \sqrt{49}} = \frac{0.5}{2 / 7} = \frac{3.5}{2} = 1.75.
		\end{align}
		
		\item Valor $p$ unilateral (cola superior):
		\begin{align}
			p\text{-valor} = P(Z > 1.75) = 1 - \Phi(1.75) = 1 - 0.9599 = 0.0401 \text{ (4.01\%).}
		\end{align}
		
		\item Evaluación al nivel estándar $\alpha = 0.05$:
		\begin{align}
			p\text{-valor} = 0.0401 < 0.05 \implies \textbf{Rechazar } H_0.
		\end{align}
		La auditoría al 5\% de significación concluye que el tiempo de espera real excede los 8 minutos prometidos por la compañía de telecomunicaciones.
		
		\item Evaluación al nivel de alta exigencia $\alpha = 0.01$:
		\begin{align}
			p\text{-valor} = 0.0401 > 0.01 \implies \textbf{No rechazar } H_0.
		\end{align}
		\textbf{Análisis crítico:} Al exigir un riesgo de error Tipo I (falsa alarma) de sólo el 1\%, la prueba requiere una divergencia muestral aún mayor para declarar rechazo ($Z > 2.326$). Este problema resalta cómo los resultados con valores $p$ intermedios (entre $0.01$ y $0.05$) son sensibles a la especificación de $\alpha$.
	\end{enumerate}
\end{solproblema}

\subsubsection*{3. Nivel Analítico (Avanzados / De Conexión)}

\begin{solproblema}[prob-t-3]
	\begin{enumerate}
		\item Cálculos aritméticos exactos de la muestra ($n=9$ cables):
		\begin{align}
			\sum X_i &= 98 + 102 + 99 + 101 + 97 + 103 + 100 + 98 + 102 = 900. \\
			\bar{X} &= \frac{900}{9} = 100 \text{ ohmios.}
		\end{align}
		Calculamos la cuasivarianza muestral insesgada con $S_{xx} = \sum (X_i - \bar{X})^2$:
		\begin{align}
			\sum (X_i - 100)^2 &= (-2)^2 + 2^2 + (-1)^2 + 1^2 + (-3)^2 + 3^2 + 0^2 + (-2)^2 + 2^2 \\
			&= 4 + 4 + 1 + 1 + 9 + 9 + 0 + 4 + 4 = 36. \\
			s^2 &= \frac{\sum (X_i - \bar{X})^2}{n - 1} = \frac{36}{9 - 1} = \frac{36}{8} = 4.5. \\
			s &= \sqrt{4.5} \approx 2.1213 \text{ ohmios.}
		\end{align}
		
		\item Hipótesis bilaterales sobre el parámetro de resistencia:
		\begin{align}
			H_0: \mu &= 100 \text{ ohmios} \\
			H_a: \mu &\neq 100 \text{ ohmios}
		\end{align}
		
		\item Estadístico de Student:
		\begin{align}
			t = \frac{\bar{X} - \mu_0}{s / \sqrt{n}} = \frac{100 - 100}{2.1213 / \sqrt{9}} = \frac{0}{0.7071} = 0.
		\end{align}
		Grados de libertad exactos: $\nu = n - 1 = 8$.
		
		\item El valor $p$ para un estadístico nulo en distribución simétrica es:
		\begin{align}
			p\text{-valor} = 2 \times P(t_8 > 0) = 2 \times 0.5000 = 1.0000.
		\end{align}
		Como $p\text{-valor} = 1.0 > 0.05 = \alpha$, \textbf{no rechazamos $H_0$}.
		
		\textbf{Conclusión:} La resistencia media de los cables muestreados coincide con exactitud absoluta con la especificación de 100 ohmios.
	\end{enumerate}
\end{solproblema}

\begin{solproblema}[prob-z-t-new2]
	\begin{enumerate}
		\item Por linealidad del operador esperanza y dado que las observaciones $X_i$ son independientemente e idénticamente distribuidas con $E[X_i] = \mu$:
		\begin{align}
			E[\bar{X}] = E\left[ \frac{1}{n} \sum_{i=1}^n X_i \right] = \frac{1}{n} \sum_{i=1}^n E[X_i] = \frac{1}{n} (n \mu) = \mu.
		\end{align}
		Por independencia de las observaciones, las varianzas son directamente aditivas ($\Var[\sum X_i] = \sum \Var[X_i]$):
		\begin{align}
			\Var[\bar{X}] = \Var\left[ \frac{1}{n} \sum_{i=1}^n X_i \right] = \frac{1}{n^2} \sum_{i=1}^n \Var[X_i] = \frac{1}{n^2} (n \sigma^2) = \frac{\sigma^2}{n}.
		\end{align}
		
		\item Sea $Z_n = \dfrac{\bar{X} - \mu}{\sigma / \sqrt{n}}$. Aplicando las propiedades del momento central a una variable lineal tipificada:
		\begin{align}
			E[Z_n] &= E\left[ \frac{\bar{X} - \mu}{\sigma/\sqrt{n}} \right] = \frac{E[\bar{X}] - \mu}{\sigma/\sqrt{n}} = \frac{\mu - \mu}{\sigma/\sqrt{n}} = 0. \\
			\Var[Z_n] &= \Var\left[ \frac{\bar{X} - \mu}{\sigma/\sqrt{n}} \right] = \left( \frac{1}{\sigma/\sqrt{n}} \right)^2 \Var[\bar{X} - \mu] = \frac{n}{\sigma^2} \Var[\bar{X}] = \frac{n}{\sigma^2} \left( \frac{\sigma^2}{n} \right) = 1.
		\end{align}
		
		\item \textbf{Validez asintótica del Teorema del Límite Central:}
		El Teorema del Límite Central establece de manera fundamental que, bajo condiciones generales de regularidad y varianza finita, la función de distribución acumulada de $Z_n$ converge uniformemente en ley hacia la normal estándar:
		\begin{align}
			\lim_{n \to \infty} P(Z_n \le z) = \Phi(z) = \frac{1}{\sqrt{2\pi}} \int_{-\infty}^z e^{-u^2/2} du, \quad \text{para todo } z \in \mathbb{R}.
		\end{align}
		Por consiguiente, cuando el tamaño de la muestra es moderadamente grande ($n \ge 30$), la distribución del promedio $\bar{X}$ pierde el sesgo y la curtosis particular de la distribución original, permitiendo calcular valores $p$ precisos usando la tabla normal tipificada $Z$.
	\end{enumerate}
\end{solproblema}

\subsubsection*{4. Nivel Desafiante (Expertos / Deducción Profunda)}

\begin{solproblema}[prob-z-t-new3]
	Sustituyamos en el cociente fundamental $T = \dfrac{Z}{\sqrt{V / \nu}}$ las expresiones analíticas exactas que relacionan a las variables muestrales con los parámetros poblacionales bajo normalidad:
	\begin{align}
		Z &= \frac{\bar{X} - \mu}{\sigma / \sqrt{n}} \sim N(0,1), \\
		V &= \frac{(n - 1)S^2}{\sigma^2} \sim \chi^2(n-1), \quad \text{con grados de libertad } \nu = n - 1.
	\end{align}
	Sustituyendo ambos términos dentro de la raíz en el denominador del estadístico $T$:
	\begin{align}
		T = \frac{\dfrac{\bar{X} - \mu}{\sigma / \sqrt{n}}}{\sqrt{\dfrac{\dfrac{(n - 1)S^2}{\sigma^2}}{n - 1}}}.
	\end{align}
	Simplificando el término $(n - 1)$ dentro del radical en el denominador:
	\begin{align}
		\sqrt{\frac{(n - 1)S^2}{\sigma^2 (n - 1)}} = \sqrt{\frac{S^2}{\sigma^2}} = \frac{S}{\sigma}.
	\end{align}
	Reescribiendo la fracción completa del estadístico de Student:
	\begin{align}
		T = \frac{\dfrac{\bar{X} - \mu}{\sigma / \sqrt{n}}}{\dfrac{S}{\sigma}} = \frac{\bar{X} - \mu}{\sigma / \sqrt{n}} \times \frac{\sigma}{S}.
	\end{align}
	Observemos cómo el parámetro de dispersión poblacional $\sigma$ en el numerador y en el denominador se cancelan aritméticamente:
	\begin{align}
		T = \frac{\bar{X} - \mu}{S / \sqrt{n}} \sim t_{n-1}.
	\end{align}
	\textbf{Implicación teórica estructural:} Esta cancelación exacta de $\sigma$ es lo que dota de poder a la inferencia de Student. Permite que el estadístico $T$ sea completamente calculable a partir exclusiva de los datos de la muestra ($\bar{X}, S, n$) y de la hipótesis bajo prueba ($\mu_0$), operando bajo una distribución de referencia matemática ($t_{n-1}$) que es 100\% independiente del verdadero valor de $\sigma$.
\end{solproblema}

\begin{solproblema}[prob-z-t-new4]
	\begin{enumerate}
		\item \textbf{Demostración analítica de varianza superior:}
		Para cualquier número de grados de libertad $\nu > 2$, la varianza de la distribución $t$ está dada por:
		\begin{align}
			\Var(T) = \frac{\nu}{\nu - 2}.
		\end{align}
		Dado que $\nu > 2 \implies \nu - 2 > 0$, podemos escribir el numerador en forma diferenciada como $\nu = (\nu - 2) + 2$:
		\begin{align}
			\Var(T) = \frac{(\nu - 2) + 2}{\nu - 2} = 1 + \frac{2}{\nu - 2}.
		\end{align}
		Como el término fraccionario residual $\dfrac{2}{\nu - 2}$ es estrictamente positivo para cualquier $\nu > 2$ finito, se demuestra formalmente que:
		\begin{align}
			\Var(T) = 1 + \frac{2}{\nu - 2} > 1 = \Var(Z).
		\end{align}
		
		\item \textbf{Explicación de incertidumbre en el parámetro de escala y leptocurtosis:}
		Cuando se conoce la verdadera varianza poblacional $\sigma^2$, la única fuente de variabilidad en la tipificación de $\bar{X}$ proviene del error de muestreo del promedio en sí mismo ($Z$). Sin embargo, cuando reemplazamos $\sigma$ por el estimador muestral $S$, introducimos una \textbf{segunda fuente de variabilidad o incertidumbre aleatoria}: el error de estimación de la cuasidesviación estándar $S$.
		
		En las muestras pequeñas donde $S$ subestima accidentalmente a $\sigma$ ($S \ll \sigma$), el cociente $T = \dfrac{\bar{X} - \mu}{S/\sqrt{n}}$ sufre una inflación numérica importante en el denominador, disparando el valor de $T$ hacia colas extremas. Para capturar matemáticamente esta doble variabilidad aleatoria sin generar un exceso falsos positivos en las pruebas, la densidad de probabilidad $t$ de Student debe ser leptocúrtica: sus colas son estrictamente más gruesas que la normal ($P(|T| > k) > P(|Z| > k)$), otorgando una compensación de prudencia inferencial exacta frente al desconocimiento de $\sigma$.
		
		\item \textbf{Límite asintótico para muestras infinitas:}
		Evaluamos el comportamiento de la varianza cuando $\nu \to \infty$:
		\begin{align}
			\lim_{\nu \to \infty} \Var(T) = \lim_{\nu \to \infty} \left( 1 + \frac{2}{\nu - 2} \right) = 1 + 0 = 1.
		\end{align}
		Este límite es perfectamente coherente y consustancial con el teorema de convergencia de Student: a medida que el tamaño muestral aumenta ($n \to \infty$), el estimador $S^2$ converge en probabilidad hacia la constante poblacional $\sigma^2$ por la Ley de los Grandes Números. Al eliminarse la incertidumbre en el denominador, la distribución de colas pesadas $t_\nu$ converge de manera unívoca e indistinguible en distribución hacia la normal estándar $Z \sim N(0,1)$.
	\end{enumerate}
\end{solproblema}
