% Recordar
\begin{problema}
	\label{prob:4e01dbd}
	Una variable aleatoria continua $X$ tiene PDF triangular $f(x) = 2x$ para $0 \le x \le 1$ y $f(x) = 0$ en otro caso.
	\begin{enumerate}
		\item Calcule la esperanza $\E[X] = \int_{0}^{1} x \cdot 2x\,dx$.
		\item Calcule $\E[X^{2}]$ y verifique la varianza $\Var(X) = \E[X^{2}] - (\E[X])^{2}$.
		\item Calcule la mediana de $X$ y compare con $\E[X]$.
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:f43c638}
	Sea $X \sim \text{Exp}(\lambda = 0.5)$ con PDF $f(x) = 0.5 e^{-0.5 x}$ para $x \ge 0$.
	\begin{enumerate}
		\item Calcule $\E[X] = 1/\lambda$ y $\Var(X) = 1/\lambda^{2}$ usando las definiciones de integral.
		\item Verifique la propiedad de \emph{ausencia de memoria}: $\E[X - a \mid X > a] = \E[X]$ para todo $a > 0$, y explique intuitivamente qué significa esta propiedad para un proceso de espera modelado con esta distribución.
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:7b147c4}
	En un experimento de física, se mide una cantidad $X = \mu + \epsilon$ donde $\mu$ es el valor verdadero y $\epsilon \sim N(0, \sigma^{2})$ es el error de medición. Se realizan $n$ mediciones independientes $X_{1}, \dots, X_{n}$.
	\begin{enumerate}
		\item Calcule $\E[\bar{X}]$ y $\Var(\bar{X})$ donde $\bar{X} = (1/n) \sum X_{i}$ es la media muestral.
		\item Demuestre que el error estándar del estimado es $\sigma/\sqrt{n}$ y decrece con $\sqrt{n}$.
		\item Para $\sigma = 0.1$, determine $n$ tal que $\sigma/\sqrt{n} < 0.01$ (error $< 10\%$ del verdadero valor).
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:de8d740}
	Sea $X$ una variable aleatoria continua con PDF $f(x)$. Use la ley total de esperanza para demostrar la \emph{fórmula de descomposición de varianza}:
	$$\Var(X) = \E[\Var(X \mid Y)] + \Var(\E[X \mid Y])$$
	donde $Y$ es otra variable aleatoria. Esta identidad es fundamental en el análisis de varianza (ANOVA) y modelos jerárquicos.
\end{problema}

\begin{problema}
	\label{prob:9d4a41b}
	Para una variable aleatoria exponencial $X \sim \text{Exp}(\lambda = 1)$:
	\begin{enumerate}
		\item Calcule el coeficiente de asimetría $\gamma_{1} = \E[(X - \mu)^{3}]/\sigma^{3}$ y la curtosis $\gamma_{2} = \E[(X - \mu)^{4}]/\sigma^{4} - 3$.
		\item Evalúe comparativamente estos valores contra los de la Normal estándar ($\gamma_{1} = 0$, $\gamma_{2} = 0$) e interprete qué implican para la forma de la distribución (dirección de la cola y peso de valores extremos).
	\end{enumerate}
\end{problema}

% Crear
\begin{problema}
	\label{prob:5c186f2}
	Diseñe un problema original que aplique la linealidad de la esperanza: un portafolio de inversión consta de tres activos con rendimientos aleatorios $R_1$, $R_2$, $R_3$ (defina sus valores esperados y las ponderaciones $w_1, w_2, w_3$ del portafolio, con $w_1+w_2+w_3=1$). Calcule el rendimiento esperado total del portafolio $\E[w_1R_1+w_2R_2+w_3R_3]$, y explique por qué esta propiedad de linealidad es válida incluso si los rendimientos de los tres activos están correlacionados entre sí.
\end{problema}

\subsection*{Soluciones detalladas}

% Recordar
\begin{solproblema}[prob:4e01dbd]
	\begin{enumerate}
		\item $\E[X] = \int_{0}^{1} x \cdot 2x\,dx = \int_{0}^{1} 2x^{2}\,dx = \left[ \dfrac{2x^{3}}{3} \right]_{0}^{1} = \dfrac{2}{3}$.
		\item $\E[X^{2}] = \int_{0}^{1} 2x^{3}\,dx = \dfrac{1}{2}$. Por lo tanto, $\Var(X) = \dfrac{1}{2} - \dfrac{4}{9} = \dfrac{1}{18} \approx 0.0556$.
		\item La mediana $m$ satisface $F(m) = m^{2} = 0.5$, dando $m = 1/\sqrt{2} \approx 0.7071$. Como $\E[X] = 2/3 \approx 0.6667 < m$, la media es menor que la mediana debido a la asimetría de la distribución triangular: la cola izquierda concentra menos masa pero se extiende comparativamente, desplazando la media por debajo de la mediana.
	\end{enumerate}
\end{solproblema}

% Comprender
\begin{solproblema}[prob:f43c638]
	\begin{enumerate}
		\item Por integración por partes con $u=x$, $dv=0.5e^{-0.5x}dx$:
		\begin{align*}
			\E[X] = \int_{0}^{\infty} x \cdot 0.5 e^{-0.5x}\,dx = \left[-xe^{-0.5x}\right]_0^\infty + \int_0^\infty e^{-0.5x}\,dx = 0+2 = 2 = 1/\lambda.
		\end{align*}
		De manera similar, $\E[X^2] = 8 = 2/\lambda^2$, así que $\Var(X) = 8-4=4=1/\lambda^2$.
		\item Condicionalmente a $X>a$, $X-a$ sigue $\text{Exp}(\lambda)$, por lo que $\E[X-a\mid X>a] = 1/\lambda = \E[X]$. Intuitivamente, esto significa que un proceso de espera exponencial \emph{no envejece}: si ya se ha esperado $a$ unidades de tiempo sin que ocurra el evento, el tiempo de espera adicional esperado es exactamente el mismo que al inicio, como si el reloj se hubiera reiniciado. Esta es la razón por la que la exponencial modela procesos sin fatiga ni desgaste acumulado (por ejemplo, tiempos entre llegadas en teoría de colas).
	\end{enumerate}
\end{solproblema}

% Aplicar
\begin{solproblema}[prob:7b147c4]
	\begin{enumerate}
		\item Cada medición es $X_i=\mu+\epsilon_i$ con $\epsilon_i\sim N(0,\sigma^2)$ independientes. La media muestral es $\bar{X}=\mu+(1/n)\sum\epsilon_i$, así que $\E[\bar{X}]=\mu+0=\mu$, y $\Var(\bar{X})=(1/n^2)\cdot n\sigma^2=\sigma^2/n$.
		\item El error estándar es $\text{SE}(\bar{X})=\sqrt{\Var(\bar{X})}=\sigma/\sqrt{n}$. Para reducir el error a la mitad, es necesario cuadruplicar $n$ (dependencia $\sqrt{n}$).
		\item Para $\sigma/\sqrt{n}<0.01$ con $\sigma=0.1$: $\sqrt{n}>10$, así $n>100$, es decir, se necesitan al menos $n=101$ mediciones. Con $n=100$ el error estándar es exactamente $0.01$ (el límite, no satisface la desigualdad estricta); con $n=101$, $\sigma/\sqrt{n}\approx0.00995<0.01$ (menos del 1\% del valor verdadero).
	\end{enumerate}
\end{solproblema}

% Analizar
\begin{solproblema}[prob:de8d740]
	Por la ley total de esperanza, $\E[X]=\E[\E[X\mid Y]]$. Expandimos $\Var(X)$:
	\begin{align*}
		\Var(X) &= \E[(X-\E[X])^2] = \E\big[\E[(X-\E[X\mid Y]+\E[X\mid Y]-\E[X])^2 \mid Y]\big] \\
		&= \E[\E[(X-\E[X\mid Y])^2\mid Y]] + \E[\E[(\E[X\mid Y]-\E[X])^2\mid Y]] \\
		&\quad + 2\E\big[\E[(X-\E[X\mid Y])(\E[X\mid Y]-\E[X])\mid Y]\big].
	\end{align*}
	El tercer término es cero por la propiedad de torre: $\E[X-\E[X\mid Y]\mid Y]=\E[X\mid Y]-\E[X\mid Y]=0$. El primer término es $\E[\Var(X\mid Y)]$ y el segundo es $\Var(\E[X\mid Y])$. Por lo tanto:
	\begin{align*}
		\Var(X) = \E[\Var(X \mid Y)] + \Var(\E[X \mid Y]).
	\end{align*}
	En ANOVA, la variabilidad total se descompone en \emph{variabilidad dentro de grupos} ($\E[\Var(X\mid Y)]$, análoga a SCE) y \emph{variabilidad entre grupos} ($\Var(\E[X\mid Y])$, análoga a SCTR).
\end{solproblema}

% Evaluar
\begin{solproblema}[prob:9d4a41b]
	Para la exponencial con $\lambda=1$: $\mu=1$, $\sigma=1$, $\E[(X-1)^3]=2$, $\E[(X-1)^4]=9$. Por lo tanto:
	\begin{align*}
		\gamma_1 = 2/1^3 = 2 \quad \text{(asimetría positiva)}, \qquad \gamma_2 = 9/1^4-3 = 6 \quad \text{(exceso de curtosis)}.
	\end{align*}
	Comparado con la Normal estándar ($\gamma_1=0$, $\gamma_2=0$), la exponencial tiene una asimetría positiva marcada: su cola derecha es mucho más larga que la izquierda (que ni siquiera existe, pues el soporte es $[0,\infty)$), reflejando que valores muy por encima de la media, aunque raros, son bastante más probables que en una distribución simétrica. El exceso de curtosis positivo ($\gamma_2=6 \gg 0$) indica una distribución \emph{leptocúrtica}: más apuntada cerca del origen y con colas más pesadas que la Normal, es decir, mayor probabilidad tanto de valores muy pequeños (cercanos a cero) como de valores extremadamente grandes, en comparación con lo que predeciría una aproximación normal.
\end{solproblema}

% Crear
\begin{solproblema}[prob:5c186f2]
	\textbf{Ejemplo:} sea un portafolio con rendimientos esperados $\E[R_1]=0.08$ (8\%, acciones), $\E[R_2]=0.05$ (5\%, bonos), $\E[R_3]=0.12$ (12\%, fondo de alto riesgo), con ponderaciones $w_1=0.40$, $w_2=0.35$, $w_3=0.25$.

	Por linealidad de la esperanza:
	\begin{align*}
		\E[w_1R_1+w_2R_2+w_3R_3] = w_1\E[R_1]+w_2\E[R_2]+w_3\E[R_3] = 0.40(0.08)+0.35(0.05)+0.25(0.12).
	\end{align*}
	\begin{align*}
		= 0.032+0.0175+0.030 = 0.0795 \text{ (7.95\%).}
	\end{align*}
	La linealidad de la esperanza no requiere independencia entre $R_1, R_2, R_3$: la demostración general $\E[aX+bY]=a\E[X]+b\E[Y]$ se basa únicamente en la linealidad de la integral (o suma) que define la esperanza, sin usar en ningún paso la factorización de una densidad conjunta que exigiría independencia. Por eso el rendimiento esperado del portafolio es válido y calculable así incluso si, en la práctica, los tres activos están correlacionados (como suele ocurrir en mercados financieros reales).
\end{solproblema}
