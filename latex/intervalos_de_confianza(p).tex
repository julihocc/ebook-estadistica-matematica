\section*{Problemas}

\subsection{Enunciados de los problemas}

\subsubsection*{1. Nivel Fundamental (Sencillos / Directos)}

\begin{problema}
	\label{prob-ic-1}
	Una muestra aleatoria de $n = 36$ estudiantes obtuvo una calificación promedio muestral de $\bar{X} = 78$ puntos en un examen de matemáticas. Suponga que la desviación estándar poblacional es conocida y exactamente igual a $\sigma = 12$ puntos.
	\begin{enumerate}
		\item Construya un intervalo de confianza del 95\% para la media poblacional $\mu$.
		\item Construya un intervalo de confianza del 99\% para la media poblacional $\mu$.
		\item ¿Qué intervalo posee una amplitud mayor? Explique el motivo teórico por el cual la amplitud varía en función del nivel de confianza.
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob-ic-2}
	Un fabricante de focos LED desea estimar la vida útil promedio de su nueva línea de productos. Toma una muestra aleatoria de $n = 16$ focos y registra un promedio de $\bar{X} = 1,200$ horas con una desviación estándar muestral $s = 80$ horas.
	\begin{enumerate}
		\item ¿Debe utilizar la distribución normal $Z$ o la distribución $t$ de Student para calcular el intervalo de confianza? Argumente su respuesta.
		\item Construya un intervalo de confianza del 95\% para la verdadera vida útil promedio.
		\item Interprete formalmente el intervalo resultante en el contexto del problema.
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob-ic-fund-new}
	Un analista de control de calidad planifica construir un intervalo de confianza paramétrico al 98\% para la resistencia media a la tensión de un hilo sintético, asumiendo una desviación estándar poblacional conocida $\sigma$ en un tamaño muestral grande ($n \ge 30$).
	\begin{enumerate}
		\item Determine la proporción de área $\alpha$ excluida y el área correspondiente a cada cola extrema ($\alpha / 2$).
		\item Calcule el valor crítico $Z_{\alpha/2}$ exacto en la distribución normal estándar para este nivel de confianza del 98\%.
	\end{enumerate}
\end{problema}

\subsubsection*{2. Nivel Operativo (Intermedios / De Desarrollo)}

\begin{problema}
	\label{prob-ic-3}
	Un investigador académico indaga sobre el tiempo promedio semanal que los estudiantes universitarios dedican al estudio autónomo. Selecciona una muestra aleatoria simple de $n = 25$ alumnos y obtiene $\bar{X} = 15.0$ horas junto con una desviación estándar muestral $s = 4.0$ horas.
	\begin{enumerate}
		\item Construya el intervalo de confianza del 90\% para la media poblacional $\mu$.
		\item Construya el intervalo de confianza del 95\% para la misma media.
		\item Si el investigador desea obtener una estimación de mayor precisión (un intervalo de confianza más angosto) conservando el nivel del 95\% de confianza, ¿qué modificaciones metodológicas concretas debe aplicar?
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob-ic-4}
	Una firma de telecomunicaciones requiere determinar el tiempo promedio real de las llamadas telefónicas de sus abonados corporativos. A partir de una auditoría sobre $n = 100$ llamadas, encuentra una media muestral $\bar{X} = 4.5$ minutos con $s = 2.0$ minutos.
	\begin{enumerate}
		\item Construya el intervalo de confianza del 95\% para la duración media $\mu$.
		\item Determine la magnitud del margen de error de estimación ($E$) actual.
		\item Si la gerencia técnica requiere reducir el margen de error máximo tolerado a $E = 0.2$ minutos al mismo nivel del 95\% de confianza, deduzca y calcule el tamaño de muestra mínimo $n$ necesario que se debe recolectar en total.
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob-ic-6}
	El departamento de ingeniería de una planta de baterías afirma en su publicidad que la vida útil de sus celdas de almacenamiento excede los 500 ciclos de recarga. Un laboratorio de certificación independiente prueba $n = 25$ unidades y obtiene $\bar{X} = 490$ ciclos con una cuasidesviación estándar $s = 30$ ciclos.
	\begin{enumerate}
		\item Construya el intervalo de confianza del 95\% para el número promedio real de ciclos $\mu$.
		\item ¿El intervalo construido contiene o excluye el umbral de 500 ciclos afirmado por la publicidad?
		\item Explique cuál es la conclusión técnica respecto a la veracidad de la afirmación y detalle cómo se relaciona este intervalo con el resultado de una prueba de hipótesis bilateral de nivel $\alpha = 0.05$.
	\end{enumerate}
\end{problema}

\subsubsection*{3. Nivel Analítico (Avanzados / De Conexión)}

\begin{problema}
	\label{prob-ic-5}
	Un estudio comparativo de nutrición clínica evalúa la ingesta energética diaria media entre dos subpoblaciones adultas (Hombres y Mujeres). Se extraen dos muestras independientes arrojando los siguientes estadísticos resumen:
	\begin{itemize}
		\item Muestra de hombres: $n_1 = 40$, $\bar{X}_1 = 2,500$ kcal, $s_1 = 300$ kcal.
		\item Muestra de mujeres: $n_2 = 45$, $\bar{X}_2 = 2,100$ kcal, $s_2 = 250$ kcal.
	\end{itemize}
	\begin{enumerate}
		\item Calcule el error estándar estimado para la diferencia de medias independientes $\text{SE}(\bar{X}_1 - \bar{X}_2)$ sin asumir que las varianzas poblacionales sean necesariamente iguales.
		\item Construya un intervalo de confianza del 95\% para la verdadera diferencia de medias poblacionales $\mu_1 - \mu_2$.
		\item Con base en si el intervalo incluye al cero, ¿existe una diferencia biológica o conductual estadísticamente significativa entre ambos géneros en su consumo energético?
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob-ic-analit-pooled}
	\textbf{Intervalo para la diferencia de medias con varianza ponderada común ($S_p^2$):}
	Un químico industrial ensaya dos catalizadores alternativos ($A$ y $B$) para mejorar el rendimiento porcentual de una reacción termoquímica en lotes pequeños. Bajo el supuesto razonable de normalidad e igualdad de varianzas poblacionales ($\sigma_1^2 = \sigma_2^2$), obtiene las siguientes mediciones:
	\begin{itemize}
		\item Catalizador A: $n_1 = 10$, $\bar{X}_1 = 85.0\%$, $s_1 = 4.0\%$.
		\item Catalizador B: $n_2 = 12$, $\bar{X}_2 = 81.0\%$, $s_2 = 5.0\%$.
	\end{itemize}
	\begin{enumerate}
		\item Deduzca y calcule la varianza combinada ponderada por los grados de libertad (cuasivarianza agrupada o \textit{pooled variance}) $S_p^2$.
		\item Construya un intervalo de confianza del 95\% para la verdadera diferencia en el rendimiento medio $\mu_A - \mu_B$ utilizando la distribución $t$ con $\nu = n_1 + n_2 - 2$ grados de libertad.
	\end{enumerate}
\end{problema}

\subsubsection*{4. Nivel Desafiante (Expertos / Deducción Profunda)}

\begin{problema}
	\label{prob-ic-desaf-muestra-bern}
	\textbf{Deducción analítica de la cota conservadora del tamaño muestral para una proporción Bernoulli:}
	Al diseñar una encuesta o sondeo de opinión para estimar la verdadera proporción poblacional $p \in (0,1)$ con un margen de error máximo admitido $E$ y una confianza $(1-\alpha)$, la expresión algebraica que define el semiancho del intervalo asintótico normal es:
	\begin{align}
		E = Z_{\alpha/2} \sqrt{\frac{p(1-p)}{n}}.
	\end{align}
	\begin{enumerate}
		\item Utilizando el cálculo diferencial o propiedades trigonométricas/algebraicas de las funciones cuadráticas, demuestre rigurosamente que la varianza muestral teórica de Bernoulli $g(p) = p(1-p)$ alcanza un máximo global absoluto en el dominio $p \in [0,1]$ exactamente en $p = 0.5$.
		\item Demuestre que si no se dispone de información a priori ni estudios piloto confiables sobre $p$, la fórmula que garantiza que el margen de error nunca excederá el límite $E$ es:
		\begin{align}
			n = \left\lceil \frac{Z_{\alpha/2}^2}{4 E^2} \right\rceil.
		\end{align}
		\item Una consultora de ciencia de datos requiere estimar la intención de voto hacia una iniciativa pública de salud con un margen de error máximo del $3\%$ ($E = 0.03$) y un nivel de confianza del 95\%. Calcule el número mínimo de encuestas válidas ($n$) que deberá garantizar.
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob-ic-desaf-epistemologia}
	\textbf{Fundamentación epistemológica y dualidad matemática del intervalo de confianza:}
	Considere la construcción formal del intervalo de confianza empírico $(1-\alpha)$ bajo normalidad:
	\begin{align}
		[L(X), \, U(X)] = \left[ \bar{X} - Z_{\alpha/2}\frac{\sigma}{\sqrt{n}}, \; \bar{X} + Z_{\alpha/2}\frac{\sigma}{\sqrt{n}} \right].
	\end{align}
	\begin{enumerate}
		\item Demuestre probabilísticamente antes de la recolección de los datos (ex-ante) que los límites del intervalo son variables aleatorias que satisfacen exactamente:
		\begin{align}
			P\left( L(X) \le \mu \le U(X) \right) = 1 - \alpha.
		\end{align}
		\item Suponga que, una vez tomada la muestra real particular de 36 estudiantes del Problema \ref{prob-ic-1}, se evalúan los datos y se publica el intervalo numérico cerrado $[74.08, \, 81.92]$. Explique analíticamente por qué, desde la axiomática estricta de la estadística frecuentista de Kolmogorov, es teóricamente erróneo enunciar que:
		\begin{quote}
			``La probabilidad de que la verdadera calificación media $\mu$ esté dentro del intervalo $[74.08, \, 81.92]$ es del $95\%$ (`$P(74.08 \le \mu \le 81.92) = 0.95$')''.
		\end{quote}
		\item Defina rigurosamente qué significa una ``confianza del 95\%'' desde la perspectiva del muestreo repetido infinito en el tiempo, y contrástelo con el concepto epistemológico del \textbf{intervalo de credibilidad Bayesiano}.
	\end{enumerate}
\end{problema}


\subsection{Soluciones detalladas}

\subsubsection*{1. Nivel Fundamental (Sencillos / Directos)}

\begin{solucion}
	\begin{enumerate}
		\item \textbf{Intervalo de confianza del 95\% ($\alpha = 0.05$):}
		El cuantil crítico en la distribución normal tipificada es $Z_{0.025} = 1.96$.
		La fórmula general del intervalo normal es:
		\begin{align}
			IC_{95\%} = \bar{X} \pm Z_{\alpha/2} \frac{\sigma}{\sqrt{n}} = 78 \pm 1.96 \left( \frac{12}{\sqrt{36}} \right).
		\end{align}
		Simplificando la desviación estándar de la media ($\text{SE} = 12 / 6 = 2$):
		\begin{align}
			IC_{95\%} = 78 \pm 1.96(2) = 78 \pm 3.92 = [74.08, \; 81.92].
		\end{align}
		
		\item \textbf{Intervalo de confianza del 99\% ($\alpha = 0.01$):}
		Para un nivel de confianza del 99\%, el valor crítico en cada cola es $Z_{0.005} = 2.576$:
		\begin{align}
			IC_{99\%} = 78 \pm 2.576(2) = 78 \pm 5.152 = [72.85, \; 83.15].
		\end{align}
		
		\item \textbf{Comparación y justificación teórica de la amplitud:}
		Calculamos la amplitud ($A = U - L$) para ambos intervalos:
		\begin{align}
			A_{95\%} &= 81.92 - 74.08 = 7.84 \text{ puntos.} \\
			A_{99\%} &= 83.15 - 72.85 = 10.30 \text{ puntos.}
		\end{align}
		El intervalo del 99\% posee una amplitud notablemente superior ($10.30 > 7.84$). Esto ocurre por el principio de incertidumbre de la inferencia frecuentista: para incrementar nuestra certeza (confianza) de no dejar afuera el verdadero parámetro $\mu$ cuando la cantidad de información observada ($n$) permanece constante, nos vemos obligados a expandir la longitud del margen de captura, abarcando una fracción más amplia del área bajo las colas de la campana normal.
	\end{enumerate}
\end{solucion}

\begin{solucion}
	\begin{enumerate}
		\item Se debe utilizar obligatoriamente la \textbf{distribución $t$ de Student}. La decisión se justifica porque confluyen dos condiciones teóricas indiscutibles:
		\begin{itemize}
			\item El tamaño de la muestra es pequeño ($n = 16 < 30$), lo que impide invocar la aproximación normal del Teorema del Límite Central.
			\item La desviación estándar poblacional $\sigma$ es desconocida, debiendo sustituirse por el estimador empírico cuasidesviación $s = 80$ horas, el cual introduce un error de estimación adicional que requiere las colas pesadas de la distribución $t$.
		\end{itemize}
		
		\item \textbf{Cálculo del intervalo al 95\% con $\nu = 15$ grados de libertad:}
		Los grados de libertad exactos son $\nu = n - 1 = 16 - 1 = 15$. Para $\alpha = 0.05$, el valor crítico de la tabla Student es $t_{0.025, \, 15} = 2.131$.
		\begin{align}
			IC_{95\%} &= \bar{X} \pm t_{\alpha/2, \, \nu} \frac{s}{\sqrt{n}} = 1,200 \pm 2.131 \left( \frac{80}{\sqrt{16}} \right) \\
			&= 1,200 \pm 2.131 \left( \frac{80}{4} \right) = 1,200 \pm 2.131(20) = 1,200 \pm 42.62 \\
			&= [1,157.38, \; 1,242.62] \text{ horas.}
		\end{align}
		
		\item \textbf{Interpretación:} Si repitiéramos este proceso de muestreo (extraer 16 focos al azar y calcular el intervalo) en una serie indefinida de ocasiones, en el $95\%$ de esas muestras la vida útil media verdadera $\mu$ de toda la producción del fabricante quedará capturada entre 1,157.38 y 1,242.62 horas de funcionamiento.
	\end{enumerate}
\end{solucion}

\begin{solucion}
	\begin{enumerate}
		\item \textbf{Cálculo de las áreas en las colas:}
		El nivel de confianza del $98\%$ corresponde a una probabilidad central de $1 - \alpha = 0.98$.
		El complemento total excluido o riesgo de error es por lo tanto:
		\begin{align}
			\alpha = 1 - 0.98 = 0.02 \text{ (2\%).}
		\end{align}
		Por la simetría par de la función normal tipificada alrededor de cero, esta masa exterior se divide en partes idénticas en ambas colas de la distribución:
		\begin{align}
			\frac{\alpha}{2} = \frac{0.02}{2} = 0.01 \text{ en cada extremo de la campana.}
		\end{align}
		
		\item \textbf{Valor crítico exacto $Z_{\alpha/2}$:}
		El valor crítico positivo $Z_{0.01}$ es la abscisa que acumula a su izquierda el $(1 - 0.01) = 0.99$ de la probabilidad total bajo la normal estándar:
		\begin{align}
			\Phi(Z_{0.01}) = 0.9900.
		\end{align}
		Consultando las tablas inversas o mediante interpolación numérica estándar de la función probit, obtenemos:
		\begin{align}
			Z_{0.01} = 2.326.
		\end{align}
	\end{enumerate}
\end{solucion}

\subsubsection*{2. Nivel Operativo (Intermedios / De Desarrollo)}

\begin{solucion}
	\begin{enumerate}
		\item \textbf{Intervalo de confianza del 90\% ($\alpha = 0.10$):}
		Con $n = 25$, los grados de libertad son $\nu = 24$. Para una confianza de 0.90, el cuantil de la tabla $t$ es $t_{0.05, \, 24} = 1.711$:
		\begin{align}
			IC_{90\%} &= \bar{X} \pm t_{0.05, \, 24} \frac{s}{\sqrt{n}} = 15.0 \pm 1.711 \left( \frac{4.0}{\sqrt{25}} \right) \\
			&= 15.0 \pm 1.711(0.8) = 15.0 \pm 1.369 = [13.63, \; 16.37] \text{ horas.}
		\end{align}
		
		\item \textbf{Intervalo de confianza del 95\% ($\alpha = 0.05$):}
		Para 0.95 de confianza en $\nu = 24$, el valor crítico asciende a $t_{0.025, \, 24} = 2.064$:
		\begin{align}
			IC_{95\%} &= 15.0 \pm 2.064(0.8) = 15.0 \pm 1.651 = [13.35, \; 16.65] \text{ horas.}
		\end{align}
		
		\item \textbf{Metodologías para mejorar la precisión y estrechar el intervalo al 95\%:}
		Para reducir la amplitud del intervalo conservando el 95\% de confiabilidad ($Z$ o $t$ fijos), el investigador dispone de dos caminos matemáticos:
		\begin{itemize}
			\item \textbf{Incrementar el tamaño de la muestra ($n$):} Como el error estándar de la estimación es inversamente proporcional a la raíz cuadrada de $n$ ($\text{SE} = s / \sqrt{n}$), cuadruplicar el tamaño muestral (ej. subir de $n=25$ a $n=100$) reduce exactamente a la mitad la incertidumbre del margen de error, produciendo un intervalo mucho más acotado.
			\item \textbf{Mejorar el diseño muestral o reducir la variabilidad empírica ($s$):} Si se estratifica la muestra por facultades o áreas del conocimiento, o se emplean técnicas estandarizadas de autoregistro horario con aplicaciones móviles para evitar sesgos de memoria, se puede reducir la dispersión interna en los datos observados ($s$), haciendo el intervalo más compacto.
		\end{itemize}
	\end{enumerate}
\end{solucion}

		Al ser una muestra grande ($n = 100 \ge 30$), el Teorema del Límite Central garantiza que la media muestral se distribuye asintóticamente normal. Con $Z_{0.025} = 1.96$:
		\begin{align}
			IC_{95\%} &= \bar{X} \pm Z_{0.025} \frac{s}{\sqrt{n}} = 4.5 \pm 1.96 \left( \frac{2.0}{\sqrt{100}} \right) \\

			&= [4.108, \; 4.892] \text{ minutos.}
		\end{align}
		
		\item \textbf{Margen de error de estimación ($E$):}
		El semiancho de incertidumbre del intervalo actual es $E = 0.392$ minutos (aproximadamente 23.5 segundos).
		
		\item \textbf{Deducción del nuevo tamaño de muestra para una cota preestablecida $E = 0.2$:}
		Partimos de la definición algebraica del margen de error:
		\begin{align}
			E = Z_{\alpha/2} \frac{s}{\sqrt{n}}.
		\end{align}
		Despejando la variable radical $\sqrt{n}$:
		\begin{align}
			\sqrt{n} = \frac{Z_{\alpha/2} \cdot s}{E}.
		\end{align}
		Elevando al cuadrado ambos miembros obtenemos la fórmula del tamaño muestral requerido:
		\begin{align}
			n = \left( \frac{Z_{\alpha/2} \cdot s}{E} \right)^2.
		\end{align}
		Sustituyendo los valores del estudio ($Z_{0.025} = 1.96$, $s = 2.0$, $E = 0.2$):
		\begin{align}
			n = \left( \frac{1.96 \times 2.0}{0.2} \right)^2 = \left( \frac{3.92}{0.2} \right)^2 = (19.6)^2 = 384.16.
		\end{align}
		Dado que no podemos muestrear una fracción de llamada, aplicamos la regla del techo ($\lceil x \rceil$) redondeando siempre hacia el entero inmediato superior:
		\begin{align}
			n_{\text{óptimo}} = \lceil 384.16 \rceil = 385 \text{ llamadas.}
		\end{align}
		La empresa requerirá recolectar una muestra de al menos 385 llamadas en total para asegurar un margen de error igual o menor a 12 segundos ($0.2$ minutos).
	\end{enumerate}
\end{solucion}

\subsubsection*{3. Nivel Analítico (Avanzados / De Conexión)}

\begin{solucion}
	\begin{enumerate}
		\item \textbf{Cálculo del error estándar de la diferencia de medias independientes sin suposición de varianzas homogéneas ($\sigma_1^2 \neq \sigma_2^2$):}
		Por la independencia de los dos grupos de estudio, la varianza de la diferencia entre medias muestrales es la suma de las varianzas individuales ($\Var(\bar{X}_1 - \bar{X}_2) = \Var(\bar{X}_1) + \Var(\bar{X}_2)$). Su raíz cuadrada es el error estándar estimado:
		\begin{align}
			\text{SE}(\bar{X}_1 - \bar{X}_2) = \sqrt{\frac{s_1^2}{n_1} + \frac{s_2^2}{n_2}}.
		\end{align}
		Sustituyendo las varianzas muestrales observadas en hombres ($s_1 = 300, n_1 = 40$) y mujeres ($s_2 = 250, n_2 = 45$):
		\begin{align}
			\text{SE}(\bar{X}_1 - \bar{X}_2) &= \sqrt{\frac{300^2}{40} + \frac{250^2}{45}} = \sqrt{\frac{90,000}{40} + \frac{62,500}{45}} \\
			&= \sqrt{2,250 + 1,388.889} = \sqrt{3,638.889} \approx 60.323 \text{ kcal/día.}
		\end{align}
		
		\item \textbf{Construcción del intervalo del 95\% para $\mu_1 - \mu_2$:}
		La diferencia elemental entre promedios muestrales es:
		\begin{align}
			\hat{D} = \bar{X}_1 - \bar{X}_2 = 2,500 - 2,100 = 400 \text{ kcal/día.}
		\end{align}
		Como las dos muestras superan holgadamente las 30 observaciones ($n_1 = 40, n_2 = 45$), invocamos la aproximación normal asintótica del Teorema del Límite Central con $Z_{0.025} = 1.96$:
		\begin{align}

		Cuando se asume que las dos poblaciones normales poseen una varianza común idéntica ($\sigma_1^2 = \sigma_2^2 = \sigma^2$), las estimaciones individuales $s_1^2$ y $s_2^2$ son dos observaciones independientes estimando el mismo parámetro $\sigma^2$. Para maximizar los grados de libertad sin introducir sesgo, se construye la **varianza agrupada o combinada ($S_p^2$, \textit{pooled variance})**, la cual es el promedio ponderado de las cuasivarianzas muestrales en función de sus respectivos grados de libertad internos ($n_i - 1$):
		\begin{align}
			S_p^2 = \frac{(n_1 - 1)s_1^2 + (n_2 - 1)s_2^2}{n_1 + n_2 - 2}.
		\end{align}
		Sustituyendo los datos de los catalizadores A ($n_1=10, s_1=4.0$) y B ($n_2=12, s_2=5.0$):
		\begin{align}
			S_p^2 &= \frac{(10 - 1)(4.0)^2 + (12 - 1)(5.0)^2}{10 + 12 - 2} = \frac{9(16.0) + 11(25.0)}{20} \\
			&= \frac{144.0 + 275.0}{20} = \frac{419.0}{20} = 20.95 \implies S_p = \sqrt{20.95} \approx 4.5771\%.
		\end{align}
		
		\item \textbf{Construcción del intervalo para la diferencia con $\nu = 20$:}
		El error estándar de la diferencia para varianza agrupada es:
		\begin{align}
			\text{SE}_p(\bar{X}_1 - \bar{X}_2) = \sqrt{S_p^2 \left( \frac{1}{n_1} + \frac{1}{n_2} \right)} = \sqrt{20.95 \left( \frac{1}{10} + \frac{1}{12} \right)} = \sqrt{20.95(0.18333)} \approx 1.9598\%.

			IC_{95\%}(\mu_A - \mu_B) &= (\bar{X}_1 - \bar{X}_2) \pm t_{0.025, \, 20} \cdot \text{SE}_p \\

\subsubsection*{4. Nivel Desafiante (Expertos / Deducción Profunda)}

\begin{solucion}
	\begin{enumerate}
		\item \textbf{Maximización analítica de la varianza elemental de Bernoulli:}
		Definamos la función de varianza paramétrica en el intervalo unitario:
		\begin{align}
			g(p) = p(1 - p) = p - p^2, \quad p \in [0, 1].
		\end{align}
		Derivamos la función respecto a la proporción $p$ e igualamos a cero para hallar puntos críticos:
		\begin{align}
			g'(p) = \frac{d}{dp}(p - p^2) = 1 - 2p = 0 \implies 2p = 1 \implies p^* = 0.5.
		\end{align}
		Comprobamos la condición suficiente de máximo absoluto mediante la segunda derivada:
		\begin{align}
			g''(p) = \frac{d^2}{dp^2}(p - p^2) = -2 < 0.
		\end{align}
		Al ser la concavidad uniformemente negativa en todo su dominio, queda demostrado en forma irrevocable que $g(p)$ alcanza un supremo global absoluto justo cuando el evento se encuentra en máxima incertidumbre ($p = 0.5$). El valor numérico de dicha cota máxima es:
		\begin{align}
			g(0.5) = 0.5(1 - 0.5) = 0.5 \times 0.5 = 0.25 = \frac{1}{4}.
		\end{align}
		
		\item \textbf{Deducción rigurosa de la cota de tamaño muestral:}
		Para garantizar que un intervalo de confianza al $(1-\alpha)$ posea un semiancho inferior o igual a un margen preestablecido $E$, se impone la condición algebraica general:
		\begin{align}
			Z_{\alpha/2} \sqrt{\frac{p(1-p)}{n}} \le E.
		\end{align}
		Elevando al cuadrado ambos lados de la inecuación:
		\begin{align}
			Z_{\alpha/2}^2 \frac{p(1-p)}{n} \le E^2 \implies n \ge \frac{Z_{\alpha/2}^2 \cdot p(1-p)}{E^2}.
		\end{align}
		Dado que la proporción poblacional real $p$ es desconocida a priori por el encuestador, para que la inecuación $n \ge \dots$ sea cierta **independientemente del verdadero valor que tome $p$ en la realidad**, se debe reemplazar el numerador variable por su cota superior uniforme extrema probada en la Parte 1 ($p(1-p) \le 0.25$):
		\begin{align}
			n \ge \frac{Z_{\alpha/2}^2 (0.25)}{E^2} = \frac{Z_{\alpha/2}^2}{4 E^2}.
		\end{align}
		Así se deduce la célebre fórmula de cota conservadora de muestras para proporciones:
		\begin{align}
			n = \left\lceil \frac{Z_{\alpha/2}^2}{4 E^2} \right\rceil.
		\end{align}
		
		\item \textbf{Cálculo aplicado para la consultora ($E = 0.03$, confianza 95\%):}
		Sustituimos $Z_{0.025} = 1.96$ y $E = 0.03$ en la ecuación deducida:
		\begin{align}
			n &= \frac{(1.96)^2}{4 (0.03)^2} = \frac{3.8416}{4 (0.0009)} = \frac{3.8416}{0.0036} = 1,067.111.
		\end{align}
		Aplicando la función de redondeo superior hacia enteros exactos para no ceder una milésima de precisión en el margen final:
		\begin{align}
			n_{\text{óptimo}} = \lceil 1,067.111 \rceil = 1,068 \text{ encuestas válidas.}
		\end{align}

		\item \textbf{Demostración probabilística ex-ante (antes de muestrear):}
		Sea $\bar{X} \sim N\left(\mu, \dfrac{\sigma^2}{n}\right)$ la variable aleatoria que describe el promedio muestral antes de tomar los datos. Por la teoría del Límite Central, la variable estandarizada cumple:
		\begin{align}
			Z = \frac{\bar{X} - \mu}{\sigma/\sqrt{n}} \sim N(0,1).
		\end{align}
		Por definición de cuantiles simétricos al nivel $(1-\alpha)$, se cumple que el evento aleatorio $Z$ esté contenido en el intervalo $[-Z_{\alpha/2}, \, +Z_{\alpha/2}]$ con probabilidad exacta:
		\begin{align}
			P\left( -Z_{\alpha/2} \le \frac{\bar{X} - \mu}{\sigma/\sqrt{n}} \le +Z_{\alpha/2} \right) = 1 - \alpha.
		\end{align}
		Multiplicamos toda la doble desigualdad en el interior del operador de probabilidad por la constante estrictamente positiva de escala de error estándar $\dfrac{\sigma}{\sqrt{n}}$:
		\begin{align}
			P\left( -Z_{\alpha/2}\frac{\sigma}{\sqrt{n}} \le \bar{X} - \mu \le +Z_{\alpha/2}\frac{\sigma}{\sqrt{n}} \right) = 1 - \alpha.
		\end{align}
		Restamos la media muestral aleatoria $\bar{X}$ en los tres miembros interiores:
		\begin{align}
			P\left( -\bar{X} - Z_{\alpha/2}\frac{\sigma}{\sqrt{n}} \le -\mu \le -\bar{X} + Z_{\alpha/2}\frac{\sigma}{\sqrt{n}} \right) = 1 - \alpha.
		\end{align}
		Multiplicamos por la constante negativa $(-1)$ en toda la inecuación, recordando que en álgebra esto invierte exactamente el sentido de las desigualdades:
		\begin{align}
			P\left( \bar{X} + Z_{\alpha/2}\frac{\sigma}{\sqrt{n}} \ge \mu \ge \bar{X} - Z_{\alpha/2}\frac{\sigma}{\sqrt{n}} \right) = 1 - \alpha.
		\end{align}
		Reordenando los extremos de menor a mayor, definimos las funciones del límite inferior $L(X) = \bar{X} - Z_{\alpha/2}\frac{\sigma}{\sqrt{n}}$ y superior $U(X) = \bar{X} + Z_{\alpha/2}\frac{\sigma}{\sqrt{n}}$, demostrando plenamente el teorema:
		\begin{align}
			P\left( L(X) \le \mu \le U(X) \right) = 1 - \alpha.
		\end{align}
		
		\item \textbf{Fallo epistemológico del enunciado probabilístico ex-post:}
		En el paradigma de la estadística Frecuentista Clásica formulado por Neyman y Pearson, el parámetro poblacional verdadero $\mu$ es una **constante de la naturaleza fija, inmutable y única**, absolutamente desprovista de azar (no es una variable aleatoria).
		
		Una vez que se extrae la muestra del mundo físico y los números se introducen en la fórmula, la variable aleatoria $\bar{X}$ colapsa a la realización observada constante $\bar{x} = 78$, convirtiendo al intervalo en el conjunto numérico rígidamente delimitado $[74.08, \, 81.92]$.
		
		En este punto ex-post, **todos los elementos de la expresión son constantes deterministas**. Por lo tanto, el evento $\mu \in [74.08, \, 81.92]$ no tiene ninguna aleatoriedad asociada. Para un observador omnisciente, solo existen dos estados matemáticamente válidos para esa probabilidad particular:
		\begin{align}
			P(74.08 \le \mu \le 81.92) = 
			\begin{cases}
				1, & \text{si el verdadero } \mu \text{ está efectivamente en este rango.} \\
				0, & \text{si el verdadero } \mu \text{ quedó fuera de esta realización.}
			\end{cases}
		\end{align}
		Asignar un número decimal o porcentual pariente como ``$0.95$'' a ese intervalo empírico específico es un absurdo conceptual formal al confundir la probabilidad del método de muestreo con la probabilidad de un parámetro fijo.
		
		\item \textbf{Definición axiomática correcta y contraste con el paradigma Bayesiano:}
		\begin{itemize}
			\item \textbf{Interpretación Frecuentista (Confianza del 95\%):} La confianza es una propiedad exclusiva del **proceso o procedimiento generador del intervalo**, no del intervalo numérico particular arrojado. Significa que si repitiéramos el experimento de muestreo aleatorio un número infinito de veces, generando miles de intervalos $[L(X_k), U(X_k)]$ con los diferentes promedios que irían surgiendo por azar, exactamente el **$95\%$ de todos los intervalos aleatorios construidos por la fórmula envolverán exitosamente al parámetro verdadero $\mu$**, mientras que el $5\%$ restante errará por efecto marginal del muestreo.
			\item \textbf{Contraste con el Intervalo de Credibilidad Bayesiano:} Si un científico o científico de datos realmente necesita hacer afirmaciones probabilísticas directas e intuitivas sobre dónde se encuentra el parámetro desconocido una vez observados los datos, debe abandonar el paradigma frecuentista y adoptar el **análisis Bayesiano**. En la inferencia Bayesiana, el parámetro $\theta$ sí se trata explícitamente como una variable aleatoria dotada de una distribución a priori $\pi(\theta)$. Al condicionar en los datos muestrales observados $X=x$, el Teorema de Bayes calcula la **distribución a posteriori** $\pi(\theta \mid X=x)$. A partir de ella se construye el *Intervalo de Credibilidad Bayesiano* $[A, B]$ tal que:
			\begin{align}
				P(A \le \theta \le B \mid X = x) = 0.95.
			\end{align}
			Aquí sí es riguroso, legal y epistemológicamente irreprochable afirmar que, dados los datos y la información previa, existe una probabilidad del $95\%$ de que el parámetro $\theta$ resida actualmente dentro de los límites calculados.
		\end{itemize}
	\end{enumerate}
\end{solucion}
