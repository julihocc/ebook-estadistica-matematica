\section{Solved Problems in Multiple Linear Regression}

\subsection{Problem Statements}

\subsubsection*{1. Fundamental Level (Simple / Direct)}

\begin{problema}
	\label{prob:reg_mult_combustible}
	We investigate the average fuel efficiency ($Y$, in km/l) of urban automobiles as a function of two continuous predictor variables: gross vehicle weight ($X_1$, in metric tons) and engine displacement ($X_2$, in liters of capacity). After collecting a random sample of $n = 30$ automobiles under standardized test conditions, the following multiple linear regression model is fitted via Ordinary Least Squares:
	\begin{align}
		\hat{Y} = 25.0 - 3.0X_1 - 1.5X_2, \quad R^2 = 0.780, \quad R_{\text{adj}}^2 = 0.763.
	\end{align}
	\begin{enumerate}
		\item Interpret the physical and geometric meaning of the partial regression coefficients $\hat{\beta}_1 = -3.0$ and $\hat{\beta}_2 = -1.5$, explicitly stating the \emph{ceteris paribus} condition.
		\item Compute the point prediction of expected fuel efficiency for a compact sedan weighing $1.5$ metric tons with a $2.0$-liter engine.
		\item If the total sample variance of observed fuel efficiencies across the 30 vehicles was $S_Y^2 = 12.0$, compute the Regression Sum of Squares ($\text{SSR}$) and the Residual Sum of Squares ($\text{SSD}$).
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:reg_mult_pruebas_t}
	In the vehicular fuel efficiency model from Problem \ref{prob:reg_mult_combustible} with $n = 30$ observations ($p = 2$ regressors plus 1 intercept), analytical software reports that the estimated standard errors for each regression parameter are:
	\begin{align}
		SE(\hat{\beta}_0) = 2.50, \quad SE(\hat{\beta}_1) = 0.80, \quad SE(\hat{\beta}_2) = 0.60.
	\end{align}
	\begin{enumerate}
		\item Formally state the null and alternative hypothesis sets to conduct individual significance tests on vehicle weight ($X_1$) and engine displacement ($X_2$).
		\item Compute Student's $t$ test statistic for each partial coefficient ($t_j = \dfrac{\hat{\beta}_j}{SE(\hat{\beta}_j)}$).
		\item At a two-tailed significance level $\alpha = 0.05$ (with critical quantile $t_{0.025, \, 27} = 2.052$), determine whether both predictors contribute individually and statistically to the model.
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:reg_mult_r2_adj_basico}
	A marketing consulting firm analyzes the impact of advertising channel diversification on corporate monthly sales ($Y$, in thousands of dollars). In a historical sample of $n = 40$ evaluated quarters using a model with three independent variables ($p = 3$: TV expenditure $X_1$, radio expenditure $X_2$, and number of sales executives $X_3$), the explained variance yielded a nominal coefficient of determination of $R^2 = 0.780$.
	\begin{enumerate}
		\item Algebraically compute the Adjusted Coefficient of Determination ($R_{\text{adj}}^2$) via the degrees of freedom correction formula:
		\begin{align}
			R_{\text{adj}}^2 = 1 - \left[ \frac{n - 1}{n - p - 1} (1 - R^2) \right].
		\end{align}
		\item Explain why, in multivariate analysis, the indicator $R_{\text{adj}}^2$ is strictly smaller than the nominal $R^2$ ($R_{\text{adj}}^2 < R^2$) whenever two or more predictors are included ($p \geq 2$).
		\item If a junior analyst proposes adding a fourth predictor $X_4$ representing the numerical zip code of corporate headquarters (exhibiting zero true marginal correlation with sales, such that $R^2$ barely rises from $0.780$ to $0.781$), quantitatively determine what happens to the new $R_{\text{adj}}^2$ of the model with $p = 4$.
	\end{enumerate}
\end{problema}

\subsubsection*{2. Operational Level (Intermediate / Development)}

\begin{problema}
	\label{prob:reg_mult_ventas_anidados}
	To optimize the quarterly sales projections ($Y$, in thousands of dollars) from Problem \ref{prob:reg_mult_r2_adj_basico} ($n = 40$), corporate management compares two nested regression model specifications:
	\begin{itemize}
		\item \textbf{Model 1 (Restricted, $p_1 = 2$):} $\hat{Y} = 50.0 + 2.5X_1 + 1.8X_2$, with $R_1^2 = 0.750$.
		\item \textbf{Model 2 (Full, $p_2 = 3$):} $\hat{Y} = 45.0 + 2.3X_1 + 1.6X_2 + 0.80X_3$, with $R_2^2 = 0.780$.
	\end{itemize}
	\begin{enumerate}
		\item Compute the respective $R_{\text{adj}}^2$ values for Model 1 and Model 2.
		\item Conduct a \textbf{Partial (Incremental) $F$-Test} ($H_0: \beta_3 = 0$) at significance level $\alpha = 0.05$ to rigorously assess whether adding the number of sales executives ($X_3$) significantly improves model fit, computing the statistic:
		\begin{align}
			F_{\text{partial}} = \frac{(R_2^2 - R_1^2) / (p_2 - p_1)}{(1 - R_2^2) / (n - p_2 - 1)},
		\end{align}
		and comparing it against Fisher's tabulated critical quantile $F_{0.05, \, 1, \, 36} = 4.113$.
		\item Formulate a comprehensive argument regarding which model is both technically and managerially recommended.
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:reg_mult_vif_viviendas}
	A multiple regression model is calibrated to predict the appraised value of residential housing properties ($Y$, in thousands of dollars) using $p = 4$ structural and locational characteristics: built area ($X_1$, in $\text{m}^2$), number of bedrooms ($X_2$), built age ($X_3$, in years), and radial distance to the urban center ($X_4$, in km). From a sample of $n = 50$ properties, the following model was obtained:
	\begin{align}
		\hat{Y} = 50.0 + 0.80X_1 + 15.0X_2 - 2.0X_3 - 5.0X_4,
	\end{align}
	and the Variance Inflation Factors ($\text{VIF}_j = \dfrac{1}{1 - R_j^2}$, where $R_j^2$ is the multiple correlation of regressing variable $X_j$ against all other $p-1$ predictors) are:
	\begin{align}
		\text{VIF}(X_1) = 4.2, \quad \text{VIF}(X_2) = 3.8, \quad \text{VIF}(X_3) = 1.5, \quad \text{VIF}(X_4) = 1.2.
	\end{align}
	\begin{enumerate}
		\item Interpret economically the meaning of the area coefficient ($\hat{\beta}_1 = 0.80$) and the distance coefficient ($\hat{\beta}_4 = -5.0$).
		\item Evaluate collinearity diagnostics via the magnitudes of $\text{VIF}_j$. Is there severe multicollinearity invalidating the model, or does it represent an acceptable moderate degree?
		\item Compute the projected appraisal value for a residence with $120\text{ m}^2$, $3$ bedrooms, an age of $10$ years, and situated exactly $8\text{ km}$ from the city commercial center.
		\item What modeling strategy (e.g., dimensionality reduction, standardization, or variable selection) would you apply if $\text{VIF}(X_1)$ exceeded the critical threshold of 10?
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:reg_mult_dummy_salarios}
	A data scientist in a human capital department models annual salary compensation ($Y$, in thousands of dollars) as a function of years of professional experience ($X_1$) and educational attainment level. Rather than making the mistake of coding education as a false ordinal variable ($1, 2, 3$), the scientist constructs a system of \textbf{indicator or dummy variables}, using professionals with \emph{High School / Secondary Education} as the reference baseline group ($D_1 = D_2 = 0$). We define:
	\begin{align}
		D_1 = \begin{cases} 1 & \text{if employee holds a Bachelor's Degree / College Degree} \\ 0 & \text{otherwise} \end{cases}, \qquad D_2 = \begin{cases} 1 & \text{if employee holds a Master's Degree / Advanced Degree} \\ 0 & \text{otherwise} \end{cases}.
	\end{align}
	Using a random sample of $n = 100$ employees, the parametric linear model is fitted:
	\begin{align}
		\hat{Y} = 22.0 + 2.5X_1 + 12.0D_1 + 25.0D_2, \quad R^2 = 0.760.
	\end{align}
	\begin{enumerate}
		\item Interpret the numerical values of the intercept $\hat{\beta}_0 = 22.0$ and categorical coefficients $\hat{\beta}_2 = 12.0$ and $\hat{\beta}_3 = 25.0$ regarding company salaries.
		\item Write the explicit regression lines for salary as a function of experience ($X_1$) separated individually for each of the three educational subgroups (High School, Bachelor's, and Master's).
		\item Predict the expected salary for a manager with $8$ years of experience who holds a Master's degree.
		\item Rigorously explain what constitutes the \textbf{Dummy Variable Trap} in linear matrix algebra and why it would have been fatal had the analyst included a third dummy variable $D_0 = 1$ for the High School group alongside the intercept $\beta_0$.
	\end{enumerate}
\end{problema}

\subsubsection*{3. Analytical Level (Advanced / Synthesis)}

\begin{problema}
	\label{prob:reg_mult_matricial_exacto}
	In an econometric study across $n = 5$ temporal observations modeling percentage return on equity ($Y$) as a function of two predictive indices ($X_1$: liquidity index, $X_2$: financial leverage), the model is specified in general Ordinary Least Squares matrix notation:
	\begin{align}
		\mathbf{Y} = \mathbf{X}\boldsymbol{\beta} + \boldsymbol{\varepsilon}.
	\end{align}
	Following raw data processing, the exact cross-product matrix $\mathbf{X}^T \mathbf{X}$ and projection vector $\mathbf{X}^T \mathbf{Y}$ are calculated:
	\begin{align}
		\mathbf{X}^T \mathbf{X} = \begin{pmatrix} 5 & 10 & 20 \\ 10 & 30 & 40 \\ 20 & 40 & 90 \end{pmatrix}, \qquad \mathbf{X}^T \mathbf{Y} = \begin{pmatrix} 50 \\ 120 \\ 230 \end{pmatrix}.
	\end{align}
	(The first column of $\mathbf{X}$ is the vector of ones $\mathbf{1}_5$ corresponding to intercept $\beta_0$).
	\begin{enumerate}
		\item Knowing that the inverse of this symmetric normal matrix is:
		\begin{align}
			(\mathbf{X}^T \mathbf{X})^{-1} = \frac{1}{50} \begin{pmatrix} 110 & -10 & -20 \\ -10 & 5 & 0 \\ -20 & 0 & 5 \end{pmatrix} = \begin{pmatrix} 2.2 & -0.2 & -0.4 \\ -0.2 & 0.1 & 0.0 \\ -0.4 & 0.0 & 0.1 \end{pmatrix},
		\end{align}
		compute the exact least squares estimator solution vector $\hat{\boldsymbol{\beta}} = (\mathbf{X}^T \mathbf{X})^{-1} \mathbf{X}^T \mathbf{Y} = (\hat{\beta}_0, \hat{\beta}_1, \hat{\beta}_2)^T$.
		\item Write the scalar predictive equation of the resulting model.
		\item Formally define the \textbf{Projection Matrix or Hat Matrix} $\mathbf{H} = \mathbf{X}(\mathbf{X}^T \mathbf{X})^{-1} \mathbf{X}^T$ and algebraically prove its two axiomatic properties: symmetry ($\mathbf{H}^T = \mathbf{H}$) and idempotency ($\mathbf{H}^2 = \mathbf{H}$). Explain how operator $\mathbf{H}$ transforms raw response vector $\mathbf{Y}$ into predicted vector $\hat{\mathbf{Y}}$.
	\end{enumerate}
\end{problema}

\begin{problema}

	For the econometric model from Problem \ref{prob:reg_mult_matricial_exacto} ($n = 5, p = 2$), suppose an extended sample of $n = 25$ firms operating under the exact same normalized covariance structure $(\mathbf{X}^T \mathbf{X})^{-1}$ yielded an Error Sum of Squares $\text{SSD} = 44.0$ and an identical estimated coefficient vector $\hat{\boldsymbol{\beta}} = (1.2, \; 2.6, \; 1.8)^T$.
	\begin{enumerate}

		\item Compute the \textbf{Estimated Parameter Variance-Covariance Matrix} $\widehat{\text{Cov}}(\hat{\boldsymbol{\beta}}) = \hat{\sigma}^2 (\mathbf{X}^T \mathbf{X})^{-1}$ and extract standard errors $SE(\hat{\beta}_0)$, $SE(\hat{\beta}_1)$, and $SE(\hat{\beta}_2)$.
		\item Construct a $95\%$ confidence interval for the true liquidity index parameter $\beta_1$ ($t_{0.025, \, 22} = 2.074$).
		\item If the Total Sum of Squares of returns in that sample was $\text{SST} = 264.0$, perform an \textbf{Overall $F$-Test for Regression Significance} ($H_0: \beta_1 = \beta_2 = 0$ vs $H_a$: at least one $\beta_j \neq 0$) at $\alpha = 0.05$ ($F_{0.05, \, 2, \, 22} = 3.443$), constructing the complete multiple regression ANOVA table.
	\end{enumerate}
\end{problema}

\subsubsection*{4. Challenging Level (Expert / Deep Deduction)}

\begin{problema}
	\label{prob:reg_mult_deduccion_matricial_mco}
	\textbf{Analytical Derivation of Vector-Matrix OLS and Population Unbiasedness in the General $p$-Regressor Model:}
	Consider the multiple linear regression model in vector space $\mathbb{R}^n$:
	\begin{align}
		\mathbf{Y} = \mathbf{X}\boldsymbol{\beta} + \boldsymbol{\varepsilon},
	\end{align}
	where $\mathbf{Y}$ is the $n \times 1$ response vector, $\mathbf{X}$ is the non-random $n \times (p+1)$ design matrix of full column rank ($\text{rank}(\mathbf{X}) = p+1 \leq n$), $\boldsymbol{\beta}$ is the $(p+1) \times 1$ population parameter vector, and $\boldsymbol{\varepsilon}$ is the error vector satisfying $E[\boldsymbol{\varepsilon}] = \mathbf{0}$ and $\text{Var}(\boldsymbol{\varepsilon}) = E[\boldsymbol{\varepsilon}\boldsymbol{\varepsilon}^T] = \sigma^2 \mathbf{I}_n$.
	\begin{enumerate}
		\item Starting from the scalar quadratic form of the error sum of squares $S(\boldsymbol{\beta}) = \boldsymbol{\varepsilon}^T \boldsymbol{\varepsilon} = (\mathbf{Y} - \mathbf{X}\boldsymbol{\beta})^T (\mathbf{Y} - \mathbf{X}\boldsymbol{\beta})$, apply vector-matrix differentiation rules over quadratic forms with respect to vector $\boldsymbol{\beta}$ to rigorously derive the normal equations $\mathbf{X}^T \mathbf{X} \hat{\boldsymbol{\beta}} = \mathbf{X}^T \mathbf{Y}$ and justify the uniqueness of the analytical solution $\hat{\boldsymbol{\beta}} = (\mathbf{X}^T \mathbf{X})^{-1} \mathbf{X}^T \mathbf{Y}$.
		\item Prove that the empirical residual vector $\hat{\boldsymbol{\varepsilon}} = \mathbf{Y} - \hat{\mathbf{Y}} = (\mathbf{I}_n - \mathbf{H})\mathbf{Y}$ is orthogonal to the entire column space generated by design matrix $\mathbf{X}$ ($\mathbf{X}^T \hat{\boldsymbol{\varepsilon}} = \mathbf{0}$), and prove that the cross-product sum between predictions and residuals vanishes identically ($\hat{\mathbf{Y}}^T \hat{\boldsymbol{\varepsilon}} = 0$).
		\item Applying expectation operator $E[\cdot]$, prove that vector $\hat{\boldsymbol{\beta}}$ is a multivariable \textbf{unbiased} estimator ($E[\hat{\boldsymbol{\beta}}] = \boldsymbol{\beta}$) and that its theoretical covariance matrix equals precisely $\text{Var}(\hat{\boldsymbol{\beta}}) = \sigma^2 (\mathbf{X}^T \mathbf{X})^{-1}$.
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:reg_mult_gauss_markov_meli}
	\textbf{Proof of the Gauss-Markov Theorem (BLUE - \emph{Best Linear Unbiased Estimator}) and $R_{\text{adj}}^2$ Penalization:}
	Within the matrix linear model framework $\mathbf{Y} = \mathbf{X}\boldsymbol{\beta} + \boldsymbol{\varepsilon}$ under verified Gauss-Markov assumptions ($E[\boldsymbol{\varepsilon}] = \mathbf{0}$, $\text{Var}(\boldsymbol{\varepsilon}) = \sigma^2 \mathbf{I}_n$):
	\begin{enumerate}
		\item Let $\tilde{\boldsymbol{\beta}} = \mathbf{C}\mathbf{Y}$ be an arbitrary competing linear estimator required to be unbiased for vector $\boldsymbol{\beta}$ ($E[\tilde{\boldsymbol{\beta}}] = \boldsymbol{\beta}$ for all $\boldsymbol{\beta}$). Express its weighting matrix as an OLS perturbation: $\mathbf{C} = (\mathbf{X}^T \mathbf{X})^{-1}\mathbf{X}^T + \mathbf{D}$. Prove step-by-step that for $\tilde{\boldsymbol{\beta}}$ to be unbiased, the perturbing matrix must strictly satisfy the orthogonality condition $\mathbf{D}\mathbf{X} = \mathbf{0}$.
		\item With unbiasedness verified, algebraically prove that the variance-covariance matrix of $\tilde{\boldsymbol{\beta}}$ decomposes additively and exactly into:
		\begin{align}
			\text{Var}(\tilde{\boldsymbol{\beta}}) = \text{Var}(\hat{\boldsymbol{\beta}}) + \sigma^2 \mathbf{D}\mathbf{D}^T.
		\end{align}
		Conclude from the positive semi-definiteness of quadratic matrix $\mathbf{D}\mathbf{D}^T \succeq \mathbf{0}$ that the OLS estimator possesses minimum variance in the Loewner matrix sense ($\text{Var}(\tilde{\boldsymbol{\beta}}) - \text{Var}(\hat{\boldsymbol{\beta}}) \succeq \mathbf{0}$), proving the Gauss-Markov Theorem.
		\item Prove that if, in a model containing $p$ true explanatory variables, one decides to incorporate an additional predictor ($X_{p+1}$) whose true population slope is zero ($\beta_{p+1} = 0$), the nominal sample coefficient $R^2$ satisfies the empirical dominance relation $R_{p+1}^2 \geq R_p^2$, whereas the Adjusted Coefficient of Determination $R_{\text{adj}, p+1}^2$ is strictly smaller than the previous $R_{\text{adj}, p}^2$ unless the Student's $t$ test statistic of that new regressor exceeds unit magnitude ($|t_{p+1}| > 1$).
	\end{enumerate}
\end{problema}


\subsection{Detailed Solutions}

\subsubsection*{1. Fundamental Level (Simple / Direct)}

\begin{solucion}
	\begin{enumerate}
		\item \textbf{Physical interpretation of partial coefficients:}
		In multiple regression, each parameter represents the rate of change of expected $Y$ with respect to one variable, \emph{holding all other explanatory variables fixed} (\emph{ceteris paribus} condition):
		\begin{itemize}
			\item $\hat{\beta}_1 = -3.0$: For automobiles possessing the \textbf{same engine displacement} ($X_2$ constant), for every additional metric ton of gross vehicle weight ($X_1$), average fuel efficiency decreases by $3.0\text{ km/l}$.
			\item $\hat{\beta}_2 = -1.5$: For automobiles of \textbf{identical vehicle weight} ($X_1$ constant), for every additional liter of engine displacement or capacity ($X_2$), average fuel efficiency decreases by $1.5\text{ km/l}$.
		\end{itemize}
		
		\item Computation of projected efficiency for an automobile with $X_1 = 1.5$ metric tons and $X_2 = 2.0$ liters:
		\begin{align}
			\hat{Y} = 25.0 - 3.0(1.5) - 1.5(2.0) = 25.0 - 4.5 - 3.0 = 17.5\text{ km/l}.
		\end{align}
		An urban fuel efficiency of $17.5$ kilometers per liter is estimated.
		
		\item With $n = 30$ total observations and an observed sample variance $S_Y^2 = 12.0$, the Total Sum of Squares of fuel efficiency is:
		\begin{align}
			\text{SST} = (n - 1) S_Y^2 = (30 - 1)(12.0) = 29 \times 12.0 = 348.0\text{ (km/l)}^2.
		\end{align}
		Using the nominal coefficient of determination $R^2 = 0.780$, we obtain the Regression Sum of Squares ($\text{SSR}$):
		\begin{align}
			\text{SSR} = R^2 \times \text{SST} = (0.780)(348.0) = 271.44\text{ (km/l)}^2.
		\end{align}
		By the algebraic principle of orthogonal error partitioning in OLS, unexplained variability is:
		\begin{align}
			\text{SSD} = \text{SST} - \text{SSR} = 348.0 - 271.44 = 76.56\text{ (km/l)}^2.
		\end{align}
	\end{enumerate}
\end{solucion}

\begin{solucion}
	\begin{enumerate}
		\item \textbf{Formal formulation of partial significance hypotheses:}
		To evaluate whether each explanatory variable contributes non-redundant predictive information to the model (given the presence of the other):
		\begin{itemize}
			\item For vehicle weight ($X_1$):
			\begin{align}
				H_{0,1}: \beta_1 &= 0 \quad (\text{Weight provides no predictive significance once displacement is controlled}). \\
				H_{a,1}: \beta_1 &\neq 0 \quad (\text{Vehicle weight significantly influences fuel efficiency}).
			\end{align}
			\item For engine displacement ($X_2$):
			\begin{align}
				H_{0,2}: \beta_2 &= 0 \quad (\text{Displacement is irrelevant for predicting consumption once weight is controlled}). \\
				H_{a,2}: \beta_2 &\neq 0 \quad (\text{Displacement has a direct, significant impact on efficiency}).
			\end{align}
		\end{itemize}
		
		\item Computation of individual Student's $t$ test statistics under null hypotheses ($H_{0,j}: \beta_j = 0$):
		Residual degrees of freedom for the model with $n=30$ and $p=2$ predictors are $\nu_E = n - p - 1 = 30 - 2 - 1 = 27$.
		\begin{align}
			t_1 &= \frac{\hat{\beta}_1}{SE(\hat{\beta}_1)} = \frac{-3.0}{0.80} = -3.750, \\
			t_2 &= \frac{\hat{\beta}_2}{SE(\hat{\beta}_2)} = \frac{-1.5}{0.60} = -2.500.
		\end{align}
		
		\item Decision rule and conclusion:
		The critical quantile in the two-tailed Student's $t$ distribution with $\nu_E = 27$ at $\alpha = 0.05$ yields an absolute cutoff value of $t_{0.025, \, 27} = 2.052$.
		\begin{itemize}
			\item For weight ($X_1$): Since $|t_1| = 3.750 > 2.052$ ($p$-value $\approx 0.0008$), \textbf{we reject $H_{0,1}$}. Net vehicle weight is a proven and robust determinant of fuel economy.
			\item For engine displacement ($X_2$): Since $|t_2| = 2.500 > 2.052$ ($p$-value $\approx 0.0188$), \textbf{we reject $H_{0,2}$}. Volumetric engine capacity contributes significantly to predicting consumption above and beyond physical vehicle weight.
		\end{itemize}
	\end{enumerate}
\end{solucion}

\begin{solucion}
	\begin{enumerate}
		\item Rigorous computation of the Adjusted Coefficient of Determination ($R_{\text{adj}}^2$):
		For $n = 40$ quarters, $p = 3$ predictors in the model, and an unadjusted coefficient $R^2 = 0.780$:
		\begin{align}
			R_{\text{adj}}^2 &= 1 - \left[ \frac{n - 1}{n - p - 1} (1 - R^2) \right] = 1 - \left[ \frac{40 - 1}{40 - 3 - 1} (1 - 0.780) \right] \\
			&= 1 - \left[ \frac{39}{36} (0.220) \right] = 1 - \left[ (1.08333)(0.220) \right] = 1 - 0.23833 \approx 0.7617 \quad (76.17\%).
		\end{align}
		
		\item \textbf{Theoretical justification and dimensional penalization:} In any Ordinary Least Squares regression fit, adding a new predictor variable (even if it consists of pure random number sequences sharing no geometric relationship with sales) invariably reduces the Residual Sum of Squares or, in the worst case, leaves it unchanged ($\text{SSD}_{p+1} \leq \text{SSD}_p$). Consequently, the \textbf{nominal $R^2$ can never decrease} upon adding variables, creating a deceptive analytical mirage that encourages overfitting complex models with noise.
		
		The \textbf{$R_{\text{adj}}^2$ mathematically corrects this optimistic bias} by replacing sums of squares with their corresponding *mean squares (sample variances)*:
		\begin{align}
			R_{\text{adj}}^2 = 1 - \frac{\text{MSE}}{\text{MST}} = 1 - \frac{\text{SSD} / (n - p - 1)}{\text{SST} / (n - 1)}.
		\end{align}
		Each time a predictor is added ($p \to p+1$), the term $(n - p - 1)$ in the error variance denominator \emph{decreases exactly by one unit}, penalizing the indicator by inflating the ratio $\text{MSE}/\text{MST}$. For net $R_{\text{adj}}^2$ to increase, the newly added variable must absorb enough true variance to overcome and compensate for the loss of that error degree of freedom.
		
		\item Numerical impact calculation when adding the spurious zip code ($X_4$) with $R_{\text{new}}^2 = 0.781$ and $p = 4$:
		\begin{align}
			R_{\text{adj, new}}^2 &= 1 - \left[ \frac{40 - 1}{40 - 4 - 1} (1 - 0.781) \right] = 1 - \left[ \frac{39}{35} (0.219) \right] \\
			&= 1 - \left[ (1.11429)(0.219) \right] = 1 - 0.24403 \approx 0.7560 \quad (75.60\%).
		\end{align}
		Although nominal $R^2$ rose superficially from $0.780$ to $0.781$, \textbf{the Adjusted Coefficient of Determination suffered a noticeable drop from $0.7617$ to $0.7560$}. The $R_{\text{adj}}^2$ diagnostic severely penalizes and rejects the use of corporate zip code when predicting sales success.
	\end{enumerate}
\end{solucion}

\subsubsection*{2. Operational Level (Intermediate / Development)}

\begin{solucion}
	\begin{enumerate}
		\item Comparative Adjusted Coefficient of Determination computation:
		\begin{itemize}
			\item For \textbf{Model 1 (Restricted, $p_1 = 2, R_1^2 = 0.750, n=40$)}:
			\begin{align}
				R_{\text{adj, 1}}^2 = 1 - \left[ \frac{39}{40 - 2 - 1} (1 - 0.750) \right] = 1 - \left[ \frac{39}{37} (0.250) \right] = 1 - 0.26351 \approx 0.7365 \quad (73.65\%).
			\end{align}
			\item For \textbf{Model 2 (Full, $p_2 = 3, R_2^2 = 0.780, n=40$)}:
			As established in Problem \ref{prob:reg_mult_r2_adj_basico}:
			\begin{align}
				R_{\text{adj, 2}}^2 = 1 - \left[ \frac{39}{36} (0.220) \right] \approx 0.7617 \quad (76.17\%).
			\end{align}
		\end{itemize}
		
		\item \textbf{Partial (Incremental) $F$-Test for Explained Sum of Squares:}
		This test evaluates with inferential rigor whether the marginal improvement in fit achieved by including sales executives ($X_3$) is significantly larger than random sampling noise. We formulate the hypotheses:
		\begin{align}
			H_0: \beta_3 &= 0 \quad (\text{The restricted two-predictor model is statistically sufficient}). \\
			H_a: \beta_3 &\neq 0 \quad (\text{The number of sales executives contributes significantly to forecasting}).
		\end{align}
		The incremental formula based on $R^2$ proportions yields:
		\begin{align}
			F_{\text{partial}} &= \frac{(R_2^2 - R_1^2) / (p_2 - p_1)}{(1 - R_2^2) / (n - p_2 - 1)} = \frac{(0.780 - 0.750) / (3 - 2)}{(1 - 0.780) / (40 - 3 - 1)} \\
			&= \frac{0.030 / 1}{0.220 / 36} = \frac{0.030}{0.006111} \approx 4.9091.
		\end{align}
		For nominal significance level $\alpha = 0.05$, with $\nu_1 = p_2 - p_1 = 1$ and $\nu_2 = n - p_2 - 1 = 36$ degrees of freedom, the critical value from the Fisher-Snedecor distribution table is:
		\begin{align}

		\begin{align}

			\begin{align}

			\item \textbf{Idempotency ($\mathbf{H}^2 = \mathbf{H}$):} Squaring the projective operator and grouping intermediate matrices:
			\begin{align}
				\mathbf{H}^2 = \mathbf{H} \cdot \mathbf{H} &= \left[ \mathbf{X}(\mathbf{X}^T \mathbf{X})^{-1} \mathbf{X}^T \right] \left[ \mathbf{X}(\mathbf{X}^T \mathbf{X})^{-1} \mathbf{X}^T \right] \\

			\end{align}

			\begin{align}
				\mathbf{H}^2 = \mathbf{X} \cdot \mathbf{I}_{p+1} \cdot (\mathbf{X}^T \mathbf{X})^{-1} \mathbf{X}^T = \mathbf{X}(\mathbf{X}^T \mathbf{X})^{-1} \mathbf{X}^T = \mathbf{H}. \quad (\text{Q.E.D.})
			\end{align}
		\end{itemize}
		Idempotency ensures physically that projecting data twice consecutively onto the regression plane leaves predictions unchanged ($\mathbf{H}\hat{\mathbf{Y}} = \hat{\mathbf{Y}}$), because fitted values already reside completely within the explained column space of the model.

\begin{solucion}
	\begin{enumerate}
		\item \textbf{Analytical derivation of matrix OLS:}
		Let the error sum of squares objective function be expressed via the Euclidean vector inner product:
		\begin{align}
			S(\boldsymbol{\beta}) &= (\mathbf{Y} - \mathbf{X}\boldsymbol{\beta})^T (\mathbf{Y} - \mathbf{X}\boldsymbol{\beta}) \\
			&= (\mathbf{Y}^T - \boldsymbol{\beta}^T \mathbf{X}^T)(\mathbf{Y} - \mathbf{X}\boldsymbol{\beta}) = \mathbf{Y}^T \mathbf{Y} - \mathbf{Y}^T \mathbf{X}\boldsymbol{\beta} - \boldsymbol{\beta}^T \mathbf{X}^T \mathbf{Y} + \boldsymbol{\beta}^T \mathbf{X}^T \mathbf{X}\boldsymbol{\beta}.
		\end{align}
		Because the intermediate term $\mathbf{Y}^T \mathbf{X}\boldsymbol{\beta}$ is a $1 \times 1$ scalar, it equals its own matrix transpose ($\mathbf{Y}^T \mathbf{X}\boldsymbol{\beta} = (\mathbf{Y}^T \mathbf{X}\boldsymbol{\beta})^T = \boldsymbol{\beta}^T \mathbf{X}^T \mathbf{Y}$). We can simplify the objective function to:
		\begin{align}
			S(\boldsymbol{\beta}) = \mathbf{Y}^T \mathbf{Y} - 2\boldsymbol{\beta}^T \mathbf{X}^T \mathbf{Y} + \boldsymbol{\beta}^T \mathbf{X}^T \mathbf{X}\boldsymbol{\beta}.
		\end{align}
		To locate the vector minimum, we take the gradient with respect to parameter vector $\boldsymbol{\beta}$, utilizing vector-matrix differentiation rules $\dfrac{\partial (\boldsymbol{\beta}^T \mathbf{a})}{\partial \boldsymbol{\beta}} = \mathbf{a}$ and $\dfrac{\partial (\boldsymbol{\beta}^T \mathbf{A} \boldsymbol{\beta})}{\partial \boldsymbol{\beta}} = 2\mathbf{A}\boldsymbol{\beta}$ for symmetric matrices:
		\begin{align}
			\frac{\partial S(\boldsymbol{\beta})}{\partial \boldsymbol{\beta}} = -2\mathbf{X}^T \mathbf{Y} + 2\mathbf{X}^T \mathbf{X}\boldsymbol{\beta} = \mathbf{0}.
		\end{align}
		Dividing both sides of the vector equation by the constant $+2$, we obtain the general system of \textbf{Gaussian Matrix Normal Equations}:
		\begin{align}
			\mathbf{X}^T \mathbf{X} \hat{\boldsymbol{\beta}} = \mathbf{X}^T \mathbf{Y}.
		\end{align}
		By hypothesis, design matrix $\mathbf{X}$ has full column rank ($\text{rank}(\mathbf{X}) = p+1$), guaranteeing by real linear algebra that the $(p+1) \times (p+1)$ square cross-product matrix $\mathbf{X}^T \mathbf{X}$ is \textbf{symmetric positive-definite and non-singular} ($\det(\mathbf{X}^T \mathbf{X}) > 0$). Therefore, a unique inverse $(\mathbf{X}^T \mathbf{X})^{-1}$ exists. Pre-multiplying the normal system by this inverse uniquely isolates the estimator vector solution:
		\begin{align}
			(\mathbf{X}^T \mathbf{X})^{-1} \mathbf{X}^T \mathbf{X} \hat{\boldsymbol{\beta}} = (\mathbf{X}^T \mathbf{X})^{-1} \mathbf{X}^T \mathbf{Y} \implies \hat{\boldsymbol{\beta}} = (\mathbf{X}^T \mathbf{X})^{-1} \mathbf{X}^T \mathbf{Y}.
		\end{align}
		
		\item \textbf{Matrix orthogonality between empirical residuals and predictors ($\mathbf{X}^T \hat{\boldsymbol{\varepsilon}} = \mathbf{0}$):}
		The OLS residual vector is $\hat{\boldsymbol{\varepsilon}} = \mathbf{Y} - \hat{\mathbf{Y}} = \mathbf{Y} - \mathbf{X}\hat{\boldsymbol{\beta}}$. Substituting $\hat{\boldsymbol{\beta}} = (\mathbf{X}^T \mathbf{X})^{-1}\mathbf{X}^T \mathbf{Y}$ and incorporating hat matrix $\mathbf{H}$:
		\begin{align}
			\hat{\boldsymbol{\varepsilon}} = \mathbf{Y} - \mathbf{H}\mathbf{Y} = (\mathbf{I}_n - \mathbf{H})\mathbf{Y}.
		\end{align}
		To prove geometric orthogonality to the column space of $\mathbf{X}$, we pre-multiply the residual vector by the transpose of design matrix $\mathbf{X}$:
		\begin{align}
			\mathbf{X}^T \hat{\boldsymbol{\varepsilon}} = \mathbf{X}^T (\mathbf{Y} - \mathbf{X}\hat{\boldsymbol{\beta}}) = \mathbf{X}^T \mathbf{Y} - \mathbf{X}^T \mathbf{X} \hat{\boldsymbol{\beta}}.
		\end{align}
		Invoking the exact identity of the Normal Equations ($\mathbf{X}^T \mathbf{X}\hat{\boldsymbol{\beta}} = \mathbf{X}^T \mathbf{Y}$), we prove conclusively:
		\begin{align}
			\mathbf{X}^T \hat{\boldsymbol{\varepsilon}} = \mathbf{X}^T \mathbf{Y} - \mathbf{X}^T \mathbf{Y} = \mathbf{0}. \quad (\text{Orthogonality to the subspace spanned by } \mathbf{X}).
		\end{align}
		Because prediction vector $\hat{\mathbf{Y}} = \mathbf{X}\hat{\boldsymbol{\beta}}$ belongs to the column space of $\mathbf{X}$, the inner product between predictions and residuals is identically zero:
		\begin{align}
			\hat{\mathbf{Y}}^T \hat{\boldsymbol{\varepsilon}} = (\mathbf{X}\hat{\boldsymbol{\beta}})^T \hat{\boldsymbol{\varepsilon}} = \hat{\boldsymbol{\beta}}^T (\mathbf{X}^T \hat{\boldsymbol{\varepsilon}}) = \hat{\boldsymbol{\beta}}^T [\mathbf{0}] = 0. \quad (\text{Complete empirical orthogonality}).
		\end{align}
		
		\item \textbf{Proof of unbiasedness and derivation of the matrix variance tensor:}
		We express $\hat{\boldsymbol{\beta}}$ by substituting true population response $\mathbf{Y} = \mathbf{X}\boldsymbol{\beta} + \boldsymbol{\varepsilon}$:
		\begin{align}
			\hat{\boldsymbol{\beta}} &= (\mathbf{X}^T \mathbf{X})^{-1} \mathbf{X}^T (\mathbf{X}\boldsymbol{\beta} + \boldsymbol{\varepsilon}) \\
			&= (\mathbf{X}^T \mathbf{X})^{-1} \mathbf{X}^T \mathbf{X} \boldsymbol{\beta} + (\mathbf{X}^T \mathbf{X})^{-1} \mathbf{X}^T \boldsymbol{\varepsilon} = \boldsymbol{\beta} + (\mathbf{X}^T \mathbf{X})^{-1} \mathbf{X}^T \boldsymbol{\varepsilon}.
		\end{align}
		Taking the unconditional matrix mathematical expectation under the non-randomness of $\mathbf{X}$:
		\begin{align}
			E[\hat{\boldsymbol{\beta}}] = E\left[ \boldsymbol{\beta} + (\mathbf{X}^T \mathbf{X})^{-1} \mathbf{X}^T \boldsymbol{\varepsilon} \right] = \boldsymbol{\beta} + (\mathbf{X}^T \mathbf{X})^{-1} \mathbf{X}^T E[\boldsymbol{\varepsilon}].
		\end{align}
		By the central Gauss-Markov probabilistic assumption ($E[\boldsymbol{\varepsilon}] = \mathbf{0}$):
		\begin{align}
			E[\hat{\boldsymbol{\beta}}] = \boldsymbol{\beta} + \mathbf{0} = \boldsymbol{\beta}. \quad (\text{Unbiased multivariable estimator}).
		\end{align}
		To derive the exact covariance matrix $\text{Var}(\hat{\boldsymbol{\beta}}) = E[(\hat{\boldsymbol{\beta}} - \boldsymbol{\beta})(\hat{\boldsymbol{\beta}} - \boldsymbol{\beta})^T]$, we note that deviation from the mean is $\hat{\boldsymbol{\beta}} - \boldsymbol{\beta} = (\mathbf{X}^T \mathbf{X})^{-1} \mathbf{X}^T \boldsymbol{\varepsilon}$:
		\begin{align}
			\text{Var}(\hat{\boldsymbol{\beta}}) &= E\left[ \left( (\mathbf{X}^T \mathbf{X})^{-1} \mathbf{X}^T \boldsymbol{\varepsilon} \right) \left( (\mathbf{X}^T \mathbf{X})^{-1} \mathbf{X}^T \boldsymbol{\varepsilon} \right)^T \right] \\
			&= E\left[ (\mathbf{X}^T \mathbf{X})^{-1} \mathbf{X}^T \boldsymbol{\varepsilon} \boldsymbol{\varepsilon}^T \mathbf{X} (\mathbf{X}^T \mathbf{X})^{-1} \right].
		\end{align}
		Extracting deterministic matrices outside the expectation operator and recognizing that under uncorrelated homoscedasticity $E[\boldsymbol{\varepsilon}\boldsymbol{\varepsilon}^T] = \sigma^2 \mathbf{I}_n$:
		\begin{align}
			\text{Var}(\hat{\boldsymbol{\beta}}) &= (\mathbf{X}^T \mathbf{X})^{-1} \mathbf{X}^T E[\boldsymbol{\varepsilon}\boldsymbol{\varepsilon}^T] \mathbf{X} (\mathbf{X}^T \mathbf{X})^{-1} \\
			&= (\mathbf{X}^T \mathbf{X})^{-1} \mathbf{X}^T (\sigma^2 \mathbf{I}_n) \mathbf{X} (\mathbf{X}^T \mathbf{X})^{-1} = \sigma^2 (\mathbf{X}^T \mathbf{X})^{-1} \left[ \mathbf{X}^T \mathbf{X} \right] (\mathbf{X}^T \mathbf{X})^{-1}.
		\end{align}
		The product in the middle of the chain cancels the penultimate term, collapsing the tensor cleanly and decisively to:
		\begin{align}
			\text{Var}(\hat{\boldsymbol{\beta}}) = \sigma^2 (\mathbf{X}^T \mathbf{X})^{-1}. \quad (\text{Q.E.D.})
		\end{align}
	\end{enumerate}
\end{solucion}

\begin{solucion}
	\begin{enumerate}
		\item \textbf{Proof of the unbiasedness condition on perturbing matrix $\mathbf{D}$:}
		Let alternative linear estimator be $\tilde{\boldsymbol{\beta}} = \mathbf{C}\mathbf{Y}$ where $\mathbf{C} = (\mathbf{X}^T \mathbf{X})^{-1}\mathbf{X}^T + \mathbf{D}$.
		Substituting population model $\mathbf{Y} = \mathbf{X}\boldsymbol{\beta} + \boldsymbol{\varepsilon}$:
		\begin{align}
			\tilde{\boldsymbol{\beta}} &= \left[ (\mathbf{X}^T \mathbf{X})^{-1}\mathbf{X}^T + \mathbf{D} \right] (\mathbf{X}\boldsymbol{\beta} + \boldsymbol{\varepsilon}) \\
			&= (\mathbf{X}^T \mathbf{X})^{-1}\mathbf{X}^T \mathbf{X}\boldsymbol{\beta} + (\mathbf{X}^T \mathbf{X})^{-1}\mathbf{X}^T \boldsymbol{\varepsilon} + \mathbf{D}\mathbf{X}\boldsymbol{\beta} + \mathbf{D}\boldsymbol{\varepsilon} \\
			&= \boldsymbol{\beta} + \mathbf{D}\mathbf{X}\boldsymbol{\beta} + \left[ (\mathbf{X}^T \mathbf{X})^{-1}\mathbf{X}^T + \mathbf{D} \right] \boldsymbol{\varepsilon}.
		\end{align}
		Taking expectation and using $E[\boldsymbol{\varepsilon}] = \mathbf{0}$:
		\begin{align}
			E[\tilde{\boldsymbol{\beta}}] = \boldsymbol{\beta} + \mathbf{D}\mathbf{X}\boldsymbol{\beta} + \mathbf{0} = \left( \mathbf{I}_{p+1} + \mathbf{D}\mathbf{X} \right) \boldsymbol{\beta}.
		\end{align}
		For the estimator to be universally and strictly unbiased ($E[\tilde{\boldsymbol{\beta}}] = \boldsymbol{\beta}$ for any conceivable true parameter vector $\boldsymbol{\beta} \in \mathbb{R}^{p+1}$), the term multiplying $\boldsymbol{\beta}$ must equal the identity matrix exactly, imposing the mandatory design orthogonality condition:
		\begin{align}
			\mathbf{I}_{p+1} + \mathbf{D}\mathbf{X} = \mathbf{I}_{p+1} \iff \mathbf{D}\mathbf{X} = \mathbf{0}.
		\end{align}
		
		\item \textbf{Algebraic proof of optimality and minimum variance (Gauss-Markov Theorem / BLUE):}
		With condition $\mathbf{D}\mathbf{X} = \mathbf{0}$ guaranteed, the deviation of $\tilde{\boldsymbol{\beta}}$ from the population mean reduces to:
		\begin{align}
			\tilde{\boldsymbol{\beta}} - \boldsymbol{\beta} = \left[ (\mathbf{X}^T \mathbf{X})^{-1}\mathbf{X}^T + \mathbf{D} \right] \boldsymbol{\varepsilon} = \mathbf{C}\boldsymbol{\varepsilon}.
		\end{align}
		The theoretical variance matrix of this general estimator is:
		\begin{align}
			\text{Var}(\tilde{\boldsymbol{\beta}}) &= E\left[ (\tilde{\boldsymbol{\beta}} - \boldsymbol{\beta})(\tilde{\boldsymbol{\beta}} - \boldsymbol{\beta})^T \right] = E\left[ \mathbf{C}\boldsymbol{\varepsilon} \boldsymbol{\varepsilon}^T \mathbf{C}^T \right] = \mathbf{C} E[\boldsymbol{\varepsilon}\boldsymbol{\varepsilon}^T] \mathbf{C}^T = \sigma^2 \mathbf{C}\mathbf{C}^T.
		\end{align}
		Substituting $\mathbf{C} = (\mathbf{X}^T \mathbf{X})^{-1}\mathbf{X}^T + \mathbf{D}$ into the quadratic product:
		\begin{align}
			\text{Var}(\tilde{\boldsymbol{\beta}}) &= \sigma^2 \left[ (\mathbf{X}^T \mathbf{X})^{-1}\mathbf{X}^T + \mathbf{D} \right] \left[ \mathbf{X}(\mathbf{X}^T \mathbf{X})^{-1} + \mathbf{D}^T \right] \\
			&= \sigma^2 \left[ (\mathbf{X}^T \mathbf{X})^{-1}\mathbf{X}^T \mathbf{X}(\mathbf{X}^T \mathbf{X})^{-1} + (\mathbf{X}^T \mathbf{X})^{-1}\mathbf{X}^T \mathbf{D}^T + \mathbf{D}\mathbf{X}(\mathbf{X}^T \mathbf{X})^{-1} + \mathbf{D}\mathbf{D}^T \right].
		\end{align}
		Let us examine the cross-products: the third term contains product $\mathbf{D}\mathbf{X}$, which by part 1 we proved is the zero matrix ($\mathbf{D}\mathbf{X} = \mathbf{0}$). Its transpose in the second term vanishes identically ($\mathbf{X}^T \mathbf{D}^T = (\mathbf{D}\mathbf{X})^T = \mathbf{0}^T = \mathbf{0}$).
		With cross-interferences canceling completely, the variance matrix of the competing estimator separates into pure additive form:
		\begin{align}
			\text{Var}(\tilde{\boldsymbol{\beta}}) = \sigma^2 (\mathbf{X}^T \mathbf{X})^{-1} + \sigma^2 \mathbf{D}\mathbf{D}^T = \text{Var}(\hat{\boldsymbol{\beta}}) + \sigma^2 \mathbf{D}\mathbf{D}^T.
		\end{align}
		For any real matrix $\mathbf{D}$, outer product $\mathbf{D}\mathbf{D}^T$ is always and under all circumstances a \textbf{positive semi-definite matrix ($\mathbf{D}\mathbf{D}^T \succeq \mathbf{0}$)}—meaning that for any non-zero vector $\mathbf{v} \in \mathbb{R}^{p+1}$, we have $\mathbf{v}^T (\mathbf{D}\mathbf{D}^T)\mathbf{v} = \|\mathbf{D}^T \mathbf{v}\|^2 \geq 0$.
		In conclusion, the variance difference between any competing linear unbiased estimator and the OLS estimator is always positive semi-definite:
		\begin{align}
			\text{Var}(\tilde{\boldsymbol{\beta}}) - \text{Var}(\hat{\boldsymbol{\beta}}) = \sigma^2 \mathbf{D}\mathbf{D}^T \succeq \mathbf{0} \implies \text{Var}(\tilde{\beta}_j) \geq \text{Var}(\hat{\beta}_j) \quad \forall j.
		\end{align}
		This proves with elegance and absolute mathematical rigor the \textbf{Gauss-Markov Theorem}: the Ordinary Least Squares estimator ($\hat{\boldsymbol{\beta}}$) is the Best Linear Unbiased Estimator (\textbf{BLUE}).
		
		\item \textbf{Proof of analytical $R_{\text{adj}}^2$ penalization when adding superfluous predictors:}
		Let $R_p^2$ be the nominal coefficient with $p$ regressors and $R_{p+1}^2$ the coefficient with $p+1$ regressors upon incorporating variable $X_{p+1}$.
		By sum of squares increment theory ($\text{SSR}_{p+1} = \text{SSR}_p + \text{SSR}(X_{p+1} | X_1,\dots,X_p)$), since every orthogonal projection sum of squares is non-negative ($\text{SSR}_{\text{extra}} \geq 0$), nominal explained variance obeys a non-decreasing monotonicity law:
		\begin{align}
			R_{p+1}^2 \geq R_p^2.
		\end{align}
		Now let us relate adjusted coefficient $R_{\text{adj}, p}^2 = 1 - \dfrac{\text{MSE}_p}{\text{MST}}$ when adding a regressor: for the new model to outperform the previous one ($R_{\text{adj}, p+1}^2 > R_{\text{adj}, p}^2$), it is necessary and sufficient that the new Mean Squared Error be strictly smaller than the prior one ($\text{MSE}_{p+1} < \text{MSE}_p$).
		The exact algebraic relationship linking the two sample residual variances after introducing variable $X_{p+1}$ with partial Student's $t$ statistic ($t_{p+1}$) follows from ANOVA partitioning:
		\begin{align}
			\text{SSD}_p = \text{SSD}_{p+1} + \frac{\text{SSD}_{p+1}}{n - p - 2} t_{p+1}^2 = \text{SSD}_{p+1} \left( 1 + \frac{t_{p+1}^2}{n - p - 2} \right).
		\end{align}
		Dividing both sides by their respective residual degrees of freedom ($\nu_p = n - p - 1$ and $\nu_{p+1} = n - p - 2$):
		\begin{align}
			(n - p - 1) \text{MSE}_p = (n - p - 2) \text{MSE}_{p+1} \left( 1 + \frac{t_{p+1}^2}{n - p - 2} \right).
		\end{align}
		Solving for the ratio of residual variances:
		\begin{align}
			\frac{\text{MSE}_p}{\text{MSE}_{p+1}} = \frac{n - p - 2 + t_{p+1}^2}{n - p - 1}.
		\end{align}
		For the new model to improve error variance ($\text{MSE}_{p+1} < \text{MSE}_p \iff \dfrac{\text{MSE}_p}{\text{MSE}_{p+1}} > 1$), we require that the numerator strictly exceed the denominator:
		\begin{align}
			n - p - 2 + t_{p+1}^2 > n - p - 1 \implies t_{p+1}^2 > 1 \iff |t_{p+1}| > 1.
		\end{align}
		\textbf{Fundamental Analytical Conclusion:} If we incorporate into a model a redundant or spurious predictor variable ($X_{p+1}$) whose empirical partial test statistic is smaller than unity in absolute value ($|t_{p+1}| < 1$, corresponding in large samples to a $p$-value greater than approximately $0.32$), the \textbf{Adjusted Coefficient of Determination ($R_{\text{adj}}^2$) will decrease inexorably} ($R_{\text{adj}, p+1}^2 < R_{\text{adj}, p}^2$). This built-in mathematical penalty acts as an objective, automated parsimony filter protecting regression specifications against overfitting and artificial variable bloat.
	\end{enumerate}
\end{solucion}
