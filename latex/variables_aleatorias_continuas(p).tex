% Recordar
\begin{problema}
	\label{prob:14b3125}
	Considere la función $f(x) = c \cdot e^{-x}$ para $x \ge 0$ y $f(x) = 0$ para $x < 0$, donde $c$ es una constante positiva.
	\begin{enumerate}
		\item Determine el valor de $c$ tal que $f(x)$ sea una función de densidad de probabilidad (PDF) válida.
		\item Calcule la probabilidad $P(1 \le X \le 3)$ integrando la PDF en el intervalo.
		\item Calcule la función de distribución acumulada (CDF) $F(x)$ para todo $x \in \mathbb{R}$.
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:287c45c}
	Sea $X$ una variable aleatoria continua con PDF $f(x) = 2x$ para $0 \le x \le 1$ y $f(x) = 0$ en otro caso.
	\begin{enumerate}
		\item Calcule la CDF $F(x) = P(X \le x)$ para todo $x \in \mathbb{R}$.
		\item Verifique las propiedades fundamentales: $F(-\infty) = 0$, $F(+\infty) = 1$, y monotonía no decreciente.
		\item Calcule $P(0.3 \le X \le 0.7)$ usando la CDF.
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:61a9dd8}
	El tiempo de espera en una fila de un banco (en minutos) sigue una distribución exponencial con media $\mu = 5$ minutos, es decir, $X \sim \text{Exp}(\lambda = 0.2)$.
	\begin{enumerate}
		\item Calcule $P(X \le 3)$ y $P(X \ge 10)$.
		\item Encuentre el cuantil $q_{0.90}$ (el tiempo bajo el cual espera el 90\% de los clientes).
		\item El banco quiere garantizar que el 95\% de los clientes espere menos de $t$ minutos. Determine $t$.
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:19f62fd}
	Para una variable aleatoria continua $X$ con PDF $f(x)$ y una función $g: \mathbb{R} \to \mathbb{R}$, demuestre el \emph{Teorema LOTUS Continuo} (Law of the Unconscious Statistician):
	$$\E[g(X)] = \int_{-\infty}^{\infty} g(x) \cdot f(x)\,dx$$
	\begin{enumerate}
		\item Demuestre el caso particular $g(X) = X^{2}$.
		\item Derive la fórmula de la varianza $\Var(X) = \E[X^{2}] - (\E[X])^{2}$ usando LOTUS con $g(x) = (x - \mu)^{2}$.
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:8ec7e10}
	La prueba de Kolmogorov-Smirnov (KS) compara la CDF empírica $F_{n}$ de una muestra con una CDF teórica $F_{0}$ mediante el estadístico:
	$$D_{n} = \sup_{x \in \mathbb{R}} |F_{n}(x) - F_{0}(x)|$$
	Para una muestra de $n = 50$ observaciones de $\text{Uniform}(0, 1)$ con $D_{50} = 0.18$, evalúe si la hipótesis nula (los datos provienen de $\text{Uniform}(0,1)$) se rechaza a nivel $\alpha = 0.05$ (valor crítico $K_{0.95} \approx 1.358$, donde $\sqrt{n} D_n \xrightarrow{d} K$ bajo $H_0$), y justifique su decisión.
\end{problema}

\begin{problema}
	\label{prob:0d490fc}
	Diseñe una función de densidad de probabilidad original definida por tramos (distinta de las exponenciales, uniformes o normales ya vistas en esta sección): el tiempo de vida útil $X$ (en años, con mínimo 1 año) de un componente electrónico de bajo costo tiene densidad de cola pesada $f(x) = k/x^{2}$ para $x \ge 1$, y $f(x) = 0$ en otro caso. Encuentre la constante de normalización $k$ y calcule $P(1 \le X \le 2)$.
\end{problema}

\subsection*{Soluciones detalladas}

% Recordar
\begin{solproblema}[prob:14b3125]
	\begin{enumerate}
		\item La condición de normalización requiere:
		\begin{align*}
			\int_{-\infty}^{\infty} f(x)\,dx = \int_{0}^{\infty} c \cdot e^{-x}\,dx = c \cdot [-e^{-x}]_{0}^{\infty} = c = 1.
		\end{align*}
		Por lo tanto, $c = 1$.
		\item $P(1 \le X \le 3) = \int_{1}^{3} e^{-x}\,dx = [-e^{-x}]_{1}^{3} = e^{-1} - e^{-3} \approx 0.3679 - 0.0498 = 0.3181$.
		\item $F(x) = \int_{-\infty}^{x} f(t)\,dt = \begin{cases} 0 & \text{si } x < 0 \\ 1 - e^{-x} & \text{si } x \ge 0 \end{cases}$, la CDF de la distribución exponencial con $\lambda=1$.
	\end{enumerate}
\end{solproblema}

% Comprender
\begin{solproblema}[prob:287c45c]
	\begin{enumerate}
		\item $F(x) = \int_{-\infty}^{x} f(t)\,dt = \begin{cases} 0 & \text{si } x < 0 \\ x^{2} & \text{si } 0 \le x \le 1 \\ 1 & \text{si } x > 1 \end{cases}$.
		\item Verificación: $F(-\infty)=0$, $F(+\infty)=1$, y $F$ es no decreciente en $[0,1]$ (crece de $0$ a $1$).
		\item $P(0.3 \le X \le 0.7) = F(0.7)-F(0.3) = 0.7^2-0.3^2 = 0.49-0.09=0.40$.
	\end{enumerate}
\end{solproblema}

% Aplicar
\begin{solproblema}[prob:61a9dd8]
	La CDF es $F(x) = 1 - e^{-0.2x}$ para $x \ge 0$.
	\begin{enumerate}
		\item $P(X \le 3) = 1-e^{-0.6}\approx1-0.5488=0.4512$. $P(X\ge10)=e^{-2}\approx0.1353$.
		\item $q_{0.90}$ satisface $1-e^{-0.2q_{0.90}}=0.9$, es decir $q_{0.90}=-5\ln(0.1)=5\ln10\approx11.51$ min.
		\item Para $P(X\le t)=0.95$: $t=-5\ln(0.05)\approx14.98$ min.
	\end{enumerate}
\end{solproblema}

% Analizar
\begin{solproblema}[prob:19f62fd]
	\begin{enumerate}
		\item Para $g(X)=X^2$, la definición operacional de LOTUS da directamente $\E[X^2]=\int_{-\infty}^{\infty}x^2f(x)\,dx$, que es simplemente la esperanza con $g(x)=x^2$.
		\item Para la varianza con $g(X)=(X-\mu)^2$ donde $\mu=\E[X]$:
		\begin{align*}
			\Var(X) &= \E[(X-\mu)^2] = \int_{-\infty}^{\infty}(x-\mu)^2f(x)\,dx = \int_{-\infty}^{\infty}(x^2-2\mu x+\mu^2)f(x)\,dx \\
			&= \E[X^2]-2\mu\E[X]+\mu^2 = \E[X^2]-2\mu^2+\mu^2 = \E[X^2]-(\E[X])^2,
		\end{align*}
		usando linealidad de la integral y la constancia de $\mu$ bajo la integral.
	\end{enumerate}
\end{solproblema}

% Evaluar
\begin{solproblema}[prob:8ec7e10]
	Bajo $H_0$, por el Teorema de Glivenko-Cantelli, la CDF empírica converge uniformemente a la CDF teórica, y la estandarización $\sqrt{n}D_n$ converge en distribución a la distribución de Kolmogorov $K$. Para $n=50$ y $D_{50}=0.18$:
	\begin{align*}
		\sqrt{50}\cdot0.18 \approx 1.273 < 1.358 = K_{0.95}.
	\end{align*}
	Como el estadístico observado ($1.273$) no supera el valor crítico ($1.358$), \textbf{no se rechaza} $H_0$ a nivel $\alpha=0.05$: no hay evidencia estadística suficiente para rechazar que los datos provienen de $\text{Uniform}(0,1)$. Es importante notar que "no rechazar" no equivale a "confirmar" la hipótesis nula: la prueba KS simplemente no detecta una discrepancia suficientemente grande entre la CDF empírica y la teórica con esta muestra de tamaño 50.
\end{solproblema}

\begin{solproblema}[prob:0d490fc]
	La condición de normalización exige:
	\begin{align*}
		\int_{1}^{\infty} \frac{k}{x^2}\,dx = k\left[-\frac{1}{x}\right]_{1}^{\infty} = k(0-(-1)) = k = 1.
	\end{align*}
	Por lo tanto $k=1$, y $f(x) = 1/x^2$ para $x\ge1$ (una distribución de Pareto continua con parámetro de forma $1$). La probabilidad solicitada es:
	\begin{align*}
		P(1 \le X \le 2) = \int_{1}^{2} \frac{1}{x^2}\,dx = \left[-\frac{1}{x}\right]_{1}^{2} = -\frac{1}{2}+1 = \frac{1}{2}.
	\end{align*}
\end{solproblema}
