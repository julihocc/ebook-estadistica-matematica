\section*{Problems}

\subsection{Problem Statements}

\subsubsection*{1. Fundamental Level (Simple / Direct)}

\begin{problema}
	\label{prob:en:1.1.1}
	Classify each of the following variables according to its scale of measurement (\emph{Nominal, Ordinal, Interval, or Ratio}):
	\begin{enumerate}
		\item The postal code (ZIP code) of a customer's residence on an e-commerce platform.
		\item A user's satisfaction rating regarding a mobile application (Very Dissatisfied, Dissatisfied, Neutral, Satisfied, Very Satisfied).
		\item The temperature in degrees Celsius (°C) recorded by server racks in a data center.
		\item The monthly salary in Mexican pesos of a data science engineer.
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:en:1.1.2}
	In a marketing study, researchers wish to investigate the average annual expenditure on video streaming subscriptions per household in Mexico. To achieve this, a digital survey is randomly sent to $2,500$ registered subscribers of a fiber-optic internet provider in Mexico City.
	\begin{enumerate}
		\item Identify the \emph{Target Population} and the \emph{Sample} of this study.
		\item Does the study incur any type of coverage bias? Briefly explain why or why not.
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:en:1.1.3}
	Determine whether the numerical metric described in the following statements represents a \textbf{Parameter} ($\mu, \sigma^2, p$) or a \textbf{Statistic} ($\bar{X}, S^2, \hat{p}$):
	\begin{enumerate}
		\item After conducting a complete census across all 32 federal entities, INEGI determined that the average household size in the country is $3.6$ people.
		\item A quality control engineer selects a random batch of 80 microprocessors and calculates a sample clock frequency variance of $0.015 \text{ GHz}^2$.
		\item In a testing subset of 5,000 banking transactions, the detected credit card fraud rate was $0.84\%$.
	\end{enumerate}
\end{problema}

\subsubsection*{2. Operational Level (Intermediate / Analytical Development)}

\begin{problema}
	\label{prob:en:1.1.4}
	A database administrator knows from historical audits that the true mean query execution time on a primary server is $\mu = 45 \text{ ms}$. During a high-concurrency peak hour, the administrator takes a random sample of $n = 60$ queries and obtains a sample mean of $\bar{X} = 51.2 \text{ ms}$.
	\begin{enumerate}
		\item Calculate the \emph{sampling error} (or simple estimation error) for this particular sample.
		\item If the administrator takes a second independent random sample of size $n = 60$ during the same peak hour, will they necessarily obtain the same sample mean $\bar{X} = 51.2 \text{ ms}$? Explain the concept of sampling variability.
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:en:1.1.5}
	A technology consulting corporation employs 4,000 personnel divided across four divisions: Engineering ($2,000$), Infrastructure ($1,000$), Sales ($600$), and Human Resources ($400$). A random sample of $n = 200$ employees must be selected to assess organizational workplace climate.
	\begin{enumerate}
		\item If simple random sampling is applied across the entire payroll without distinction, what is the potential statistical risk regarding the Human Resources division?
		\item Design a \emph{proportional stratified random sampling} scheme and determine the exact number of employees to be interviewed within each of the four divisions to guarantee equitable representation.
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:en:1.1.6}
	To evaluate the success of a newly deployed online banking portal, a UX department sends a mass email broadcast to 100,000 clients inviting them to voluntarily complete an online usability questionnaire. A total of 1,200 clients respond, and $88\%$ of them state that the new portal is confusing and difficult to navigate.
	\begin{enumerate}
		\item What methodological sampling bias predominates in this data collection process, and why does it severely distort the true population proportion ($p$) of confused clients?
		\item Propose a redesign of the sampling protocol to obtain an unbiased and reliable estimate of web usability without increasing data collection expenditures.
	\end{enumerate}
\end{problema}

\subsubsection*{3. Analytical Level (Advanced / Theoretical Connection)}

\begin{problema}
	\label{prob:en:1.1.7}
	In 1936, the magazine \emph{Literary Digest} conducted a massive sampling campaign by mailing 10 million straw ballots to predict the U.S. presidential election between Franklin D. Roosevelt and Alf Landon, using telephone directories and automobile registration lists as their sampling frame. Despite receiving a colossal sample of $n \approx 2.4 \text{ million}$ responses, they predicted a resounding landslide victory for Landon, whereas Roosevelt won by a historic margin. Simultaneously, George Gallup accurately predicted Roosevelt's victory by surveying a much smaller sample of only $n = 50,000$ individuals.
	
	Demonstrate conceptually and algebraically how, if the sampling selection mechanism exhibits a systematic bias $B = \mathbb{E}[\bar{X}] - \mu \neq 0$, increasing the sample size indefinitely ($n \to \infty$) does not reduce the total estimation error, but instead causes the estimate to converge with certainty toward an incorrect value.
\end{problema}

\begin{problema}
	\label{prob:en:1.1.8}
	In inferential statistics, the standard error of the sample mean when estimating a true population mean $\mu$ is given by the formula:
	\begin{align}
		\sigma_{\bar{X}} = \dfrac{\sigma}{\sqrt{n}},
	\end{align}
	where $\sigma$ is the population standard deviation and $n$ is the sample size.
	\begin{enumerate}
		\item If an analytics team wishes to reduce the sample standard error by exactly half, by what factor must they multiply their current sample size $n$? Demonstrate algebraically.
		\item Discuss the implications of the law of diminishing returns on Big Data collection budgets based on this mathematical result.
	\end{enumerate}
\end{problema}

\subsubsection*{4. Challenging Level (Expert / Exploration)}

\begin{problema}
	\label{prob:en:1.1.9}
	In a financial Machine Learning project, a data engineering team extracts automated server logs to model the duration of user sessions on a high-frequency trading portal. However, the architecture automatically interrupts and discards the log entries (\emph{timeout logs}) of any session that exceeds 15 minutes of inactivity.
	\begin{enumerate}
		\item Demonstrate how this truncation or survival bias (\emph{survival bias}) distorts the observed empirical distribution and shifts the sample mean $\bar{X}$ relative to the true population mean session duration $\mu$.
		\item If the dataset additionally contains extreme outliers caused by automated telemetry bots that generate session bursts of $0.001 \text{ seconds}$, why does traditional descriptive statistics based solely on the arithmetic mean become highly fragile? Justify which robust statistical measures should be preferred prior to predictive modeling.
	\end{enumerate}
\end{problema}

\begin{problema}
	\label{prob:en:1.1.10}
	Suppose a non-proportional stratified random sample of total size $n$ has been drawn from $K$ population strata. Let $N_k$ be the true size of stratum $k$ in the full census ($N = \sum_{k=1}^K N_k$), and let $n_k$ be the sample size taken from that specific stratum ($n = \sum_{k=1}^K n_k$). If $\bar{X}_k$ denotes the sample mean within stratum $k$, demonstrate algebraically that the post-stratified unbiased estimator of the total population mean $\mu$ is the weighted average:
	\begin{align}
		\bar{X}_{est} = \sum_{k=1}^K w_k \bar{X}_k, \quad \text{where } w_k = \dfrac{N_k}{N}.
	\end{align}
	Verify what occurs to the weights $w_k$ when the original sampling design is strictly proportional to stratum size ($n_k = n \cdot \frac{N_k}{N}$).
\end{problema}

\subsection{Detailed Solutions and Hints}

\subsubsection*{1. Fundamental Level (Simple / Direct)}

\begin{solproblem}
	The classification by scale of measurement is justified as follows:
	\begin{enumerate}
		\item \textbf{Postal code $\to$ Nominal:} The numbers act purely as geographical labels or identification tags. There is no natural ranking, magnitude, or mathematical validity in adding or subtracting ZIP codes.
		\item \textbf{User satisfaction rating $\to$ Ordinal:} An intrinsic hierarchy exists (Very Dissatisfied $<$ Dissatisfied $<$ Neutral $<$ Satisfied $<$ Very Satisfied), but the conceptual distance between adjacent categories is not necessarily uniform or numerically quantifiable.
		\item \textbf{Temperature in °C $\to$ Interval:} The numerical difference in degrees carries a constant physical meaning ($30^\circ\text{C} - 20^\circ\text{C} = 20^\circ\text{C} - 10^\circ\text{C}$), but the zero point is arbitrary ($0^\circ\text{C}$ represents the freezing point of water, not the absolute absence of thermal energy). Therefore, we cannot validly state that $40^\circ\text{C}$ is "twice as hot" as $20^\circ\text{C}$.
		\item \textbf{Monthly salary $\to$ Ratio:} Possesses uniform intervals and a true absolute zero point ($0$ pesos implies complete absence of salary income). Multiplicative ratios are meaningful: a salary of $\$60,000$ is exactly twice that of $\$30,000$.
	\end{enumerate}
\end{solproblem}

\begin{solproblem}
	From the conceptual analysis of the sampling design:
	\begin{enumerate}
		\item \textbf{Target Population ($N$):} All households across Mexico that consume or are potential consumers of video streaming subscription services. \textbf{Sample ($n$):} The $2,500$ registered subscribers of the fiber-optic internet provider in Mexico City who received (or responded to) the survey.
		\item \textbf{Coverage bias:} Yes, the study incurs severe coverage and selection bias. By surveying only high-speed residential internet (fiber-optic) customers in the nation's capital, the methodology systematically excludes rural areas, mid-sized cities, households without fiber connections, and consumers who stream content exclusively over mobile data networks. Consequently, the estimated average expenditure will be significantly higher than the true national average.
	\end{enumerate}
\end{solproblem}

\begin{solproblem}
	The distinction between parameter and statistic depends on whether the metric reflects the entire target universe or only a sampled portion:
	\begin{enumerate}
		\item \textbf{Average of $3.6$ people per household $\to$ Parameter ($\mu$):} Derived from an exhaustive national census encompassing all 32 federal entities.
		\item \textbf{Variance of $0.015 \text{ GHz}^2$ $\to$ Statistic ($S^2$):} Calculated specifically and exclusively from the random sample of 80 inspected microprocessors.
		\item \textbf{Fraud rate of $0.84\%$ $\to$ Statistic ($\hat{p}$):} An estimated proportion computed over the testing subset of 5,000 sampled transactions.
	\end{enumerate}
\end{solproblem}

\subsubsection*{2. Operational Level (Intermediate / Analytical Development)}

\begin{solproblem}
	For sampling calculation and variability:
	\begin{enumerate}
		\item The simple estimation error (sampling discrepancy) is formally calculated as:
		\begin{align}
			e = \bar{X} - \mu = 51.2 \text{ ms} - 45.0 \text{ ms} = +6.2 \text{ ms}.
		\end{align}
		This particular sample overestimates the true population mean execution time by $6.2 \text{ ms}$.
		\item No, a second independent random sample of size $n = 60$ will almost certainly yield a different value of $\bar{X}$ (e.g., $43.8 \text{ ms}$ or $47.5 \text{ ms}$). This behavior embodies \textbf{sampling variability}: because the randomly chosen elements differ from one sample to the next, the sample mean $\bar{X}$ is fundamentally a random variable whose sampling distribution fluctuates around the population parameter $\mu$.
	\end{enumerate}
\end{solproblem}

\begin{solproblem}
	From the stratified sampling design:
	\begin{enumerate}
		\item If simple random sampling ($n = 200$ from $N = 4,000$) is used across the entire payroll, because Human Resources is the smallest division ($400/4000 = 10\%$ of the workforce), there is a statistical risk that random sampling fluctuations may result in very few or even zero HR personnel being selected, severely compromising their representation in the organizational climate assessment.
		\item Under proportional stratified random sampling, the sample size drawn from each stratum $k$ is directly proportional to its relative weight in the total workforce: $n_k = n \cdot \left(\frac{N_k}{N}\right) = 200 \cdot \left(\frac{N_k}{4,000}\right)$. Substituting:
		\begin{itemize}
			\item \textbf{Engineering:} $n_1 = 200 \cdot \left(\frac{2,000}{4,000}\right) = 200(0.50) = 100 \text{ employees.}$
			\item \textbf{Infrastructure:} $n_2 = 200 \cdot \left(\frac{1,000}{4,000}\right) = 200(0.25) = 50 \text{ employees.}$
			\item \textbf{Sales:} $n_3 = 200 \cdot \left(\frac{600}{4,000}\right) = 200(0.15) = 30 \text{ employees.}$
			\item \textbf{Human Resources:} $n_4 = 200 \cdot \left(\frac{400}{4,000}\right) = 200(0.10) = 20 \text{ employees.}$
		\end{itemize}
		The sum confirms $100 + 50 + 30 + 20 = 200$ total interviews across all strata with exact structural parity guaranteed.
	\end{enumerate}
\end{solproblem}

\begin{solproblem}
	From the analysis of the web usability broadcast:
	\begin{enumerate}
		\item The predominant bias is \textbf{voluntary response bias (self-selection bias)}. Users who experienced severe technical frustration with the new portal possess a significantly stronger psychological motivation to open the survey and express dissatisfaction compared to satisfied or indifferent users who completed their banking transactions without issues. Therefore, $88\%$ represents a massive overestimation of the true population proportion ($p$) of confused clients.
		\item \textbf{Redesign without extra budget:} Instead of passively sending an email broadcast to 100,000 clients, the UX team should select a much smaller, rigorously random sample (e.g., $n = 500$ clients) and implement active, controlled follow-ups (such as an in-session intercept pop-up triggered upon transaction completion or scheduled direct outreach), thereby ensuring a high response rate free from self-selection frustration bias.
	\end{enumerate}
\end{solproblem}

\subsubsection*{3. Analytical Level (Advanced / Theoretical Connection)}

\begin{solproblem}
	Let $\bar{X}_n$ be the sample mean estimator based on a sample of size $n$. By definition, the Mean Squared Error (MSE) of an estimator decomposes the total estimation inaccuracy into two distinct algebraic components: sample variance and squared bias:
	\begin{align}
		\text{MSE}(\bar{X}_n) = \mathbb{E}\left[(\bar{X}_n - \mu)^2\right] = \text{Var}(\bar{X}_n) + (\text{Bias})^2 = \dfrac{\sigma^2}{n} + B^2,
	\end{align}
	where $B = \mathbb{E}[\bar{X}_n] - \mu$ represents the systematic methodological bias derived from the sampling frame selection mechanism (e.g., in \emph{Literary Digest}, surveying wealthy individuals with telephones and automobiles during the Great Depression).
	
	As the sample size increases indefinitely ($n \to \infty$), the variance component approaches zero ($\lim_{n \to \infty} \frac{\sigma^2}{n} = 0$) by virtue of the Law of Large Numbers. However, the systematic bias component remains completely invariant:
	\begin{align}
		\lim_{n \to \infty} \text{MSE}(\bar{X}_n) = 0 + B^2 = B^2 > 0.
	\end{align}
	This proves mathematically that a colossal sample of millions of biased records ($n = 2.4 \text{ million}$) merely reduces the sampling variance around an incorrect value, creating an illusion of absolute certainty regarding an erroneous estimate. Conversely, a smaller, well-designed random sample like Gallup's ($n = 50,000$), where $B = 0$, achieves unbiased convergence exactly toward the true population parameter $\mu$.
\end{solproblem}

\begin{solproblem}
	From the standard error and diminishing returns analysis:
	\begin{enumerate}
		\item We require a new standard error $\sigma_{\bar{X}}'$ such that $\sigma_{\bar{X}}' = \frac{1}{2}\sigma_{\bar{X}}$. Let $k$ be the multiplicative factor applied to the original sample size such that the new sample size is $n' = k \cdot n$. Then:
		\begin{align}
			\sigma_{\bar{X}}' = \dfrac{\sigma}{\sqrt{n'}} = \dfrac{\sigma}{\sqrt{k \cdot n}} = \dfrac{1}{\sqrt{k}} \cdot \left(\dfrac{\sigma}{\sqrt{n}}\right) = \dfrac{1}{\sqrt{k}} \sigma_{\bar{X}}.
		\end{align}
		Equating the scaling factors:
		\begin{align}
			\dfrac{1}{\sqrt{k}} = \dfrac{1}{2} \implies \sqrt{k} = 2 \implies k = 4.
		\end{align}
		Therefore, to cut the sampling standard error exactly in half, it is mandatory to \textbf{quadruple} the sample size ($n' = 4n$).
		\item \textbf{Big Data implications:} The inverse square-root relationship dictates a severe law of diminishing returns. If scaling from $n=1,000$ to $n=4,000$ (adding $3,000$ records) halves the error, achieving another equivalent halving requires scaling to $n=16,000$ (adding an extra $12,000$ records). In massive Big Data pipelines, collecting and processing additional petabytes incurs exponential computational and financial costs while yielding only marginal improvements in standard error. Consequently, investing in rigorous data quality and sampling frame representativeness (eliminating bias $B$) is vastly more cost-effective than blindly inflating raw data volume.
	\end{enumerate}
\end{solproblem}

\subsubsection*{4. Challenging Level (Expert / Exploration)}

\begin{solproblem}
	From the algorithmic and log truncation analysis:
	\begin{enumerate}
		\item By truncating sessions with duration $X > 15 \text{ min}$, the empirical distribution observed by the data pipeline is no longer the true population distribution $f(x)$, but rather the right-truncated conditional distribution:
		\begin{align}
			f_{\text{obs}}(x) = f(x \mid X \le 15) = \dfrac{f(x)}{P(X \le 15)} \quad \text{for } 0 \le x \le 15.
		\end{align}
		Because the true population expected value integrates across the entire unbounded domain ($0 \le x < \infty$):
		\begin{align}
			\mu = \int_{0}^{15} x f(x) \, dx + \int_{15}^{\infty} x f(x) \, dx,
		\end{align}
		the sample mean computed exclusively from the observed logs ($\bar{X}_{\text{obs}}$) converges to the truncated expectation $\mu_{\text{obs}} = \mathbb{E}[X \mid X \le 15]$, whose integral omits the heavy right tail of prolonged user sessions. Consequently, the observed sample mean systematically underestimates the true population mean session duration ($\mathbb{E}[\bar{X}_{\text{obs}}] < \mu$).
		\item The arithmetic mean is a statistical operator with a \textbf{zero breakdown point} (\emph{zero breakdown point}), meaning that a single extreme outlier or corrupted record ($x_i \to 0$ due to $0.001 \text{ s}$ telemetry bursts or an artificially inflated log entry) can drag the sample mean arbitrarily far. Therefore, in noisy Big Data pipelines, data scientists must deploy \textbf{robust descriptive measures}, such as the \textbf{sample median} (which features a $50\%$ breakdown point) or the \textbf{trimmed/truncated mean} (e.g., trimming $5\%$ or $10\%$ from both tails), which successfully isolate extreme anomalies without distorting the underlying central tendency.
	\end{enumerate}
\end{solproblem}

\begin{solproblem}
	From the post-stratification weighting derivation:
	
	By fundamental mathematical definition, the total population mean $\mu$ of a population of size $N$ partitioned into $K$ mutually exclusive strata is expressed as the weighted average of the individual stratum population means ($\mu_k$):
	\begin{align}
		\mu = \dfrac{1}{N} \sum_{i=1}^N X_i = \sum_{k=1}^K \dfrac{N_k}{N} \mu_k = \sum_{k=1}^K w_k \mu_k,
	\end{align}
	where $w_k = \frac{N_k}{N}$ denotes the true proportion of stratum $k$ in the total census ($\sum w_k = 1$).
	
	Since the sample mean $\bar{X}_k = \frac{1}{n_k} \sum_{j=1}^{n_k} X_{kj}$ calculated within stratum $k$ is an unbiased estimator of the within-stratum population mean ($\mathbb{E}[\bar{X}_k] = \mu_k$), we apply the expectation operator by linearity to the proposed estimator $\bar{X}_{est}$:
	\begin{align}
		\mathbb{E}[\bar{X}_{est}] = \mathbb{E}\left[ \sum_{k=1}^K w_k \bar{X}_k \right] = \sum_{k=1}^K w_k \mathbb{E}[\bar{X}_k] = \sum_{k=1}^K w_k \mu_k = \mu.
	\end{align}
	This rigorously proves that $\bar{X}_{est}$ is an unbiased estimator of the overall population mean, even if the sample was collected across strata in disproportionate numbers.
	
	\textbf{Verification under proportional allocation:}
	If the sampling design is strictly proportional by construction, then $n_k = n \left(\frac{N_k}{N}\right) = n \cdot w_k$, meaning that the stratum weight equals the exact sampling fraction: $w_k = \frac{n_k}{n}$. Substituting this into the post-stratified formula yields:
	\begin{align}
		\bar{X}_{est} = \sum_{k=1}^K \left( \dfrac{n_k}{n} \right) \bar{X}_k = \dfrac{1}{n} \sum_{k=1}^K n_k \left( \dfrac{1}{n_k} \sum_{j=1}^{n_k} X_{kj} \right) = \dfrac{1}{n} \sum_{k=1}^K \sum_{j=1}^{n_k} X_{kj} = \bar{X}.
	\end{align}
	In other words, under proportional allocation, the weighted post-stratified average simplifies elegantly to the simple arithmetic mean of all $n$ sample data points.
\end{solproblem}
