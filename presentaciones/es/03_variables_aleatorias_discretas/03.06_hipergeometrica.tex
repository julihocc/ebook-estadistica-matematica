% Slide 3: Motivación en Ciencia de Datos
\begin{frame}{El Rol del Muestreo en Poblaciones Finitas}
  \footnotesize
  \begin{motivacion}[¿Por qué no podemos usar la Binomial siempre?]
    Cuando auditamos un lote industrial de 50 procesadores o revisamos transacciones contables sospechosas, cada elemento extraído y verificado \emph{no regresa} al lote original. La suposición de probabilidad de éxito constante ($p$) se rompe, y los ensayos se vuelven lógicamente dependientes.
  \end{motivacion}
  
  \vspace{0.15cm}
  \pause
  \begin{itemize}\itemsep=0.1cm
    \item \textbf{Muestreo con Reemplazo (Binomial):} La probabilidad $p$ es estrictamente invariable y los ensayos son mutuamente independientes ($P(A \cap B) = P(A)P(B)$). \pause
    \item \textbf{Muestreo sin Reemplazo (Hipergeométrica):} La población disminuye ($N \to N-1$). El resultado del primer intento modifica drásticamente la probabilidad del segundo, generando una \textbf{covarianza negativa} que reduce la incertidumbre total del sistema.
  \end{itemize}
\end{frame}

% Slide 4: Definición Formal PMF
\begin{frame}{Definición Formal de la Distribución Hipergeométrica}
  \footnotesize
  Consideremos una población de tamaño $N$ dividida en $K$ éxitos y $N-K$ fracasos. Si extraemos una muestra aleatoria de tamaño $n$ \emph{sin reemplazo}, el número de éxitos $X$ sigue una ley hipergeométrica: \pause

  \vspace{0.1cm}
  \begin{block}{Función de Masa de Probabilidad (PMF)}
    $$P(X = k) = \dfrac{\comb{K}{k} \comb{N-K}{n-k}}{\comb{N}{n}}, \quad \text{para } k \in \mathcal{S}_X$$
  \end{block}

  \vspace{0.1cm}
  \pause
  \begin{alertblock}{Interpretación Combinatoria y Parametrización en Python}
    \begin{itemize}\itemsep=0.06cm
      \item \textbf{Numerador:} Formas de elegir $k$ éxitos de los $K$ disponibles por las formas de elegir $n-k$ fracasos de los $N-K$ restantes.
      \item \textbf{Denominador:} Total combinatorio de muestras de tamaño $n$ extraídas de la población $N$.
      \item \textbf{En SciPy (`scipy.stats`):} Se utiliza \texttt{hypergeom.pmf(k, M=N, n=K, N=n)}.
    \end{itemize}
  \end{alertblock}
\end{frame}

% Slide 5: Soporte Combinatorio Exacto
\begin{frame}{Soporte Combinatorio y Normalización de Vandermonde}
  \footnotesize
  El intervalo de valores físicamente posibles para $X$ no siempre es $\{0, 1, \dots, n\}$ debido a las restricciones físicas de la población finita: \pause

  \vspace{0.1cm}
  \begin{block}{Soporte Exacto $\mathcal{S}_X$}
    $$\mathcal{S}_X = \left\{ k \in \Z \;\Big|\; \max(0, \, n - (N - K)) \le k \le \min(n, \, K) \right\}$$
  \end{block}

  \vspace{0.1cm}
  \pause
  \begin{alertblock}{Identidad de Vandermonde (Normalización Unitaria)}
    Para verificar la suma unitaria de la probabilidad, empleamos la identidad clásica algebraicamente idéntica a sumar sobre todas las particiones del tamaño muestral $n$:
    $$\sum_{k \in \mathcal{S}_X} \comb{K}{k} \comb{N-K}{n-k} = \comb{N}{n} \implies \sum_{k \in \mathcal{S}_X} P(X = k) = 1$$
  \end{alertblock}
\end{frame}

% Slide 6: Esperanza Matemática por Linealidad
\begin{frame}{Teorema: Esperanza Matemática e Invariancia de Muestreo}
  \footnotesize
  Descomponemos el conteo total como suma de variables indicadoras $X = \sum_{i=1}^n I_i$, donde $I_i = 1$ si la $i$-ésima extraction resulta en un éxito y $0$ en caso contrario: \pause

  \vspace{0.1cm}
  \begin{block}{Esperanza Condicional e Invariancia Marginal}
    Por simetría combinatoria, la probabilidad marginal de que cualquier intento $i$ sea un éxito es idéntica a la primera extraction:
    $$P(I_i = 1) = \dfrac{K}{N} = p \implies \E[I_i] = \dfrac{K}{N}$$
  \end{block}

  \vspace{0.1cm}
  \pause
  \begin{alertblock}{Demostración por Linealidad Universal}
    La linealidad de la esperanza es universalmente válida, \textbf{incluso con variables dependientes}:
    $$\E[X] = \E\left[ \sum_{i=1}^n I_i \right] = \sum_{i=1}^n \E[I_i] = n \dfrac{K}{N} = np$$
  \end{alertblock}
\end{frame}

% Slide 7: Varianza, Covarianza Negativa y Factor FPCF
\begin{frame}{Teorema: Varianza y el Factor de Corrección (`FPCF`)}
  \footnotesize
  A diferencia de los ensayos de Bernoulli independientes donde $\Var(\sum I_i) = \sum \Var(I_i)$, el muestreo sin reemplazo introduce una covarianza entre extracciones distintas ($i \neq j$): \pause

  \vspace{0.08cm}
  \begin{block}{Covarianza Negativa de Extracción}
    $$\cov(I_i, I_j) = \Prob(I_i=1 \cap I_j=1) - \Prob(I_i=1)\Prob(I_j=1) = \dfrac{K(K-1)}{N(N-1)} - \left(\dfrac{K}{N}\right)^2 = -\dfrac{K(N-K)}{N^2(N-1)}$$
  \end{block}

  \vspace{0.08cm}
  \pause
  \begin{alertblock}{Varianza Hipergeométrica y Reducción Muestral}
    Al sumar todas las varianzas y las $n(n-1)$ covarianzas cruzadas negativas, obtenemos:
    $$\Var(X) = n \dfrac{K}{N} \left(1 - \dfrac{K}{N}\right) \left[ \dfrac{N-n}{N-1} \right] = np(1-p) \cdot \mathbf{\text{FPCF}}$$
  \end{alertblock}
\end{frame}

% Slide 8: Análisis de Sub-dispersión y Límites Muestrales
\begin{frame}{Análisis de Sub-dispersión: Censo vs. Muestreo}
  \footnotesize
  El Factor de Corrección por Población Finita (`FPCF`), definido como $\dfrac{N-n}{N-1}$, dicta la contracción en la dispersión según el tamaño de la muestra $n$: \pause

  \vspace{0.1cm}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Caso Límite 1: Muestra Unitaria ($n=1$)}
        $$\text{FPCF} = \dfrac{N-1}{N-1} = 1$$
        Al extraer un solo elemento, no hay reemplazo que afecte al futuro. La varianza coincide con una variable de Bernoulli $p(1-p)$.
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Caso Límite 2: Censo Total ($n=N$)}
        $$\text{FPCF} = \dfrac{N-N}{N-1} = 0 \implies \Var(X) = 0$$
        Al auditar toda la población, conocemos el conteo exacto ($X=K$) sin ninguna incertidumbre residual.
      \end{alertblock}
    \end{column}
  \end{columns}

  \vspace{0.15cm}
  \pause
  \begin{center}
    \scriptsize\textbf{Propiedad Crítica:} La hipergeométrica es siempre \textbf{sub-dispersa} ($\Var(X) < \E[X]$).
  \end{center}
\end{frame}

% Slide 9: Convergencia Asintótica y la Regla del 5%
\begin{frame}{Convergencia Asintótica a la Binomial (Regla del 5\%)}
  \footnotesize
  Cuando el tamaño de la población $N$ crece indefinidamente manteniendo la proporción inicial de éxitos fija ($p = K/N$), la pérdida de un elemento se vuelve casi insignificante: \pause

  \vspace{0.1cm}
  \begin{block}{Límite Asintótico de Vandermonde ($N \to \infty$)}
    $$\lim_{N \to \infty} \comb{K}{k}\comb{N-K}{n-k} \Big/ \comb{N}{n} = \comb{n}{k} p^k (1-p)^{n-k}$$
    El modelo hipergeométrico converge exactly al modelo binomial $\text{Bin}(n, p)$.
  \end{block}

  \vspace{0.1cm}
  \pause
  \begin{alertblock}{Regla de Oro en Ingeniería de Calidad ($\dfrac{n}{N} < 0.05$)}
    En auditorías reales, si la muestra $n$ representa menos del $5\%$ del lote total $N$, el `FPCF` satisface $\dfrac{N-n}{N-1} > 0.95$. En tal caso, se permite usar la Binomial para simplificar cálculos sin error apreciable.
  \end{alertblock}
\end{frame}

  \begin{block}{Problema 3.6.2 — Aceptación de Lotes de Calidad Industrial (Enunciado)}

  \end{block}

  \vspace{0.2cm}
  \pause
  \begin{alertblock}{Planteamiento del Modelo Exacto}
    La variable aleatoria $X$ representa el número de sensores defectuosos en la muestra. Dado que se extraen sin reemplazo de una población finita, sigue una ley hipergeométrica:
    $$X \sim \text{Hyper}(N=30, \; K=6, \; n=5)$$ \pause
    La probabilidad total de aceptación corresponde a la suma probabilística para $X \le 1$:
    $$\Prob(\text{Aceptar}) = \Prob(X = 0) + \Prob(X = 1)$$
  \end{alertblock}

\begin{frame}{Ejercicio en Clase — Nivel 2: Operativo (1/2)}
  \footnotesize
  \begin{block}{Problema 3.6.4 — Reducción de Varianza por FPCF (Enunciado)}
    Una empresa de envíos tiene una flota de $N = 50$ camiones de reparto, de los cuales $K = 10$ tienen problemas en el sistema de frenos ($p = 0.20$). El inspector selecciona al azar $n = 20$ camiones para una revisión detallada sin reemplazo. Compara cuantitativamente la varianza de camiones con fallas bajo el modelo hipergeométrico frente al modelo binomial.
  \end{block}

  \vspace{0.2cm}
  \pause
  \begin{alertblock}{Varianza Bajo Supuesto Binomial (Con Reemplazo)}
    Si el muestreo se realizara con reemplazo (o si la población fuera infinita), la varianza respondería a los ensayos independientes de Bernoulli:
    $$\Var_{\text{Bin}}(X) = n p (1-p) = 20(0.20)(0.80) = \mathbf{3.20}$$
  \end{alertblock}

% Slide 13B: Ejercicio en Clase — Nivel 2: Operativo (Resolución)
\begin{frame}{Ejercicio en Clase — Nivel 2: Operativo (2/2)}
  \footnotesize
  Ahora incorporamos la naturaleza exacta del muestreo en población finita sin reemplazo: \pause

  \vspace{0.15cm}
  \begin{block}{Factor de Corrección y Varianza Exacta}
    \begin{itemize}\itemsep=0.15cm
      \item \textbf{Factor de Corrección por Población Finita (FPCF):}

  \end{block}

  \vspace{0.15cm}
  \pause
  \begin{alertblock}{Análisis de Reducción de Incertidumbre}
    El muestreo sin reemplazo reduce la incertidumbre de estimación en un exacto $\mathbf{38.78\%}$ frente al modelo binomial convencional, al eliminar el doble conteo de camiones.
  \end{alertblock}

% Slide 14A: Ejercicio en Clase — Nivel 3: Analítico (Planteamiento)
\begin{frame}{Ejercicio en Clase — Nivel 3: Analítico (1/2)}
  \footnotesize
  \begin{block}{Problema 3.6.7 — Demostración de Momento Factorial (Enunciado)}
    A partir de la identidad combinatoria fundamental $k \comb{K}{k} = K \comb{K-1}{k-1}$, demuestra algebraicamente que el primer momento factorial de una variable hipergeométrica es exactly $\E[X] = n \dfrac{K}{N}$.
  \end{block}

  \vspace{0.1cm}
  \pause
  \begin{alertblock}{Paso 1: Expansión de la Definición de Esperanza}
    \scriptsize
    Partimos de la suma ponderada de probabilidades sobre el soporte $k > 0$:
    $$\E[X] = \sum_{k=1}^{\min(n,K)} k \dfrac{\comb{K}{k}\comb{N-K}{n-k}}{\comb{N}{n}}$$ \pause
    Aplicamos absorción en el numerador $k\comb{K}{k} = K\comb{K-1}{k-1}$ y en el denominador $n\comb{N}{n} = N\comb{N-1}{n-1} \implies \comb{N}{n} = \frac{N}{n}\comb{N-1}{n-1}$.
  \end{alertblock}
\end{frame}

% Slide 14B: Ejercicio en Clase — Nivel 3: Analítico (Resolución)
\begin{frame}{Ejercicio en Clase — Nivel 3: Analítico (2/2)}
  \scriptsize
  Al sustituir ambas identidades combinatorias en la expresión de la esperanza obtenemos: \pause

  \vspace{-0.15cm}
  \begin{block}{Paso 2: Extracción y Suma de Vandermonde}
    \scriptsize
    $\E[X] = \dfrac{n K}{N} \sum_{k=1}^{\min(n,K)} \dfrac{\comb{K-1}{k-1}\comb{N-K}{n-k}}{\comb{N-1}{n-1}}$ \pause \\
    \vspace{0.02cm}
    Con el índice $j = k-1$, la suma abarca todas las muestras de tamaño $n-1$: \\
    \vspace{0.02cm}
    $\E[X] = \dfrac{n K}{N} \sum_{j=0}^{\min(n-1,K-1)} \dfrac{\comb{K-1}{j}\comb{(N-1)-(K-1)}{(n-1)-j}}{\comb{N-1}{n-1}}$
  \end{block}

  \vspace{-0.15cm}
  \pause
  \begin{alertblock}{Conclusión por Normalización Unitaria}
    \scriptsize
    La sumatoria es la probabilidad total de $\text{Hyper}(N-1, K-1, n-1)$ y vale $1 \implies \E[X] = n \dfrac{K}{N} \cdot 1 = np \quad \blacksquare$
  \end{alertblock}
\end{frame}

% Slide 15A: Ejercicio en Clase — Nivel 4: Desafiante (Enunciado)
\begin{frame}{Ejercicio en Clase — Nivel 4: Desafiante (1/2)}
  \footnotesize
  \begin{block}{Problema 3.6.10 — Prueba Exacta de Fisher en Ensayos Biomédicos}
    Un ensayo clínico de cardiología evalúa una nueva terapia antitrombótica con $N = 25$ pacientes severos. Se asignan aleatoriamente $K = 12$ pacientes al fármaco activo y los $13$ restantes al placebo. Al finalizar el ensayo, exactly $n = 8$ pacientes desarrollaron trombosis profunda. Si el fármaco fuera ineficaz ($H_0$), las 8 trombosis se distribuirían al azar.
  \end{block}

  \vspace{0.2cm}
  \pause
  \begin{alertblock}{Planteamiento de la Hipótesis e Inferencia Exacta}
    Sea $X$ el número de pacientes en el grupo activo que desarrollaron trombosis. Bajo la hipótesis nula $H_0$, la variable es estrictamente hipergeométrica:
    $$X \sim \text{Hyper}(N=25, \; K=12, \; n=8)$$ \pause
    En nuestro ensayo clínico, observamos únicamente $X = 1$ trombosis en el grupo activo (y las otras 7 en el grupo placebo). Evaluemos la significancia estadística exacta.
  \end{alertblock}
\end{frame}

% Slide 15B: Ejercicio en Clase — Nivel 4: Desafiante (Resolución)
\begin{frame}{Ejercicio en Clase — Nivel 4: Desafiante (2/2)}
  \scriptsize
  Para determinar significancia a nivel $\alpha = 0.05$, calculamos el $p$-value exacto de una cola ($\Prob(X \le 1)$ bajo $H_0$): \pause

  \vspace{0.04cm}
  \begin{block}{Cálculo del $p$-value por la Ley Hipergeométrica}
    \scriptsize
    $p\text{-value} = \Prob(X = 0) + \Prob(X = 1) = \dfrac{\comb{12}{0}\comb{13}{8}}{\comb{25}{8}} + \dfrac{\comb{12}{1}\comb{13}{7}}{\comb{25}{8}}$ \pause
    \begin{itemize}\itemsep=0.01cm
      \item $\Prob(X=0) = \frac{1 \times 1,287}{1,081,575} \approx 0.00119$ \quad | \quad $\Prob(X=1) = \frac{12 \times 1,716}{1,081,575} \approx 0.01904$
    \end{itemize}
  \end{block}

  \vspace{0.04cm}
  \pause
  \begin{alertblock}{Decisión Clínica ($p\text{-value} < \alpha$)}
    \scriptsize
    Como $p\text{-value} = 0.00119 + 0.01904 = \mathbf{0.02023}$ ($2.02\%$) es menor que $\alpha = 0.05$, rechazamos $H_0$. Concluimos que la reducción de trombosis en la terapia activa es clínicamente significativa.
  \end{alertblock}
\end{frame}

% Slide 16: Cierre de Sección
\begin{frame}{Cierre de Sección y Síntesis Combinatoria}
  \begin{reflexion}[El Poder del Muestreo sin Reemplazo]
    La distribución hipergeométrica nos enseña una valiosa lección de ingeniería y ciencia de datos: extraer información sin reemplazo de una población finita es intrínsecamente más preciso que muestrear con reemplazo. Gracias a la covarianza negativa entre observaciones, el Factor de Corrección por Población Finita (`FPCF`) reduce la varianza muestral, permitiéndonos tomar decisiones rigurosas con muestras más pequeñas.
  \end{reflexion}

  \vspace{0.4cm}
  \pause
  \begin{block}{Lecciones Clave del Muestreo Combinatorio}
    \begin{itemize}\itemsep=0.15cm
      \item \textbf{Invariancia Marginal:} $\E[I_i] = K/N$ se mantiene intacta en cualquier extraction $i$.
      \item \textbf{Sub-dispersión:} La varianza siempre es menor que el valor esperado.
      \item \textbf{Pruebas Exactas:} Es la base matemática incontrovertible del test exacto de Fisher.
    \end{itemize}
  \end{block}
\end{frame}
