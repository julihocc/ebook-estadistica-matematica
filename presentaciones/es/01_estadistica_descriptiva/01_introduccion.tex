% Slide 2: Motivación del Mundo Real (1 solo entorno + pausas)
\begin{frame}{¿Por qué necesitamos la Estadística Descriptiva?}
  \begin{motivacion}
    Imagina que eres Científico de Datos en \textbf{Mercado Libre} o \textbf{Netflix}. Cada segundo se generan millones de clics, compras y transacciones.
  \end{motivacion}
  
  \vspace{0.6em}
  La \textbf{Estadística Descriptiva} es nuestra herramienta fundamental de compresión de información:
  \begin{itemize}
    \item \textbf{Resumir:} Convierte gigabytes de datos brutos en 3 o 4 métricas interpretables. \pause
    \item \textbf{Detectar:} Saca a la luz patrones ocultos, anomalías graves o fraudes. \pause
    \item \textbf{Comunicar:} Permite reportar hallazgos de forma clara y ejecutiva a directivos e ingenieros.
  \end{itemize}
\end{frame}

% Slide 3: Reflexión para el Aula (1 solo entorno + pausa)
\begin{frame}{Reflexión para el Aula: El Riesgo del Sesgo (2 min)}
  \begin{reflexion}
    Un analista quiere conocer el salario promedio de los ingenieros de software en México. Como no tiene acceso a la nómina de todas las empresas del país, entrevista a \textbf{200 ingenieros} afuera de una torre corporativa en una zona financiera exclusiva.
  \end{reflexion}
  
  \vspace{0.8em}
  \pause
  \begin{itemize}
    \item ¿Qué problemas metodológicos observas en esta muestra? \pause
    \item ¿El resultado representará fielmente la realidad nacional? \pause
    \item \textbf{Reflexión clave:} La diferencia crítica entre lo que queremos medir (\emph{Población}) y lo que realmente podemos observar (\emph{Muestra}).
  \end{itemize}
\end{frame}

% Slide 4: Conceptos Clave - Descriptiva vs Inferencial (Sin sobrecarga)
\begin{frame}{Dos Ramas Complementarias del Análisis de Datos}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      {\large \color{TecAzulOscuro}\textbf{Estadística Descriptiva}}
      \vspace{0.3em}
      \begin{itemize}
        \item \textbf{Enfoque:} Resumir y organizar el conjunto de datos observado. \pause
        \item \textbf{Herramientas:} Tablas, histogramas, promedios, medidas de dispersión. \pause
        \item \textbf{Certeza:} 100\% sobre los datos medidos; no hace suposiciones sobre lo no visto.
      \end{itemize}
    \end{column}
    
    \begin{column}{0.48\textwidth}
      {\large \color{TecRojo}\textbf{Estadística Inferencial}}
      \vspace{0.3em}
      \begin{itemize}
        \item \textbf{Enfoque:} Utiliza los datos medidos para hacer predicciones y generalizaciones. \pause
        \item \textbf{Herramientas:} Intervalos de confianza, pruebas de hipótesis, regresión. \pause
        \item \textbf{Incertidumbre:} Cuantifica rigurosamente el margen de error y la probabilidad.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

% Slide 5: Población vs Muestra & Parámetro vs Estadístico
\begin{frame}{Población vs. Muestra y Parámetro vs. Estadístico}
  Para no perdernos en la notación matemática, recordemos este puente fundamental:
  
  \vspace{0.4em}
  \begin{center}
    \small
    \begin{tabular}{p{3.2cm} p{4.6cm} p{4.6cm}}
      \toprule
      \textbf{Concepto} & \textbf{Población (Universo Total)} & \textbf{Muestra (Subconjunto)} \\
      \midrule
      \textbf{Definición} & Todos los individuos de interés & Parte selecta y representativa \\
      \textbf{Medida asociada} & \textbf{Parámetro} (Fijo, oculto) & \textbf{Estadístico} (Calculado) \\
      \textbf{Media (Promedio)} & $\mu$ (Letra griega mu) & $\bar{X}$ o $\bar{x}$ (X barra) \\
      \textbf{Varianza} & $\sigma^2$ (Sigma al cuadrado) & $S^2$ o $s^2$ (S al cuadrado) \\
      \textbf{Proporción} & $p$ o $\pi$ & $\hat{p}$ (p gorro) \\
      \bottomrule
    \end{tabular}
  \end{center}

  \vspace{0.4em}
  \pause
  \begin{itemize}
    \item \textbf{Regla de oro:} Nunca calculamos $\mu$ exactamente a menos que hagamos un censo global. ¡Usamos $\bar{X}$ como un telescopio para estimar $\mu$!
  \end{itemize}
\end{frame}

% Slide 7: Reto Rápido en Equipo
\begin{frame}{Reto Rápido en Equipo: ¿Parámetro o Estadístico?}
  \begin{retoclase}
    Clasifica cada valor como un \textbf{Parámetro ($\mu, \sigma^2, p$)} o un \textbf{Estadístico ($\bar{X}, S^2, \hat{p}$)}:
  \end{retoclase}
  
  \vspace{0.5em}
  \begin{enumerate}
    \item Se escogen al azar 50 transacciones en un servidor y se calcula un monto promedio de \textbf{\$1,450 MXN}. \pause
    \item El Instituto Nacional de Estadística censó las 35,000 secundarias del país y determinó que el \textbf{12\%} no tiene laboratorio de cómputo. \pause
    \item Un algoritmo de ML entrena con un batch de 512 imágenes y calcula una pérdida (\emph{loss}) promedio de \textbf{0.34}. \pause
    \item Por diseño de ingeniería, una máquina embotelladora llena latas con un volumen promedio teórico exacto de \textbf{355 ml}.
  \end{enumerate}
\end{frame}

% Slide 8: Resumen y Conexión
\begin{frame}{Resumen y Conexión con la Sección 1.2}
  \begin{itemize}
    \item La \textbf{Estadística Descriptiva} organiza y resume los datos observados de manera rigurosa y clara. \pause
    \item Siempre debemos distinguir entre el \textbf{Parámetro} (la verdad oculta de la Población) y el \textbf{Estadístico} (el resumen que obtenemos de nuestra Muestra). \pause
    \item La representatividad muestral es la clave: un cálculo perfecto sobre una muestra sesgada produce conclusiones erróneas.
  \end{itemize}

  \vspace{0.8em}
  \pause
  {\large \color{TecAzulOscuro}\textbf{¿Qué veremos en la Sección 1.2?}}
  \vspace{0.3em}
  \begin{itemize}
    \item ¿Cuál es el mejor "resumen de un solo número" para un conjunto de datos? Exploraremos las \textbf{Medidas de Tendencia Central} (\emph{Media, Mediana y Moda}) y por qué el promedio suele engañarnos en presencia de valores extremos (\emph{outliers}).
  \end{itemize}
\end{frame}
