		\begin{column}{0.48\textwidth}
			\begin{block}{The Quantitative Imperative}
				\footnotesize
				To apply the rigorous machinery of mathematical analysis and linear algebra, every qualitative outcome $\omega \in \Omega$ must be **projected** onto a well-defined numerical value along the real line $\mathbb{R}$.
				
				\vspace{0.05cm}
				This projection quantifies the exact operational metric of interest to engineers and data scientists.
			\end{block}

	\end{columns}

	\begin{alertblock}{Core Modeling Principle}
		\footnotesize

% SLIDE 05: Formal Definition of Random Variable
\begin{frame}{Formal Definition: Discrete Random Variable}
	\begin{block}{Definition of a Random Variable}
		\small
		A \textbf{random variable} (or \emph{stochastic function}) is a measurable mapping $X: \Omega \to \mathbb{R}$ that assigns a unique real number $X(\omega) = x$ to every sample point $\omega \in \Omega$.
	\end{block}
	\vspace{-0.1cm}
	\begin{block}{Discrete Characterization}
		\small
		A random variable $X$ is classified as **discrete** if its range (the set of values it can assume with positive probability) is finite or countably infinite. This set is formally termed the **support** of $X$, denoted by:
		\begin{align*}
			\mathcal{S}_X = \set{x_1, x_2, \dots, x_k, \dots} \subset \mathbb{R}.
		\end{align*}
	\end{block}
	\vspace{-0.15cm}
	\begin{alertblock}{Notational Convention}
		\footnotesize
		By strict mathematical convention, uppercase letters ($X, Y, Z$) denote random variables, while lowercase letters ($x, y, z$) represent specific scalar realizations.
	\end{alertblock}
\end{frame}

% SLIDE 06: Support of a Random Variable
\begin{frame}{Measurable Support and Discrete Cardinality}
	\begin{columns}[T]
		\begin{column}{0.48\textwidth}
			\begin{block}{Finite Support}
				\small
				The support $\mathcal{S}_X$ contains a bounded number of elements ($|\mathcal{S}_X| = n < \infty$).
				\begin{itemize}
					\item \textbf{Sum of two dice:} $\mathcal{S}_X = \set{2, 3, \dots, 12}$ ($11$ elements).
					\item \textbf{Batch defects:} $\mathcal{S}_X = \set{0, 1, \dots, N}$.
				\end{itemize}
			\end{block}
		\end{column}
		\begin{column}{0.48\textwidth}
			\begin{block}{Countably Infinite Support}
				\small
				The support $\mathcal{S}_X$ can be put in bijection with the natural numbers ($\mathbb{N}$ or $\mathbb{Z}^+$).
				\begin{itemize}
					\item \textbf{Trials to first success:} $\mathcal{S}_X = \set{1, 2, 3, \dots}$ (Geometric).
					\item \textbf{Server arrivals:} $\mathcal{S}_X = \set{0, 1, 2, \dots}$ (Poisson).
				\end{itemize}
			\end{block}
		\end{column}
	\end{columns}
	\vspace{-0.15cm}
	\begin{block}{Exclusion of Null Points}
		\footnotesize
		By rigorous measure-theoretic definition, the support excludes any real number whose point probability is zero: $\mathcal{S}_X = \set{x \in \mathbb{R} : P(X = x) > 0}$.
	\end{block}
\end{frame}

% SLIDE 07: Probability Mass Function (PMF)
\begin{frame}{Probability Mass Function (PMF)}
	\begin{block}{Formal Definition of the PMF}
		\footnotesize
		Let $X$ be a discrete random variable with support $\mathcal{S}_X$. The **Probability Mass Function** (or *discrete probability function*), denoted by $p_X(x)$ or simply $f(x)$, assigns to every real number $x$ the exact probability that $X$ equals $x$:
		\begin{align*}
			f(x) = P(X = x) = P\left(\set{\omega \in \Omega : X(\omega) = x}\right).
		\end{align*}
	\end{block}
	\vspace{-0.2cm}
	\begin{block}{Extension to the Entire Real Line $\mathbb{R}$}
		\footnotesize
		To ensure full domain coverage across $\mathbb{R}$, the function is defined to vanish outside its support:
		\begin{align*}
			f(x) = \begin{cases}
				P(X = x_k) & \text{if } x = x_k \in \mathcal{S}_X, \\[2pt]
				0 & \text{otherwise ($x \notin \mathcal{S}_X$).}
			\end{cases}
		\end{align*}
	\end{block}
\end{frame}

% SLIDE 08: Kolmogorov Axioms & Normalization
\begin{frame}{Kolmogorov Axioms for Discrete Functions}
	\begin{block}{Existence and Legitimacy Theorem for PMFs}
		\small
		A real-valued function $f(x)$ constitutes a mathematically legitimate probability mass function if and only if it strictly satisfies two core Kolmogorov axioms:
		\begin{enumerate}
			\itemsep=0.15cm
			\item \textbf{Non-negativity:} For all $x \in \mathbb{R}$, the probability mass must be non-negative:
			\begin{align*}
				f(x) \ge 0.
			\end{align*}
			\item \textbf{Normalization (Unitary Sum):} The total probability mass summed across all points in the support must equal exactly one:
			\begin{align*}
				\sum_{x \in \mathcal{S}_X} f(x) = 1.
			\end{align*}
		\end{enumerate}
	\end{block}
\end{frame}

% SLIDE 09: Canonical Textbook Examples I
\begin{frame}{Canonical Textbook Examples: Coins and Dice}
	\begin{columns}[T]
		\begin{column}{0.48\textwidth}
			\begin{block}{Example 2.6.2: Two Coins ($X = \text{Heads}$)}
				\footnotesize
				Space $\Omega = \set{HH, HT, TH, TT}$ equiprobable. Support $\mathcal{S}_X = \set{0, 1, 2}$.
				\begin{align*}
					f(x) = \begin{cases}
						1/4 & x = 0 \text{ ($TT$)} \\
						1/2 & x = 1 \text{ ($HT, TH$)} \\
						1/4 & x = 2 \text{ ($HH$)} \\
						0 & \text{otherwise.}
					\end{cases}
				\end{align*}
			\end{block}
		\end{column}
		\begin{column}{0.48\textwidth}
			\begin{block}{Example 2.6.3: Sum of Two Dice}
				\footnotesize
				Space $|\Omega| = 36$. Support $\mathcal{S}_X = \set{2, \dots, 12}$.
				Mass is proportional to the number of combinations summing to $x$:
				\begin{align*}
					f(x) = \frac{6 - |x - 7|}{36}, \quad x \in \set{2, \dots, 12}.
				\end{align*}
				For instance: $f(7) = 6/36$, $f(2) = f(12) = 1/36$.
			\end{block}
		\end{column}
	\end{columns}
	\vspace{-0.15cm}
	\begin{alertblock}{Axiomatic Verification}
		\scriptsize
		Both formulas satisfy non-negativity and unitary summation ($\sum f(x) = \frac{1+2+1}{4} = 1$ and $\sum f(x) = \frac{36}{36} = 1$).
	\end{alertblock}
\end{frame}

\begin{frame}{Problem Synthesis --- Fundamental Level (Direct / Introductory)}
	\begin{columns}[T]
		\begin{column}{0.48\textwidth}
			\begin{block}{Problem 3.1.1: Polynomial Normalization}
				\small
				Given $f(x) = c x^2$ over $\mathcal{S}_X = \set{1, 2, 3, 4}$, the unitary summation condition requires:
				\begin{align*}
					c(1 + 4 + 9 + 16) = 30c = 1 \implies c = \frac{1}{30}.
				\end{align*}
				The conditional probability evaluates to:
				\begin{align*}
					P(X \text{ even} \mid X \ge 2) = \frac{20/30}{29/30} = \frac{20}{29}.
				\end{align*}
			\end{block}
		\end{column}
		\begin{column}{0.48\textwidth}
			\begin{block}{Problem 3.1.2: Three Coins \& Absolute Difference}
				\small
				If $X = |n_H - n_T|$ in 3 coin tosses, the support is strictly $\mathcal{S}_X = \set{1, 3}$.
				\begin{align*}
					f(1) = \frac{6}{8} = \frac{3}{4}, \quad f(3) = \frac{2}{8} = \frac{1}{4}.
				\end{align*}
				The sum verifies precisely as $3/4 + 1/4 = 1$.
			\end{block}
		\end{column}
	\end{columns}
\end{frame}

% SLIDE 16: Problem Synthesis --- Operational Level
\begin{frame}{Problem Synthesis --- Operational Level (Intermediate / Applied)}
	\begin{columns}[T]
		\begin{column}{0.48\textwidth}
			\begin{block}{Problem 3.1.5: Absolute Transformation}
				\small
				Let $X$ be distributed on $\set{-2, -1, 0, 1, 2}$ and define $Y = |X - 1|$. The induced support is $\mathcal{S}_Y = \set{0, 1, 2, 3}$.
				
				\vspace{0.05cm}
				Preimage probabilities aggregate cleanly:
				\begin{align*}
					p_Y(1) = P(X=0) + P(X=2) = 0.30.
				\end{align*}
			\end{block}
		\end{column}
		\begin{column}{0.48\textwidth}
			\begin{block}{Problem 3.1.6: Maximum of Two Spheres}
				\footnotesize
				Drawing two spheres without replacement from $1$ to $6$, the maximum $M = \max(X_1, X_2)$ has mass:
				\begin{align*}
					f(m) = \frac{m-1}{15}, \quad m \in \set{2, \dots, 6}.
				\end{align*}
				The upper tail is $P(M \ge 5) = \frac{4+5}{15} = \frac{3}{5}$.
			\end{block}
		\end{column}
	\end{columns}
\end{frame}

% SLIDE 17: Problem Synthesis --- Analytical Level
\begin{frame}{Problem Synthesis --- Analytical Level (Advanced / Proofs)}
	\begin{block}{Problem 3.1.7: Telescoping Sums and Normalization over $\mathcal{S}_N = \set{1, \dots, N}$}
		\footnotesize
		Given the family $f_N(x) = \dfrac{c_N}{x(x+1)}$, partial fraction decomposition yields:
		\begin{align*}
			\sum_{x=1}^{N} \frac{c_N}{x(x+1)} &= c_N \sum_{x=1}^{N} \left( \frac{1}{x} - \frac{1}{x+1} \right) \\
			&= c_N \left( 1 - \frac{1}{N+1} \right) = c_N \left( \frac{N}{N+1} \right) = 1.
		\end{align*}
		This rigorously proves that the normalization constant is exactly $c_N = \dfrac{N+1}{N}$.
	\end{block}
	\vspace{-0.15cm}
	\begin{block}{Problem 3.1.8: Quadratic Mapping of a Symmetric Uniform Variable}
		\footnotesize
		For $X$ uniform over $\set{-m, \dots, m}$ ($2m+1$ points), the variable $Z = X^2$ concentrates twice the probability mass on positive squares ($p_Z(k^2) = \frac{2}{2m+1}$ from $\pm k$) and single mass at zero ($p_Z(0) = \frac{1}{2m+1}$).
	\end{block}
\end{frame}

% SLIDE 18: Problem Synthesis --- Challenging Level
\begin{frame}{Problem Synthesis --- Challenging Level (Experts / Paradoxes)}
	\begin{columns}[T]
		\begin{column}{0.48\textwidth}
			\begin{block}{Problem 3.1.9: Zipf Law and Pareto Tails}
				\small
				For $f(k) = \dfrac{1}{\zeta(s) k^s}$ across $\mathbb{Z}^+$, unitary normalization is guaranteed by Riemann zeta convergence.
				
				\vspace{0.05cm}
				Integral bounding establishes heavy-tail polynomial decay:
				\begin{align*}
					P(X > M) \sim \frac{1}{(s-1) M^{s-1}}.
				\end{align*}
			\end{block}
		\end{column}
		\begin{column}{0.48\textwidth}
			\begin{block}{Problem 3.1.10: Discrete Redundancy}
				\small
				For two independent subsystems with failure probabilities $p_A, p_B$, the time to first failure $T = \min(N_A, N_B)$ satisfies:
				\begin{align*}
					P(T > n) = \left[ (1-p_A)(1-p_B) \right]^n.
				\end{align*}
				$T$ is exactly Geometric with failure parameter $p_{\text{eq}} = p_A + p_B - p_A p_B$.
			\end{block}
		\end{column}
	\end{columns}
\end{frame}

% SLIDE 19: Conclusions
\begin{frame}{Conclusions and Summary}
	\begin{block}{Key Takeaways from Section 03.01}
		\small
		\begin{itemize}
			\itemsep=0.1cm
			\item \textbf{Discrete Random Variable:} A measurable function projecting an abstract probability space onto a finite or countably infinite numerical support $\mathcal{S}_X$.
			\item \textbf{Legitimate PMF:} Any discrete mass function $f(x) = P(X = x)$ is uniquely characterized by the fundamental Kolmogorov axioms: $f(x) \ge 0$ and $\sum_{x \in \mathcal{S}_X} f(x) = 1$.
			\item \textbf{Support Transformations:} Under a mapping $g(X)$, the probabilities of all disjoint preimages sum algebraically to generate the induced PMF.
		\end{itemize}
	\end{block}
