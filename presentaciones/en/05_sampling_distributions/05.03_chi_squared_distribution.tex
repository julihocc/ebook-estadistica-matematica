% SLIDE 03: Motivation
\begin{frame}{From Known $\sigma$ to Unknown $\sigma$}
	\footnotesize
	\begin{motivacion}
		The CLT (Section 05.02) approximates $\bar{X}$ with a Normal when $\sigma$ is \emph{known}. In practice we almost always must estimate $\sigma$ with $S$. For that we need to understand the exact distribution of $S^2$ --- and this is where the chi-squared comes in.
	\end{motivacion}
	\vspace{0.15cm}
	\pause
	\begin{itemize}\itemsep=0.1cm
		\item \textbf{Origin:} the chi-squared arises from summing squares of independent standard normals. \pause
		\item \textbf{Connection to 04.07:} we already saw that $\chi^2_\nu = \text{Gamma}(\nu/2, 2)$. \pause
		\item \textbf{Central question:} what is the exact distribution of $(n-1)S^2/\sigma^2$?
	\end{itemize}
\end{frame}

% SLIDE 03B: Definition and Properties
\begin{frame}{Definition and Properties of the Chi-Squared Distribution}
	\vspace{-0.15cm}
	\begin{columns}[T]
		\begin{column}{0.48\textwidth}
			\begin{block}{Definition and Density}
				\footnotesize
				$\chi^2_\nu = \sum_{i=1}^\nu Z_i^2$ for iid $Z_i\sim N(0,1)$. Density: $f(x)=\frac{1}{2^{\nu/2}\Gamma(\nu/2)}x^{\nu/2-1}e^{-x/2}$.
			\end{block}
		\end{column}
		\begin{column}{0.48\textwidth}
			\begin{alertblock}{Properties}
				\footnotesize
				$E(\chi^2_\nu)=\nu$, $\Var(\chi^2_\nu)=2\nu$; special case of Gamma$(\nu/2,2)$; reproductive: $\chi^2_{\nu_1}+\chi^2_{\nu_2}\sim\chi^2_{\nu_1+\nu_2}$.
			\end{alertblock}
		\end{column}
	\end{columns}
\end{frame}

% SLIDE 04: Fisher's Theorem
\begin{frame}{Fisher's Theorem}
	\vspace{-0.15cm}
	\begin{columns}[T]
		\begin{column}{0.48\textwidth}
			\begin{block}{Statement}
				\footnotesize
				For normal iid $X_1,\ldots,X_n$: (1) $\bar{X}$ and $S^2$ are independent; (2) $(n-1)S^2/\sigma^2 \sim \chi^2_{n-1}$.
			\end{block}
		\end{column}
		\begin{column}{0.48\textwidth}
			\begin{alertblock}{Why Surprising?}
				\footnotesize
				$S^2$ is built from $(X_i-\bar X)$, yet is independent of $\bar X$. An orthogonal transformation separates $\bar X$'s direction from the $n-1$ orthogonal directions capturing $(n-1)S^2$ (a case of Cochran's theorem).
			\end{alertblock}
		\end{column}
	\end{columns}
\end{frame}

% SLIDE 05: Confidence Interval for sigma^2
\begin{frame}{Exact Confidence Interval for $\sigma^2$}
	\vspace{-0.15cm}
	\begin{columns}[T]
		\begin{column}{0.48\textwidth}
			\begin{block}{Interval Formula}
				\footnotesize
				$\left[\dfrac{(n-1)S^2}{\chi^2_{n-1,1-\alpha/2}},\; \dfrac{(n-1)S^2}{\chi^2_{n-1,\alpha/2}}\right]$. Unlike Normal-based intervals, this one is \emph{asymmetric} around $S^2$.
			\end{block}
		\end{column}
		\begin{column}{0.48\textwidth}
			\begin{alertblock}{Hypothesis Testing for $\sigma^2$}
				\footnotesize
				The same statistic $(n-1)S^2/\sigma_0^2$ tests $H_0:\sigma^2=\sigma_0^2$ against $\chi^2_{n-1}$ quantiles.
			\end{alertblock}
		\end{column}
	\end{columns}
\end{frame}

% SLIDE 06: Applications
\begin{frame}{Applications in Inference and Quality Control}
	\vspace{-0.1cm}
	\begin{itemize}\itemsep=0.1cm
		\item \textbf{Quality control:} tests about manufacturing process variability against a target $\sigma^2$. \pause
		\item \textbf{Goodness-of-fit:} the $\chi^2$ test compares observed vs. expected frequencies. \pause
		\item \textbf{Preview 05.04:} the Student's $t$-distribution is built by combining a Normal with an independent chi-squared. \pause
		\item \textbf{Preview 05.05:} the $F$-distribution compares two variances via a ratio of two chi-squared variables.
	\end{itemize}
\end{frame}

	\begin{block}{Problem 5.3.1 --- Mean and Variance of $\chi^2_{15}$ (Statement)}
		Let $X \sim \chi^2_{15}$. Compute the mean $E(X)$ and the variance $\Var(X)$.
	\end{block}
	\vspace{0.15cm}
	\pause
	\begin{alertblock}{Model Setup}
		\begin{itemize}\itemsep=0.04cm
			\item Use directly $E(\chi^2_\nu)=\nu$ and $\Var(\chi^2_\nu)=2\nu$.
		\end{itemize}
	\end{alertblock}
\end{frame}

% SLIDE 12: Exercise Level 1 (Solution)
\begin{frame}{In-Class Exercise --- Level 1: Fundamental (2/2)}
	\vspace{-0.05cm}
	\scriptsize
	Substitute $\nu=15$ directly: \pause
	\begin{block}{Calculation}
		$$E(X) = 15, \qquad \Var(X) = 2\cdot 15 = 30.$$
	\end{block}
	\vspace{0.05cm}
	\pause
	\begin{alertblock}{Interpretation}
		\scriptsize
		Unlike the Normal, the chi-squared has mean and variance that grow together with $\nu$: more degrees of freedom means more absolute spread, though the distribution becomes relatively more symmetric (approaching Normal by the CLT).
	\end{alertblock}
\end{frame}

\begin{frame}{In-Class Exercise --- Level 2: Operational (1/2)}

	\begin{block}{Problem 5.3.6 --- Hypothesis Test for $\sigma^2$ (Statement)}
		A bottling machine should have variance $\sigma_0^2=10$. A sample of $n=16$ bottles yields $S^2=18.2$. Compute $(n-1)S^2/\sigma_0^2$ and decide whether to reject $H_0:\sigma^2=10$ at $\alpha=0.05$, using $\chi^2_{15,0.95}\approx 24.996$.
	\end{block}
	\vspace{0.15cm}
	\pause
	\begin{alertblock}{Strategy}
		\scriptsize
		Compare the computed statistic against the given critical value.
	\end{alertblock}
\end{frame}

% SLIDE 14: Exercise Level 2 (Solution)
\begin{frame}{In-Class Exercise --- Level 2: Operational (2/2)}
	\vspace{-0.05cm}
	\scriptsize
	Compute the statistic and compare: \pause
	\begin{block}{Calculation}
		$$\frac{(n-1)S^2}{\sigma_0^2} = \frac{15\cdot 18.2}{10} = 27.3$$
	\end{block}
	\vspace{0.05cm}
	\pause
	\begin{alertblock}{Decision}
		\scriptsize
		Since $27.3 > 24.996 = \chi^2_{15,0.95}$, we \textbf{reject} $H_0:\sigma^2=10$ at $\alpha=0.05$: there is statistical evidence that the machine's true variance exceeds the target of $10$.
	\end{alertblock}
\end{frame}

% SLIDE 15: Exercise Level 3 (Statement)
\begin{frame}{In-Class Exercise --- Level 3: Analytical (1/2)}
	\vspace{-0.1cm}
	\begin{block}{Problem 5.3.7 --- Deriving $E(\chi^2_\nu)$ and $\Var(\chi^2_\nu)$ (Statement)}
		Starting from $\chi^2_\nu=\text{Gamma}(\nu/2,2)$ and $\mu_X=\alpha\beta$, $\sigma_X^2=\alpha\beta^2$ (Section 04.07), algebraically derive that $E(\chi^2_\nu)=\nu$ and $\Var(\chi^2_\nu)=2\nu$.
	\end{block}
	\vspace{0.1cm}
	\pause
	\begin{alertblock}{Strategy}
		\scriptsize
		Substitute $\alpha=\nu/2$ and $\beta=2$ directly into the general Gamma formulas.
	\end{alertblock}
\end{frame}

% SLIDE 16: Exercise Level 3 (Solution)
\begin{frame}{In-Class Exercise --- Level 3: Analytical (2/2)}
	\vspace{-0.05cm}
	\scriptsize
	Substitute $\alpha=\nu/2$, $\beta=2$: \pause
	\begin{block}{Derivation}
		\begin{align*}
			E(\chi^2_\nu) = \alpha\beta = \frac{\nu}{2}\cdot 2 = \nu, \qquad
			\Var(\chi^2_\nu) = \alpha\beta^2 = \frac{\nu}{2}\cdot 4 = 2\nu. \quad \qedhere
		\end{align*}
	\end{block}
	\vspace{0.05cm}
	\pause
	\begin{alertblock}{Interpretation}
		\scriptsize
		This derivation connects directly to Section 04.07: every property already proven for the general Gamma family is automatically inherited by its special cases, including the chi-squared.
	\end{alertblock}
\end{frame}

% SLIDE 17: Exercise Level 4 (Statement)
\begin{frame}{In-Class Exercise --- Level 4: Challenging (1/2)}
	\vspace{-0.1cm}
	\begin{block}{Problem 5.3.9 --- From Normal to Student's $t$ (Statement)}
		Using Fisher's theorem, show that $T = \dfrac{\bar X-\mu}{S/\sqrt n} = \dfrac{Z}{\sqrt{\chi^2_{n-1}/(n-1)}}$, where $Z=(\bar X-\mu)/(\sigma/\sqrt n)\sim N(0,1)$. Why is this decomposition relevant when $\sigma$ is unknown?
	\end{block}
	\vspace{0.15cm}
	\pause
	\begin{alertblock}{Strategy}
		\scriptsize
		Divide numerator and denominator of $T$ by $\sigma/\sqrt n$, and recognize $S^2/\sigma^2$ in terms of Fisher's theorem.
	\end{alertblock}
\end{frame}

% SLIDE 18: Exercise Level 4 (Solution)
\begin{frame}{In-Class Exercise --- Level 4: Challenging (2/2)}
	\vspace{-0.05cm}
	\scriptsize
	Divide numerator and denominator: \pause
	\begin{block}{Derivation}
		\scriptsize
		\begin{align*}
			T = \frac{\bar X-\mu}{S/\sqrt n} = \frac{(\bar X-\mu)/(\sigma/\sqrt n)}{S/\sigma} = \frac{Z}{\sqrt{S^2/\sigma^2}} = \frac{Z}{\sqrt{\chi^2_{n-1}/(n-1)}}. \quad \qedhere
		\end{align*}
	\end{block}
	\vspace{0.05cm}
	\pause
	\begin{alertblock}{Relevance}
		\scriptsize
		When $\sigma$ is unknown, $T$ is no longer exactly Normal: it is the ratio of a standard Normal to the square root of an independent chi-squared divided by its degrees of freedom --- precisely the definition of Student's $t$-distribution studied in Section 05.04.
	\end{alertblock}
\end{frame}

% SLIDE 19: Closing and Chapter Perspective
\begin{frame}{Closing Section and Chapter Perspective}
	\vspace{-0.2cm}
	\begin{block}{Synthesis: Chi-Squared and Fisher's Theorem}
		\scriptsize
		Fisher's theorem is the bridge between Normal distribution theory and practical inference with unknown $\sigma$: it tells us exactly how $S^2$ is distributed, and that it does so independently of $\bar X$. Without this result, we could not build exact confidence intervals for $\sigma^2$ nor derive Student's $t$-distribution.
	\end{block}
	\vspace{0.05cm}
	\pause
	\begin{alertblock}{Key Lessons and Next Step}
		\begin{itemize}\itemsep=0.05cm
			\item \textbf{Chi-squared:} $\chi^2_\nu=\sum Z_i^2$, with $E=\nu$, $\Var=2\nu$, a special case of the Gamma.
			\item \textbf{Fisher's theorem:} for normal samples, $\bar X \perp S^2$ and $(n-1)S^2/\sigma^2\sim\chi^2_{n-1}$.
			\item \textbf{Application:} exact (asymmetric) confidence intervals and hypothesis tests for $\sigma^2$.
			\item \textbf{Next Section:} \textcolor{TecRojo}{Student's $t$-Distribution and Small Samples}.
		\end{itemize}
	\end{alertblock}
