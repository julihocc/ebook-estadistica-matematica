% SLIDE 03: Motivation
\begin{frame}{From PDF to Moments: Integration as a Natural Operation}
	\footnotesize
	\begin{motivacion}[Why continuous expectation?]
		We know the PDF and CDF of a continuous variable. Now we need numerical summaries that capture the essential properties: the \emph{expectation} $\E[X]$ (center of mass of the distribution) and the \emph{variance} $\Var(X)$ (dispersion around the center). These moments guide decisions in inference, simulation, and risk analysis.
	\end{motivacion}
	\vspace{0.15cm}
	\pause
	\begin{itemize}\itemsep=0.1cm
		\item \textbf{Expectation:} $\E[X] = \int x f(x)\,dx$, a natural generalization of $\E[X] = \sum x P(X=x)$ via Riemann sums. \pause
		\item \textbf{Variance:} $\Var(X) = \E[(X - \mu)^{2}] = \E[X^{2}] - (\E[X])^{2}$, measures mean squared dispersion. \pause
		\item \textbf{LOTUS:} $\E[g(X)] = \int g(x) f(x)\,dx$ avoids deriving the distribution of $g(X)$.
	\end{itemize}
\end{frame}

% SLIDE 03B: Definition and Properties
\begin{frame}{Definition of Continuous Expectation and Variance}
	\vspace{-0.15cm}
	\begin{columns}[T]
		\begin{column}{0.48\textwidth}
			\begin{block}{Expectation}
				\footnotesize
				$\E[X] = \int_{-\infty}^{\infty} x f(x)\,dx$ under convergence $\int |x| f(x)\,dx < \infty$. For light-tailed distributions, the integral converges; for Cauchy, it diverges.
			\end{block}
		\end{column}
		\begin{column}{0.48\textwidth}
			\begin{alertblock}{Variance and LOTUS}
				\footnotesize
				$\Var(X) = \E[(X-\mu)^{2}] = \E[X^{2}] - (\E[X])^{2}$. LOTUS: $\E[g(X)] = \int g(x) f(x)\,dx$ without deriving the distribution of $g(X)$. Linearity: $\E[aX+b] = a\E[X]+b$.
			\end{alertblock}
		\end{column}
	\end{columns}
\end{frame}

% SLIDE 03C: Linearity and Variance Properties
\begin{frame}{Linearity of Expectation and Variance Properties}
	\vspace{-0.15cm}
	\begin{columns}[T]
		\begin{column}{0.48\textwidth}
			\begin{block}{Variance Properties}
				\footnotesize
				\begin{itemize}\itemsep=0.03cm
					\item $\Var(X) \ge 0$ always, with equality iff $X$ is constant.
					\item $\Var(aX + b) = a^{2} \Var(X)$ (affine linearity).
				\end{itemize}
			\end{block}
		\end{column}
		\begin{column}{0.48\textwidth}
			\begin{alertblock}{Linearity Theorem}
				\footnotesize
				$\E[aX+b] = a\E[X]+b$ and $\E[X+Y] = \E[X]+\E[Y]$, even for \emph{dependent} $X, Y$. In contrast, $\Var(X+Y) = \Var(X) + \Var(Y) + 2\cov(X,Y)$ requires independence to drop the covariance term.
			\end{alertblock}
		\end{column}
	\end{columns}
\end{frame}

% SLIDE 04: Central Moments and Law of Total Variance
\begin{frame}{Central Moments and Law of Total Variance}
	\vspace{-0.15cm}
	\begin{columns}[T]
		\begin{column}{0.48\textwidth}
			\begin{block}{Skewness and Kurtosis}
				\footnotesize
				Skewness: $\gamma_{1} = \E[(X-\mu)^{3}]/\sigma^{3}$ measures asymmetry. Kurtosis: $\gamma_{2} = \E[(X-\mu)^{4}]/\sigma^{4} - 3$ measures tail heaviness. For Exponential(1): $\gamma_{1} = 2$, $\gamma_{2} = 6$.
			\end{block}
		\end{column}
		\begin{column}{0.48\textwidth}
			\begin{alertblock}{Law of Total Variance}
				\footnotesize
				$\Var(X) = \E[\Var(X \mid Y)] + \Var(\E[X \mid Y])$. Fundamental in ANOVA: decomposes total variance into within-group and between-group components. For hierarchical models with $Y$ as group index.
			\end{alertblock}
		\end{column}
	\end{columns}
\end{frame}

% SLIDE 05: Applications
\begin{frame}{Applications in Engineering, Science, and Data Science}
	\vspace{-0.1cm}
	\begin{itemize}\itemsep=0.1cm
		\item \textbf{Uncertainty Propagation:} If $X = \mu + \epsilon$ with $\epsilon \sim N(0, \sigma^2)$, the standard error of the mean of $n$ measurements is $\sigma/\sqrt{n}$. \pause
		\item \textbf{Risk Quantification:} VaR at 95\% is the quantile $q_{0.05}$ of the loss distribution; CVaR is $\E[X \mid X \le q_{0.05}]$. \pause
		\item \textbf{Regression Analysis:} Residuals must have zero mean and constant variance (homoscedasticity). \pause
		\item \textbf{Quality Control:} $\bar{x}$ charts use $\mu \pm 3\sigma/\sqrt{n}$ as control limits.
	\end{itemize}
\end{frame}

\begin{frame}{In-Class Exercise --- Level 1: Fundamental (1/2)}
	\vspace{-0.1cm}
	\begin{block}{Problem 4.3.1 --- Expectation and Variance of Triangular PDF (Statement)}
		Let $X$ have triangular PDF $f(x) = 2x$ for $0 \le x \le 1$. Calculate $\E[X] = \int_{0}^{1} 2x^{2}\,dx$, $\E[X^{2}] = \int_{0}^{1} 2x^{3}\,dx$, and verify $\Var(X) = \E[X^{2}] - (\E[X])^{2}$.
	\end{block}
	\vspace{0.15cm}
	\pause
	\begin{alertblock}{Model Setup}
		\begin{itemize}\itemsep=0.04cm
			\item Support: $[0, 1]$ (bounded).
			\item Expectation: $\E[X] = \int x f(x)\,dx = \int_{0}^{1} 2x^{2}\,dx$.
			\item Variance by identity: $\Var(X) = \E[X^{2}] - (\E[X])^{2}$.
		\end{itemize}
	\end{alertblock}
\end{frame}

% SLIDE 11: Exercise Level 1 (Solution)
\begin{frame}{In-Class Exercise --- Level 1: Fundamental (2/2)}
	\vspace{-0.05cm}
	\scriptsize
	Compute the integrals and verify: \pause
	\begin{block}{Expectation and Second Moment}
		\begin{align*}
			\E[X] &= \int_{0}^{1} 2x^{2}\,dx = \left[ \dfrac{2x^{3}}{3} \right]_{0}^{1} = \dfrac{2}{3}. \\
			\E[X^{2}] &= \int_{0}^{1} 2x^{3}\,dx = \left[ \dfrac{x^{4}}{2} \right]_{0}^{1} = \dfrac{1}{2}.
		\end{align*}
	\end{block}
	\vspace{0.05cm}
	\pause
	\begin{alertblock}{Variance and Comparison with Median}
		\scriptsize
		$\Var(X) = \E[X^{2}] - (\E[X])^{2} = \dfrac{1}{2} - \dfrac{4}{9} = \dfrac{1}{18} \approx 0.0556$. The median is $m = 1/\sqrt{2} \approx 0.7071 > \E[X] \approx 0.6667$, difference due to asymmetry of the triangular PDF.
	\end{alertblock}
\end{frame}

% SLIDE 13: Exercise Level 2 (Solution)
\begin{frame}{In-Class Exercise --- Level 2: Operational (2/2)}
	\vspace{-0.05cm}
	\scriptsize
	Compute the central moments and compare: \pause
	\begin{block}{Central Moments of Exponential(1)}
		\begin{align*}
			\mu = 1, \quad \sigma = 1, \quad \E[(X-1)^{3}] = 2, \quad \E[(X-1)^{4}] = 9.
		\end{align*}
	\end{block}
	\vspace{0.05cm}
	\pause
	\begin{alertblock}{Skewness and Kurtosis}
		\scriptsize
		\begin{align*}
			\gamma_{1} = \dfrac{2}{1^{3}} = 2 \quad \text{(positive skew, right tail)}, \quad
			\gamma_{2} = \dfrac{9}{1^{4}} - 3 = 6 \quad \text{(excess kurtosis)}.
		\end{align*}
		Compared with the Normal ($\gamma_{1} = 0$, $\gamma_{2} = 0$), the Exponential is \emph{more asymmetric} and has \emph{heavier tails}. Monte Carlo: $\hat{\gamma}_{1} \approx 1.99$, $\hat{\gamma}_{2} \approx 5.87$ (with $N = 250{,}000$).
	\end{alertblock}
\end{frame}

% SLIDE 14: Exercise Level 3 (Statement)
\begin{frame}{In-Class Exercise --- Level 3: Analytical (1/2)}
	\vspace{-0.1cm}
	\begin{block}{Problem 4.3.7 --- Linearity of Expectation (Statement)}
		Let $X$ be continuous with PDF $f(x)$ and $\E[X] = \mu$. Prove that $\E[aX + b] = a\mu + b$ and $\E[X + Y] = \E[X] + \E[Y]$ for constants $a, b$ and random variables $X, Y$.
	\end{block}
	\vspace{0.1cm}
	\pause
	\begin{alertblock}{Deduction Strategy}
		\scriptsize
		Use the linearity of integration and Fubini's theorem for the sum. Expectation is linear even for dependent variables (unlike variance).
	\end{alertblock}
\end{frame}

% SLIDE 15: Exercise Level 3 (Solution)
\begin{frame}{In-Class Exercise --- Level 3: Analytical (2/2)}
	\vspace{-0.05cm}
	\scriptsize
	Demonstrate linearity via integration: \pause
	\begin{block}{Proof}
		\scriptsize
		\begin{align*}
			\E[aX + b] &= \int (ax + b) f(x)\,dx = a \int x f(x)\,dx + b \int f(x)\,dx = a\mu + b.
		\end{align*}
		For $\E[X + Y]$ with joint PDF $f_{X,Y}(x, y)$:
		\begin{align*}
			\E[X + Y] &= \int\int (x + y) f_{X,Y}(x, y)\,dx\,dy = \E[X] + \E[Y]
		\end{align*}
		by Fubini's theorem. Linearity holds \emph{even for dependent variables} (unlike $\Var(X+Y) = \Var(X) + \Var(Y) + 2\cov(X,Y)$). \quad \qedhere
	\end{block}
\end{frame}

% SLIDE 16: Exercise Level 4 (Statement)
\begin{frame}{In-Class Exercise --- Level 4: Challenging (1/2)}
	\vspace{-0.1cm}
	\begin{block}{Problem 4.3.9 --- Law of Total Variance (Statement)}
		Let $X$ be continuous with PDF $f(x)$ and $Y$ another variable. Prove the variance decomposition identity:
		$$\Var(X) = \E[\Var(X \mid Y)] + \Var(\E[X \mid Y])$$
		This identity is fundamental in ANOVA and hierarchical models.
	\end{block}
	\vspace{0.15cm}
	\pause
	\begin{alertblock}{Application}
		\scriptsize
		In one-way ANOVA, total variability decomposes into \emph{between-treatment} ($\Var(\E[X \mid Y])$) and \emph{within-treatment} ($\E[\Var(X \mid Y)]$) variability.
	\end{alertblock}
\end{frame}

% SLIDE 17: Exercise Level 4 (Solution)
\begin{frame}{In-Class Exercise --- Level 4: Challenging (2/2)}
	\vspace{-0.05cm}
	\scriptsize
	Prove via the law of total expectation: \pause
	\begin{block}{Formal Proof}
		\scriptsize
		By the law of total expectation, $\E[X] = \E[\E[X \mid Y]]$. Expanding:
		\begin{align*}
			\Var(X) &= \E[(X - \E[X])^{2}] = \E[\E[(X - \E[X \mid Y] + \E[X \mid Y] - \E[X])^{2} \mid Y]] \\
			&= \E[\Var(X \mid Y)] + \E[\Var(\E[X \mid Y] \mid Y)] + 2\E[\E[(X - \E[X \mid Y])(\E[X \mid Y] - \E[X]) \mid Y]].
		\end{align*}
		The third term is $0$ (tower property). The first two terms are $\E[\Var(X \mid Y)]$ and $\Var(\E[X \mid Y])$, respectively. Therefore: $\Var(X) = \E[\Var(X \mid Y)] + \Var(\E[X \mid Y])$. \quad \qedhere
	\end{block}
\end{frame}
