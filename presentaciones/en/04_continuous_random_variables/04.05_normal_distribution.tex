% SLIDE 03: Motivation
\begin{frame}{The Gaussian Bell: Universality of the Normal}
	\footnotesize
	\begin{motivacion}[Why the Normal distribution?]
		The Normal distribution $N(\mu, \sigma^{2})$ is arguably the most important in all of statistics. Its bell shape models measurement errors, signal noise, human heights, exam scores, and many other natural phenomena. By the Central Limit Theorem, the sum (or average) of i.i.d. variables converges to the Normal regardless of the original distribution.
	\end{motivacion}
	\vspace{0.15cm}
	\pause
	\begin{itemize}\itemsep=0.1cm
		\item \textbf{Bell shape:} Symmetric around $\mu$ with dispersion controlled by $\sigma$. \pause
		\item \textbf{Standardization:} Any Normal reduces to $N(0,1)$ via $Z = (X-\mu)/\sigma$. \pause
		\item \textbf{Universality:} Appears in the CLT as the limit of sums; foundation of statistical inference.
	\end{itemize}
\end{frame}

% SLIDE 03B: Definition and Z-Score
\begin{frame}{Definition and Z-Score Standardization}
	\vspace{-0.15cm}
	\begin{columns}[T]
		\begin{column}{0.48\textwidth}
			\begin{block}{PDF and Symmetry}
				\footnotesize
				$f(x) = \frac{1}{\sigma\sqrt{2\pi}} \exp\left(-\frac{(x-\mu)^{2}}{2\sigma^{2}}\right)$. Bell-shaped, symmetric around $\mu$. Support $\mathbb{R}$. Mean, median, and mode all equal $\mu$.
			\end{block}
		\end{column}
		\begin{column}{0.48\textwidth}
			\begin{alertblock}{Z-Score and Empirical Rule}
				\footnotesize
				$Z = (X-\mu)/\sigma \sim N(0,1)$. Empirical rule: $P(|X-\mu| \le 1\sigma) \approx 68\%$, $P(|X-\mu| \le 2\sigma) \approx 95\%$, $P(|X-\mu| \le 3\sigma) \approx 99.7\%$.
			\end{alertblock}
		\end{column}
	\end{columns}
\end{frame}

% SLIDE 04: MGF and CLT
\begin{frame}{MGF, Sum-Closure, and Central Limit Theorem}
	\vspace{-0.15cm}
	\begin{columns}[T]
		\begin{column}{0.48\textwidth}
			\begin{block}{MGF and Sum-Closure}
				\footnotesize
				$M_{X}(t) = e^{\mu t + \sigma^{2}t^{2}/2}$, with $M'(0)=\mu$ and $M''(0)=\mu^{2}+\sigma^{2}$. Sum of independent normals is normal: $N(\mu_{1}, \sigma_{1}^{2}) + N(\mu_{2}, \sigma_{2}^{2}) \sim N(\mu_{1}+\mu_{2}, \sigma_{1}^{2}+\sigma_{2}^{2})$.
			\end{block}
		\end{column}
		\begin{column}{0.48\textwidth}
			\begin{alertblock}{Central Limit Theorem}
				\footnotesize
				For i.i.d. with finite mean and variance, $Z_{n} = (\bar{X}-\mu)/(\sigma/\sqrt{n}) \xrightarrow{d} N(0,1)$. Universality: limit distribution is normal regardless of original distribution.
			\end{alertblock}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{In-Class Exercise --- Level 1: Fundamental (1/2)}
	\vspace{-0.1cm}
	\begin{block}{Problem 4.6.1 --- Normal Distribution $N(100, 15^{2})$ (Statement)}
		Let $X \sim N(100, 15^{2})$. Write the PDF $f(x) = \frac{1}{15\sqrt{2\pi}} e^{-(x-100)^{2}/450}$, and calculate $P(85 \le X \le 115)$, $P(X > 130)$, and $P(X < 70)$ using the standardization $Z = (X - 100)/15$.
	\end{block}
	\vspace{0.15cm}
	\pause
	\begin{alertblock}{Model Setup}
		\begin{itemize}\itemsep=0.04cm
			\item Mean $\mu = 100$ and standard deviation $\sigma = 15$.
			\item Standardization: $Z = (X - 100)/15 \sim N(0, 1)$.
			\item Probability: $P(a \le X \le b) = P((a-100)/15 \le Z \le (b-100)/15)$.
		\end{itemize}
	\end{alertblock}
\end{frame}

% SLIDE 11: Exercise Level 1 (Solution)
\begin{frame}{In-Class Exercise --- Level 1: Fundamental (2/2)}
	\vspace{-0.05cm}
	\scriptsize
	Standardize and use the standard Normal CDF: \pause
	\begin{block}{Probability Calculation}
		\begin{itemize}\itemsep=0.05cm
			\item $P(85 \le X \le 115) = P(-1 \le Z \le 1) = 2\Phi(1) - 1 \approx 0.6827$ (68.27\% of mass).
			\item $P(X > 130) = P(Z > 2) = 1 - \Phi(2) \approx 0.0228$ (2.28\% right tail).
			\item $P(X < 70) = P(Z < -2) = \Phi(-2) \approx 0.0228$ (2.28\% left tail).
		\end{itemize}
	\end{block}
	\vspace{0.05cm}
	\pause
	\begin{alertblock}{Interpretation}
		\scriptsize
		68.27\% of observations fall in $[85, 115]$ (one standard deviation around the mean). Only 4.56\% of the total mass lies outside $\pm 2\sigma$ (i.e., outside $[70, 130]$), confirming the 68-95-99.7 rule.
	\end{alertblock}
\end{frame}

% SLIDE 14: Exercise Level 3 (Statement)
\begin{frame}{In-Class Exercise --- Level 3: Analytical (1/2)}
	\vspace{-0.1cm}
	\begin{block}{Problem 4.6.7 --- Normal MGF Derivation (Statement)}
		Prove that if $X \sim N(\mu, \sigma^{2})$, its moment generating function is $M_{X}(t) = e^{\mu t + \sigma^{2}t^{2}/2}$. Verify that $M'(0) = \mu$ and $M''(0) = \mu^{2} + \sigma^{2}$.
	\end{block}
	\vspace{0.1cm}
	\pause
	\begin{alertblock}{Deduction Strategy}
		\scriptsize
		Complete the square in the exponent of the integral: $tx - (x-\mu)^{2}/(2\sigma^{2}) = -(x - (\mu + \sigma^{2}t))^{2}/(2\sigma^{2}) + \mu t + \sigma^{2}t^{2}/2$. Factor the exponential and use the Gaussian integral.
	\end{alertblock}
\end{frame}

% SLIDE 15: Exercise Level 3 (Solution)
\begin{frame}{In-Class Exercise --- Level 3: Analytical (2/2)}
	\vspace{-0.05cm}
	\scriptsize
	Complete the square and differentiate: \pause
	\begin{block}{Formal Proof}
		\scriptsize
		\begin{align*}
			M_{X}(t) &= \int e^{tx} \cdot \dfrac{1}{\sigma\sqrt{2\pi}} e^{-(x-\mu)^{2}/(2\sigma^{2})}\,dx \\
			&= e^{\mu t + \sigma^{2}t^{2}/2} \int \dfrac{1}{\sigma\sqrt{2\pi}} e^{-(x - (\mu + \sigma^{2}t))^{2}/(2\sigma^{2})}\,dx = e^{\mu t + \sigma^{2}t^{2}/2},
		\end{align*}
		where the integral is exactly 1 (Normal PDF with mean $\mu + \sigma^{2}t$ and variance $\sigma^{2}$). Differentiating: $M'(0) = \mu$, $M''(0) = \mu^{2} + \sigma^{2}$. \quad \qedhere
	\end{block}
\end{frame}

		For $X \sim \text{Bin}(400, 0.25)$ use the Normal approximation with Yates correction to calculate $P(X \ge 120)$, compare with the exact value, and determine $n$ required for $\text{SE}(\hat{p}) < 0.01$.
	\end{block}
	\vspace{0.15cm}
	\pause
	\begin{alertblock}{Application}
		\scriptsize
		In a standardized exam with $n = 400$ multiple-choice questions with 4 options, the number of correct answers $X$ follows $\text{Bin}(400, 0.25)$. The Normal approximation avoids computing tail probabilities with 400 terms.
	\end{alertblock}
\end{frame}

% SLIDE 17: Exercise Level 4 (Solution)
\begin{frame}{In-Class Exercise --- Level 4: Challenging (2/2)}
	\vspace{-0.05cm}
	\scriptsize
	Apply the Normal approximation with Yates correction: \pause
	\begin{block}{Numerical Calculation}
		\scriptsize
		With $\mu = np = 100$ and $\sigma = \sqrt{np(1-p)} = \sqrt{75} \approx 8.66$:
		\begin{align*}
			P(X \ge 120) \approx P(Y \ge 119.5) = 1 - \Phi\left(\dfrac{119.5 - 100}{8.66}\right) = 1 - \Phi(2.25) \approx 0.0122.
		\end{align*}
		The exact value is $P(X \ge 120) = 1 - \text{binom.cdf}(119, 400, 0.25) \approx 0.0133$, giving an absolute error of $0.0011$.
	\end{block}
	\vspace{0.05cm}
	\pause
	\begin{alertblock}{Required Sample Size}
		\scriptsize
		For $\text{SE}(\hat{p}) = \sqrt{p(1-p)/n} < 0.01$: $n > 0.25 \cdot 0.75/0.0001 = 1875$. At least 1875 observations are needed for a standard error smaller than 1\%.
	\end{alertblock}
