% SLIDE 03: Motivation
\begin{frame}{From Discrete to Continuous: The Density Function}
	\footnotesize
	\begin{motivacion}[Why continuous random variables?]
		In Chapter 03 we worked with variables taking discrete values: number of successes, defects, calls. But many phenomena vary continuously: waiting time, measurement error, voltage in a circuit, weight of a product. For these variables, probability mass ``spreads'' over a continuum, and the \emph{probability density} $f(x)$ encodes how that mass concentrates.
	\end{motivacion}
	\vspace{0.15cm}
	\pause
	\begin{itemize}\itemsep=0.1cm
		\item \textbf{Continuity:} $X$ takes values on an interval (or union of intervals), not a discrete set. \pause
		\item \textbf{Probability as area:} $P(a \le X \le b) = \int_{a}^{b} f(x)\,dx$, not a sum of point masses. \pause
		\item \textbf{Technical condition:} $\int_{-\infty}^{\infty} f(x)\,dx = 1$ guarantees the total mass equals one.
	\end{itemize}
\end{frame}

% SLIDE 03B: Formal Definition and CDF
\begin{frame}{Formal Definition: PDF and Continuous CDF}
	\vspace{-0.15cm}
	\begin{columns}[T]
		\begin{column}{0.48\textwidth}
			\begin{block}{Probability Density Function (PDF)}
				\footnotesize
				A continuous random variable $X$ has PDF $f: \mathbb{R} \to [0, \infty)$ with $f(x) \ge 0$ and $\int f(x)\,dx = 1$. Probabilities are areas:
				$$P(a \le X \le b) = \int_{a}^{b} f(x)\,dx$$
			\end{block}
		\end{column}
		\begin{column}{0.48\textwidth}
			\begin{alertblock}{Continuous CDF}
				\footnotesize
				$F(x) = \int_{-\infty}^{x} f(t)\,dt$ with $F(-\infty) = 0$ and $F(+\infty) = 1$. By the Fundamental Theorem of Calculus, $f(x) = F'(x)$.
			\end{alertblock}
		\end{column}
	\end{columns}
\end{frame}

% SLIDE 03C: Continuous Cumulative Distribution Function
\begin{frame}{Continuous CDF and Interval Probabilities}
\scriptsize
\begin{block}{Definition}
For a continuous variable with density $f_X$, the cumulative distribution is
\[
F_X(x)=P(X\le x)=\int_{-\infty}^{x}f_X(t)\,dt,
\qquad P(a<X\le b)=F_X(b)-F_X(a).
\]
\end{block}
\begin{alertblock}{Density--CDF Link}
Where $F_X$ is differentiable, $F_X'(x)=f_X(x)$ and $P(X=x)=0$ for every point.
\end{alertblock}
\end{frame}

% SLIDE 04: Support and Continuous LOTUS
\begin{frame}{Continuous Support, Expectation, and LOTUS}
	\vspace{-0.15cm}
	\begin{columns}[T]
		\begin{column}{0.48\textwidth}
			\begin{block}{Continuous Support}
				\footnotesize
				$\mathcal{S}_X = \{x \in \mathbb{R} : f(x) > 0\}$ can be bounded $[a,b]$, semi-bounded $[a, \infty)$ or $(-\infty, b]$, or all $\mathbb{R}$. Examples: Uniform$([0,1])$, Exp$(\lambda) \to [0, \infty)$, Normal $\to \mathbb{R}$.
			\end{block}
		\end{column}
		\begin{column}{0.48\textwidth}
			\begin{alertblock}{Continuous LOTUS}
				\footnotesize
				$\E[g(X)] = \int g(x) f(x)\,dx$ under regularity conditions. Special cases: $g(x) = x$ gives $\E[X]$; $g(x) = (x - \mu)^{2}$ gives $\Var(X) = \E[X^{2}] - (\E[X])^{2}$.
			\end{alertblock}
		\end{column}
	\end{columns}
\end{frame}

% SLIDE 04B: Zero-Probability Points and Continuous Expectation
\begin{frame}{Fundamental Property and Continuous Expectation}
	\vspace{-0.15cm}
	\begin{columns}[T]
		\begin{column}{0.48\textwidth}
			\begin{alertblock}{Fundamental Property: $P(X = x) = 0$}
				\footnotesize
				\begin{itemize}\itemsep=0.05cm
					\item For a continuous variable, any single point has zero probability.
					\item Probabilities are computed over \emph{intervals}, not points.
					\item Hence $P(a \le X \le b) = P(a < X \le b) = P(a \le X < b) = P(a < X < b)$.
				\end{itemize}
			\end{alertblock}
		\end{column}
		\begin{column}{0.48\textwidth}
			\begin{block}{Continuous Expectation}
				\footnotesize
				$$\E[X] = \int_{-\infty}^{\infty} x \cdot f(x)\,dx$$
				provided the integral exists ($\int |x| f(x)\,dx < \infty$). Computationally, $\Var(X) = \E[X^{2}] - (\E[X])^{2}$ and $\sigma = \sqrt{\Var(X)}$.
			\end{block}
		\end{column}
	\end{columns}
\end{frame}

% SLIDE 04C: Continuous LOTUS
\begin{frame}{Continuous LOTUS: Expectations of Functions}
\scriptsize
\begin{block}{Law of the Unconscious Statistician}
For any integrable function $g$,
\[
\E[g(X)]=\int_{\mathcal{S}_X}g(x)f_X(x)\,dx.
\]
This avoids deriving the density of $Y=g(X)$ explicitly.
\end{block}
\begin{exampleblock}{Example}
Taking $g(x)=x^2$ gives the second moment $\E[X^2]$ and therefore
$\Var(X)=\E[X^2]-\E[X]^2$.
\end{exampleblock}
\end{frame}

\begin{frame}{Closing Section and Modular Perspective}
	\vspace{-0.2cm}
	\begin{block}{Synthesis: The Bridge from Discrete to Continuous}
		\scriptsize
		The Probability Density Function $f(x)$ is the continuous analogue of the PMF: instead of mass at points, it distributes probability over a continuum. The formal properties are analogous (non-negativity, integral = 1), but the calculation techniques require integration instead of summation.
	\end{block}
	\vspace{0.05cm}
	\pause
	\begin{alertblock}{Key Lessons and Next Step}
		\begin{itemize}\itemsep=0.05cm
			\item \textbf{Normalization:} $\int f = 1$ guarantees total mass; PDFs are relative, not absolute.
			\item \textbf{Support:} Determines where to focus computation; can be bounded, semi-bounded, or unbounded.
			\item \textbf{LOTUS:} $\E[g(X)] = \int g(x) f(x)\,dx$ avoids deriving the distribution of $g(X)$.
			\item \textbf{Next Section:} \textcolor{TecRojo}{Continuous CDF and Quantiles}.
		\end{itemize}
	\end{alertblock}
