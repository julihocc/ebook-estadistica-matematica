% Slide 03: Motivation — Conditional Inversion
\begin{frame}{Motivation --- Inverting Conditionals}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Forward Direction (Likelihood)}
        \small
        \begin{itemize}\itemsep=0.1cm
          \item In physical and industrial modeling, we know the probability of observing an \textbf{effect or evidence} $B$ given an underlying cause or hypothesis $A$:
          $$P(\text{Effect} \mid \text{Cause}) = P(B \mid A)$$
          \item \emph{Example:} The probability that a vibration sensor triggers an alarm ($B$) when a turbine suffers mechanical failure ($A$).
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Inverse Direction (Posterior)}
        \small
        \begin{itemize}\itemsep=0.1cm
          \item In clinical diagnostics, robotics, and data science, the real problem operates in reverse: we observe evidence $B$ and must infer the cause $A$:
          $$P(\text{Cause} \mid \text{Effect}) = P(A \mid B)$$
          \item \emph{Example:} Given that the alarm triggered ($B$), what is the true probability of a catastrophic mechanical failure ($A$)?
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

% Slide 04: Deriving Bayes' Theorem (Simple Form)
\begin{frame}{Deriving Bayes' Theorem (Simple Form)}
  \fontsize{7.5pt}{9pt}\selectfont
  \begin{block}{Symmetry of the Joint Intersection}
    \fontsize{7.5pt}{9pt}\selectfont
    Let $A$ and $B$ be two events in the sample space $S$ with strictly positive probabilities ($P(A)>0, P(B)>0$). By the multiplication rule of conditional probability, the joint probability $P(A \cap B)$ is algebraically symmetric:
    $$P(A \cap B) = P(B \mid A)P(A) = P(A \mid B)P(B)$$
    Solving directly for $P(A \mid B)$ yields the elementary formulation of \textbf{Bayes' Theorem}:
    $$P(A \mid B) = \frac{P(B \mid A)P(A)}{P(B)}$$
  \end{block}
  \vspace{-0.15cm}
  \pause
  \begin{alertblock}{Anatomy and Odds Form Formulation}
    \fontsize{7.5pt}{9pt}\selectfont
    \begin{itemize}\itemsep=0.08cm
      \item $P(A)$: \textbf{Prior Probability} (initial belief or domain knowledge before observing evidence).
      \item $P(B \mid A)$: \textbf{Likelihood} (compatibility of the observed evidence with the hypothesis).
      \item $P(A \mid B)$: \textbf{Posterior Probability} (updated belief after incorporating data $B$).
      \item \textbf{Odds Form:} For two competing hypotheses $A_1, A_2$: $\displaystyle \frac{P(A_1 \mid B)}{P(A_2 \mid B)} = \frac{P(B \mid A_1)}{P(B \mid A_2)} \cdot \frac{P(A_1)}{P(A_2)}$.
    \end{itemize}
  \end{alertblock}
\end{frame}

% Slide 05: Bayes' Theorem for Exhaustive Partitions
\begin{frame}{Bayes' Theorem for Exhaustive Partitions}
  \fontsize{7.5pt}{9pt}\selectfont
  \begin{block}{Expansion via the Law of Total Probability}
    \fontsize{7.5pt}{9pt}\selectfont
    In complex engineering systems, the sample space is partitioned into a \textbf{finite, exhaustive set} of mutually exclusive causes $\{B_1, B_2, \ldots, B_k\}$ such that $S = \bigcup_{i=1}^k B_i$. To evaluate the probability of any particular state $B_j$ given the observable event $A$, we expand the denominator $P(A)$:
    $$P(B_j \mid A) = \frac{P(A \mid B_j)P(B_j)}{\sum_{i=1}^k P(A \mid B_i)P(B_i)}, \quad \text{for all } j \in \{1, 2, \ldots, k\}$$
  \end{block}
  \vspace{-0.15cm}
  \pause
  \begin{alertblock}{Interpretation of the Denominator (Normalization Constant)}
    \fontsize{7.5pt}{9pt}\selectfont
    The denominator $\sum_{i=1}^k P(A \mid B_i)P(B_i)$ represents the \textbf{marginal probability of the observable event} $P(A)$. It acts as a strict normalization constant ensuring that the sum of the posterior probabilities across all possible causes equals exactly unity: $\sum_{j=1}^k P(B_j \mid A) = 1$.
  \end{alertblock}
\end{frame}

% Slide 06: Clinical Diagnostics and the Base Rate Fallacy
\begin{frame}{Clinical Diagnostics and the Base Rate Fallacy}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Conditional Test Metrics}
        \small
        Let $E$ be the presence of a disease and $T^+$ a positive clinical test:
        \begin{itemize}\itemsep=0.1cm
          \item \textbf{Sensitivity ($P(T^+ \mid E)$):} Probability of detecting the disease when present.
          \item \textbf{Specificity ($P(T^- \mid E')$):} Probability of testing negative when healthy.
          \item \textbf{False Positive Rate:} $P(T^+ \mid E') = 1 - \text{Specificity}$.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{The Base Rate Fallacy (Prevalence)}
        \small
        \begin{itemize}\itemsep=0.1cm
          \item The \textbf{Positive Predictive Value (PPV)} is:
          $$P(E \mid T^+) = \frac{P(T^+ \mid E)P(E)}{P(T^+)}$$
          \item If prior prevalence $P(E)$ is extremely low ($1\%$), the volume of false positives across the healthy population overwhelms true positives, keeping $P(E \mid T^+) \ll 50\%$ despite high sensitivity.
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

% Slide 07: Sequential Bayesian Updating
\begin{frame}{Sequential Bayesian Updating in Engineering}
  \fontsize{7.5pt}{9pt}\selectfont
  \begin{block}{Today's Posterior is Tomorrow's Prior Distribution}
    \fontsize{7.5pt}{9pt}\selectfont
    When an industrial system or autonomous robot receives a continuous stream of independent observations given the hypotheses ($B_1, B_2, \ldots, B_t$), Bayesian inference operates recursively without re-evaluating the entire history from scratch:
    $$P(A \mid B_1) = \frac{P(B_1 \mid A)P(A)}{P(B_1)} \implies P(A \mid B_1 \cap B_2) = \frac{P(B_2 \mid A)P(A \mid B_1)}{P(B_2 \mid B_1)}$$
  \end{block}
  \vspace{-0.15cm}
  \pause
  \begin{alertblock}{Convergence Toward Truth in Robotics and AI}
    \fontsize{7.5pt}{9pt}\selectfont
    \begin{itemize}\itemsep=0.08cm
      \item At each iteration $t$, the new evidence $B_t$ adjusts the likelihood ratio to update the belief over the underlying system state $\theta$.
      \item This recursive architecture forms the mathematical foundation of \textbf{Kalman Filters}, simultaneous localization and mapping (SLAM), and parameter estimation in modern probabilistic models.
    \end{itemize}
  \end{alertblock}
\end{frame}

% Slide 08: Naive Bayes Classification in High Dimensions
\begin{frame}{Naive Bayes Classification in High Dimensions}
  \fontsize{7.5pt}{9pt}\selectfont
  \begin{block}{The Conditional Independence Assumption (Naive Bayes)}
    \fontsize{7.5pt}{9pt}\selectfont
    In text classification (e.g., spam filters) or multi-symptom diagnostics $W = W_1 \cap W_2 \cap \ldots \cap W_m$, computing the exact joint likelihood $P(W \mid C)$ is computationally intractable. The \textbf{Naive Bayes} algorithm assumes that attributes are conditionally independent given the class $C$:
    $$P(W_1 \cap W_2 \cap \ldots \cap W_m \mid C) = \prod_{i=1}^m P(W_i \mid C)$$
  \end{block}
  \vspace{-0.15cm}
  \pause
  \begin{alertblock}{Maximum A Posteriori Decision Rule}
    \fontsize{7.5pt}{9pt}\selectfont
    The posterior probability of belonging to class $C_k$ given the observed attribute vector $W$ is evaluated via the product of marginal likelihoods:
    $$P(C_k \mid W) = \frac{P(C_k) \prod_{i=1}^m P(W_i \mid C_k)}{\sum_{j} P(C_j) \prod_{i=1}^m P(W_i \mid C_j)}$$
    Despite the simplifying assumption of independence, Naive Bayes demonstrates remarkable empirical accuracy in real-time decision engines.
  \end{alertblock}
\end{frame}

% Slide 14: Synthesis and Conclusion
\begin{frame}{Synthesis and Conclusion --- Bayes' Theorem and Inversion}
  \fontsize{7.5pt}{9pt}\selectfont
  \begin{block}{Conceptual Pillars of Section 02.05}
    \fontsize{7.5pt}{9pt}\selectfont
    \begin{itemize}\itemsep=0.12cm
      \item \textbf{Rigorous Conditional Inversion:} Bayes' Theorem provides the exact algebraic mechanism to infer causal probabilities ($P(A \mid B)$) from forward empirical likelihoods ($P(B \mid A)$).
      \item \textbf{The Critical Weight of the Prior:} Disregarding the prior marginal probability $P(A)$ inevitably leads to severe cognitive biases and the base rate fallacy across technical diagnostics and medical screening.
      \item \textbf{Sequential Updating and Robotics:} The recursive transition where one cycle's posterior distribution becomes the prior belief for the next cycle is the foundation of AI algorithms and real-time state estimation.
    \end{itemize}
  \end{block}
\end{frame}
