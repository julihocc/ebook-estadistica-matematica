% Slide 03: Motivation — Conditional Inversion
\begin{frame}{Motivation --- Inverting Conditionals}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Forward Direction (Likelihood)}
        \small
        \begin{itemize}\itemsep=0.1cm
          \item In physical and industrial modeling, we know the probability of observing an \textbf{effect or evidence} $B$ given an underlying cause or hypothesis $A$:
          $$P(\text{Effect} \mid \text{Cause}) = P(B \mid A)$$
          \item \emph{Example:} The probability that a vibration sensor triggers an alarm ($B$) when a turbine suffers mechanical failure ($A$).
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Inverse Direction (Posterior)}
        \small
        \begin{itemize}\itemsep=0.1cm
          \item In clinical diagnostics, robotics, and data science, the real problem operates in reverse: we observe evidence $B$ and must infer the cause $A$:
          $$P(\text{Cause} \mid \text{Effect}) = P(A \mid B)$$
          \item \emph{Example:} Given that the alarm triggered ($B$), what is the true probability of a catastrophic mechanical failure ($A$)?
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

% Slide 04: Deriving Bayes' Theorem (Simple Form)
\begin{frame}{Deriving Bayes' Theorem (Simple Form)}
  \fontsize{7.5pt}{9pt}\selectfont
  \begin{block}{Symmetry of the Joint Intersection}
    \fontsize{7.5pt}{9pt}\selectfont
    Let $A$ and $B$ be two events in the sample space $S$ with strictly positive probabilities ($P(A)>0, P(B)>0$). By the multiplication rule of conditional probability, the joint probability $P(A \cap B)$ is algebraically symmetric:
    $$P(A \cap B) = P(B \mid A)P(A) = P(A \mid B)P(B)$$
    Solving directly for $P(A \mid B)$ yields the elementary formulation of \textbf{Bayes' Theorem}:
    $$P(A \mid B) = \frac{P(B \mid A)P(A)}{P(B)}$$
  \end{block}
  \vspace{-0.15cm}
  \pause
  \begin{alertblock}{Anatomy and Odds Form Formulation}
    \fontsize{7.5pt}{9pt}\selectfont
    \begin{itemize}\itemsep=0.08cm
      \item $P(A)$: \textbf{Prior Probability} (initial belief or domain knowledge before observing evidence).
      \item $P(B \mid A)$: \textbf{Likelihood} (compatibility of the observed evidence with the hypothesis).
      \item $P(A \mid B)$: \textbf{Posterior Probability} (updated belief after incorporating data $B$).
      \item \textbf{Odds Form:} For two competing hypotheses $A_1, A_2$: $\displaystyle \frac{P(A_1 \mid B)}{P(A_2 \mid B)} = \frac{P(B \mid A_1)}{P(B \mid A_2)} \cdot \frac{P(A_1)}{P(A_2)}$.
    \end{itemize}
  \end{alertblock}
\end{frame}

% Slide 05: Bayes' Theorem for Exhaustive Partitions
\begin{frame}{Bayes' Theorem for Exhaustive Partitions}
  \fontsize{7.5pt}{9pt}\selectfont
  \begin{block}{Expansion via the Law of Total Probability}
    \fontsize{7.5pt}{9pt}\selectfont
    In complex engineering systems, the sample space is partitioned into a \textbf{finite, exhaustive set} of mutually exclusive causes $\{B_1, B_2, \ldots, B_k\}$ such that $S = \bigcup_{i=1}^k B_i$. To evaluate the probability of any particular state $B_j$ given the observable event $A$, we expand the denominator $P(A)$:
    $$P(B_j \mid A) = \frac{P(A \mid B_j)P(B_j)}{\sum_{i=1}^k P(A \mid B_i)P(B_i)}, \quad \text{for all } j \in \{1, 2, \ldots, k\}$$
  \end{block}
  \vspace{-0.15cm}
  \pause
  \begin{alertblock}{Interpretation of the Denominator (Normalization Constant)}
    \fontsize{7.5pt}{9pt}\selectfont
    The denominator $\sum_{i=1}^k P(A \mid B_i)P(B_i)$ represents the \textbf{marginal probability of the observable event} $P(A)$. It acts as a strict normalization constant ensuring that the sum of the posterior probabilities across all possible causes equals exactly unity: $\sum_{j=1}^k P(B_j \mid A) = 1$.
  \end{alertblock}
\end{frame}

% Slide 06: Clinical Diagnostics and the Base Rate Fallacy
\begin{frame}{Clinical Diagnostics and the Base Rate Fallacy}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Conditional Test Metrics}
        \small
        Let $E$ be the presence of a disease and $T^+$ a positive clinical test:
        \begin{itemize}\itemsep=0.1cm
          \item \textbf{Sensitivity ($P(T^+ \mid E)$):} Probability of detecting the disease when present.
          \item \textbf{Specificity ($P(T^- \mid E')$):} Probability of testing negative when healthy.
          \item \textbf{False Positive Rate:} $P(T^+ \mid E') = 1 - \text{Specificity}$.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{The Base Rate Fallacy (Prevalence)}
        \small
        \begin{itemize}\itemsep=0.1cm
          \item The \textbf{Positive Predictive Value (PPV)} is:
          $$P(E \mid T^+) = \frac{P(T^+ \mid E)P(E)}{P(T^+)}$$
          \item If prior prevalence $P(E)$ is extremely low ($1\%$), the volume of false positives across the healthy population overwhelms true positives, keeping $P(E \mid T^+) \ll 50\%$ despite high sensitivity.
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

% Slide 07: Sequential Bayesian Updating
\begin{frame}{Sequential Bayesian Updating in Engineering}
  \fontsize{7.5pt}{9pt}\selectfont
  \begin{block}{Today's Posterior is Tomorrow's Prior Distribution}
    \fontsize{7.5pt}{9pt}\selectfont
    When an industrial system or autonomous robot receives a continuous stream of independent observations given the hypotheses ($B_1, B_2, \ldots, B_t$), Bayesian inference operates recursively without re-evaluating the entire history from scratch:
    $$P(A \mid B_1) = \frac{P(B_1 \mid A)P(A)}{P(B_1)} \implies P(A \mid B_1 \cap B_2) = \frac{P(B_2 \mid A)P(A \mid B_1)}{P(B_2 \mid B_1)}$$
  \end{block}
  \vspace{-0.15cm}
  \pause
  \begin{alertblock}{Convergence Toward Truth in Robotics and AI}
    \fontsize{7.5pt}{9pt}\selectfont
    \begin{itemize}\itemsep=0.08cm
      \item At each iteration $t$, the new evidence $B_t$ adjusts the likelihood ratio to update the belief over the underlying system state $\theta$.
      \item This recursive architecture forms the mathematical foundation of \textbf{Kalman Filters}, simultaneous localization and mapping (SLAM), and parameter estimation in modern probabilistic models.
    \end{itemize}
  \end{alertblock}
\end{frame}

% Slide 08: Naive Bayes Classification in High Dimensions
\begin{frame}{Naive Bayes Classification in High Dimensions}
  \fontsize{7.5pt}{9pt}\selectfont
  \begin{block}{The Conditional Independence Assumption (Naive Bayes)}
    \fontsize{7.5pt}{9pt}\selectfont
    In text classification (e.g., spam filters) or multi-symptom diagnostics $W = W_1 \cap W_2 \cap \ldots \cap W_m$, computing the exact joint likelihood $P(W \mid C)$ is computationally intractable. The \textbf{Naive Bayes} algorithm assumes that attributes are conditionally independent given the class $C$:
    $$P(W_1 \cap W_2 \cap \ldots \cap W_m \mid C) = \prod_{i=1}^m P(W_i \mid C)$$
  \end{block}
  \vspace{-0.15cm}
  \pause
  \begin{alertblock}{Maximum A Posteriori Decision Rule}
    \fontsize{7.5pt}{9pt}\selectfont
    The posterior probability of belonging to class $C_k$ given the observed attribute vector $W$ is evaluated via the product of marginal likelihoods:
    $$P(C_k \mid W) = \frac{P(C_k) \prod_{i=1}^m P(W_i \mid C_k)}{\sum_{j} P(C_j) \prod_{i=1}^m P(W_i \mid C_j)}$$
    Despite the simplifying assumption of independence, Naive Bayes demonstrates remarkable empirical accuracy in real-time decision engines.
  \end{alertblock}
\end{frame}

\begin{frame}{In-Class Exercise --- Tier 1: Fundamental (1/2)}
  \begin{block}{Problem 2.5.4 --- Studying and Posterior Probability of Success}
    \footnotesize
    A university student prepares for a comprehensive general examination. Statistical records indicate that:
    \begin{itemize}\itemsep=0.1cm
      \item If the student studies diligently ($E$), the conditional probability of passing ($A$) is $90\%$: $P(A \mid E) = 0.90$.
      \item If the student does not study ($E'$), the conditional probability of passing is only $30\%$: $P(A \mid E') = 0.30$.
      \item The prior probability that the student prepares rigorously is $70\%$: $P(E) = 0.70$.
    \end{itemize}
    \begin{enumerate}\itemsep=0.15cm
      \item Calculate the total marginal probability that the student passes the exam ($P(A)$).
      \item Given that the student successfully passed ($A$), what is the posterior conditional probability that they actually studied ($P(E \mid A)$)?
    \end{enumerate}
  \end{block}
\end{frame}

% Slide 10B: In-Class Exercise — Tier 1: Fundamental (Solution)
\begin{frame}{In-Class Exercise --- Tier 1: Fundamental (2/2)}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{1. Total Probability of Passing ($P(A)$)}
        \fontsize{6.5pt}{7.8pt}\selectfont
        The exhaustive hypotheses are $\{E, E'\}$ with $P(E)=0.70$ and $P(E')=0.30$. By the Law of Total Probability:
        $$P(A) = P(A \mid E)P(E) + P(A \mid E')P(E')$$
        Substituting the given parameters:
        $$P(A) = (0.90)(0.70) + (0.30)(0.30)$$
        $$P(A) = 0.63 + 0.09 = \mathbf{0.72 \text{ (72\%)}}$$
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{2. Bayesian Inversion ($P(E \mid A)$)}
        \fontsize{6.5pt}{7.8pt}\selectfont
        Applying Bayes' Theorem to invert the conditional direction:
        $$P(E \mid A) = \frac{P(A \mid E)P(E)}{P(A)}$$
        Substituting the numerator and the evaluated denominator:
        $$P(E \mid A) = \frac{0.63}{0.72} = \frac{7}{8} = \mathbf{0.875 \text{ (87.5\%)}}$$
      \end{alertblock}
    \end{column}
  \end{columns}
  \vspace{-0.1cm}
  \pause
  \begin{exampleblock}{Pedagogical Interpretation}
    \fontsize{6.5pt}{7.8pt}\selectfont
    Knowing that the student passed increases our certainty regarding their academic preparation from $70\%$ a priori to $87.5\%$ a posteriori. Observed success provides rigorous empirical evidence favoring diligent effort.
  \end{exampleblock}
\end{frame}

% Slide 11A: In-Class Exercise — Tier 2: Operational (Statement)
\begin{frame}{In-Class Exercise --- Tier 2: Operational (1/2)}
  \begin{block}{Problem 2.5.2 --- Clinical Diagnostic Test Evaluation (PPV \& NPV)}
    \footnotesize
    A clinical laboratory evaluates a new molecular screening test for a low-prevalence pathology. The epidemiological and technical test parameters are:
    \begin{itemize}\itemsep=0.1cm
      \item \textbf{Prior Prevalence:} In the general population, $2\%$ of individuals carry the pathology ($P(E) = 0.02$).
      \item \textbf{Sensitivity:} If a patient is diseased, the probability that the test tests positive is $0.95$ ($P(+ \mid E) = 0.95$).
      \item \textbf{Specificity:} If a patient is healthy ($E'$), the probability that the test tests negative is $0.90$ ($P(- \mid E') = 0.90$). Thus, the false positive rate is $P(+ \mid E') = 0.10$.
    \end{itemize}
    \begin{enumerate}\itemsep=0.15cm
      \item Calculate the Positive Predictive Value (PPV: $P(E \mid +)$) and explain why it is surprisingly low.
      \item Calculate the Negative Predictive Value (NPV: $P(E' \mid -)$).
    \end{enumerate}
  \end{block}
\end{frame}

% Slide 11B: In-Class Exercise — Tier 2: Operational (Solution)
\begin{frame}{In-Class Exercise --- Tier 2: Operational (2/2)}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{1. Positive Predictive Value (PPV)}
        \fontsize{6.2pt}{7.4pt}\selectfont
        Compute the total marginal probability of a positive test $P(+)$:
        $$P(+) = P(+ \mid E)P(E) + P(+ \mid E')P(E')$$
        $$P(+) = (0.95)(0.02) + (0.10)(0.98) = 0.019 + 0.098 = 0.117$$
        Applying Bayes' Theorem to obtain the PPV ($P(E \mid +)$):
        $$P(E \mid +) = \frac{0.019}{0.117} \approx \mathbf{0.16239 \text{ (16.24\%)}}$$
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{2. Negative Predictive Value (NPV)}
        \fontsize{6.2pt}{7.4pt}\selectfont
        Compute the total marginal probability of a negative test $P(-) = 1 - P(+) = 0.883$:
        $$P(E' \mid -) = \frac{P(- \mid E')P(E')}{P(-)}$$
        $$P(E' \mid -) = \frac{(0.90)(0.98)}{0.883} = \frac{0.882}{0.883} \approx \mathbf{0.99887 \text{ (99.89\%)}}$$
      \end{alertblock}
    \end{column}
  \end{columns}
  \vspace{-0.1cm}
  \pause
  \begin{exampleblock}{Base Rate Fallacy Analysis}
    \fontsize{6.2pt}{7.4pt}\selectfont
    Exactly $83.76\%$ of individuals who receive a positive screening test are false alarms! This occurs not because the test lacks sensitivity ($95\%$), but because the vast demographic mass of healthy individuals ($98\%$) multiplied by the false positive error ($10\%$) generates a volume of false positives ($0.098$) exceeding true positives ($0.019$) by over 5 to 1.
  \end{exampleblock}
\end{frame}

\begin{frame}{In-Class Exercise --- Tier 3: Analytical (1/2)}
  \begin{block}{Problem 2.5.3 --- Bayesian Audit in Semiconductor Manufacturing}
    \footnotesize
    A semiconductor assembly plant operates three concurrent production lines ($M_1, M_2, M_3$) contributing $50\%$, $30\%$, and $20\%$ of the total output, respectively. The structural defect rates ($D$) verified by quality assurance are:
    $$P(D \mid M_1) = 0.03, \qquad P(D \mid M_2) = 0.05, \qquad P(D \mid M_3) = 0.07$$
    \begin{enumerate}\itemsep=0.15cm
      \item If an external quality auditor selects a defective microchip from the final storage bin, determine the posterior probability vector $P(M_i \mid D)$ to isolate the manufacturing source.
      \item Prove analytically which manufacturing line is statistically the most likely origin of the defect, explaining the trade-off between production volume and error rate.
    \end{enumerate}
  \end{block}
\end{frame}

% Slide 12B: In-Class Exercise — Tier 3: Analytical (Solution)
\begin{frame}{In-Class Exercise --- Tier 3: Analytical (2/2)}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{1. Marginal Rate and Posterior Vector}
        \fontsize{6.2pt}{7.4pt}\selectfont
        Calculate the total marginal defect rate in the plant:
        $$P(D) = \sum_{i=1}^3 P(D \mid M_i)P(M_i)$$
        $$P(D) = (0.03)(0.50) + (0.05)(0.30) + (0.07)(0.20)$$
        $$P(D) = 0.015 + 0.015 + 0.014 = \mathbf{0.044 \text{ (4.4\%)}}$$
        Inverting with Bayes for each line $M_i$:
        \begin{itemize}\itemsep=0.04cm
          \item $P(M_1 \mid D) = \frac{0.015}{0.044} = \frac{15}{44} \approx \mathbf{34.09\%}$.
          \item $P(M_2 \mid D) = \frac{0.015}{0.044} = \frac{15}{44} \approx \mathbf{34.09\%}$.
          \item $P(M_3 \mid D) = \frac{0.014}{0.044} = \frac{14}{44} \approx \mathbf{31.82\%}$.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{2. Tie Analysis and Audit Recommendation}
        \fontsize{6.2pt}{7.4pt}\selectfont
        \textbf{Conclusion:} Manufacturing lines $M_1$ and $M_2$ tie exactly at $34.09\%$ probability as the most likely source of any randomly isolated defective component.
        \vspace{0.1cm}
        
        \textbf{Algorithmic Justification:} Even though line $M_1$ maintains the lowest defect rate ($3\%$) compared to $M_2$ ($5\%$), its massive manufacturing volume ($50\%$ vs $30\%$) produces exactly the same absolute quantity of failed chips into the total inventory:
        $$P(D \cap M_1) = P(D \cap M_2) = 0.015$$
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

    \footnotesize
    \begin{itemize}\itemsep=0.15cm
      \item \textbf{Prosecutor's Fallacy (2.5.9):} In a forensic investigation, a crime scene DNA sample matches a suspect ($E$). The random match probability in a population of $N=5,000,000$ is $\alpha=10^{-6}$. The prosecutor claims the suspect has a $99.9999\%$ probability of guilt. Use Bayesian odds formulation to evaluate the true posterior probability $P(\text{Guilty} \mid E)$.
      \item \textbf{Naive Bayes Spam Filter (2.5.10):} A server processes $60\%$ spam ($S$) and $40\%$ legitimate emails ($S'$). Under conditional independence, evaluate a message containing keywords $W_1 \cap W_2 \cap W_3$ with likelihoods: $P(W \mid S)=(0.80)(0.70)(0.60)=0.336$ and $P(W \mid S')=(0.10)(0.20)(0.10)=0.002$. Compute $P(S \mid W)$.
    \end{itemize}

% Slide 13B: In-Class Exercise — Tier 4: Challenging (Solution)
\begin{frame}{In-Class Exercise --- Tier 4: Challenging (2/2)}
  \vspace{-0.15cm}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{1. Prosecutor's Fallacy in Odds ($N=5 \times 10^6$)}
        \fontsize{5.5pt}{6.4pt}\selectfont
        Prior odds in a population of 5 million potential suspects:
        $$\text{Odds}(G) = \frac{1 / 5,000,000}{1} = 2 \times 10^{-7}$$
        Likelihood Ratio ($\text{LR}$) given a DNA match:
        $$\text{LR} = \frac{P(E \mid G)}{P(E \mid G')} = \frac{1}{10^{-6}} = 1,000,000$$
        Posterior odds and true probability of guilt:
        $$\text{Odds}(G \mid E) = 10^6 \times (2 \times 10^{-7}) = 0.20 = \frac{1}{5}$$
        $$P(G \mid E) = \frac{0.20}{1 + 0.20} = \frac{1}{6} \approx \mathbf{16.67\%}$$
        The prosecutor's assertion ($99.9999\%$) is a logical fallacy ignoring the marginal base rate!
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{2. Naive Bayes Spam Inversion ($W$)}
        \fontsize{5.5pt}{6.4pt}\selectfont
        Compute the joint marginal probability of keyword vector $W$:
        $$P(W) = P(W \mid S)P(S) + P(W \mid S')P(S')$$
        $$P(W) = (0.336)(0.60) + (0.002)(0.40) = 0.2016 + 0.0008 = 0.2024$$
        Evaluate the exact posterior probability that the message is spam ($S$):
        $$P(S \mid W) = \frac{0.2016}{0.2024} = \frac{252}{253} \approx \mathbf{99.6047\%}$$
        The simultaneous presence of all three indicators classifies the email with near-certainty.
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Synthesis and Conclusion --- Bayes' Theorem and Inversion}
  \fontsize{7.5pt}{9pt}\selectfont
  \begin{block}{Conceptual Pillars of Section 02.05}
    \fontsize{7.5pt}{9pt}\selectfont
    \begin{itemize}\itemsep=0.12cm
      \item \textbf{Rigorous Conditional Inversion:} Bayes' Theorem provides the exact algebraic mechanism to infer causal probabilities ($P(A \mid B)$) from forward empirical likelihoods ($P(B \mid A)$).
      \item \textbf{The Critical Weight of the Prior:} Disregarding the prior marginal probability $P(A)$ inevitably leads to severe cognitive biases and the base rate fallacy across technical diagnostics and medical screening.
      \item \textbf{Sequential Updating and Robotics:} The recursive transition where one cycle's posterior distribution becomes the prior belief for the next cycle is the foundation of AI algorithms and real-time state estimation.
    \end{itemize}
  \end{block}
\end{frame}
